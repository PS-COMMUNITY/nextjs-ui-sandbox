   176→// ============ الحفظ المؤجل ============
   177→let persistTimer: ReturnType<typeof setTimeout> | null = null;
   178→
   179→function schedulePersist(get: () => LabStore) {
   180→  if (persistTimer) clearTimeout(persistTimer);
   181→  persistTimer = setTimeout(() => {
   182→    const s = get();
   183→    if (!s.hydrated) return;
   184→    const shape: PersistShape = {
   185→      version: APP_VERSION,
   186→      simulation: s.sim,
   187→      tech: s.tech,
   188→      logs: s.logs,
   189→      quizScores: s.quizScores,
   190→      missionsDone: s.missionsDone,
   191→      lessonsRead: s.lessonsRead,
   192→      settings: { quality: s.quality },
   193→      savedAt: Date.now(),
   194→    };
   195→    const res = saveAll(shape);
   196→    if (res.ok) {
   197→      useLabStore.setState({ saveStatus: { available: true, lastSavedAt: Date.now(), lastErrorAr: null } });
   198→    } else {
   199→      useLabStore.setState({ saveStatus: { available: true, lastSavedAt: null, lastErrorAr: res.messageAr } });
   200→    }
   201→  }, 400);
   202→}

   225→  tech: null,
   226→  logs: [],
   227→  quizScores: {},
   228→  missionsDone: [],
   229→  lessonsRead: [],
   230→  activeMissionId: null,

   234→  hydrate: () => {
   235→    if (!storageAvailable()) {
   236→      set({ hydrated: true, saveStatus: { available: false, lastSavedAt: null, lastErrorAr: 'تخزين المتصفح غير متاح — لن يُحفظ التقدم' } });
   237→      return;
   238→    }
   239→    const data = loadAll();
   240→    if (data) {
   241→      const sim = normalizeSimulationState(data.simulation);
   242→      // استئناف جلسة محفوظة والجهاز يعمل فيها: نُرجع ساعة التمهيد إلى الماضي
   243→      // حتى تظهر نتيجتها المحسومة مباشرة بدل الت stuck على شعار البدء
   244→      const bootStartedAt =
   245→        sim.flags.powerOn && sim.flags.acConnected && !sim.peripherals.psuDead
   246→          ? Date.now() - 9999
   247→          : null;
   248→      set({
   249→        sim,
   250→        bootStartedAt,
   251→        tech: data.tech ?? null,
   252→        logs: data.logs ?? [],
   253→        quizScores: data.quizScores ?? {},
   254→        missionsDone: data.missionsDone ?? [],
   255→        lessonsRead: data.lessonsRead ?? [],
   256→        quality: data.settings?.quality ?? 'medium',
   257→        hydrated: true,
   258→        saveStatus: { available: true, lastSavedAt: data.savedAt, lastErrorAr: null },
   259→      });
   260→    } else {
   261→      set({ hydrated: true });
   262→    }
   263→  },
   264→
   265→  setScreen: (screen) => set({ screen, selected: screen === 'lab' ? get().selected : null }),
   266→  select: (selected) => set({ selected }),
   267→  toggleXray: () => set({ xray: !get().xray }),
   268→  toggleExploded: () => set({ exploded: !get().exploded }),
   269→  setIsolated: (isolated) => set({ isolated }),

   273→  },
   274→  setSidePanelsHidden: (sidePanelsHidden) => set({ sidePanelsHidden }),
   275→
   276→  toggleFlag: (f) => {
   277→    const s = get();
   278→    const check = canToggleFlag(s.sim, f);
   279→    const info = FLAG_INFO[f];
   280→    if (!check.ok) {
   281→      s.logEvent('flag', `${info.labelAr}: مرفوض`, false, check.messageAr);
   282→      toast.error(check.messageAr);
   283→      return;
   284→    }
   285→    const sim = applyToggleFlag(s.sim, f);
   286→    // ساعة تمهيد الشاشة: تضبط لحظة التشغيل وتُصفّر عند الإطفاء
   287→    const bootStartedAt = f === 'powerOn' ? (sim.flags.powerOn ? Date.now() : null) : s.bootStartedAt;
   288→    set({ sim, bootStartedAt });
   289→    s.logEvent('flag', check.messageAr || info.labelAr, true, info.hintAr);
   290→    toast.success(check.messageAr || info.labelAr);
   291→    schedulePersist(get);
   292→  },
   293→
   294→  // ============ آلة الحالات: فصل جزئي → استخراج → التقاط → تركيب ============
   295→  detachPart: (id) => {
   296→    const s = get();
   297→    if (id === 'case') return;
   298→    if (isCable(id)) {
   299→      // الكابلات تفصل بحركة واحدة — التوجيه للمسار الموحد
   300→      s.removePart(id);
   301→      return;
   302→    }
   303→    if (s.sim.parts[id] !== 'installed') {
   304→      toast.error(`${nameOf(id)}: ليست في حالة تسمح بالفصل الجزئي`);
   305→      return;
   306→    }
   307→    const check = canRemove(s.sim, id);
   308→    if (!check.ok) {
   309→      s.logEvent('remove', `${nameOf(id)}: فصل مرفوض`, false, check.messageAr);
   310→      toast.error(check.messageAr);
   311→      return;
   312→    }
   313→    const sim = applyDetach(s.sim, id);
   314→    set({ sim });
   315→    s.logEvent('remove', `فصل جزئي: ${nameOf(id)}`, true, 'القطعة ما زالت في مكانها — استخرجها إلى الصينية عند الجاهزية');
   316→    toast.success(`${nameOf(id)}: فُصلت جزئيًا — استخرجها إلى الصينية`);
   317→    schedulePersist(get);
   318→  },
   319→
   320→  extractPart: (id) => {
   321→    const s = get();
   322→    if (s.sim.parts[id] !== 'separated') {
   323→      toast.error(`${nameOf(id)}: يجب فصلها جزئيًا أولًا قبل استخراجها`);
   324→      return;
   325→    }
   326→    const sim = applyExtract(s.sim, id);
   327→    set({ sim });
   328→    s.logEvent('remove', `استخراج ${nameOf(id)} إلى صينية القطع`, true);
   329→    toast.success(`${nameOf(id)}: نُقلت إلى صينية القطع`);
   330→    schedulePersist(get);
   331→  },
   332→
   333→  preparePart: (id) => {
   334→    const s = get();
   335→    if (s.sim.parts[id] !== 'removed') {
   336→      toast.error(`${nameOf(id)}: القطعة ليست في الصينية`);
   337→      return;
   338→    }
   339→    const sim = applyPrepare(s.sim, id);
   340→    set({ sim, selected: id });
   341→    s.logEvent('install', `التقاط ${nameOf(id)} للتركيب`, true, 'راجع شروط التركيب ثم اضغط «ركّب»');
   342→    toast.success(`${nameOf(id)}: ملتقطة — راجع شروط التركيب ثم ركّبها`);
   343→    schedulePersist(get);
   344→  },
   345→
   346→  installPart: (id) => {
   347→    const s = get();
   348→    const status = s.sim.parts[id];
   349→    if (status === 'installed') {
   350→      toast.error(`${nameOf(id)}: مثبتة أصلًا`);
   351→      return;
   352→    }
   353→    const check = canInstall(s.sim, id);
   354→    if (!check.ok) {
   355→      // INSTALL_REJECTED: حدث رفض تعليمي — القطعة تبقى بانتظار التركيب
   356→      s.logEvent('install', `${nameOf(id)}: تركيب مرفوض`, false, check.messageAr);
   357→      toast.error(check.messageAr);
   358→      return;
   359→    }
   360→    const sim = applyInstall(s.sim, id);
   361→    set({ sim });
   362→    s.logEvent('install', `تركيب ${nameOf(id)}`, true, 'أُعيدت القطعة إلى موضع تثبيتها — أغلق المزالج والبراغي إن وجدت');
   363→    toast.success(`${nameOf(id)}: رُكّبت في موضعها`);
   364→    schedulePersist(get);
   365→  },
   366→
   367→  removePart: (id) => {
   368→    const s = get();
   369→    const name = nameOf(id);
   370→    if (id === 'case') {
   371→      toast.error('الهيكل هو جسم الجهاز؛ استخدم زر فتح/إغلاق الغطاء الجانبي.');
   372→      return;
   373→    }
   374→    const check = canRemove(s.sim, id);
   375→    if (!check.ok) {
   376→      s.logEvent('remove', `${name}: فك مرفوض`, false, check.messageAr);
   377→      toast.error(check.messageAr);
   378→      return;
   379→    }
   380→    // الكابلات: فصل مباشر إلى الصينية — القطع الكبيرة: فصل جزئي ثم استخراج
   381→    const sim = isCable(id) ? applyRemove(s.sim, id) : applyDetach(s.sim, id);
   382→    set({ sim });
   383→    if (isCable(id)) {
   384→      s.logEvent('remove', `فصل ${name}`, true, 'نُقل إلى صينية القطع');
   385→      toast.success(`${name}: فُصل إلى صينية القطع`);
   386→    } else {
   387→      s.logEvent('remove', `فصل جزئي: ${name}`, true, 'أكمل الاستخراج إلى الصينية');
   388→      toast.success(`${name}: فُصل جزئيًا — أكمل الاستخراج`);
   389→    }
   390→    schedulePersist(get);
   391→  },
   392→
   393→  damagePart: (id) => {
   394→    const s = get();
   395→    if (id === 'case') return;
   396→    const sim = applyDamage(s.sim, id);
   397→    set({ sim, lastDiagnosis: null });
   398→    s.logEvent('other', `تلف تدريبي مقصود: ${nameOf(id)}`, false, 'وضع الاستكشاف: القطعة تالفة — استبدلها من المخزن التعليمي');
   399→    toast.warning(`${nameOf(id)}: تلف تدريبي — استبدلها من المخزن`);
   400→    schedulePersist(get);
   401→  },
   402→
   403→  repairPart: (id) => {
   404→    const s = get();
   405→    if (s.sim.parts[id] !== 'damaged') {
   406→      toast.error(`${nameOf(id)}: ليست تالفة`);
   407→      return;
   408→    }
   409→    const sim = applyRepair(s.sim, id);
   410→    set({ sim });
   411→    s.logEvent('install', `استبدال ${nameOf(id)} بقطعة سليمة`, true, 'من المخزن التعليمي');
   412→    toast.success(`${nameOf(id)}: استُبدلت بقطعة سليمة (في الصينية)`);
   413→    schedulePersist(get);
   414→  },
   415→
   416→  // ============ الطرفيات ============
   417→  toggleMonitor: () => {
   418→    const s = get();
   419→    const sim = { ...s.sim, peripherals: { ...s.sim.peripherals, monitorOn: !s.sim.peripherals.monitorOn } };
   420→    set({ sim });
   421→    s.logEvent('flag', sim.peripherals.monitorOn ? 'تشغيل الشاشة' : 'إطفاء الشاشة', true);
   422→    schedulePersist(get);
   423→  },
   424→
   425→  setNotepad: (text) => set({ notepad: text.slice(-400) }),
   426→
   427→  typeKey: (ch) => {
   428→    const s = get();
   429→    if (!s.notepad.length && ch.length === 1) {
   430→      // أول حرف — لا شيء خاص
   431→    }
   432→    let next = s.notepad;
   433→    if (ch === 'BACK') next = next.slice(0, -1);
   434→    else if (ch === 'SPACE') next = `${next} `;
   435→    else if (ch === 'ENTER') next = `${next}\n`;
   436→    else if (ch === 'CLR') next = '';
   437→    else next = next + ch;
   438→    set({ notepad: next.slice(-400) });
   439→  },
   440→
   441→  clearNotepad: () => set({ notepad: '' }),
   442→
   443→  setHintsVisible: (v) => set({ hintsVisible: v }),
   444→
   445→  setAirflow: (v) => set({ airflowVisible: v }),
   446→
   447→  setInputMode: (inputMode) => set({ inputMode }),
   448→
   449→  // ============ V6-C/D: النظام الافتراضي والأقراص والبرنامج الثابت ============
   450→
   451→  insertUsb: (family) => {
   452→    const s = get();

   865→  },

   867→  logEvent: (kind, labelAr, ok, detailAr) => {

   874→/** عدّاد القطع المثبتة (للشارات) — الهيكل مستثنى */
   875→export function installedCount(sim: SimulationState): number {
   876→  let n = 0;
   877→  for (const id of Object.keys(sim.parts) as ComponentId[]) {
   878→    if (id === 'case') continue;
   879→    if (isInstalled(sim.parts[id] as PartStatus)) n++;
   880→  }
