    79→/** هل الأصول الحقيقية جاهزة؟ (يُستخدم للقرارات التشخيصية) */

    84→/** تحميل كل النماذج مسبقًا — ينجح دائمًا (الفشل الفردي يتخطى النموذج) */
    85→export function preloadGlbAssets(): Promise<void> {
    86→  if (!preloadPromise) {
    87→    const loader = new GLTFLoader();
    88→    loader.setMeshoptDecoder(MeshoptDecoder);
    89→    const jobs = Object.values(DEFS).map(
    90→      (def) =>
    91→        new Promise<void>((resolve) => {
    92→          loader.load(
    93→            def.url,
    94→            (gltf) => {
    95→              const scene = gltf.scene;
    96→              const mats = new Set<THREE.Material>();
    97→              scene.traverse((obj) => {
    98→                if (!(obj instanceof THREE.Mesh)) return;
    99→                const arr = Array.isArray(obj.material) ? obj.material : [obj.material];
   100→                for (const m of arr) {
   101→                  if (def.forceTransparent?.some((n) => (m.name || '').toLowerCase().includes(n))) {
   102→                    m.transparent = true;
   103→                    m.depthWrite = false;
   104→                  }
   105→                  if (def.doubleSide) m.side = THREE.DoubleSide;
   106→                  if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = Math.min((m as THREE.MeshStandardMaterial).emissiveIntensity ?? 0, 1);
   107→                  mats.add(m);
   108→                }
   109→              });
   110→              cache.set(def.url, scene);
   111→              console.info('[glb] loaded:', def.url);
   112→              resolve();
   113→            },
   114→            undefined,
   115→            (err) => {
   116→              console.warn('[glb] FAILED:', def.url, String(err));
   117→              resolve(); // فشل التحميل → يبقى المجسم الإجرائي
   118→            },
   119→          );
   120→        }),
   121→    );
   122→    preloadPromise = Promise.all(jobs).then(() => undefined);
   123→  }
   124→  return preloadPromise;
   125→}

   131→function vec3(v: Vec3): THREE.Vector3 {
   132→  return new THREE.Vector3(v[0], v[1], v[2]);
   133→}
   134→
   135→/**
   136→ * يطبع نسخة النموذج على المرساة: يعيد (الجذر + الإزاحة المطبقة).
   137→ * التطبيع: world = R·S·p ثم إزاحة بحيث تقع نقطة المرساة عند الأصل.
   138→ */
   139→function normalize(src: THREE.Object3D, def: PartDef): { root: THREE.Group; offset: THREE.Vector3 } {
   140→  const root = new THREE.Group();
   141→  root.name = 'glb-normalized';
   142→  root.quaternion.setFromRotationMatrix(new THREE.Matrix4().makeBasis(vec3(def.basis[0]), vec3(def.basis[1]), vec3(def.basis[2])));
   143→  root.scale.set(def.scale[0], def.scale[1], def.scale[2]);
   144→  root.add(src);
   145→  root.updateMatrixWorld(true);
   146→  const bbox = new THREE.Box3().setFromObject(root);
   147→  const offset = new THREE.Vector3();
   148→  if (def.anchor === 'center') offset.copy(bbox.getCenter(new THREE.Vector3())).negate();
   149→  else if (def.anchor === 'minAll') offset.copy(bbox.min).negate();
   150→  else {
   151→    // minY: القاع عند Y=0 ومركز X/Z عند الصفر
   152→    const c = bbox.getCenter(new THREE.Vector3());
   153→    offset.set(c.x, bbox.min.y, c.z).negate();
   154→  }
   155→  // الإزاحة على الجذر نفسه (تُطبَّق بعد R·S في فضاء العالم) — لا تلمس src المحلي
   156→  root.position.copy(offset);
   157→  return { root, offset };
   158→}

   160→/** تفريغ مجموعة من أبنائها مع تحرير الهندسات (الخامات مشتركة من الـcache — لا تُحرر) */
   161→function clearGroup(g: THREE.Object3D, keep: THREE.Object3D[] = []): void {
   162→  for (const child of [...g.children]) {
   163→    if (keep.includes(child)) continue;
   164→    g.remove(child);
   165→    child.traverse((o) => {
   166→      if (o instanceof THREE.Mesh) o.geometry.dispose();
   167→    });
   168→  }
   169→}

   172→function setShadows(root: THREE.Object3D, cast: boolean, receive = false): void {
   173→  root.traverse((o) => {
   174→    if (o instanceof THREE.Mesh) {
   175→      o.castShadow = cast;
   176→      if (receive) o.receiveShadow = true;
   177→    }
   178→  });
   179→}

   181→/**
   182→ * غلاف pivot على مستوى المجموعة الديناميكية (فضاءها = فضاء العالم قبل التموضع):
   183→ * يسمح بتدوير الرافعة/المزلاج حول محور عالمي صحيح حتى لو كان الأب المباشر مُدارًا.
   184→ */
   185→function pivotAround(node: THREE.Object3D, group: THREE.Object3D, localY: Vec3): THREE.Object3D {
   186→  group.updateMatrixWorld(true);
   187→  const pivot = new THREE.Object3D();
   188→  pivot.name = `${node.name || 'node'}-pivot`;
   189→  node.getWorldPosition(pivot.position);
   190→  group.worldToLocal(pivot.position);
   191→  pivot.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), vec3(localY).normalize());
   192→  group.add(pivot);
   193→  pivot.attach(node);
   194→  return pivot;
   195→}

   197→/** برغي إجرائي صغير (للقطع التي لا تحمل براغٍ في النموذج الأصلي — أداة تفاعل الفك) */
   198→function makeScrewAt(axis: 'x' | 'y' | 'z', r: number, len: number): THREE.Mesh {
   199→  const geo = new THREE.CylinderGeometry(r, r, len, 10);
   200→  const m = new THREE.MeshStandardMaterial({ color: 0x9aa3ad, metalness: 0.85, roughness: 0.35 });
   201→  const mesh = new THREE.Mesh(geo, m);
   202→  if (axis === 'x') mesh.rotation.z = Math.PI / 2;
   203→  else if (axis === 'z') mesh.rotation.x = Math.PI / 2;
   204→  return mesh;
   205→}

   219→  let found: THREE.Mesh | null = null;

   221→    if (found || !(o instanceof THREE.Mesh)) return;

   258→/** سجل الارتكازات المقاسة لحظة التشغيل (يقرأه PCScene للتقارير والفحوص) */
   259→export interface GpuSeatReport {
   260→  seated: boolean;
   261→  /** مركز شق PCIe x16 في فضاء المشهد */

   269→  source: 'glb-anchors' | 'procedural-fallback';

   271→
   272→export const gpuSeat: GpuSeatReport = {
   273→  seated: false, slotPoint: null, bracketZ: null, gpuBox: null, clearance: null,
   274→  source: 'procedural-fallback',
   275→};

   277→/** مركز صندوق إحاطة عقدة باسمها داخل مجموعة (فضاء المجموعة الأب) */
   278→function anchorCenter(group: THREE.Object3D, name: string, out: THREE.Vector3): boolean {
   279→  const node = findNode(group, name);
   280→  if (!node) return false;
   281→  group.updateMatrixWorld(true);
   282→  const box = new THREE.Box3().setFromObject(node);
   283→  if (!isFinite(box.min.x)) {
   284→    // عقدة بلا شبكة: استخدم موضعها
   285→    node.getWorldPosition(out);
   286→    group.parent?.worldToLocal(out);
   287→    return true;
   288→  }
   289→  box.getCenter(out);
   290→  group.parent?.worldToLocal(out);
   291→  return true;
   292→}
   293→
   294→/**
   295→ * يجلس البطاقة في شق PCIe x16 الحقيقي (توجيه V6 §2.3):
   296→ *  1) مركز الشق من عقدة anchor_pcie_x16 في لوحة الأم الحقيقية (بديل إجرائي عند الفشل).
   297→ *  2) مستوى البراكيت من anchor_slot_1 في هيكل الصندوق الحقيقي.
   298→ *  3) القيود: الأطراف الذهبية (الحافة +X للبطاقة) داخل الشق، أعلى البطاقة بمستوى أعلى الشق،
   299→ *     والبراكيت على مستوى فتحات التوسعة — ثم يفحص التخلوص من المبرد والذاكرة ويسجل النتيجة.
   300→ */
   301→function alignGpuToSlot(visuals: PartVisuals): void {
   302→  const gpuGroup = visuals.dynamic.gpu;
   303→  const boardGroup = visuals.dynamic.motherboard;
   304→  gpuSeat.seated = false;
   305→  gpuSeat.clearance = null;
   306→
   307→  const slot = new THREE.Vector3();
   308→  const fromGlb = anchorCenter(boardGroup, 'anchor_pcie_x16', slot);
   309→  // مستوى فتحات التوسعة الخلفية من الهيكل الحقيقي
   310→  const bracket = new THREE.Vector3();
   311→  const caseOk = anchorCenter(visuals.caseShell, 'anchor_slot_1', bracket);
   312→
   313→  let seatX: number;
   314→  let topY: number;
   315→  let bracketZ: number;
   316→  if (fromGlb) {
   317→    // الأطراف تدخل قناة الشق ~2سم (الشق بارز من وجه اللوحة)
   318→    seatX = slot.x + 0.02;
   319→    topY = slot.y + 0.012;
   320→    gpuSeat.source = 'glb-anchors';
   321→  } else {
   322→    // بديل إجرائي: فتحة x16 في مصنع القطع (Y=2.0، وجه اللوحة 1.82)
   323→    seatX = 1.79;
   324→    topY = 2.01;
   325→    slot.set(1.79, 2.0, 1.42);
   326→  }
   327→  bracketZ = caseOk ? bracket.z : 0.12;
   328→  gpuSeat.slotPoint = [slot.x, slot.y, slot.z];
   329→  gpuSeat.bracketZ = bracketZ;
   330→
   331→  gpuGroup.updateMatrixWorld(true);
   332→  const box = new THREE.Box3().setFromObject(gpuGroup);
   333→  if (isFinite(box.min.x)) {
   334→    const dx = seatX - box.max.x;           // دفع الأطراف نحو الشق
   335→    const dy = topY - box.max.y;            // إنزال البطاقة حتى مستوى الشق
   336→    const dz = bracketZ + 0.004 - box.min.z; // البراكيت على مستوى الفتحات
   337→    gpuGroup.position.x += dx;
   338→    gpuGroup.position.y += dy;
   339→    gpuGroup.position.z += dz;
   340→  } else {
   341→    // bbox غير متاح (مجسم إجرائي) — استخدم MOUNTS كما هي
   342→    gpuGroup.position.set(1.47, 2.2, 1.42);
   343→  }
   344→  gpuGroup.updateMatrixWorld(true);
   345→
   346→  const box2 = new THREE.Box3().setFromObject(gpuGroup);
   347→  gpuSeat.gpuBox = { min: [box2.min.x, box2.min.y, box2.min.z], max: [box2.max.x, box2.max.y, box2.max.z] };
   348→  gpuSeat.seated = true;
   349→
   350→  // فحص تصادم تقريبي (توجيه V6 §2.3.7): مبرد CPU وRAM مقابل جسم البطاقة
   351→  const coolerBox = new THREE.Box3().setFromObject(visuals.dynamic.cooler);
   352→  const ramBox = new THREE.Box3().setFromObject(visuals.dynamic.ram);
   353→  const gapTo = (other: THREE.Box3): number => {
   354→    if (!isFinite(other.min.x)) return 99;
   355→    const dy = Math.max(other.min.y - box2.max.y, box2.min.y - other.max.y);
   356→    if (dy >= 0) return dy;
   357→    // تداخل رأسي: قس الفجوة الأفقية
   358→    const dz = Math.max(other.min.z - box2.max.z, box2.min.z - other.max.z);
   359→    const dx2 = Math.max(other.min.x - box2.max.x, box2.min.x - other.max.x);
   360→    return Math.max(dz, dx2);
   361→  };
   362→  gpuSeat.clearance = { toCooler: gapTo(coolerBox), toRam: gapTo(ramBox) };
   363→  console.info(
   364→    `[v6] GPU seated at slot (${slot.x.toFixed(2)}, ${slot.y.toFixed(2)}, ${slot.z.toFixed(2)}), bracketZ=${bracketZ.toFixed(2)}, ` +
   365→    `clearance cooler=${(gpuSeat.clearance.toCooler * 10).toFixed(1)}cm ram=${(gpuSeat.clearance.toRam * 10).toFixed(1)}cm`,
   366→  );
   367→}

   371→function swapCase(visuals: PartVisuals): void {
   372→  const def = DEFS.case;
   373→  const model = cache.get(def.url);
   374→  if (!model) return;
   375→
   376→  // 1) الهيكل: كل العقد ما عدا الزجاج

   380→  const shellRoot = new THREE.Group();
   381→  shellRoot.add(shellClone);
   382→  const shellN = normalize(shellRoot, def);
   383→  clearGroup(visuals.caseShell);
   384→  setShadows(shellN.root, true, true);
   385→  visuals.caseShell.add(shellN.root);
   386→
   387→  // لمبة الطاقة الحقيقية — خامتها تُستنسخ لعزل التحكم بالإطفاء/التشغيل
   388→  const led = findNode(visuals.caseShell, 'case_power_led');
   389→  if (led instanceof THREE.Mesh) {
   390→    const mats = Array.isArray(led.material) ? [led.material[0]] : [led.material];
   391→    led.material = (mats[0] as THREE.MeshStandardMaterial).clone();
   392→    visuals.flags.powerLed = led;
   393→  }
   394→
   395→  // 2) الغطاء الزجاجي داخل المجموعة المفصلية (محور المفصلة عند (0,2.25,0.06))
   396→  // الهدف: عند زاوية الدوران 0 يكون الزجاج في مكانه الحقيقي في الإحداثيات العالمية:
   397→  //   world = hinge · translate(offset − hinge) · R·S
   398→  const glassSource = findNode(model, 'case_glass');
   399→  if (glassSource) {
   400→    const hinge = new THREE.Vector3(0, 2.25, 0.06);
   401→    const wrap = new THREE.Group();
   402→    wrap.quaternion.setFromRotationMatrix(new THREE.Matrix4().makeBasis(vec3(def.basis[0]), vec3(def.basis[1]), vec3(def.basis[2])));
   403→    wrap.scale.set(def.scale[0], def.scale[1], def.scale[2]);
   404→    wrap.position.copy(shellN.offset).sub(hinge);
   405→    wrap.add(glassSource.clone(true));
   406→    clearGroup(visuals.sidePanel);
   407→    visuals.sidePanel.add(wrap);
   408→    setShadows(wrap, false);
   409→  }
   410→}

   672→  mon.root.traverse((o) => {
   673→    if (!(o instanceof THREE.Mesh)) return;
   674→    const mats = Array.isArray(o.material) ? o.material : [o.material];

   676→      if (targets.includes(m.name || '')) {
   677→        const std = m as THREE.MeshStandardMaterial;
   678→        std.color.set(0x1c222a);
   679→        std.metalness = 0.55;
   680→        std.roughness = 0.45;
   681→      }
   682→    }
   683→  });
   684→

   686→  const screenMesh = findScreenMesh(mon.root);

   699→      plane.position.set(center.x, center.y, center.z + size.z * 0.75);
   700→    } else if (thinAxis === 1) {
   701→      plane.rotation.x = Math.PI / 2;
   702→      plane.position.set(center.x, center.y + size.y * 0.75, center.z);
   703→    } else {
   704→      plane.rotation.y = Math.PI / 2;
   705→      plane.position.set(center.x + size.x * 0.75, center.y, center.z);
   706→    }
   707→    plane.name = 'monitor-screen';
   708→    group.add(plane);
