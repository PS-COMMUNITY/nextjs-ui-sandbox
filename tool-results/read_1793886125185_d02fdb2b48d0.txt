    50→interface MatOpts {
    51→  map?: THREE.Texture | null;
    52→  emissive?: number;
    53→  emissiveIntensity?: number;
    54→  transparent?: boolean;
    55→  opacity?: number;
    56→  side?: THREE.Side;

    59→/** cache للمواد القياسية — مفاتيح بنطاق القطعة */
    60→const matCache = new Map<string, THREE.MeshStandardMaterial>();
    61→
    62→function mat(color: number, metalness: number, roughness: number, opts?: MatOpts): THREE.MeshStandardMaterial {
    63→  const mapId = opts?.map ? opts.map.uuid : 'n';
    64→  const key = `${partScope}|${color}|${metalness}|${roughness}|${mapId}|${opts?.emissive ?? ''}|${opts?.emissiveIntensity ?? ''}|${opts?.transparent ?? ''}|${opts?.opacity ?? ''}|${opts?.side ?? ''}`;
    65→  const cached = matCache.get(key);
    66→  if (cached) return cached;
    67→  const m = new THREE.MeshStandardMaterial({
    68→    color,
    69→    metalness,
    70→    roughness,
    71→    map: opts?.map ?? null,
    72→    emissive: opts?.emissive ?? 0x000000,
    73→    emissiveIntensity: opts?.emissiveIntensity ?? 1,
    74→    transparent: opts?.transparent ?? false,
    75→    opacity: opts?.opacity ?? 1,
    76→    side: opts?.side ?? THREE.FrontSide,
    77→    flatShading: false,
    78→  });
    79→  matCache.set(key, m);
    80→  return m;
    81→}

   104→/** صندوق سريع مع ضبط الظلال حسب الجودة */
   105→function box(
   106→  w: number, h: number, d: number, material: THREE.Material,
   107→  x = 0, y = 0, z = 0, cast = true, receive = false,
   108→): THREE.Mesh {
   109→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), material);
   110→  m.position.set(x, y, z);
   111→  if (castEnabled && cast) m.castShadow = true;
   112→  if (receive) m.receiveShadow = true;
   113→  return m;
   114→}

   116→type Axis = 'x' | 'y' | 'z';
   117→
   118→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
   119→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
   120→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
   121→  if (axis === 'x') m.rotation.z = Math.PI / 2;
   122→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
   123→  m.position.set(x, y, z);
   124→  if (castEnabled && cast) m.castShadow = true;
   125→  return m;
   126→}

   130→  return cyl(r, length, mat(COLORS.aluminum, 0.85, 0.35), axis);

   133→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
   134→function makeCable(points: THREE.Vector3[], radius: number, color = 0x181d24): THREE.Mesh {
   135→  const curve = new THREE.CatmullRomCurve3(points);
   136→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(color, 0.3, 0.6));
   137→  if (castEnabled) m.castShadow = true;
   138→  return m;
   139→}

   141→/** تحويل نقاط كابل عالمية إلى محلية بالنسبة لموضع تثبيت القطعة */
   142→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
   143→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));
   144→}

   146→/** مستطيل مستوٍ مواجه لمحور محدد (للملصقات والشبكات) */
   147→function plane(w: number, h: number, material: THREE.Material, axis: 'x' | 'y' | 'z', x: number, y: number, z: number): THREE.Mesh {
   148→  const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), material);
   149→  if (axis === 'x') m.rotation.y = -Math.PI / 2; // مواجه -X
   150→  else if (axis === 'y') m.rotation.x = -Math.PI / 2;
   151→  m.position.set(x, y, z);
   152→  return m;
   153→}

   155→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   156→
   157→/**
   158→ * مروحة واقعية: إطار مربع كامل بحواف وأركان دائرية + فتحات براغي + حلقة داخلية
   159→ * + 9 شفرات مائلة + محور بغطاء. محور الدوران الافتراضي = Y المحلي.
   160→ */
   161→function makeFan(radius: number): FanParts {
   162→  const depth = Math.max(0.14, radius * 0.5);
   163→  const half = radius * 1.32;
   164→
   165→  // الإطار: 4 حواف كاملة + 4 أركان أسطوانية
   166→  const frame = new THREE.Group();
   167→  const frameMat = mat(COLORS.plastic, 0.3, 0.62);
   168→  const barGeoX = new THREE.BoxGeometry(half * 2, depth, 0.1);
   169→  const barGeoZ = new THREE.BoxGeometry(0.1, depth, half * 2);
   170→  for (const s of [-1, 1]) {
   171→    const barX = new THREE.Mesh(barGeoX, frameMat);
   172→    barX.position.set(0, 0, s * half);
   173→    const barZ = new THREE.Mesh(barGeoZ, frameMat);
   174→    barZ.position.set(s * half, 0, 0);
   175→    if (castEnabled) { barX.castShadow = true; barZ.castShadow = true; }
   176→    frame.add(barX, barZ);
   177→  }
   178→  const cornerGeo = new THREE.CylinderGeometry(0.055, 0.055, depth, 10);
   179→  const holeGeo = new THREE.CylinderGeometry(0.026, 0.026, 0.02, 8);
   180→  const holeMat = mat(COLORS.black, 0.2, 0.6);
   181→  for (const sx of [-1, 1]) {
   182→    for (const sz of [-1, 1]) {
   183→      const corner = new THREE.Mesh(cornerGeo, frameMat);
   184→      corner.position.set(sx * (half - 0.05), 0, sz * (half - 0.05));
   185→      if (castEnabled) corner.castShadow = true;
   186→      frame.add(corner);
   187→      // فتحة برغي مظلمة على الوجه العلوي
   188→      const hole = new THREE.Mesh(holeGeo, holeMat);
   189→      hole.position.set(sx * half, depth / 2, sz * half);
   190→      frame.add(hole);
   191→    }
   192→  }
   193→  // حلقة داخلية محيطية
   194→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.99, 0.016, 6, 28), mat(COLORS.metalDark, 0.5, 0.5));
   195→  ring.rotation.x = Math.PI / 2;
   196→  if (castEnabled) ring.castShadow = true;
   197→  frame.add(ring);
   198→  // دعامات خلفية (ستراهنة بين الحواف خلف الشفرات)
   199→  const strutGeo = new THREE.BoxGeometry(0.05, 0.02, radius * 1.7);
   200→  for (const sz of [-1, 1]) {
   201→    const strut = new THREE.Mesh(strutGeo, frameMat);
   202→    strut.position.set(sz * radius * 0.72, -depth * 0.28, 0);
   203→    strut.rotation.y = Math.PI / 2;
   204→    frame.add(strut);
   205→  }
   206→
   207→  // الشفرات: 9 شفرات مائلة 35° داخل مجموعة blades (تُدفع إلى spinners)
   208→  const blades = new THREE.Group();
   209→  const bladeGeo = new THREE.BoxGeometry(0.016, radius * 0.8, radius * 0.32);
   210→  const bladeMat = mat(0x242c36, 0.25, 0.45);
   211→  for (let i = 0; i < 9; i++) {
   212→    const arm = new THREE.Object3D();
   213→    arm.rotation.y = (i / 9) * Math.PI * 2;
   214→    const pitch = new THREE.Object3D();
   215→    pitch.position.set(radius * 0.52, 0, 0);
   216→    pitch.rotation.x = 0.62; // ميلان الشفرة ≈ 35°
   217→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   218→    blade.rotation.z = Math.PI / 2;
   219→    if (castEnabled) blade.castShadow = true;
   220→    pitch.add(blade);
   221→    arm.add(pitch);
   222→    blades.add(arm);
   223→  }
   224→  const hub = new THREE.Mesh(
   225→    new THREE.CylinderGeometry(radius * 0.17, radius * 0.17, depth * 0.72, 14),
   226→    mat(COLORS.metalDark, 0.5, 0.4),
   227→  );
   228→  if (castEnabled) hub.castShadow = true;
   229→  blades.add(hub);
   230→  // غطاء محور بحلقة داكنة
   231→  const cap = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.1, radius * 0.1, depth * 0.76, 12), mat(0x11161c, 0.3, 0.5));
   232→  blades.add(cap);
   233→  const capRing = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.11, 0.008, 6, 16), mat(COLORS.metalMid, 0.6, 0.4));
   234→  capRing.rotation.x = Math.PI / 2;
   235→  capRing.position.y = depth * 0.38;
   236→  blades.add(capRing);
   237→
   238→  return { frame, blades };
   239→}

   259→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   260→  scene.background = new THREE.Color(COLORS.bg);
   261→  scene.fog = new THREE.Fog(COLORS.bg, 20, 52);
   262→
   263→  // أرضية المختبر — 75سم أسفل سطح مكتب الورشة (توجيه v4 أ.1)
   264→  const ground = new THREE.Mesh(
   265→    new THREE.CircleGeometry(30, 28),
   266→    new THREE.MeshStandardMaterial({ color: 0x131820, metalness: 0, roughness: 0.94, flatShading: false }),
   267→  );
   268→  ground.rotation.x = -Math.PI / 2;
   269→  ground.position.y = DESK.floorY;
   270→  ground.receiveShadow = true;
   271→  scene.add(ground);
   272→
   273→  // شبكة أرضية خفيفة
   274→  const grid = new THREE.GridHelper(40, 40, 0x232c38, 0x171d26);
   275→  grid.position.y = DESK.floorY + 0.01;
   276→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   277→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   278→  for (const gm of gridMats) {
   279→    gm.transparent = true;
   280→    gm.opacity = 0.35;
   281→  }
   282→  scene.add(grid);
   283→
   284→  scene.add(new THREE.AmbientLight(0xffffff, 0.6));
   285→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x1b2028, 0.6));
   286→
   287→  // شمس دافئة مع ظلال (حسب الجودة) — تغطي المكتب كاملًا
   288→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.6);
   289→  dir.position.set(-7, 10, 8);
   290→  if (quality !== 'low') {
   291→    dir.castShadow = true;
   292→    const size = quality === 'high' ? 2048 : 1024;
   293→    dir.shadow.mapSize.set(size, size);
   294→    dir.shadow.camera.left = -12;
   295→    dir.shadow.camera.right = 12;
   296→    dir.shadow.camera.top = 12;
   297→    dir.shadow.camera.bottom = -12;
   298→    dir.shadow.camera.near = 2;
   299→    dir.shadow.camera.far = 45;
   300→    dir.shadow.camera.updateProjectionMatrix();
   301→    dir.shadow.bias = -0.0004;
   302→    dir.shadow.normalBias = 0.02;
   303→  }
   304→  scene.add(dir);
   305→
   306→  const fill = new THREE.PointLight(0xffffff, 0.6, 0, 0);
   307→  fill.position.set(4, 7, -2);
   308→  scene.add(fill);
   309→  const front = new THREE.PointLight(0xfff4e0, 1.6, 0, 0);
   310→  front.position.set(-5, 4.5, 6);
   311→  scene.add(front);
   312→
   313→  // محطة العمل: مكتب + جدار ورشة + إضاءة سقفية + مشترك كهرباء
   314→  scene.add(buildWorkstation());
   315→}

   321→function buildWorkstation(): THREE.Group {
   322→  scope('desk');
   323→  const g = new THREE.Group();
   324→  g.name = 'workstation';
   325→
   326→  const w = DESK.maxX - DESK.minX; // 15
   327→  const d = DESK.maxZ - DESK.minZ; // 10.6
   328→  const cx = (DESK.minX + DESK.maxX) / 2;
   329→  const cz = (DESK.minZ + DESK.maxZ) / 2;
   330→
   331→  // --- سطح المكتب: لوح سميك بحافة ألمنيوم على الأطراف ---
   332→  const topMat = mat(0x4a525c, 0.25, 0.65, { map: brushedTexture('#525a64') });
   333→  const top = box(w, DESK.thickness, d, topMat, cx, DESK.topY - DESK.thickness / 2, cz, true, true);
   334→  top.name = 'desk-top';
   335→  g.add(top);
   336→  const trimMat = mat(0x565f6a, 0.7, 0.4);
   337→  // أربعة أشرطة حواف معدنية رفيعة (تفصل السطح عن الأطراف كطاولات الورش الحقيقية)
   338→  g.add(box(w, 0.06, 0.06, trimMat, cx, DESK.topY - 0.03, DESK.minZ + 0.03, false));
   339→  g.add(box(w, 0.06, 0.06, trimMat, cx, DESK.topY - 0.03, DESK.maxZ - 0.03, false));
   340→  g.add(box(0.06, 0.06, d, trimMat, DESK.minX + 0.03, DESK.topY - 0.03, cz, false));
   341→  g.add(box(0.06, 0.06, d, trimMat, DESK.maxX - 0.03, DESK.topY - 0.03, cz, false));
   342→
   343→  // --- أرجل فولاذية على شكل حرف L + عارضات عرضية + رف سفلي ---
   344→  const legMat = mat(COLORS.deskSteel, 0.6, 0.45, { map: brushedTexture('#2f3740') });
   345→  const legH = DESK.topY - DESK.thickness - DESK.floorY; // 7
   346→  const legCy = DESK.floorY + legH / 2;
   347→  for (const lx of [DESK.minX + DESK.legInset, DESK.maxX - DESK.legInset]) {
   348→    for (const lz of [DESK.minZ + DESK.legInset, DESK.maxZ - DESK.legInset]) {
   349→      g.add(box(0.24, legH, 0.24, legMat, lx, legCy, lz, true));
   350→      // قدم مطاطية سوداء
   351→      g.add(box(0.3, 0.08, 0.3, mat(0x0c0f14, 0.2, 0.7), lx, DESK.floorY + 0.04, lz, false));
   352→    }
   353→  }
   354→  // عارضتان طوليتان تربطان الأرجل + رف سفلي كامل
   355→  for (const lz of [DESK.minZ + DESK.legInset, DESK.maxZ - DESK.legInset]) {
   356→    g.add(box(w - DESK.legInset * 2, 0.14, 0.1, legMat, cx, DESK.floorY + 1.1, lz, false));
   357→  }
   358→  g.add(box(w - 1.4, 0.1, d - 1.4, mat(0x232a32, 0.5, 0.55), cx, DESK.floorY + 3.6, cz, true, true));
   359→
   360→  // --- جدار الورشة خلف المكتب + لوح تعليق أدوات + رفان ---
   361→  const wallMat = mat(COLORS.wall, 0.15, 0.85);
   362→  const wall = box(34, 26, 0.4, wallMat, cx, DESK.floorY + 13, DESK.minZ - 0.8, false, true);
   363→  wall.name = 'workshop-wall';
   364→  g.add(wall);
   365→  // إيزود السقف/الجدار: شريط داكن أفقياً + لوح تعليق خلف الجهاز
   366→  g.add(box(34, 0.5, 0.06, mat(0x11161d, 0.2, 0.8), cx, DESK.floorY + 6.4, DESK.minZ - 0.58, false));
   367→  const pegboard = box(6.4, 5.2, 0.06, mat(0x171d26, 0.3, 0.75, { map: ventMeshTexture('#1c242e', '#10161d') }), 1.05, DESK.floorY + 9.4, DESK.minZ - 0.56, false);
   368→  pegboard.name = 'pegboard';
   369→  g.add(pegboard);
   370→  // رفان معدنيان على الجدار مع علب صغيرة
   371→  for (const sy of [DESK.floorY + 10.6, DESK.floorY + 12.2]) {
   372→    g.add(box(5.2, 0.1, 1.2, legMat, -3.6, sy, DESK.minZ - 0.2, false));
   373→    g.add(box(5.2, 0.1, 1.2, legMat, 6.8, sy, DESK.minZ - 0.2, false));
   374→    for (let i = 0; i < 3; i++) {
   375→      g.add(box(0.9, 0.8, 0.9, mat(i % 2 ? 0x2c3641 : 0x39434e, 0.2, 0.7), -4.8 + i * 1.2, sy + 0.45, DESK.minZ - 0.2, false));
   376→      g.add(box(0.9, 0.8, 0.9, mat(i % 2 ? 0x39434e : 0x2c3641, 0.2, 0.7), 5.9 + i * 1.1, sy + 0.45, DESK.minZ - 0.2, false));
   377→    }
   378→  }
   379→

   822→function buildCpu(): THREE.Group {
   823→  scope('cpu');
   824→  const g = new THREE.Group();
   825→  g.name = 'part-cpu';
   826→
   827→  // قاعدة خضراء (substrate)
   828→  g.add(box(0.05, 0.4, 0.4, mat(0x1e6b3a, 0.3, 0.55), 0, 0, 0, true, true));

   830→  const goldEdge = mat(0xc9a23f, 0.9, 0.3);

   832→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, 0.193, false));
   833→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, -0.193, 0, false));
   834→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, 0.193, 0, false));
   835→
   836→  // غطاء IHS بأركان مشطوفة (مستطيل مشطوف عبر Cylinder ثماني مسطح؟) — نستخدم Extrude
   837→  const ihsShape = new THREE.Shape();
   838→  const s = 0.17;
   839→  const r = 0.045;
   840→  ihsShape.moveTo(-s + r, -s);
   841→  ihsShape.lineTo(s - r, -s);
   842→  ihsShape.quadraticCurveTo(s, -s, s, -s + r);
   843→  ihsShape.lineTo(s, s - r);
   844→  ihsShape.quadraticCurveTo(s, s, s - r, s);
   845→  ihsShape.lineTo(-s + r, s);
   846→  ihsShape.quadraticCurveTo(-s, s, -s, s - r);
   847→  ihsShape.lineTo(-s, -s + r);
   848→  ihsShape.quadraticCurveTo(-s, -s, -s + r, -s);
   849→  const ihsGeo = new THREE.ExtrudeGeometry(ihsShape, { depth: 0.035, bevelEnabled: false });
   850→  const ihs = new THREE.Mesh(ihsGeo, mat(0xc2c9d1, 0.9, 0.3));
   851→  ihs.rotation.y = -Math.PI / 2; // البثق على المحور X نحو -X
   852→  ihs.position.set(-0.045, 0, 0);
   853→  if (castEnabled) ihs.castShadow = true;
   854→  g.add(ihs);
   855→
   856→  // شقّا اتجاه على حافتين (توجيه المعالج) — على وجه IHS الأمامي
   857→  const notchMat = mat(0x6b7280, 0.6, 0.4);
   858→  g.add(box(0.02, 0.03, 0.07, notchMat, -0.082, 0.155, 0.05, false));
   859→  g.add(box(0.02, 0.03, 0.07, notchMat, -0.082, -0.155, -0.05, false));

   862→  const tri = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.05, 0.008, 3), mat(0xd4af37, 0.85, 0.3));
   863→  tri.rotation.z = Math.PI / 2;

   865→  g.add(tri);
   866→
   867→  setComponentId(g, 'cpu');

   902→    const pipe = new THREE.Mesh(new THREE.TubeGeometry(curve, 22, 0.028, 7, false), copper);

   907→    startCap.position.copy(pts[0]);
   908→    const endCap = new THREE.Mesh(new THREE.SphereGeometry(0.028, 8, 8), copper);

   912→
   913→  // كتلة زعانف: 14 لوحًا رفيعًا متقاربًا

   926→  wrapper.position.set(-0.38, 0, 0);

   928→  const fan = makeFan(0.25);

   931→  fan.blades.name = 'cooler_fan_rotor';
   932→  fan.blades.userData.spinAxis = 'y'; // الريش الإجرائي يدور حول Y المحلي للغلاف
   933→  g.add(wrapper);
   934→  g.add(box(0.015, 0.03, 0.6, mat(COLORS.metalMid, 0.6, 0.4), -0.38, 0.385, 0, false));
   935→  g.add(box(0.015, 0.03, 0.6, mat(COLORS.metalMid, 0.6, 0.4), -0.38, -0.385, 0, false));
   936→
   937→  setComponentId(g, 'cooler');
   938→  return g;
   939→}
