   139→export function isWebGLAvailable(): boolean {
   140→  try {
   141→    const canvas = document.createElement('canvas');
   142→    return Boolean(canvas.getContext('webgl2') || canvas.getContext('webgl'));
   143→  } catch {
   144→    return false;
   145→  }
   146→}

   148→/** تخلّص من مادة وكل خرائطها النسيجية (يُستخدم في dispose) */
   149→function disposeMaterial(mat: THREE.Material): void {
   150→  for (const value of Object.values(mat) as unknown[]) {
   151→    if (value instanceof THREE.Texture) value.dispose();
   152→  }
   153→  mat.dispose();
   154→}

   227→  constructor(init: { container: HTMLDivElement; quality: SceneQuality; callbacks: PCSceneCallbacks }) {
   228→    this.container = init.container;
   229→    this.quality = init.quality;
   230→    this.callbacks = init.callbacks;
   231→
   232→    // (1) فحص WebGL قبل أي شيء — عند الفشل نعلّم failed ونبلّغ دون بناء
   233→    if (!isWebGLAvailable()) {
   234→      this.fail(WEBGL_ERROR_AR);
   235→      return;
   236→    }
   237→
   238→    try {
   239→      const w = Math.max(1, this.container.clientWidth);
   240→      const h = Math.max(1, this.container.clientHeight);
   241→
   242→      // (2) المُصيِّر — نغمة سينمائية ACES لدرجات لون واقعية
   243→      const renderer = new THREE.WebGLRenderer({ antialias: true });
   244→      renderer.setClearColor(COLORS.bg);
   245→      renderer.shadowMap.enabled = this.quality !== 'low';
   246→      renderer.shadowMap.type = THREE.PCFSoftShadowMap;
   247→      renderer.toneMapping = THREE.ACESFilmicToneMapping;
   248→      renderer.toneMappingExposure = TONE_EXPOSURE;
   249→      renderer.setSize(w, h, false);
   250→      renderer.domElement.style.width = '100%';
   251→      renderer.domElement.style.height = '100%';
   252→      renderer.domElement.style.display = 'block';
   253→      this.renderer = renderer;
   254→      // الحاوية position:relative يديرها الاستدعاء — اللوحة تملأها بالكامل
   255→      this.container.appendChild(renderer.domElement);
   256→
   257→      // (3) المشهد والكاميرا والتحكم المداري
   258→      this.scene = new THREE.Scene();
   259→      // خريطة بيئية (IBL) عبر PMREM — تجعل المعادن تعكس مثل الواقع بدل أن تبدو سوداء
   260→      const pmrem = new THREE.PMREMGenerator(renderer);
   261→      this.scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
   262→      this.scene.environmentIntensity = ENV_INTENSITY;
   263→      pmrem.dispose();
   264→      this.camera = new THREE.PerspectiveCamera(40, w / h, 0.1, 100);
   265→      this.camera.position.set(...DEFAULT_CAMERA.pos);
   266→      this.controls = new OrbitControls(this.camera, renderer.domElement);
   267→      this.controls.target.set(...DEFAULT_CAMERA.target);
   268→      this.controls.enableDamping = true;
   269→      this.controls.dampingFactor = 0.08;
   270→      this.controls.minDistance = 2.2;
   271→      this.controls.maxDistance = 14;
   272→      this.controls.maxPolarAngle = 1.45;
   273→      this.controls.update();
   274→      // تفاعل المستخدم يلغي أي تنعيم كاميرا جارٍ
   275→      this.controls.addEventListener('start', () => {
   276→        this.camTween = null;
   277→      });
   278→
   279→      // (4) البيئة ثم القطع — بنداءات المصنع المتفق عليها
   280→      buildEnvironment(this.scene, this.quality);
   281→      const visuals = buildAllParts(this.quality);
   282→      this.scene.add(visuals.root);
   283→      this.visuals = visuals;
   284→
   285→      // ضوء اتجاهي (أضافه buildEnvironment) للتحكم به عند تبديل الظلال
   286→      this.scene.traverse((obj) => {
   287→        if (!this.dirLight && obj instanceof THREE.DirectionalLight) this.dirLight = obj;
   288→      });
   289→
   290→      // تسجيل المعرفات دفاعيًا (المصنع يفعل ذلك عادةً — إعادة النداء غير ضارة)
   291→      for (const id of COMPONENT_IDS) setComponentId(visuals.dynamic[id], id);
   292→      setComponentId(visuals.caseShell, 'case');
   293→      setComponentId(visuals.sidePanel, 'case');
   294→
   295→      this.initDynamicState(visuals);
   296→      this.initFlagRecords(visuals);
   297→
   298→      // (الجلسة 4) الشاشة الافتراضية: نسيج ديناميكي + مرجع كابل الإشارة
   299→      this.painter = createScreenPainter();
   300→      if (this.painter) {
   301→        const screenMesh = visuals.monitor.getObjectByName('monitor-screen');
   302→        if (screenMesh instanceof THREE.Mesh) {
   303→          const smat = screenMesh.material;
   304→          if (!Array.isArray(smat) && 'map' in smat) {
   305→            (smat as THREE.MeshBasicMaterial).map = this.painter.texture;
   306→            (smat as THREE.MeshBasicMaterial).color.set(0xffffff);
   307→            (smat as THREE.MeshBasicMaterial).needsUpdate = true;
   308→          }
   309→        }
   310→        this.drawMonitor(true);
   311→      }
   312→      this.monitorCableGroup = visuals.monitor.getObjectByName('monitor-cable') ?? null;
   313→
   314→      // (7)(15) مستمعو المؤشر + مراقب الحجم
   315→      renderer.domElement.addEventListener('pointerdown', this.onPointerDown);
   316→      renderer.domElement.addEventListener('pointerup', this.onPointerUp);
   317→      renderer.domElement.addEventListener('pointermove', this.onPointerMove);
   318→      this.resizeObserver = new ResizeObserver(() => this.onResize());
   319→      this.resizeObserver.observe(this.container);
   320→
   321→      // (13) ضبط كثافة البكسل حسب الجودة (الظلال مضبوطة أعلاه بنفس القيمة)
   322→      this.setQuality(this.quality);
   323→
   324→      // (6) بدء حلقة الرسم
   325→      this.rafId = requestAnimationFrame(this.tick);
   326→      this.callbacks.onReady?.();
   327→    } catch {
   328→      // المعالجة نفسها: رسالة + failed=true
   329→      this.fail(WEBGL_ERROR_AR);
   330→    }
   331→  }

   337→  /** (13) تبديل الجودة: كثافة البكسل + تشغيل/إيقاف الظلال */
   338→  setQuality(q: SceneQuality): void {
   339→    if (this.blocked) return;
   340→    this.quality = q;
   341→    this.renderer.setPixelRatio(this.pixelRatioFor(q));
   342→    const wantShadows = q !== 'low';
   343→    if (this.renderer.shadowMap.enabled !== wantShadows) {
   344→      this.renderer.shadowMap.enabled = wantShadows;
   345→      this.renderer.shadowMap.needsUpdate = true;
   346→      if (this.dirLight) this.dirLight.castShadow = wantShadows;
   347→    }
   348→  }

   374→  /** (5) مزامنة حالة المحاكاة: أهداف الحركة (تنعيم) + تطبيق الأعلام فورًا */
   375→  syncState(s: SimulationState): void {
   376→    if (this.blocked) return;
   377→    // نسخة دفاعية حتى لا تؤثر طفرات المتصل على أهدافنا
   378→    this.lastState = {
   379→      flags: { ...s.flags },
   380→      parts: { ...s.parts },
   381→      peripherals: { ...s.peripherals },
   382→    };
   383→    this.recomputeTargets();
   384→    this.applyFlags(s.flags, true);
   385→    this.updateRuntimeVisuals(s);
   386→    this.drawMonitor(true);
   387→  }

   415→  /** (8) إبراز مكوّن بتعديل emissive — إبراز واحد نشط في كل مرة */
   416→  highlight(id: ComponentId | null): void {
   417→    if (this.blocked) return;
   418→    if (this.highlightedId === id) return;
   419→    const previous = this.highlightedId;
   420→    this.highlightedId = id;
   421→    if (previous !== null) this.restoreHighlight();
   422→    if (id === null) return;
   423→    this.parts[id].group.traverse((obj) => {
   424→      if (!(obj instanceof THREE.Mesh)) return;
   425→      const mat = obj.material;
   426→      if (Array.isArray(mat) || !('emissive' in mat)) return;
   427→      const std = mat as THREE.MeshStandardMaterial;
   428→      // نلتقط الأصل مرة واحدة لكل جلسة إبراز، ونستعيده عند الإلغاء
   429→      if (!this.emissiveOriginals.has(obj)) {
   430→        this.emissiveOriginals.set(obj, { hex: std.emissive.getHex(), intensity: std.emissiveIntensity });
   431→      }
   432→      std.emissive.setHex(HIGHLIGHT_HEX);
   433→      std.emissiveIntensity = HIGHLIGHT_INTENSITY;
   434→    });
   435→  }

   437→  /** (9) عزل مكوّن: إظهاره وحده مع هيكل شبه شفاف — null يعيد الجميع */
   438→  isolate(id: ComponentId | null): void {
   439→    if (this.blocked) return;
   440→    if (this.isolatedId === id) return;
   441→    this.isolatedId = id;
   442→    for (const cid of COMPONENT_IDS) this.parts[cid].group.visible = cid === id;
   443→    // الهيكل يبقى ظاهرًا (شبه شفاف عند عزل مكوّن داخلي، معتمًا عند عزل الهيكل نفسه)
   444→    this.visuals.caseShell.visible = true;
   445→    this.visuals.sidePanel.visible = id === 'case' || id === null;
   446→    this.refreshShellMaterials();
   447→  }

   449→  /** (11) إعادة الكاميرا إلى الوضع الافتراضي خلال 0.6 ثانية */
   450→  resetCamera(): void {
   451→    if (this.blocked) return;
   452→    const toPos = new THREE.Vector3(...DEFAULT_CAMERA.pos);
   453→    const toTarget = new THREE.Vector3(...DEFAULT_CAMERA.target);
   454→    this.startCamTween(toPos, toTarget, 0.6);
   455→  }

   457→  /** (12) تقرّب الكاميرا نحو مكوّن: الهدف مركز المجموعة والمسافة 3.2 خلال 0.5 ثانية */
   458→  focusComponent(id: ComponentId): void {
   459→    if (this.blocked) return;
   460→    const center = this.parts[id].group.getWorldPosition(this._v2);
   461→    const dir = this._v1.copy(this.camera.position).sub(center);
   462→    if (dir.lengthSq() < 1e-8) dir.set(-1, 0.35, 1); // اتجاه احتياطي إذا تطابقت النقطة
   463→    dir.normalize().multiplyScalar(3.2).add(center);
   464→    this.startCamTween(this._v1, this._v2, 0.5);
   465→  }

   467→  /** (14) إتلاف كامل: إيقاف الحلقة، فصل المستمعين، تحرير الهندسات والمواد، إزالة اللوحة */
   468→  dispose(): void {
   469→    if (this.disposed) return;
   470→    this.disposed = true;
   471→    if (this.rafId !== 0) cancelAnimationFrame(this.rafId);
   472→    this.rafId = 0;
   473→    this.controls?.dispose();
   474→    const dom = this.renderer?.domElement;
   475→    if (dom) {
   476→      dom.removeEventListener('pointerdown', this.onPointerDown);
   477→      dom.removeEventListener('pointerup', this.onPointerUp);
   478→      dom.removeEventListener('pointermove', this.onPointerMove);
   479→    }
   480→    this.resizeObserver?.disconnect();
   481→    this.resizeObserver = null;
   482→    if (this.scene) {
   483→      this.scene.traverse((obj) => {
   484→        const maybeMesh = obj as THREE.Mesh;
   485→        if (maybeMesh.geometry) maybeMesh.geometry.dispose();
   486→        const mat: THREE.Material | THREE.Material[] | undefined = maybeMesh.material;
   487→        if (Array.isArray(mat)) {
   488→          for (const m of mat) disposeMaterial(m);
   489→        } else if (mat) {
   490→          disposeMaterial(mat);
   491→        }
   492→      });
   493→      if (this.scene.environment instanceof THREE.Texture) this.scene.environment.dispose();
   494→      if (this.scene.background instanceof THREE.Texture) this.scene.background.dispose();
   495→    }
   496→    this.renderer?.dispose();
   497→    dom?.remove();
   498→    this.emissiveOriginals.clear();
   499→    this.shellOriginals.clear();
   500→    this.damagedOriginals.clear();
   501→    this.painter?.dispose();
   502→    this.painter = null;
   503→  }

   518→  /** (6) حلقة الرسم: دوّارات، اقتراب القطع، تنعيم الغطاء، تنعيم الكاميرا، ثم الرسم */
   519→  private readonly tick = (): void => {
   520→    if (this.disposed || this.failed) return;
   521→    this.rafId = requestAnimationFrame(this.tick);
   522→    const dt = Math.min(this.clock.getDelta(), 0.05); // حماية من قفزات التبويبات الخاملة
   523→    const k = 1 - Math.exp(-LERP_RATE * dt);
   524→
   525→    // دوران شفرات المراوح حول محورها المحفوظ — فقط والجهاز يعمل فعلاً
   526→    if (this.spinning) {
   527→      for (const spinner of this.spinners) {
   528→        spinner.obj.rotateOnAxis(spinner.axis, SPIN_SPEED * dt);
   529→      }
   530→    }
   531→
   532→    // اقتراب القطع نحو أهدافها (الموضع lerp والدوران slerp) — عدا الهيكل الثابت
   533→    for (const id of COMPONENT_IDS) {
   534→      if (id === 'case') continue;
   535→      const part = this.parts[id];
   536→      part.group.position.lerp(part.targetPos, k);
   537→      part.group.quaternion.slerp(part.targetQuat, k);
   538→    }
   539→
   540→    // تطفو القطعة الملتقطة (awaiting) فوق الصينية بلطف — إشارة بصرية لجاهزية التركيب
   541→    const elapsed = this.clock.getElapsedTime();
   542→    for (const id of COMPONENT_IDS) {
   543→      if (id === 'case') continue;
   544→      if (this.lastState && this.lastState.parts[id] === 'awaiting') {
   545→        const part = this.parts[id];
   546→        part.group.position.y = part.trayPos.y + 0.16 + Math.sin(elapsed * 3) * 0.04;
   547→      }
   548→    }
   549→
   550→    // تنعيم دوران الغطاء الجانبي نحو الهدف
   551→    const panel = this.visuals.sidePanel;
   552→    panel.rotation.y += (this.panelTarget - panel.rotation.y) * k;
   553→
   554→    // استكمال تنعيم الكاميرا (resetCamera / focusComponent)
   555→    const tween = this.camTween;
   556→    if (tween) {
   557→      tween.t = Math.min(tween.t + dt / tween.dur, 1);
   558→      const eased = PCScene.easeInOut(tween.t);
   559→      this.camera.position.lerpVectors(tween.fromPos, tween.toPos, eased);
   560→      this.controls.target.lerpVectors(tween.fromTarget, tween.toTarget, eased);
   561→      if (tween.t >= 1) this.camTween = null;
   562→    }
   563→
   564→    this.controls.update();
   565→
   566→    // إعادة رسم نسيج الشاشة الافتراضية بمعدل مقيد
   567→    if (this.painter && nowMs() - this.lastMonitorDrawAt >= MONITOR_REDRAW_MS) {
   568→      this.drawMonitor(false);
   569→    }
   570→
   571→    this.renderer.render(this.scene, this.camera);
   572→  };
   573→
   574→  /** تنعيم easeInOut: t<0.5 ? 2t² : 1 − (−2t+2)²/2 */
   575→  private static easeInOut(t: number): number {
   576→    return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
   577→  }

   579→  /** إعادة حساب أهداف القطع من آخر حالة متزامنة + علم التفجير الحالي */
   580→  private recomputeTargets(): void {
   581→    const parts = this.lastState?.parts;
   582→    for (const id of COMPONENT_IDS) {
   583→      if (id === 'case') continue;
   584→      const part = this.parts[id];
   585→      const status: string | undefined = parts ? parts[id] : undefined;
   586→      if (status === 'removed' || status === 'damaged') {
   587→        part.targetPos.copy(part.trayPos);
   588→        part.targetRot.copy(part.trayRot);
   589→        if (status === 'damaged') {
   590→          // القطعة التالفة تستلقي مائلة على الصينية
   591→          part.targetRot.z += 0.35;
   592→        }
   593→      } else if (status === 'awaiting') {
   594→        // ملتقطة من الصينية — ترتفع قليلاً (وتطفو في حلقة الرسم)
   595→        part.targetPos.copy(part.trayPos);
   596→        part.targetPos.y += 0.16;
   597→        part.targetRot.copy(part.trayRot);
   598→      } else if (status === 'separated') {
   599→        // فصل جزئي: ما زالت في موضعها مع إزاحة صغيرة نحو اتجاه الفك
   600→        part.targetPos.copy(part.basePos);
   601→        part.targetPos.addScaledVector(part.explodeVec, SEPARATE_FACTOR);
   602→        part.targetRot.set(0, 0, 0);
   603→      } else {
   604→        part.targetPos.copy(part.basePos);
   605→        if (this.exploded) part.targetPos.add(part.explodeVec);
   606→        part.targetRot.set(0, 0, 0);
   607→      }
   608→      part.targetQuat.setFromEuler(part.targetRot);
   609→    }
   610→  }

   612→  /** نسخ الأهداف فورًا إلى المجسمات (للتنعيم المعطَّل animate=false) */
   613→  private snapToTargets(): void {
   614→    for (const id of COMPONENT_IDS) {
   615→      if (id === 'case') continue;
   616→      const part = this.parts[id];
   617→      part.group.position.copy(part.targetPos);
   618→      part.group.quaternion.copy(part.targetQuat);
   619→    }

   621→  }

   623→  /**
   624→   * (5) تطبيق الأعلام فورًا على المجسمات الفرعية.
   625→   * افتراضات المحاور (لأن بناء المصنع موازٍ ولم يُثبَّت بعد):
   626→   *  - مزلاج PCIe ورافعة CPU: دوران حول Y المحلي (اللوح عمودي يواجه -X).
   627→   *  - مزالج RAM: دوران حول Z بإشارة متناوبة بين الطرفين.
   628→   *  - البراغي: ترتفع على +Y (مسامير اللوحة تُسحب على -X) مع دوران طفيف حول Z.
   629→   *  - القوابس: إزاحة -X 0.12 عند الفصل.
   630→   * كل الإزاحات نسبية إلى القيم الأصلية المحفوظة عند البناء.
   631→   */
   632→  private applyFlags(flags: Record<MachineFlag, boolean>, animate: boolean): void {
   633→    // الغطاء الجانبي — عبر هدف تنعيمي تتبعه الحلقة
   634→    this.panelTarget = flags.panelOpen ? PANEL_OPEN_ANGLE : 0;
   635→    if (!animate) this.visuals.sidePanel.rotation.y = this.panelTarget;
   636→
   637→    // مزلاج PCIe
   638→    if (this.latchRec) {
   639→      this.latchRec.obj.rotation.y = this.latchRec.baseRot + (flags.pcieLatchOpen ? LATCH_OPEN_ANGLE : 0);
   640→    }
   641→
   642→    // مزالج RAM
   643→    for (const clip of this.clipRecs) {
   644→      clip.obj.rotation.z = clip.baseRotZ + (flags.ramClipsOpen ? clip.sign * CLIP_OPEN_ANGLE : 0);
   645→    }
   646→
   647→    // رافعة مقبس CPU
   648→    if (this.leverRec) {
   649→      this.leverRec.obj.rotation.y = this.leverRec.baseRot + (flags.cpuLeverOpen ? LEVER_OPEN_ANGLE : 0);
   650→    }
   651→
   652→    // البراغي (gpu/psu/fan/m2/mobo)
   653→    for (const [flag, recs] of this.screwMap) {
   654→      const out = flags[flag];
   655→      for (const rec of recs) {
   656→        rec.obj.position.copy(rec.basePos);
   657→        if (out) {
   658→          rec.obj.position[rec.axis] += rec.delta;
   659→          rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN;
   660→        } else {
   661→          rec.obj.rotation.z = rec.baseRotZ;
   662→        }
   663→      }
   664→    }
   665→
   666→    // قوابس الموصلات (fan headers / front panel)
   667→    for (const plug of this.plugRecs) {
   668→      plug.obj.position.x = plug.baseX - (flags[plug.flag] ? PLUG_PULL : 0);
   669→    }
   670→
   671→    // لمبة الطاقة — مادة خاصة بها (cached) — تنطفئ مع مزود ميت
   672→    const led = this.visuals.flags.powerLed;
   673→    if (led) {
   674→      const mat = led.material;
   675→      if (!Array.isArray(mat) && 'emissive' in mat) {
   676→        const std = mat as THREE.MeshStandardMaterial;
   677→        const alive = flags.powerOn && !(this.lastState?.peripherals.psuDead ?? false);
   678→        std.emissive.set(alive ? COLORS.green : 0x000000);
   679→        std.emissiveIntensity = LED_INTENSITY;
   680→      }
   681→    }
   682→  }

   688→  /** إعادة حساب مواد الهيكل من الحالة المجمّعة (xray يتقدم على العزل) */
   689→  private refreshShellMaterials(): void {
   690→    for (const [mat, saved] of this.shellOriginals) {
   691→      mat.opacity = saved.opacity;
   692→      mat.transparent = saved.transparent;
   693→      mat.depthWrite = saved.depthWrite;
   694→    }
   695→    this.shellOriginals.clear();
   696→    if (this.xrayOn) {
   697→      this.applyShellOpacity(this.visuals.caseShell, 0.12);
   698→      this.applyShellOpacity(this.visuals.sidePanel, 0.06);
   699→    } else if (this.isolatedId !== null && this.isolatedId !== 'case') {
   700→      this.applyShellOpacity(this.visuals.caseShell, 0.08);
   701→    }
   702→  }

   704→  /** تطبيق شفافية على مواد مجموعة (الزجاج شفاف أصلًا فيبقى كما هو) */
   705→  private applyShellOpacity(root: THREE.Object3D, opacity: number): void {
   706→    root.traverse((obj) => {
   707→      if (!(obj instanceof THREE.Mesh)) return;
   708→      const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   709→      for (const mat of mats) {
   710→        if (mat.transparent) continue; // الزجاج/الشفاف الأصلي لا يُمسّ
   711→        if (!this.shellOriginals.has(mat)) {
   712→          this.shellOriginals.set(mat, {
   713→            opacity: mat.opacity,
   714→            transparent: mat.transparent,
   715→            depthWrite: mat.depthWrite,
   716→          });
   717→        }
   718→        mat.transparent = true;
   719→        mat.opacity = opacity;
   720→        mat.depthWrite = false;
   721→      }
   722→    });
   723→  }

   742→  private drawMonitor(force: boolean): void {
   743→    if (!this.painter || this.blocked) return;
   744→    if (!force && nowMs() - this.lastMonitorDrawAt < MONITOR_REDRAW_MS) return;
   745→    this.lastMonitorDrawAt = nowMs();
   746→    const sim = this.lastState;
   747→    const elapsed = this.bootStartedAt !== null ? Math.max(0, Date.now() - this.bootStartedAt) : 0;
   748→    if (!sim) {
   749→      this.painter.draw(
   750→        { screen: 'noSignal', bootProgress: 0, postLinesAr: [], bootDevicesAr: [], canType: false },
   751→        this.notepad,
   752→        Date.now(),
   753→      );
   754→      return;
   755→    }
   756→    const view = computeMonitorView(sim, elapsed);
   757→    this.painter.draw(view, this.notepad, Date.now());
   758→  }

   760→  /** تحديثات بصرية لحظية: دوران المراوح/كابل الشاشة/توهج القطع التالفة */
   761→  private updateRuntimeVisuals(s: SimulationState): void {
   762→    // المراوح تدور فقط مع كهرباء موصولة وجهاز يعمل ومزود حي
   763→    this.spinning = s.flags.powerOn && s.flags.acConnected && !s.peripherals.psuDead;
   764→
   765→    // كابل إشارة الشاشة يختفي عند فصله (المواصفة: الكابلات لا تطفو في الفراغ)
   766→    if (this.monitorCableGroup) {
   767→      this.monitorCableGroup.visible = s.peripherals.monitorCableConnected;
   768→    }
   769→
   770→    // توهج أحمر ثابت للقطع التالفة تدريبيًا
   771→    for (const id of COMPONENT_IDS) {
   772→      const isDamaged = s.parts[id] === 'damaged';
   773→      const wasDamaged = this.damagedIds.has(id);
   774→      if (isDamaged && !wasDamaged) {
   775→        this.parts[id].group.traverse((obj) => {
   776→          if (!(obj instanceof THREE.Mesh)) return;
   777→          const mat = obj.material;
   778→          if (Array.isArray(mat) || !('emissive' in mat)) return;
   779→          const std = mat as THREE.MeshStandardMaterial;
   780→          if (!this.damagedOriginals.has(obj)) {
   781→            this.damagedOriginals.set(obj, { hex: std.emissive.getHex(), intensity: std.emissiveIntensity });
   782→          }
   783→          std.emissive.setHex(DAMAGED_HEX);
   784→          std.emissiveIntensity = 0.45;
   785→        });
   786→        this.damagedIds.add(id);
   787→      } else if (!isDamaged && wasDamaged) {
   788→        this.parts[id].group.traverse((obj) => {
   789→          const saved = this.damagedOriginals.get(obj as THREE.Mesh);
   790→          if (!saved) return;
   791→          const mesh = obj as THREE.Mesh;
   792→          const mat = mesh.material;
   793→          if (Array.isArray(mat) || !('emissive' in mat)) return;
   794→          (mat as THREE.MeshStandardMaterial).emissive.setHex(saved.hex);
   795→          (mat as THREE.MeshStandardMaterial).emissiveIntensity = saved.intensity;
   796→          this.damagedOriginals.delete(mesh);
   797→        });
   798→        this.damagedIds.delete(id);
   799→      }
   800→    }
   801→  }

   804→    this.camTween = {

   806→      toPos,
   807→      fromTarget: this.controls.target.clone(),
   808→      toTarget,
   809→      t: 0,

   818→  /** إيجاد الهدف عند نقطة شاشة: أول إصابة مرئية غير شفافة — قطعة PC أو طرفية */
   819→  private pickAt(clientX: number, clientY: number): PickedInfo {
   820→    const rect = this.renderer.domElement.getBoundingClientRect();
   821→    const w = rect.width > 0 ? rect.width : 1;
   822→    const h = rect.height > 0 ? rect.height : 1;
   823→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
   824→    this.raycaster.setFromCamera(this._ndc, this.camera);
   825→    const hits = this.raycaster.intersectObjects(this.visuals.root.children, true);
   826→    for (const hit of hits) {
   827→      if (this.isHidden(hit.object)) continue;        // تجاهل ما خفيه العزل
   828→      if (this.isSeeThrough(hit.object)) continue;    // الزجاج/الشفاف لا يحجب الاختيار
   829→      let cur: THREE.Object3D | null = hit.object;
   830→      while (cur) {
   831→        // الطرفيات أولًا (مفاتيح لوحة المفاتيح تحمل تفاصيلها بنفسها)
   832→        const per: unknown = cur.userData['peripheral'];
   833→        if (typeof per === 'string') {
   834→          const kind = per as PeripheralKind;
   835→          const detail = typeof cur.userData['key'] === 'string'
   836→            ? (cur.userData['key'] as string)
   837→            : typeof cur.userData['btn'] === 'string'
   838→              ? (cur.userData['btn'] as string)
   839→              : undefined;
   840→          return { type: 'periph', kind, detail };
   841→        }
   842→        const cid: unknown = cur.userData['componentId'];
   843→        if (typeof cid === 'string' && (COMPONENT_IDS as readonly string[]).includes(cid)) {
   844→          return { type: 'part', id: cid as ComponentId };
   845→        }
   846→        cur = cur.parent;
   847→      }
   848→      return null; // أول إصابة مرئية بلا معرّف → تعامل كفراغ
   849→    }
   850→    return null;
   851→  }

   853→  /** هل الجسم (أو أحد أسلافه) مخفي؟ */
   854→  private isHidden(obj: THREE.Object3D): boolean {
   855→    let cur: THREE.Object3D | null = obj;
   856→    while (cur) {
   857→      if (!cur.visible) return true;
   858→      cur = cur.parent;
   859→    }
   860→    return false;
   861→  }

   863→  /** هل الشبك بمادة شفافة؟ (الزجاج والهيكل الشفاف يسمحان بمرور النقر) */
   864→  private isSeeThrough(obj: THREE.Object3D): boolean {
   865→    if (!(obj instanceof THREE.Mesh)) return false;
   866→    const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   867→    for (const mat of mats) {
   868→      if (mat.transparent) return true;
   869→    }
   870→    return false;
   871→  }

   898→  /** التقاط القيم الأصلية لمجسمات الأعلام مرة واحدة عند البناء */
   899→  private initFlagRecords(visuals: PartVisuals): void {
   900→    const flags: FlagVisuals = visuals.flags;
   901→
   902→    if (flags.pcieLatch) this.latchRec = { obj: flags.pcieLatch, baseRot: flags.pcieLatch.rotation.y };
   903→    if (flags.cpuLever) this.leverRec = { obj: flags.cpuLever, baseRot: flags.cpuLever.rotation.y };
   904→
   905→    this.clipRecs = flags.ramClips.map((obj, index): ClipRec => ({
   906→      obj,
   907→      baseRotZ: obj.rotation.z,
   908→      sign: index % 2 === 0 ? 1 : -1,
   909→    }));
   910→
   911→    const screwDefs: Array<{ flag: MachineFlag; objs: THREE.Object3D[]; axis: 'x' | 'y'; delta: number }> = [
   912→      { flag: 'gpuScrewsOut', objs: flags.gpuScrews, axis: 'y', delta: SCREW_LIFT },
   913→      { flag: 'psuScrewsOut', objs: flags.psuScrews, axis: 'y', delta: SCREW_LIFT },
   914→      { flag: 'fanScrewsOut', objs: flags.fanScrews, axis: 'y', delta: SCREW_LIFT },
   915→      { flag: 'm2ScrewOut', objs: flags.m2Screw ? [flags.m2Screw] : [], axis: 'y', delta: SCREW_LIFT },
   916→      // مسامير اللوحة تُسحب بعيدًا عن الجدار الأيمن (-X)
   917→      { flag: 'moboScrewsOut', objs: flags.moboScrews, axis: 'x', delta: -SCREW_LIFT },
   918→    ];
   919→    for (const def of screwDefs) {

   922→        def.objs.map((obj): ScrewRec => ({

   927→          delta: def.delta,

   931→

   937→    }
   938→
   939→    // محور دوران كل دوّار يُستنتج من هندسته (أصغر بُعد = عمود قرص المروحة)
   940→    this.spinners = visuals.spinners.map((obj): SpinnerRec => ({ obj, axis: this.computeSpinnerAxis(obj) }));

   943→  /** استنتاج محور الدوران: أصغر بُعد في صندوق الإحاطة للشبك الأول هو عمود القرص */
   944→  private computeSpinnerAxis(spinner: THREE.Object3D): THREE.Vector3 {
   945→    const fallback = new THREE.Vector3(0, 1, 0);
   946→    const mesh = this.firstMesh(spinner);
   947→    if (!mesh || !mesh.geometry) return fallback;
   948→    const geometry = mesh.geometry;
   949→    if (!geometry.boundingBox) geometry.computeBoundingBox();
   950→    const bb = geometry.boundingBox;
   951→    if (!bb) return fallback;
   952→    const ex = bb.max.x - bb.min.x;
   953→    const ey = bb.max.y - bb.min.y;
   954→    const ez = bb.max.z - bb.min.z;
   955→    const axis = new THREE.Vector3(
   956→      ex <= ey && ex <= ez ? 1 : 0,
   957→      ey <= ex && ey <= ez ? 1 : 0,
   958→      ez <= ex && ey <= ez ? 1 : 0,
   959→    );
   960→    if (axis.lengthSq() === 0) return fallback;
   961→    // تحويل المحور من فضاء الشبك إلى الفضاء المحلي للدوّار إن كان ابنًا مُدارًا
   962→    if (mesh !== spinner) {
   963→      spinner.updateWorldMatrix(true, false);
   964→      mesh.updateWorldMatrix(true, false);
   965→      const qSpinner = new THREE.Quaternion();
   966→      const qMesh = new THREE.Quaternion();
   967→      spinner.getWorldQuaternion(qSpinner);
   968→      mesh.getWorldQuaternion(qMesh);
   969→      axis.applyQuaternion(qSpinner.invert().multiply(qMesh));
   970→    }
   971→    return axis.normalize();
   972→  }

   974→  /** أول Mesh داخل شجرة كائن (بدون تخصيصات) */
   975→  private firstMesh(obj: THREE.Object3D): THREE.Mesh | null {
   976→    if (obj instanceof THREE.Mesh) return obj;
   977→    for (const child of obj.children) {
   978→      const found = this.firstMesh(child);
   979→      if (found) return found;
   980→    }
   981→    return null;
   982→  }

   988→  /** (13)(15) كثافة البكسل حسب الجودة */
   989→  private pixelRatioFor(q: SceneQuality): number {
   990→    const dpr = typeof window !== 'undefined' ? window.devicePixelRatio || 1 : 1;
   991→    if (q === 'low') return 1;
   992→    if (q === 'medium') return Math.min(dpr, 1.5);
   993→    return Math.min(dpr, 2);

   996→  /** (15) استجابة تغيّر حجم الحاوية */
   997→  private onResize(): void {
   998→    if (this.blocked) return;
   999→    const w = this.container.clientWidth;
  1000→    const h = this.container.clientHeight;
  1001→    if (w <= 0 || h <= 0) return;
  1002→    this.camera.aspect = w / h;
  1003→    this.camera.updateProjectionMatrix();
  1004→    this.renderer.setSize(w, h, false); // updateStyle=false — CSS 100% يتكفل بالعرض
  1005→  }

  1007→  private readonly onPointerDown = (event: PointerEvent): void => {
  1008→    if (this.blocked) return;
  1009→    this.pointerDownActive = true;
  1010→    this.downX = event.clientX;
  1011→    this.downY = event.clientY;

  1014→  /** (7) نقرة فقط إذا كانت المسافة بين down/up أقل من 6px */
  1015→  private readonly onPointerUp = (event: PointerEvent): void => {
  1016→    if (this.blocked || !this.pointerDownActive) return;
  1017→    this.pointerDownActive = false;
  1018→    const dx = event.clientX - this.downX;
  1019→    const dy = event.clientY - this.downY;
  1020→    if (dx * dx + dy * dy >= CLICK_MAX_DIST_SQ) return; // سحب لا نقرة
  1021→    const picked = this.pickAt(event.clientX, event.clientY);
  1022→    if (picked && picked.type === 'periph') {
  1023→      this.callbacks.onPeripheral?.(picked.kind, picked.detail);
  1024→      return;
  1025→    }
  1026→    this.callbacks.onSelect(picked ? picked.id : null);
  1027→  };

  1029→  /** (7) تغيير المؤشر فوق مكوّن — بفحص مُقيَّد زمنيًا (100ms) */
  1030→  private readonly onPointerMove = (event: PointerEvent): void => {
  1031→    if (this.blocked) return;
  1032→    const now = performance.now();
  1033→    if (now - this.lastHoverAt < HOVER_THROTTLE_MS) return;
  1034→    this.lastHoverAt = now;
  1035→    this.renderer.domElement.style.cursor = this.pickAt(event.clientX, event.clientY) ? 'pointer' : '';
  1036→  };
  1037→}
