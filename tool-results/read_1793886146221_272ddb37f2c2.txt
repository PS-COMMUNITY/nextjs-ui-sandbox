    76→    76→function chrome(os: VirtualOS, ctx: CanvasRenderingContext2D, family: WinFamily | null, titleAr: string): number {
    77→    77→  const themeBg: Record<string, [string, string]> = {

    82→    82→    modern11: ['#1a2c3a', '#0b141c'],
    83→    83→  };

   100→   100→function bigButton(os: VirtualOS, ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, label: string, onClick: () => void, primary = true): void {
   101→   101→  const hovered = os.hoverAt(x, y, w, h);

   104→   104→  ctx.fill();

   107→   107→  ctx.stroke();
   108→   108→  ctx.fillStyle = primary ? '#04231f' : '#e8eef5';
   109→   109→  ctx.font = 'bold 14px sans-serif';
   110→   110→  ctx.textAlign = 'center';
   111→   111→  ctx.fillText(label, x + w / 2, y + h / 2 + 5);
   112→   112→  ctx.textAlign = 'left';
   113→   113→  os.addRect('button', x, y, w, h, { onClick });
