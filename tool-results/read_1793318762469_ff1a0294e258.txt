   148→export function isWebGLAvailable(): boolean {
   149→  try {
   150→    const canvas = document.createElement('canvas');
   151→    return Boolean(canvas.getContext('webgl2') || canvas.getContext('webgl'));
   152→  } catch {
   153→    return false;
   154→  }
   155→}

   157→/** تخلّص من مادة وكل خرائطها النسيجية (يُستخدم في dispose) */
   158→function disposeMaterial(mat: THREE.Material): void {
   159→  for (const value of Object.values(mat) as unknown[]) {
   160→    if (value instanceof THREE.Texture) value.dispose();
   161→  }
   162→  mat.dispose();
   163→}

   244→  constructor(init: { container: HTMLDivElement; quality: SceneQuality; callbacks: PCSceneCallbacks }) {
   245→    this.container = init.container;
   246→    this.quality = init.quality;
   247→    this.callbacks = init.callbacks;
   248→
   249→    // (1) فحص WebGL قبل أي شيء — عند الفشل نعلّم failed ونبلّغ دون بناء
   250→    if (!isWebGLAvailable()) {
   251→      this.fail(WEBGL_ERROR_AR);
   252→      return;
   253→    }
   254→
   255→    try {
   256→      const w = Math.max(1, this.container.clientWidth);
   257→      const h = Math.max(1, this.container.clientHeight);
   258→
   259→      // (2) المُصيِّر — نغمة سينمائية ACES لدرجات لون واقعية
   260→      const renderer = new THREE.WebGLRenderer({ antialias: true });
   261→      renderer.setClearColor(COLORS.bg);
   262→      renderer.shadowMap.enabled = this.quality !== 'low';
   263→      renderer.shadowMap.type = THREE.PCFSoftShadowMap;
   264→      renderer.toneMapping = THREE.ACESFilmicToneMapping;
   265→      renderer.toneMappingExposure = TONE_EXPOSURE;
   266→      renderer.setSize(w, h, false);
   267→      renderer.domElement.style.width = '100%';
   268→      renderer.domElement.style.height = '100%';
   269→      renderer.domElement.style.display = 'block';
   270→      this.renderer = renderer;
   271→      // الحاوية position:relative يديرها الاستدعاء — اللوحة تملأها بالكامل
   272→      this.container.appendChild(renderer.domElement);
   273→
   274→      // (3) المشهد والكاميرا والتحكم المداري
   275→      this.scene = new THREE.Scene();
   276→      // خريطة بيئية (IBL) عبر PMREM — تجعل المعادن تعكس مثل الواقع بدل أن تبدو سوداء
   277→      const pmrem = new THREE.PMREMGenerator(renderer);
   278→      this.scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
   279→      this.scene.environmentIntensity = ENV_INTENSITY;
   280→      pmrem.dispose();
   281→      this.camera = new THREE.PerspectiveCamera(40, w / h, 0.1, 100);
   282→      this.camera.position.set(...DEFAULT_CAMERA.pos);
   283→      this.controls = new OrbitControls(this.camera, renderer.domElement);
   284→      this.controls.target.set(...DEFAULT_CAMERA.target);
   285→      this.controls.enableDamping = true;
   286→      this.controls.dampingFactor = 0.08;
   287→      this.controls.minDistance = 2.2;
   288→      this.controls.maxDistance = 14;
   289→      this.controls.maxPolarAngle = 1.45;
   290→      this.controls.update();
   291→      // تفاعل المستخدم يلغي أي تنعيم كاميرا جارٍ
   292→      this.controls.addEventListener('start', () => {
   293→        this.camTween = null;
   294→      });
   295→
   296→      // (4) البيئة ثم القطع — بنداءات المصنع المتفق عليها
   297→      buildEnvironment(this.scene, this.quality);
   298→      const visuals = buildAllParts(this.quality);
   299→      this.scene.add(visuals.root);
   300→      this.visuals = visuals;
   301→
   302→      // ضوء اتجاهي (أضافه buildEnvironment) للتحكم به عند تبديل الظلال
   303→      this.scene.traverse((obj) => {
   304→        if (!this.dirLight && obj instanceof THREE.DirectionalLight) this.dirLight = obj;
   305→      });
   306→
   307→      // تسجيل المعرفات دفاعيًا (المصنع يفعل ذلك عادةً — إعادة النداء غير ضارة)
   308→      for (const id of COMPONENT_IDS) setComponentId(visuals.dynamic[id], id);
   309→      setComponentId(visuals.caseShell, 'case');
   310→      setComponentId(visuals.sidePanel, 'case');
   311→
   312→      this.initDynamicState(visuals);
   313→      this.initFlagRecords(visuals);
   314→
   315→      // (الجلسة 4) الشاشة الافتراضية: نسيج ديناميكي + مرجع كابل الإشارة
   316→      this.painter = createScreenPainter();
   317→      if (this.painter) {
   318→        const screenMesh = visuals.monitor.getObjectByName('monitor-screen');
   319→        if (screenMesh instanceof THREE.Mesh) {
   320→          const smat = screenMesh.material;
   321→          if (!Array.isArray(smat) && 'map' in smat) {
   322→            (smat as THREE.MeshBasicMaterial).map = this.painter.texture;
   323→            (smat as THREE.MeshBasicMaterial).color.set(0xffffff);
   324→            (smat as THREE.MeshBasicMaterial).needsUpdate = true;
   325→          }
   326→        }
   327→        this.drawMonitor(true);
   328→      }
   329→      this.monitorCableGroup = visuals.monitor.getObjectByName('monitor-cable') ?? null;
   330→
   331→      // أصل حقيقي CC0 (Poly Haven): مسح لوحة دوائر كأساس PCB واقعي + مفك تدريبي
   332→      this.loadRealAssets(visuals);
   333→
   334→      // (7)(15) مستمعو المؤشر + مراقب الحجم
   335→      renderer.domElement.addEventListener('pointerdown', this.onPointerDown);
   336→      renderer.domElement.addEventListener('pointerup', this.onPointerUp);
   337→      renderer.domElement.addEventListener('pointermove', this.onPointerMove);
   338→      this.resizeObserver = new ResizeObserver(() => this.onResize());
   339→      this.resizeObserver.observe(this.container);
   340→
   341→      // (13) ضبط كثافة البكسل حسب الجودة (الظلال مضبوطة أعلاه بنفس القيمة)
   342→      this.setQuality(this.quality);
   343→
   344→      // (6) بدء حلقة الرسم
   345→      this.rafId = requestAnimationFrame(this.tick);
   346→      this.callbacks.onReady?.();
   347→    } catch {
   348→      // المعالجة نفسها: رسالة + failed=true
   349→      this.fail(WEBGL_ERROR_AR);
   350→    }
   351→  }

   357→  /** (13) تبديل الجودة: كثافة البكسل + تشغيل/إيقاف الظلال */
   358→  setQuality(q: SceneQuality): void {
   359→    if (this.blocked) return;
   360→    this.quality = q;
   361→    this.renderer.setPixelRatio(this.pixelRatioFor(q));
   362→    const wantShadows = q !== 'low';
   363→    if (this.renderer.shadowMap.enabled !== wantShadows) {
   364→      this.renderer.shadowMap.enabled = wantShadows;
   365→      this.renderer.shadowMap.needsUpdate = true;
   366→      if (this.dirLight) this.dirLight.castShadow = wantShadows;
   367→    }
   368→  }

   370→  /** (10) شفافية الهيكل والغطاء (X-Ray) مع استعادة الأصول عند الإيقاف */
   371→  setXray(on: boolean): void {
   372→    if (this.blocked) return;
   373→    if (this.xrayOn === on) return;
   374→    this.xrayOn = on;
   375→    this.refreshShellMaterials();
   376→  }

   378→  /** (5) فتح/إغلاق الغطاء الجانبي حول مفصلته الخلفية */
   379→  setPanelOpen(on: boolean, animate?: boolean): void {
   380→    if (this.blocked) return;
   381→    this.panelTarget = on ? PANEL_OPEN_ANGLE : 0;
   382→    if (this.lastState) this.lastState.flags.panelOpen = on;
   383→    if (animate === false) this.visuals.sidePanel.rotation.y = this.panelTarget;
   384→  }

   386→  /** (5) تشغيل/إيقاف العرض التفجيري — يعيد حساب الأهداف من آخر حالة متزامنة */
   387→  setExploded(on: boolean, animate?: boolean): void {
   388→    if (this.blocked) return;
   389→    this.exploded = on;
   390→    this.recomputeTargets();
   391→    if (animate === false) this.snapToTargets();
   392→  }

   394→  /** (5) مزامنة حالة المحاكاة: أهداف الحركة (تنعيم) + تطبيق الأعلام فورًا */
   395→  syncState(s: SimulationState): void {
   396→    if (this.blocked) return;
   397→    // نسخة دفاعية حتى لا تؤثر طفرات المتصل على أهدافنا
   398→    this.lastState = {
   399→      flags: { ...s.flags },
   400→      parts: { ...s.parts },
   401→      peripherals: { ...s.peripherals },
   402→    };
   403→    this.recomputeTargets();
   404→    this.applyFlags(s.flags, true);
   405→    this.updateRuntimeVisuals(s);
   406→    this.drawMonitor(true);
   407→  }

   411→    if (this.blocked) return;
   412→    this.bootStartedAt = startedAt;
   413→    this.drawMonitor(true);

   511→  dispose(): void {

   521→      dom.removeEventListener('pointermove', this.onPointerMove);
   522→    }
   523→    this.resizeObserver?.disconnect();

   530→        if (Array.isArray(mat)) {
   531→          for (const m of mat) disposeMaterial(m);
   532→        } else if (mat) {
   533→          disposeMaterial(mat);
   534→        }

   536→      if (this.scene.environment instanceof THREE.Texture) this.scene.environment.dispose();
   537→      if (this.scene.background instanceof THREE.Texture) this.scene.background.dispose();
   538→    }
   539→    this.renderer?.dispose();
   540→    dom?.remove();

   542→    this.shellOriginals.clear();
   543→    this.damagedOriginals.clear();

   561→  /** (6) حلقة الرسم: دوّارات، اقتراب القطع، تنعيم الغطاء، تنعيم الكاميرا، ثم الرسم */
   562→  private readonly tick = (): void => {
   563→    if (this.disposed || this.failed) return;
   564→    this.rafId = requestAnimationFrame(this.tick);
   565→    const dt = Math.min(this.clock.getDelta(), 0.05); // حماية من قفزات التبويبات الخاملة
   566→    const k = 1 - Math.exp(-LERP_RATE * dt);
   567→
   568→    // دوران شفرات المراوح حول محورها المحفوظ — فقط والجهاز يعمل فعلاً
   569→    if (this.spinning) {
   570→      for (const spinner of this.spinners) {
   571→        spinner.obj.rotateOnAxis(spinner.axis, SPIN_SPEED * dt);
   572→      }
   573→    }
   574→
   575→    // اقتراب القطع نحو أهدافها (الموضع lerp والدوران slerp) — عدا الهيكل الثابت
   576→    for (const id of COMPONENT_IDS) {
   577→      if (id === 'case') continue;
   578→      const part = this.parts[id];
   579→      part.group.position.lerp(part.targetPos, k);
   580→      part.group.quaternion.slerp(part.targetQuat, k);
   581→    }
   582→
   583→    // تطفو القطعة الملتقطة (awaiting) فوق الصينية بلطف — إشارة بصرية لجاهزية التركيب
   584→    const elapsed = this.clock.getElapsedTime();
   585→    for (const id of COMPONENT_IDS) {
   586→      if (id === 'case') continue;
   587→      if (this.lastState && this.lastState.parts[id] === 'awaiting' && this.dragId !== id) {
   588→        const part = this.parts[id];
   589→        part.group.position.y = part.trayPos.y + 0.16 + Math.sin(elapsed * 3) * 0.04;
   590→      }
   591→    }
   592→
   593→    // تنعيم دوران الغطاء الجانبي نحو الهدف
   594→    const panel = this.visuals.sidePanel;
   595→    panel.rotation.y += (this.panelTarget - panel.rotation.y) * k;
   596→
   597→    // استكمال تنعيم الكاميرا (resetCamera / focusComponent)
   598→    const tween = this.camTween;
   599→    if (tween) {
   600→      tween.t = Math.min(tween.t + dt / tween.dur, 1);
   601→      const eased = PCScene.easeInOut(tween.t);
   602→      this.camera.position.lerpVectors(tween.fromPos, tween.toPos, eased);
   603→      this.controls.target.lerpVectors(tween.fromTarget, tween.toTarget, eased);
   604→      if (tween.t >= 1) this.camTween = null;
   605→    }
   606→
   607→    this.controls.update();
   608→
   609→    // إعادة رسم نسيج الشاشة الافتراضية بمعدل مقيد
   610→    if (this.painter && nowMs() - this.lastMonitorDrawAt >= MONITOR_REDRAW_MS) {
   611→      this.drawMonitor(false);
   612→    }
   613→
   614→    this.renderer.render(this.scene, this.camera);
   615→  };
   616→
   617→  /** تنعيم easeInOut: t<0.5 ? 2t² : 1 − (−2t+2)²/2 */
   618→  private static easeInOut(t: number): number {
   619→    return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
   620→  }

   622→  /** إعادة حساب أهداف القطع من آخر حالة متزامنة + علم التفجير الحالي */
   623→  private recomputeTargets(): void {
   624→    const parts = this.lastState?.parts;
   625→    for (const id of COMPONENT_IDS) {
   626→      if (id === 'case') continue;
   627→      const part = this.parts[id];
   628→      const status: string | undefined = parts ? parts[id] : undefined;
   629→      if (status === 'removed' || status === 'damaged') {
   630→        part.targetPos.copy(part.trayPos);
   631→        part.targetRot.copy(part.trayRot);
   632→        if (status === 'damaged') {
   633→          // القطعة التالفة تستلقي مائلة على الصينية
   634→          part.targetRot.z += 0.35;
   635→        }
   636→      } else if (status === 'awaiting') {
   637→        // ملتقطة من الصينية — ترتفع قليلاً (وتطفو في حلقة الرسم)
   638→        part.targetPos.copy(part.trayPos);
   639→        part.targetPos.y += 0.16;
   640→        part.targetRot.copy(part.trayRot);
   641→      } else if (status === 'separated') {
   642→        // فصل جزئي: ما زالت في موضعها مع إزاحة صغيرة نحو اتجاه الفك
   643→        part.targetPos.copy(part.basePos);
   644→        part.targetPos.addScaledVector(part.explodeVec, SEPARATE_FACTOR);
   645→        part.targetRot.set(0, 0, 0);
   646→      } else {
   647→        part.targetPos.copy(part.basePos);
   648→        if (this.exploded) part.targetPos.add(part.explodeVec);
   649→        part.targetRot.set(0, 0, 0);
   650→      }
   651→      part.targetQuat.setFromEuler(part.targetRot);
   652→    }
   653→  }

   655→  /** نسخ الأهداف فورًا إلى المجسمات (للتنعيم المعطَّل animate=false) */

   666→  /**
   667→   * (5) تطبيق الأعلام فورًا على المجسمات الفرعية.
   668→   * افتراضات المحاور (لأن بناء المصنع موازٍ ولم يُثبَّت بعد):
   669→   *  - مزلاج PCIe ورافعة CPU: دوران حول Y المحلي (اللوح عمودي يواجه -X).
   670→   *  - مزالج RAM: دوران حول Z بإشارة متناوبة بين الطرفين.
   671→   *  - البراغي: ترتفع على +Y (مسامير اللوحة تُسحب على -X) مع دوران طفيف حول Z.
   672→   *  - القوابس: إزاحة -X 0.12 عند الفصل.
   673→   * كل الإزاحات نسبية إلى القيم الأصلية المحفوظة عند البناء.
   674→   */
   675→  private applyFlags(flags: Record<MachineFlag, boolean>, animate: boolean): void {
   676→    // الغطاء الجانبي — عبر هدف تنعيمي تتبعه الحلقة
   677→    this.panelTarget = flags.panelOpen ? PANEL_OPEN_ANGLE : 0;
   678→    if (!animate) this.visuals.sidePanel.rotation.y = this.panelTarget;
   679→
   680→    // مزلاج PCIe
   681→    if (this.latchRec) {
   682→      this.latchRec.obj.rotation.y = this.latchRec.baseRot + (flags.pcieLatchOpen ? LATCH_OPEN_ANGLE : 0);
   683→    }
   684→
   685→    // مزالج RAM
   686→    for (const clip of this.clipRecs) {
   687→      clip.obj.rotation.z = clip.baseRotZ + (flags.ramClipsOpen ? clip.sign * CLIP_OPEN_ANGLE : 0);
   688→    }
   689→
   690→    // رافعة مقبس CPU
   691→    if (this.leverRec) {
   692→      this.leverRec.obj.rotation.y = this.leverRec.baseRot + (flags.cpuLeverOpen ? LEVER_OPEN_ANGLE : 0);
   693→    }
   694→
   695→    // البراغي (gpu/psu/fan/m2/mobo)
   696→    for (const [flag, recs] of this.screwMap) {
   697→      const out = flags[flag];
   698→      for (const rec of recs) {
   699→        rec.obj.position.copy(rec.basePos);
   700→        if (out) {
   701→          rec.obj.position[rec.axis] += rec.delta;
   702→          rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN;
   703→        } else {
   704→          rec.obj.rotation.z = rec.baseRotZ;
   705→        }
   706→      }
   707→    }
   708→
   709→    // قوابس الموصلات (fan headers / front panel)
   710→    for (const plug of this.plugRecs) {
   711→      plug.obj.position.x = plug.baseX - (flags[plug.flag] ? PLUG_PULL : 0);
   712→    }
   713→
   714→    // لمبة الطاقة — مادة خاصة بها (cached) — تنطفئ مع مزود ميت
   715→    const led = this.visuals.flags.powerLed;
   716→    if (led) {
   717→      const mat = led.material;
   718→      if (!Array.isArray(mat) && 'emissive' in mat) {
   719→        const std = mat as THREE.MeshStandardMaterial;
   720→        const alive = flags.powerOn && !(this.lastState?.peripherals.psuDead ?? false);
   721→        std.emissive.set(alive ? COLORS.green : 0x000000);
   722→        std.emissiveIntensity = LED_INTENSITY;
   723→      }
   724→    }
   725→  }

   731→  /** إعادة حساب مواد الهيكل من الحالة المجمّعة (xray يتقدم على العزل) */

   747→  /** تطبيق شفافية على مواد مجموعة (الزجاج شفاف أصلًا فيبقى كما هو) */
   748→  private applyShellOpacity(root: THREE.Object3D, opacity: number): void {
   749→    root.traverse((obj) => {
   750→      if (!(obj instanceof THREE.Mesh)) return;
   751→      const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   752→      for (const mat of mats) {
   753→        if (mat.transparent) continue; // الزجاج/الشفاف الأصلي لا يُمسّ
   754→        if (!this.shellOriginals.has(mat)) {
   755→          this.shellOriginals.set(mat, {
   756→            opacity: mat.opacity,
   757→            transparent: mat.transparent,
   758→            depthWrite: mat.depthWrite,
   759→          });
   760→        }
   761→        mat.transparent = true;
   762→        mat.opacity = opacity;
   763→        mat.depthWrite = false;
   764→      }
   765→    });
   766→  }

   784→  /**
   785→   * تحميل الأصول الحقيقية المرخصة CC0 (توجيه v3 §3) — فشلها لا يعطل المشهد:
   786→   *  1) مسح لوحة دوائر (Poly Haven — circuit_board): يستبدل وجه PCB الإجرائي للوحة الأم
   787→   *     بمسح ضوئي واقعي بخامات PBR كاملة (Diffuse/Normal/ARM).
   788→   *  2) مفك مسطّح (Poly Haven — screwdriver): أداة تدريب ثانوية على طاولة صينية القطع.
   789→   */
   790→  private loadRealAssets(visuals: PartVisuals): void {
   791→    const loader = new GLTFLoader();
   792→    const root = visuals.root;
   793→
   794→    // 1) أساس PCB الحقيقي — يستبدل وجه اللوحة الأمامي فقط، ويبقى الباقي إجرائيًا تفاعليًا
   795→    loader.load(
   796→      '/models/circuit-board/circuit_board_1k.gltf',
   797→      (gltf) => {
   798→        if (this.disposed || this.failed) return;
   799→        const pcb = visuals.dynamic.motherboard.getObjectByName('mobo-pcb');
   800→        if (!pcb) return;
   801→        const scan = gltf.scene;
   802→        scan.traverse((obj) => {
   803→          if (!(obj instanceof THREE.Mesh)) return;
   804→          obj.castShadow = false;
   805→          obj.receiveShadow = true;
   806→          // توحيد الخامات: منع أي emissive قد يأتي من الأصل (يبقى الإبراز صادقًا)
   807→          const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   808→          for (const m of mats) {
   809→            if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = 0;
   810→          }
   811→        });
   812→        // bbox الأصل (م): 0.404×0.078×0.398 — رأسيًا مُدار بحيث مكوّناته نحو -X (وجه المشاهد)
   813→        const bbox = new THREE.Box3().setFromObject(scan);
   814→        const size = bbox.getSize(new THREE.Vector3());
   815→        const targetH = BOARD.yMax - BOARD.yMin;      // 3.0
   816→        const targetD = BOARD.zMax - BOARD.zMin;      // 2.44
   817→        scan.rotation.z = Math.PI / 2;                // +Y المحلي → -X عالمي
   818→        scan.scale.set(
   819→          BOARD.thickness / Math.max(size.y, 1e-6),   // سماكة اللوحة
   820→          targetH / Math.max(size.x, 1e-6),           // ارتفاع اللوحة
   821→          targetD / Math.max(size.z, 1e-6),           // عمق اللوحة
   822→        );
   823→        // إعادة قياس bbox بعد الدوران ثم تمركز اللوحة على مستوى المجموعة

   943→  /** إيجاد الهدف عند نقطة شاشة: أول إصابة مرئية غير شفافة — قطعة PC أو طرفية */
   944→  private pickAt(clientX: number, clientY: number): PickedInfo {
   945→    const rect = this.renderer.domElement.getBoundingClientRect();
   946→    const w = rect.width > 0 ? rect.width : 1;
   947→    const h = rect.height > 0 ? rect.height : 1;
   948→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
   949→    this.raycaster.setFromCamera(this._ndc, this.camera);
   950→    const hits = this.raycaster.intersectObjects(this.visuals.root.children, true);
   951→    for (const hit of hits) {
   952→      if (this.isHidden(hit.object)) continue;        // تجاهل ما خفيه العزل
   953→      if (this.isSeeThrough(hit.object)) continue;    // الزجاج/الشفاف لا يحجب الاختيار
   954→      let cur: THREE.Object3D | null = hit.object;
   955→      while (cur) {
   956→        // الطرفيات (زر الشاشة / زر طاقة الشاشة) — لا كيبورد ولا ماوس (توجيه v3 أ.1)
   957→        const per: unknown = cur.userData['peripheral'];
   958→        if (typeof per === 'string') {
   959→          return { type: 'periph', kind: per as PeripheralKind };
   960→        }
   961→        const cid: unknown = cur.userData['componentId'];
   962→        if (typeof cid === 'string' && (COMPONENT_IDS as readonly string[]).includes(cid)) {
   963→          return { type: 'part', id: cid as ComponentId };
   964→        }
   965→        cur = cur.parent;
   966→      }
   967→      return null; // أول إصابة مرئية بلا معرّف → تعامل كفراغ (الأدوات الثانوية غير قابلة للنقر)
   968→    }
   969→    return null;
   970→  }

  1062→  /** استنتاج محور الدوران: أصغر بُعد في صندوق الإحاطة للشبك الأول هو عمود القرص */
  1063→  private computeSpinnerAxis(spinner: THREE.Object3D): THREE.Vector3 {
  1064→    const fallback = new THREE.Vector3(0, 1, 0);
  1065→    const mesh = this.firstMesh(spinner);
  1066→    if (!mesh || !mesh.geometry) return fallback;
  1067→    const geometry = mesh.geometry;
  1068→    if (!geometry.boundingBox) geometry.computeBoundingBox();
  1069→    const bb = geometry.boundingBox;
  1070→    if (!bb) return fallback;
  1071→    const ex = bb.max.x - bb.min.x;
  1072→    const ey = bb.max.y - bb.min.y;
  1073→    const ez = bb.max.z - bb.min.z;
  1074→    const axis = new THREE.Vector3(
  1075→      ex <= ey && ex <= ez ? 1 : 0,
  1076→      ey <= ex && ey <= ez ? 1 : 0,
  1077→      ez <= ex && ey <= ez ? 1 : 0,
  1078→    );
  1079→    if (axis.lengthSq() === 0) return fallback;
  1080→    // تحويل المحور من فضاء الشبك إلى الفضاء المحلي للدوّار إن كان ابنًا مُدارًا
  1081→    if (mesh !== spinner) {
  1082→      spinner.updateWorldMatrix(true, false);
  1083→      mesh.updateWorldMatrix(true, false);
  1084→      const qSpinner = new THREE.Quaternion();
  1085→      const qMesh = new THREE.Quaternion();
  1086→      spinner.getWorldQuaternion(qSpinner);
  1087→      mesh.getWorldQuaternion(qMesh);
  1088→      axis.applyQuaternion(qSpinner.invert().multiply(qMesh));
  1089→    }
  1090→    return axis.normalize();
  1091→  }

  1152→    const dy = event.clientY - this.downY;
  1153→    if (dx * dx + dy * dy >= CLICK_MAX_DIST_SQ) return; // سحب لا نقرة
  1154→    const picked = this.pickAt(event.clientX, event.clientY);
  1155→    if (picked && picked.type === 'periph') {
  1156→      this.callbacks.onPeripheral?.(picked.kind, picked.detail);

  1158→    }
  1159→    this.callbacks.onSelect(picked ? picked.id : null);

  1220→  /** إنهاء السحب: تحديد منطقة الإفلات ثم إبلاغ المنطق (التحقق أولًا ثم الحركة — توجيه v3 §6) */
  1221→  private finishDrag(clientX: number, clientY: number): void {

  1225→    this.renderer.domElement.style.cursor = '';
  1226→    if (!id) return;
  1227→    const hit = this.rayPlaneHit(clientX, clientY);
  1228→    let zone: DropZone = null;
  1229→    if (hit) {
  1230→      const inTray = hit.x >= TRAY.minX && hit.x <= TRAY.maxX && hit.z >= TRAY.minZ && hit.z <= TRAY.maxZ;
  1231→      const inCase = hit.x >= CASE.minX && hit.x <= CASE.maxX && hit.z >= CASE.minZ && hit.z <= CASE.maxZ;
  1232→      zone = inTray ? 'tray' : inCase ? 'case' : null;
  1233→    }
  1234→    // المرور عبر منطق المتجر: إن رُفض الإجراء تعاد القطعة لموضع راحتها عبر syncState
  1235→    this.callbacks.onPartDrop?.(id, zone);
  1236→  }

  1238→  /** إلغاء أي سحب جارٍ (Esc / إيقاف الوضع) */
  1239→  private cancelDrag(): void {
  1240→    if (this.dragId) {
  1241→      const part = this.parts[this.dragId];
  1242→      this.recomputeTargets(); // يعيد الهدف الصحيح من آخر حالة
  1243→      if (part) part.group.position.copy(part.targetPos);
  1244→    }
  1245→    this.dragId = null;
  1246→    this.pendingDrag = null;
  1247→    if (this.controls) this.controls.enabled = true;
  1248→    if (this.renderer) this.renderer.domElement.style.cursor = '';
  1249→  }

  1251→  /** تقاطع شعاع المؤشر مع سطح السحب الحالي */
  1252→  private rayPlaneHit(clientX: number, clientY: number): THREE.Vector3 | null {
  1253→    const rect = this.renderer.domElement.getBoundingClientRect();
  1254→    const w = rect.width > 0 ? rect.width : 1;
  1255→    const h = rect.height > 0 ? rect.height : 1;
  1256→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1257→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1258→    const hit = this._v2;
  1259→    return this.raycaster.ray.intersectPlane(this.dragPlane, hit) ? hit : null;
  1260→  }
