    76→function chrome(os: VirtualOS, ctx: CanvasRenderingContext2D, family: WinFamily | null, titleAr: string): number {
    77→  const themeBg: Record<string, [string, string]> = {
    78→    blueText: ['#0a1a56', '#0a1030'],
    79→    aero: ['#123a63', '#0a1c33'],
    80→    metro: ['#0b2b3d', '#081826'],
    81→    modern: ['#12303a', '#081a20'],
    82→    modern11: ['#1a2c3a', '#0b141c'],
    83→  };
    84→  const [c1, c2] = themeBg[family?.theme ?? 'modern'];
    85→  const g = ctx.createLinearGradient(0, 0, os.W, os.H);
    86→  g.addColorStop(0, c1);
    87→  g.addColorStop(1, c2);
    88→  ctx.fillStyle = g;
    89→  ctx.fillRect(0, 0, os.W, os.H);
    90→
    91→  ctx.fillStyle = 'rgba(255,255,255,0.92)';
    92→  ctx.font = 'bold 22px sans-serif';
    93→  ctx.fillText(titleAr, 36, 46);
    94→  ctx.fillStyle = 'rgba(255,255,255,0.55)';
    95→  ctx.font = '13px sans-serif';
    96→  ctx.fillText('مثبّت محاكاة تعليمية — PC Lab (ليس وسائط Windows حقيقية ولا يطلب مفاتيح فعلية)', 36, 68);
    97→  return 96;
    98→}

   100→function bigButton(os: VirtualOS, ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, label: string, onClick: () => void, primary = true): void {
   101→  const hovered = os.hoverAt(x, y, w, h);
   102→  ctx.fillStyle = primary ? (hovered ? '#57d9c9' : '#2aa89a') : hovered ? 'rgba(255,255,255,0.12)' : 'rgba(255,255,255,0.06)';
   103→  roundRect(ctx, x, y, w, h, 7);
   104→  ctx.fill();
   105→  ctx.strokeStyle = 'rgba(255,255,255,0.35)';
   106→  roundRect(ctx, x, y, w, h, 7);
   107→  ctx.stroke();
   108→  ctx.fillStyle = primary ? '#04231f' : '#e8eef5';
   109→  ctx.font = 'bold 14px sans-serif';
   110→  ctx.textAlign = 'center';
   111→  ctx.fillText(label, x + w / 2, y + h / 2 + 5);
   112→  ctx.textAlign = 'left';
   113→  os.addRect('button', x, y, w, h, { onClick });
   114→}

   116→function selectable(os: VirtualOS, ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, label: string, subAr: string | null, selected: boolean, onSelect: () => void): void {
   117→  const hovered = os.hoverAt(x, y, w, h);
   118→  ctx.fillStyle = selected ? 'rgba(57,199,184,0.20)' : hovered ? 'rgba(255,255,255,0.07)' : 'rgba(255,255,255,0.035)';
   119→  roundRect(ctx, x, y, w, h, 7);
   120→  ctx.fill();
   121→  ctx.strokeStyle = selected ? '#39c7b8' : 'rgba(255,255,255,0.18)';
   122→  roundRect(ctx, x, y, w, h, 7);
   123→  ctx.stroke();
   124→  ctx.fillStyle = '#e8eef5';
   125→  ctx.font = 'bold 14.5px sans-serif';
   126→  ctx.fillText(label, x + 16, y + 25);
   127→  if (subAr) {
   128→    ctx.fillStyle = 'rgba(232,238,245,0.65)';
   129→    ctx.font = '12.5px sans-serif';
   130→    ctx.fillText(subAr, x + 16, y + h - 12);
   131→  }
   132→  if (selected) {
   133→    ctx.fillStyle = '#39c7b8';
   134→    ctx.fillText('✓', x + w - 26, y + 26);
   135→  }
   136→  os.addRect('row', x, y, w, h, { onClick: onSelect });
   137→}

   141→export function drawInstaller(os: VirtualOS, ctx: CanvasRenderingContext2D, now: number): void {
   142→  const store = useLabStore.getState();
   143→  const family = setup.family ? getFamily(setup.family) : null;
   144→
   145→  // اختيار العائلة أولًا (قائمة وسائط الإقلاع)
   146→  if (!setup.family) {

   151→    let y = top + 36;
   152→    const cols = 2;
   153→    const bw = Math.min(430, (os.W - 90) / cols);
   154→    let i = 0;
   155→    for (const fam of WINDOWS_FAMILIES) {
   156→      const cx = 36 + (i % cols) * (bw + 18);
   157→      const cy = y + Math.floor(i / cols) * 62;

   159→        setup.family = fam.id;
   160→        setup.step = 1;
   161→        setup.errorAr = null;
   162→        os.dirty = true;
   163→      });
   164→      i++;
   165→    }
   166→    y += Math.ceil(WINDOWS_FAMILIES.length / cols) * 62 + 6;
   167→    ctx.fillStyle = 'rgba(255,255,255,0.55)';
   168→    ctx.font = '12.5px sans-serif';
   169→    os.wrapText(ctx, 'ملاحظة أمانة: هذا معالج محاكاة داخل مختبر افتراضي — لا يوزع ملفات Windows ولا يطلب مفاتيح حقيقية ولا يمس أقراص جهازك.', 36, y, os.W - 72, 17);

   171→  }
   172→  if (!family) return;
   173→

   175→    // إنهاء المثبّت والعودة (إعادة تشغيل بدون وسيط)

   181→  // ===== 1) اللغة والمنطقة =====

   188→      y += 62;
   189→    }
   190→    bigButton(os, ctx, 36, y + 10, 140, 38, 'Next ←', () => { setup.step = 2; os.dirty = true; });

   196→  if (setup.step === 2) {

   203→    ctx.fillStyle = 'rgba(0,0,0,0.35)';

   208→    ctx.stroke();
   209→    ctx.fillStyle = '#e8eef5';

   231→      setup.step = 3;
   232→      os.dirty = true;
   233→    }, keyValid);
   234→    bigButton(os, ctx, 190, y, 110, 38, 'رجوع →', () => { setup.step = 1; os.dirty = true; }, false);

   236→      ctx.fillStyle = '#ffb4b4';
   237→      ctx.font = '13px sans-serif';
   238→      os.wrapText(ctx, '!! ' + setup.errorAr, 36, y + 60, os.W - 72, 17);

   242→

   244→  if (setup.step === 3) {
   245→    const top = chrome(os, ctx, family, 'Select Edition — اختر النسخة والبنية');

   247→    let idx = 0;
   248→    for (const ed of family.editions) {

   253→          setup.editionId = ed.id;
   254→          setup.arch = arch;
   255→          setup.errorAr = null;
   256→          os.dirty = true;

   259→      }

   262→    const y = top + 4 + rows * 58;

   266→      setup.step = 4;

   272→      ctx.font = '13px sans-serif';

   293→    for (const l of lines) { ctx.fillText(l, 52, ly); ly += 22; }

   299→      setup.step = 5;

   315→      setup.installType = 'clean';

   317→      setup.step = 6;

   321→      ctx.fillStyle = '#ffd9a0';
   322→      ctx.font = '13px sans-serif';
   323→      os.wrapText(ctx, '!! ' + setup.errorAr, 36, top + 156, os.W - 72, 17);

   336→    let y = top + 4;
   337→
   338→    // فحص متطلبات Windows 11 الموثقة

   351→    if (!platformBootsFamily) {
   352→      ctx.fillStyle = '#ffb4b4';

   354→      ctx.fillText(`!! ${family.nameEn} تقلع بوضع ${family.bootModes.join('/').toUpperCase()} فقط — المنصة حاليًا ${boot === 'uefi' ? 'UEFI' : 'BIOS/Legacy'}: بدّل الوضع من إعدادات البرنامج الثابت (تبويب Boot).`, 36, y);
   355→      y += 22;
   356→    }
   357→    ctx.fillStyle = 'rgba(255,255,255,0.75)';
   358→    ctx.font = '12.5px sans-serif';
   359→    ctx.fillText(`وضع إقلاع الجهاز: ${boot === 'uefi' ? 'UEFI' : 'BIOS/Legacy'} — الأنماط المسموحة لهذه العائلة في هذا الوضع: ${allowed.join(' / ').toUpperCase() || 'لا شيء (تعارض وضع الإقلاع)'}`, 36, y);
