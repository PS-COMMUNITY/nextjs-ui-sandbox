   176→let persistTimer: ReturnType<typeof setTimeout> | null = null;
   177→let persistPending: (() => void) | null = null;
   178→
   179→function runPersist(get: () => LabStore) {
   180→  const s = get();
   181→  if (!s.hydrated) return;
   182→  const shape: PersistShape = {
   183→    version: APP_VERSION,
   184→    simulation: s.sim,
   185→    tech: s.tech,
   186→    logs: s.logs,
   187→    quizScores: s.quizScores,
   188→    missionsDone: s.missionsDone,
   189→    lessonsRead: s.lessonsRead,
   190→    // (إصلاح U10) المهمة النشطة وخطوتها ودفتر النظام تُحفظ أيضًا
   191→    notepad: s.notepad,
   192→    activeMissionId: s.activeMissionId,
   193→    missionStepIndex: s.missionStepIndex,
   194→    settings: { quality: s.quality },
   195→    savedAt: Date.now(),
   196→  };
   197→  const res = saveAll(shape);
   198→  if (res.ok) {
   199→    useLabStore.setState({ saveStatus: { available: true, lastSavedAt: Date.now(), lastErrorAr: null } });
   200→  } else {
   201→    useLabStore.setState({ saveStatus: { available: true, lastSavedAt: null, lastErrorAr: res.messageAr } });
   202→  }
   203→}
   204→
   205→function schedulePersist(get: () => LabStore) {
   206→  if (persistTimer) clearTimeout(persistTimer);
   207→  persistPending = () => runPersist(get);
   208→  persistTimer = setTimeout(() => {
   209→    persistTimer = null;
   210→    const fn = persistPending;
   211→    persistPending = null;
   212→    fn?.();
   213→  }, 400);
   214→}
   215→
   216→/**
   217→ * (إصلاح U7) تنفيذ الحفظ المعلق فورًا إن وُجد — يُستدعى عند pagehide وعند إخفاء
   218→ * التبويب (visibilitychange) حتى لا يضيع آخر 400ms من التقدم عند إغلاق الصفحة.
   219→ * السلوك المؤجل الافتراضي (400ms) يبقى كما هو في بقية الأحوال.
   220→ */
   221→export function flushPersist(): void {
   222→  if (persistTimer) {
   223→    clearTimeout(persistTimer);
   224→    persistTimer = null;
   225→  }
   226→  const fn = persistPending;
   227→  persistPending = null;
   228→  fn?.();
   229→}

   231→// تسجيل مستمعي التفريغ مرة واحدة فقط (حماية SSR وHMR بعلم على مستوى الوحدة)
   232→let persistFlushBound = false;
   233→if (typeof window !== 'undefined' && typeof document !== 'undefined' && !persistFlushBound) {
   234→  persistFlushBound = true;
   235→  window.addEventListener('pagehide', () => flushPersist());
   236→  document.addEventListener('visibilitychange', () => {
   237→    if (document.visibilityState === 'hidden') flushPersist();
   238→  });
   239→}

   258→export const useLabStore = create<LabStore>((set, get) => ({

   276→  quizScores: {},
   277→  missionsDone: [],
   278→  lessonsRead: [],

   283→  hydrate: () => {
   284→    if (!storageAvailable()) {
   285→      set({ hydrated: true, saveStatus: { available: false, lastSavedAt: null, lastErrorAr: 'تخزين المتصفح غير متاح — لن يُحفظ التقدم' } });
   286→      return;

   291→      // استئناف جلسة محفوظة والجهاز يعمل فيها: نُرجع ساعة التمهيد إلى الماضي
   292→      // حتى تظهر نتيجتها المحسومة مباشرة بدل الت stuck على شعار البدء
   293→      const bootStartedAt =
   294→        sim.flags.powerOn && sim.flags.acConnected && !sim.peripherals.psuDead
   295→          ? Date.now() - 9999
   296→          : null;
   297→      set({
   298→        sim,
   299→        bootStartedAt,
   300→        tech: data.tech ?? null,
   301→        logs: data.logs ?? [],
   302→        quizScores: data.quizScores ?? {},
   303→        missionsDone: data.missionsDone ?? [],
   304→        lessonsRead: data.lessonsRead ?? [],
   305→        // (إصلاح U10) استعادة المهمة النشطة وخطوتها ودفتر النظام من الحفظ
   306→        notepad: data.notepad ?? '',
   307→        activeMissionId: data.activeMissionId ?? null,
   308→        missionStepIndex: data.missionStepIndex ?? 0,
   309→        quality: data.settings?.quality ?? 'medium',
   310→        hydrated: true,
   311→        saveStatus: { available: true, lastSavedAt: data.savedAt, lastErrorAr: null },
   312→      });
   313→    } else {
   314→      // (إصلاح U4) كان فشل التحميل صامتًا: إن وُجد حفظ سابق فعلًا لكنه تالف أو من
   315→      // إصدار غير مدعوم، نُعلم المستخدم مرة واحدة بدل التجاهل. لا استيراد تلقائي.
   316→      const meta = loadRawMeta();
   317→      if (meta.exists && !discardedSaveNotified) {
   318→        discardedSaveNotified = true;
   319→        toast.error('تم تجاهل حفظ سابق بسبب تعارض إصدار أو بيانات غير صالحة — بدأنا جلسة جديدة');
   320→      }
   321→      set({ hydrated: true });
   322→    }
   323→  },

   353→
   354→  // ============ آلة الحالات: فصل جزئي → استخراج → التقاط → تركيب ============
   355→  detachPart: (id) => {
   356→    const s = get();
   357→    if (id === 'case') return;
   358→    if (isCable(id)) {
   359→      // الكابلات تفصل بحركة واحدة — التوجيه للمسار الموحد
   360→      s.removePart(id);
   361→      return;
   362→    }
   363→    if (s.sim.parts[id] !== 'installed') {
   364→      toast.error(`${nameOf(id)}: ليست في حالة تسمح بالفصل الجزئي`);
   365→      return;
   366→    }
   367→    const check = canRemove(s.sim, id);
   368→    if (!check.ok) {
   369→      s.logEvent('remove', `${nameOf(id)}: فصل مرفوض`, false, check.messageAr);
   370→      toast.error(check.messageAr);
   371→      return;
   372→    }
   373→    const sim = applyDetach(s.sim, id);
   374→    set({ sim });
   375→    s.logEvent('remove', `فصل جزئي: ${nameOf(id)}`, true, 'القطعة ما زالت في مكانها — استخرجها إلى الصينية عند الجاهزية');
   376→    toast.success(`${nameOf(id)}: فُصلت جزئيًا — استخرجها إلى الصينية`);
   377→    schedulePersist(get);
   378→  },
   379→
   380→  extractPart: (id) => {
   381→    const s = get();
   382→    if (s.sim.parts[id] !== 'separated') {
   383→      toast.error(`${nameOf(id)}: يجب فصلها جزئيًا أولًا قبل استخراجها`);
   384→      return;
   385→    }
   386→    const sim = applyExtract(s.sim, id);
   387→    set({ sim });
   388→    s.logEvent('remove', `استخراج ${nameOf(id)} إلى صينية القطع`, true);
   389→    toast.success(`${nameOf(id)}: نُقلت إلى صينية القطع`);
   390→    schedulePersist(get);
   391→  },
   392→
   393→  preparePart: (id) => {
   394→    const s = get();
   395→    if (s.sim.parts[id] !== 'removed') {
   396→      toast.error(`${nameOf(id)}: القطعة ليست في الصينية`);
   397→      return;
   398→    }
   399→    const sim = applyPrepare(s.sim, id);
   400→    set({ sim, selected: id });
   401→    s.logEvent('install', `التقاط ${nameOf(id)} للتركيب`, true, 'راجع شروط التركيب ثم اضغط «ركّب»');
   402→    toast.success(`${nameOf(id)}: ملتقطة — راجع شروط التركيب ثم ركّبها`);
   403→    schedulePersist(get);
   404→  },
   405→
   406→  installPart: (id) => {
   407→    const s = get();
   408→    const status = s.sim.parts[id];
   409→    if (status === 'installed') {
   410→      toast.error(`${nameOf(id)}: مثبتة أصلًا`);
   411→      return;
   412→    }
   413→    const check = canInstall(s.sim, id);
   414→    if (!check.ok) {
   415→      // INSTALL_REJECTED: حدث رفض تعليمي — القطعة تبقى بانتظار التركيب
   416→      s.logEvent('install', `${nameOf(id)}: تركيب مرفوض`, false, check.messageAr);
   417→      toast.error(check.messageAr);
   418→      return;
   419→    }
   420→    const sim = applyInstall(s.sim, id);
   421→    set({ sim });
   422→    s.logEvent('install', `تركيب ${nameOf(id)}`, true, 'أُعيدت القطعة إلى موضع تثبيتها — أغلق المزالج والبراغي إن وجدت');
   423→    toast.success(`${nameOf(id)}: رُكّبت في موضعها`);
   424→    schedulePersist(get);
   425→  },
   426→
   427→  removePart: (id) => {
   428→    const s = get();
   429→    const name = nameOf(id);
   430→    if (id === 'case') {
   431→      toast.error('الهيكل هو جسم الجهاز؛ استخدم زر فتح/إغلاق الغطاء الجانبي.');
   432→      return;
   433→    }
   434→    const check = canRemove(s.sim, id);
   435→    if (!check.ok) {
   436→      s.logEvent('remove', `${name}: فك مرفوض`, false, check.messageAr);
   437→      toast.error(check.messageAr);
   438→      return;
   439→    }
   440→    // الكابلات: فصل مباشر إلى الصينية — القطع الكبيرة: فصل جزئي ثم استخراج
   441→    const sim = isCable(id) ? applyRemove(s.sim, id) : applyDetach(s.sim, id);
   442→    set({ sim });
   443→    if (isCable(id)) {
   444→      s.logEvent('remove', `فصل ${name}`, true, 'نُقل إلى صينية القطع');
   445→      toast.success(`${name}: فُصل إلى صينية القطع`);
   446→    } else {
