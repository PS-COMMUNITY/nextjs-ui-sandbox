    49→interface MatOpts {
    50→  map?: THREE.Texture | null;
    51→  emissive?: number;
    52→  emissiveIntensity?: number;
    53→  transparent?: boolean;
    54→  opacity?: number;

    58→/** cache للمواد القياسية — مفاتيح بنطاق القطعة */
    59→const matCache = new Map<string, THREE.MeshStandardMaterial>();
    60→
    61→function mat(color: number, metalness: number, roughness: number, opts?: MatOpts): THREE.MeshStandardMaterial {
    62→  const mapId = opts?.map ? opts.map.uuid : 'n';
    63→  const key = `${partScope}|${color}|${metalness}|${roughness}|${mapId}|${opts?.emissive ?? ''}|${opts?.emissiveIntensity ?? ''}|${opts?.transparent ?? ''}|${opts?.opacity ?? ''}|${opts?.side ?? ''}`;
    64→  const cached = matCache.get(key);
    65→  if (cached) return cached;
    66→  const m = new THREE.MeshStandardMaterial({
    67→    color,
    68→    metalness,
    69→    roughness,
    70→    map: opts?.map ?? null,
    71→    emissive: opts?.emissive ?? 0x000000,
    72→    emissiveIntensity: opts?.emissiveIntensity ?? 1,
    73→    transparent: opts?.transparent ?? false,
    74→    opacity: opts?.opacity ?? 1,
    75→    side: opts?.side ?? THREE.FrontSide,
    76→    flatShading: false,
    77→  });
    78→  matCache.set(key, m);
    79→  return m;
    80→}

   103→/** صندوق سريع مع ضبط الظلال حسب الجودة */
   104→function box(
   105→  w: number, h: number, d: number, material: THREE.Material,
   106→  x = 0, y = 0, z = 0, cast = true, receive = false,
   107→): THREE.Mesh {
   108→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), material);
   109→  m.position.set(x, y, z);
   110→  if (castEnabled && cast) m.castShadow = true;
   111→  if (receive) m.receiveShadow = true;
   112→  return m;
   113→}

   115→type Axis = 'x' | 'y' | 'z';

   117→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
   118→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
   119→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
   120→  if (axis === 'x') m.rotation.z = Math.PI / 2;
   121→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
   122→  m.position.set(x, y, z);
   123→  if (castEnabled && cast) m.castShadow = true;
   124→  return m;
   125→}

   132→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
   133→function makeCable(points: THREE.Vector3[], radius: number, color = 0x181d24): THREE.Mesh {
   134→  const curve = new THREE.CatmullRomCurve3(points);
   135→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(color, 0.3, 0.6));
   136→  if (castEnabled) m.castShadow = true;
   137→  return m;
   138→}

   140→/** تحويل نقاط كابل عالمية إلى محلية بالنسبة لموضع تثبيت القطعة */
   141→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
   142→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));
   143→}

   145→/** مستطيل مستوٍ مواجه لمحور محدد (للملصقات والشبكات) */
   146→function plane(w: number, h: number, material: THREE.Material, axis: 'x' | 'y' | 'z', x: number, y: number, z: number): THREE.Mesh {
   147→  const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), material);
   148→  if (axis === 'x') m.rotation.y = -Math.PI / 2; // مواجه -X
   149→  else if (axis === 'y') m.rotation.x = -Math.PI / 2;
   150→  m.position.set(x, y, z);
   151→  return m;
   152→}

   154→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   155→
   156→/**
   157→ * مروحة واقعية: إطار مربع كامل بحواف وأركان دائرية + فتحات براغي + حلقة داخلية
   158→ * + 9 شفرات مائلة + محور بغطاء. محور الدوران الافتراضي = Y المحلي.
   159→ */
   160→function makeFan(radius: number): FanParts {
   161→  const depth = Math.max(0.14, radius * 0.5);
   162→  const half = radius * 1.32;
   163→
   164→  // الإطار: 4 حواف كاملة + 4 أركان أسطوانية
   165→  const frame = new THREE.Group();
   166→  const frameMat = mat(COLORS.plastic, 0.3, 0.62);
   167→  const barGeoX = new THREE.BoxGeometry(half * 2, depth, 0.1);
   168→  const barGeoZ = new THREE.BoxGeometry(0.1, depth, half * 2);
   169→  for (const s of [-1, 1]) {
   170→    const barX = new THREE.Mesh(barGeoX, frameMat);
   171→    barX.position.set(0, 0, s * half);
   172→    const barZ = new THREE.Mesh(barGeoZ, frameMat);
   173→    barZ.position.set(s * half, 0, 0);
   174→    if (castEnabled) { barX.castShadow = true; barZ.castShadow = true; }
   175→    frame.add(barX, barZ);
   176→  }
   177→  const cornerGeo = new THREE.CylinderGeometry(0.055, 0.055, depth, 10);
   178→  const holeGeo = new THREE.CylinderGeometry(0.026, 0.026, 0.02, 8);
   179→  const holeMat = mat(COLORS.black, 0.2, 0.6);
   180→  for (const sx of [-1, 1]) {
   181→    for (const sz of [-1, 1]) {
   182→      const corner = new THREE.Mesh(cornerGeo, frameMat);
   183→      corner.position.set(sx * (half - 0.05), 0, sz * (half - 0.05));
   184→      if (castEnabled) corner.castShadow = true;
   185→      frame.add(corner);
   186→      // فتحة برغي مظلمة على الوجه العلوي
   187→      const hole = new THREE.Mesh(holeGeo, holeMat);
   188→      hole.position.set(sx * half, depth / 2, sz * half);
   189→      frame.add(hole);
   190→    }
   191→  }
   192→  // حلقة داخلية محيطية
   193→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.99, 0.016, 6, 28), mat(COLORS.metalDark, 0.5, 0.5));
   194→  ring.rotation.x = Math.PI / 2;
   195→  if (castEnabled) ring.castShadow = true;
   196→  frame.add(ring);
   197→  // دعامات خلفية (ستراهنة بين الحواف خلف الشفرات)
   198→  const strutGeo = new THREE.BoxGeometry(0.05, 0.02, radius * 1.7);
   199→  for (const sz of [-1, 1]) {
   200→    const strut = new THREE.Mesh(strutGeo, frameMat);
   201→    strut.position.set(sz * radius * 0.72, -depth * 0.28, 0);
   202→    strut.rotation.y = Math.PI / 2;
   203→    frame.add(strut);
   204→  }
   205→
   206→  // الشفرات: 9 شفرات مائلة 35° داخل مجموعة blades (تُدفع إلى spinners)
   207→  const blades = new THREE.Group();
   208→  const bladeGeo = new THREE.BoxGeometry(0.016, radius * 0.8, radius * 0.32);
   209→  const bladeMat = mat(0x242c36, 0.25, 0.45);
   210→  for (let i = 0; i < 9; i++) {
   211→    const arm = new THREE.Object3D();
   212→    arm.rotation.y = (i / 9) * Math.PI * 2;
   213→    const pitch = new THREE.Object3D();
   214→    pitch.position.set(radius * 0.52, 0, 0);
   215→    pitch.rotation.x = 0.62; // ميلان الشفرة ≈ 35°
   216→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   217→    blade.rotation.z = Math.PI / 2;
   218→    if (castEnabled) blade.castShadow = true;
   219→    pitch.add(blade);
   220→    arm.add(pitch);
   221→    blades.add(arm);
   222→  }
   223→  const hub = new THREE.Mesh(
   224→    new THREE.CylinderGeometry(radius * 0.17, radius * 0.17, depth * 0.72, 14),
   225→    mat(COLORS.metalDark, 0.5, 0.4),
   226→  );
   227→  if (castEnabled) hub.castShadow = true;
   228→  blades.add(hub);
   229→  // غطاء محور بحلقة داكنة
   230→  const cap = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.1, radius * 0.1, depth * 0.76, 12), mat(0x11161c, 0.3, 0.5));
   231→  blades.add(cap);
   232→  const capRing = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.11, 0.008, 6, 16), mat(COLORS.metalMid, 0.6, 0.4));
   233→  capRing.rotation.x = Math.PI / 2;
   234→  capRing.position.y = depth * 0.38;
   235→  blades.add(capRing);
   236→
   237→  return { frame, blades };
   238→}

   258→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   259→  scene.background = new THREE.Color(COLORS.bg);
   260→  scene.fog = new THREE.Fog(COLORS.bg, 18, 42);
   261→
   262→  // أرضية دائرية داكنة تستقبل الظلال
   263→  const ground = new THREE.Mesh(
   264→    new THREE.CircleGeometry(22, 24),
   265→    new THREE.MeshStandardMaterial({ color: 0x141a22, metalness: 0, roughness: 0.92, flatShading: false }),
   266→  );
   267→  ground.rotation.x = -Math.PI / 2;
   268→  ground.position.y = 0;
   269→  ground.receiveShadow = true;
   270→  scene.add(ground);
   271→
   272→  // شبكة أرضية خفيفة
   273→  const grid = new THREE.GridHelper(30, 30, 0x2a3340, 0x1a212b);
   274→  grid.position.y = 0.01;
   275→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   276→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   277→  for (const gm of gridMats) {
   278→    gm.transparent = true;
   279→    gm.opacity = 0.4;
   280→  }
   281→  scene.add(grid);
   282→
   283→  scene.add(new THREE.AmbientLight(0xffffff, 0.45));
   284→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x20242c, 0.5));
   285→
   286→  // شمس دافئة مع ظلال (حسب الجودة)
   287→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.0);
   288→  dir.position.set(-6, 8, 6);
   289→  if (quality !== 'low') {
   290→    dir.castShadow = true;
   291→    const size = quality === 'high' ? 2048 : 1024;
   292→    dir.shadow.mapSize.set(size, size);
   293→    dir.shadow.camera.left = -6;
   294→    dir.shadow.camera.right = 6;
   295→    dir.shadow.camera.top = 6;
   296→    dir.shadow.camera.bottom = -6;
   297→    dir.shadow.camera.near = 2;
   298→    dir.shadow.camera.far = 30;
   299→    dir.shadow.camera.updateProjectionMatrix();
   300→    dir.shadow.bias = -0.0004;
   301→    dir.shadow.normalBias = 0.02;
   302→  }
   303→  scene.add(dir);
   304→
   305→  const fill = new THREE.PointLight(0xffffff, 0.5, 0, 0);
   306→  fill.position.set(3, 3, -2);
   307→  scene.add(fill);
   308→  const front = new THREE.PointLight(0xfff4e0, 1.0, 0, 0);
   309→  front.position.set(-5, 3.5, 3);
   310→  scene.add(front);
   311→}

   322→  const cx = (CASE.minX + CASE.maxX) / 2; // 1.05
   323→  const cy = (CASE.minY + CASE.maxY) / 2; // 2.25

   325→  const wX = CASE.maxX - CASE.minX;       // 2.1
   326→  const hY = CASE.maxY - CASE.minY;       // 4.5

   628→
   629→  // 12) كتلة front-panel + قابسها

   633→  flags.frontPanelPlug = fpPlug;
   634→  g.add(fpPlug);
   635→
   636→  setComponentId(g, 'motherboard');
   637→  return g;

   643→
   644→function buildCpu(): THREE.Group {
   645→  scope('cpu');
   646→  const g = new THREE.Group();
   647→  g.name = 'part-cpu';
   648→
   649→  // قاعدة خضراء (substrate)
   650→  g.add(box(0.05, 0.4, 0.4, mat(0x1e6b3a, 0.3, 0.55), 0, 0, 0, true, true));
   651→  // أطراف ذهبية محيطية على الوجه الأمامي (إشارة إلى تلامسات LGA)
   652→  const goldEdge = mat(0xc9a23f, 0.9, 0.3);
   653→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, -0.193, false));
   654→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, 0.193, false));
   655→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, -0.193, 0, false));
   656→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, 0.193, 0, false));
   657→
   658→  // غطاء IHS بأركان مشطوفة (مستطيل مشطوف عبر Cylinder ثماني مسطح؟) — نستخدم Extrude
   659→  const ihsShape = new THREE.Shape();
   660→  const s = 0.17;
   661→  const r = 0.045;
   662→  ihsShape.moveTo(-s + r, -s);
   663→  ihsShape.lineTo(s - r, -s);
   664→  ihsShape.quadraticCurveTo(s, -s, s, -s + r);
   665→  ihsShape.lineTo(s, s - r);
   666→  ihsShape.quadraticCurveTo(s, s, s - r, s);
   667→  ihsShape.lineTo(-s + r, s);
   668→  ihsShape.quadraticCurveTo(-s, s, -s, s - r);
   669→  ihsShape.lineTo(-s, -s + r);
