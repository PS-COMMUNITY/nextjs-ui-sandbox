    83→    83→export function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
    84→    84→  const rr = Math.min(r, w / 2, h / 2);
    85→    85→  ctx.beginPath();
    86→    86→  ctx.moveTo(x + rr, y);
    87→    87→  ctx.arcTo(x + w, y, x + w, y + h, rr);
    88→    88→  ctx.arcTo(x + w, y + h, x, y + h, rr);
    89→    89→  ctx.arcTo(x, y + h, x, y, rr);
    90→    90→  ctx.arcTo(x, y, x + w, y, rr);
    91→    91→  ctx.closePath();
    92→    92→}

   108→   108→export class UiFrame {
   109→   109→  rects: HotRect[] = [];
   110→   110→  keyHandler: ((ev: VosKeyEvent) => boolean) | null = null; // true = استُهلك
   111→   111→  hoverId: string | null = null;
   112→   112→  textCursor: boolean = false;
   113→   113→
   114→   114→  add(r: HotRect): void {
   115→   115→    this.rects.push(r);
   116→   116→  }
   117→   117→
   118→   118→  hit(x: number, y: number): HotRect | null {
   119→   119→    // الأعلى z أولًا وعند التساوي الأحدث (ترتيب الإضافة يطابق ترتيب الرسم فوق بعضه)
   120→   120→    let best: HotRect | null = null;
   121→   121→    for (const r of this.rects) {
   122→   122→      if (x >= r.x && x <= r.x + r.w && y >= r.y && y <= r.y + r.h) {
   123→   123→        if (!best || r.z >= best.z) best = r;
   124→   124→      }
   125→   125→    }
   126→   126→    return best;
   127→   127→  }
   128→   128→}

   202→   202→  openApp(appId: string, args?: Record<string, unknown>): VosWindow | null {
   203→   203→    const app = this.apps.get(appId);
   204→   204→    if (!app) return null;
   205→   205→    if (app.fullscreenOnly) {
   206→   206→      // تطبيقات ملء الشاشة (المثبّت) تُدار كأوضاع لا نوافذ
   207→   207→      this.mode = 'setup';
   208→   208→      this.dirty = true;
   209→   209→      return null;
   210→   210→    }
   211→   211→    const existing = this.windows.find((w) => w.appId === appId);
   212→   212→    if (existing && !args) {
   213→   213→      existing.minimized = false;
   214→   214→      existing.z = ++this.zCounter;
   215→   215→      this.dirty = true;
   216→   216→      return existing;
   217→   217→    }
   218→   218→    const offset = (this.windows.length % 5) * 26;
   219→   219→    const win: VosWindow = {
   220→   220→      id: `win-${this.nextWinId++}`,
   221→   221→      appId,
   222→   222→      titleAr: app.titleAr,
   223→   223→      x: Math.round(this.W * 0.18) + offset,
   224→   224→      y: Math.round(this.H * 0.12) + offset,
   225→   225→      w: Math.min(app.defaultW, this.W - 40),
   226→   226→      h: Math.min(app.defaultH, this.H - TASKBAR_H - 40),
   227→   227→      minimized: false,
   228→   228→      maximized: false,
   229→   229→      z: ++this.zCounter,
   230→   230→      state: { ...(args ?? {}) },
   231→   231→    };
   232→   232→    this.windows.push(win);
   233→   233→    this.dirty = true;
   234→   234→    return win;
   235→   235→  }

   313→   313→  pointer(type: 'move' | 'down' | 'up' | 'dbl' | 'right', x: number, y: number, opts: VosPointerOpts = {}): void {
   314→   314→    this.cursor.x = Math.max(0, Math.min(this.W - 1, x));
   315→   315→    this.cursor.y = Math.max(0, Math.min(this.H - 1, y));
   316→   316→    const f = this.lastFrame;
   317→   317→    if (!f) return;
   318→   318→
   319→   319→    if (type === 'move') {
   320→   320→      const hit = f.hit(x, y);
   321→   321→      this.cursorKind = hit ? hit.cursor : 'default';
   322→   322→      f.hoverId = hit?.id ?? null;
   323→   323→      this.dirty = true;
   324→   324→      return;
   325→   325→    }
   326→   326→
   327→   327→    if (type === 'down') {
   328→   328→      const hit = f.hit(x, y);
   329→   329→      this.activeId = hit?.id ?? null;
   330→   330→      this.activeStart = { x, y };
   331→   331→      this.downAt = Date.now();
   332→   332→      if (!hit && this.startOpen) this.startOpen = false;
   333→   333→      if (this.ctxMenu && (!hit || !hit.id.startsWith('ctx'))) this.ctxMenu = null;
   334→   334→      if (hit?.kind === 'title') {
   335→   335→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   336→   336→        if (win) {
   337→   337→          this.focusWindow(win.id);
   338→   338→          if (!win.maximized) this.dragWin = { id: win.id, dx: x - win.x, dy: y - win.y };
   339→   339→        }
   340→   340→      }
   341→   341→      if (hit?.kind === 'resize') {
   342→   342→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   343→   343→        if (win) this.dragResize = { id: win.id, sx: x, sy: y, w0: win.w, h0: win.h };
   344→   344→      }
   345→   345→      this.dirty = true;
   346→   346→      return;
   347→   347→    }
   348→   348→
   349→   349→    if (type === 'up') {
   350→   350→      const hit = f.hit(x, y);
   351→   351→      const now = Date.now();
   352→   352→      if (this.dragWin) {
   353→   353→        // إن لم يتحرك المؤشر فعليًا فهي نقرة عنوان (تركيز فقط)
   354→   354→        const moved = Math.abs(x - this.activeStart.x) + Math.abs(y - this.activeStart.y) > 4;
   355→   355→        if (!moved) this.dragWin = null;
   356→   356→      }
   357→   357→      this.dragWin = null;
   358→   358→      this.dragResize = null;
   359→   359→      if (this.activeId && hit && hit.id === this.activeId) {
   360→   360→        if (this.lastClickId === hit.id && now - this.lastClickAt < 380) {
   361→   361→          hit.onDbl?.();
   362→   362→          this.lastClickId = null;
   363→   363→        } else {
   364→   364→          hit.onClick?.(opts);
   365→   365→          this.lastClickId = hit.id;
   366→   366→          this.lastClickAt = now;
   367→   367→        }
   368→   368→      }
   369→   369→      this.activeId = null;
   370→   370→      this.dirty = true;
   371→   371→      return;
   372→   372→    }
   373→   373→
   374→   374→    if (type === 'dbl') {
   375→   375→      const hit = f.hit(x, y);
   376→   376→      hit?.onDbl?.();
   377→   377→      return;
   378→   378→    }
   379→   379→
   380→   380→    if (type === 'right') {
   381→   381→      const hit = f.hit(x, y);
   382→   382→      if (hit?.onRight) hit.onRight();
   383→   383→      else if (!hit) this.desktopContextMenu(x, y);
   384→   384→      this.dirty = true;
   385→   385→      return;
   386→   386→    }
   387→   387→  }
   388→   388→
   389→   389→  wheel(x: number, y: number, dy: number): void {
   390→   390→    const f = this.lastFrame;
   391→   391→    if (!f) return;
   392→   392→    const hit = f.hit(x, y);
   393→   393→    hit?.onScroll?.(dy);
   394→   394→    this.dirty = true;
   395→   395→  }

   397→   397→  /** لوحة المفاتيح الحقيقية — توجيه V6 §4.3: Enter/Backspace/Tab/Shift/Ctrl/الأسهم/Esc */
   398→   398→  key(ev: VosKeyEvent): void {
   399→   399→    // مربع نص مركّز يلتقط الكتابة أولًا
   400→   400→    if (this.focusedTextbox) {
   401→   401→      const tb = this.focusedTextbox;
   402→   402→      if (ev.key === 'Enter') {
   403→   403→        tb.onEnter?.();
   404→   404→        this.dirty = true;
   405→   405→        return;
   406→   406→      }
   407→   407→      if (ev.key === 'Backspace') {
   408→   408→        tb.set(tb.get().slice(0, -1));
   409→   409→        this.dirty = true;
   410→   410→        return;
   411→   411→      }
   412→   412→      if (ev.key.length === 1 && !ev.ctrl && !ev.alt && !ev.meta) {
   413→   413→        tb.set(tb.get() + (ev.shift ? ev.key.toUpperCase() : ev.key));
   414→   414→        this.dirty = true;
   415→   415→        return;
   416→   416→      }
   417→   417→      if (ev.key === 'Escape') {
   418→   418→        this.focusedTextbox = null;
   419→   419→        this.dirty = true;
   420→   420→        return;
   421→   421→      }
   422→   422→    }
