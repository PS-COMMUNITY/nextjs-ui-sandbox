    80→/** تحميل كل النماذج مسبقًا — ينجح دائمًا (الفشل الفردي يتخطى النموذج) */
    81→export function preloadGlbAssets(): Promise<void> {
    82→  if (!preloadPromise) {
    83→    const loader = new GLTFLoader();
    84→    loader.setMeshoptDecoder(MeshoptDecoder);
    85→    const jobs = Object.values(DEFS).map(
    86→      (def) =>
    87→        new Promise<void>((resolve) => {
    88→          loader.load(
    89→            def.url,
    90→            (gltf) => {
    91→              const scene = gltf.scene;
    92→              const mats = new Set<THREE.Material>();
    93→              scene.traverse((obj) => {
    94→                if (!(obj instanceof THREE.Mesh)) return;
    95→                const arr = Array.isArray(obj.material) ? obj.material : [obj.material];
    96→                for (const m of arr) {
    97→                  if (def.forceTransparent?.some((n) => (m.name || '').toLowerCase().includes(n))) {
    98→                    m.transparent = true;
    99→                    m.depthWrite = false;
   100→                  }
   101→                  if (def.doubleSide) m.side = THREE.DoubleSide;
   102→                  if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = Math.min((m as THREE.MeshStandardMaterial).emissiveIntensity ?? 0, 1);
   103→                  mats.add(m);
   104→                }
   105→              });
   106→              cache.set(def.url, scene);
   107→              console.info('[glb] loaded:', def.url);
   108→              resolve();
   109→            },
   110→            undefined,
   111→            (err) => {
   112→              console.warn('[glb] FAILED:', def.url, String(err));
   113→              resolve(); // فشل التحميل → يبقى المجسم الإجرائي
   114→            },
   115→          );
   116→        }),
   117→    );
   118→    preloadPromise = Promise.all(jobs).then(() => undefined);
   119→  }
   120→  return preloadPromise;
   121→}

   127→function vec3(v: Vec3): THREE.Vector3 {
   128→  return new THREE.Vector3(v[0], v[1], v[2]);
   129→}
   130→
   131→/**
   132→ * يطبع نسخة النموذج على المرساة: يعيد (الجذر + الإزاحة المطبقة).
   133→ * التطبيع: world = R·S·p ثم إزاحة بحيث تقع نقطة المرساة عند الأصل.
   134→ */
   135→function normalize(src: THREE.Object3D, def: PartDef): { root: THREE.Group; offset: THREE.Vector3 } {
   136→  const root = new THREE.Group();
   137→  root.name = 'glb-normalized';
   138→  root.quaternion.setFromRotationMatrix(new THREE.Matrix4().makeBasis(vec3(def.basis[0]), vec3(def.basis[1]), vec3(def.basis[2])));
   139→  root.scale.set(def.scale[0], def.scale[1], def.scale[2]);
   140→  root.add(src);
   141→  root.updateMatrixWorld(true);
   142→  const bbox = new THREE.Box3().setFromObject(root);
   143→  const offset = new THREE.Vector3();
   144→  if (def.anchor === 'center') offset.copy(bbox.getCenter(new THREE.Vector3())).negate();
   145→  else if (def.anchor === 'minAll') offset.copy(bbox.min).negate();
   146→  else {
   147→    // minY: القاع عند Y=0 ومركز X/Z عند الصفر
   148→    const c = bbox.getCenter(new THREE.Vector3());
   149→    offset.set(c.x, bbox.min.y, c.z).negate();
   150→  }
   151→  // الإزاحة على الجذر نفسه (تُطبَّق بعد R·S في فضاء العالم) — لا تلمس src المحلي
   152→  root.position.copy(offset);
   153→  return { root, offset };
   154→}

   156→/** تفريغ مجموعة من أبنائها مع تحرير الهندسات (الخامات مشتركة من الـcache — لا تُحرر) */
   157→function clearGroup(g: THREE.Object3D, keep: THREE.Object3D[] = []): void {
   158→  for (const child of [...g.children]) {
   159→    if (keep.includes(child)) continue;
   160→    g.remove(child);
   161→    child.traverse((o) => {
   162→      if (o instanceof THREE.Mesh) o.geometry.dispose();
   163→    });
   164→  }
   165→}

   168→function setShadows(root: THREE.Object3D, cast: boolean, receive = false): void {
   169→  root.traverse((o) => {
   170→    if (o instanceof THREE.Mesh) {
   171→      o.castShadow = cast;
   172→      if (receive) o.receiveShadow = true;
   173→    }
   174→  });

   177→/**
   178→ * غلاف pivot على مستوى المجموعة الديناميكية (فضاءها = فضاء العالم قبل التموضع):
   179→ * يسمح بتدوير الرافعة/المزلاج حول محور عالمي صحيح حتى لو كان الأب المباشر مُدارًا.
   180→ */
   181→function pivotAround(node: THREE.Object3D, group: THREE.Object3D, localY: Vec3): THREE.Object3D {
   182→  group.updateMatrixWorld(true);
   183→  const pivot = new THREE.Object3D();
   184→  pivot.name = `${node.name || 'node'}-pivot`;
   185→  node.getWorldPosition(pivot.position);
   186→  group.worldToLocal(pivot.position);
   187→  pivot.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), vec3(localY).normalize());
   188→  group.add(pivot);
   189→  pivot.attach(node);
   190→  return pivot;
   191→}

   193→/** برغي إجرائي صغير (للقطع التي لا تحمل براغٍ في النموذج الأصلي — أداة تفاعل الفك) */
   194→function makeScrewAt(axis: 'x' | 'y' | 'z', r: number, len: number): THREE.Mesh {
   195→  const geo = new THREE.CylinderGeometry(r, r, len, 10);
   196→  const m = new THREE.MeshStandardMaterial({ color: 0x9aa3ad, metalness: 0.85, roughness: 0.35 });
   197→  const mesh = new THREE.Mesh(geo, m);
   198→  if (axis === 'x') mesh.rotation.z = Math.PI / 2;
   199→  else if (axis === 'z') mesh.rotation.x = Math.PI / 2;
   200→  return mesh;
   201→}

   268→export const gpuSeat: GpuSeatReport = {

   273→/** مركز صندوق إحاطة عقدة باسمها داخل مجموعة (فضاء المجموعة الأب) */
   274→function anchorCenter(group: THREE.Object3D, name: string, out: THREE.Vector3): boolean {
   275→  const node = findNode(group, name);
   276→  if (!node) return false;
   277→  group.updateMatrixWorld(true);
   278→  const box = new THREE.Box3().setFromObject(node);
   279→  if (!isFinite(box.min.x)) {
   280→    // عقدة بلا شبكة: استخدم موضعها
   281→    node.getWorldPosition(out);
   282→    group.parent?.worldToLocal(out);
   283→    return true;
   284→  }
   285→  box.getCenter(out);
   286→  group.parent?.worldToLocal(out);
   287→  return true;
   288→}

   290→/**
   291→ * يجلس البطاقة في شق PCIe x16 الحقيقي (توجيه V6 §2.3):
   292→ *  1) مركز الشق من عقدة anchor_pcie_x16 في لوحة الأم الحقيقية (بديل إجرائي عند الفشل).
   293→ *  2) مستوى البراكيت من anchor_slot_1 في هيكل الصندوق الحقيقي.
   294→ *  3) القيود: الأطراف الذهبية (الحافة +X للبطاقة) داخل الشق، أعلى البطاقة بمستوى أعلى الشق،
   295→ *     والبراكيت على مستوى فتحات التوسعة — ثم يفحص التخلوص من المبرد والذاكرة ويسجل النتيجة.
   296→ */
   297→function alignGpuToSlot(visuals: PartVisuals): void {
   298→  const gpuGroup = visuals.dynamic.gpu;
   299→  const boardGroup = visuals.dynamic.motherboard;
   300→  gpuSeat.seated = false;
   301→  gpuSeat.clearance = null;
   302→  // تصفير كامل حتى لا يقرأ alignGpuPowerCable صناديق قديمة إن فشل التطبيق قبل الجلوس
   303→  gpuSeat.gpuBox = null;
   304→  gpuSeat.slotPoint = null;
   305→  gpuSeat.bracketZ = null;
   306→
   307→  const slot = new THREE.Vector3();
   308→  const fromGlb = anchorCenter(boardGroup, 'anchor_pcie_x16', slot);
   309→  // مستوى فتحات التوسعة الخلفية من الهيكل الحقيقي
   310→  const bracket = new THREE.Vector3();
   311→  const caseOk = anchorCenter(visuals.caseShell, 'anchor_slot_1', bracket);
   312→
   313→  let seatX: number;
   314→  let topY: number;
   315→  let bracketZ: number;
   316→  if (fromGlb) {
   317→    // الأطراف تدخل قناة الشق ~2سم (الشق بارز من وجه اللوحة)
   318→    seatX = slot.x + 0.02;
   319→    topY = slot.y + 0.012;
   320→    gpuSeat.source = 'glb-anchors';
   321→  } else {
   322→    // بديل إجرائي: فتحة x16 في مصنع القطع (Y=2.0، وجه اللوحة 1.82)
   323→    seatX = 1.79;
   324→    topY = 2.01;
   325→    slot.set(1.79, 2.0, 1.42);
   326→  }
   327→  bracketZ = caseOk ? bracket.z : 0.12;
   328→  gpuSeat.slotPoint = [slot.x, slot.y, slot.z];
   329→  gpuSeat.bracketZ = bracketZ;
   330→
   331→  gpuGroup.updateMatrixWorld(true);
   332→  const box = new THREE.Box3().setFromObject(gpuGroup);
   333→  if (isFinite(box.min.x)) {
   334→    const dx = seatX - box.max.x;           // دفع الأطراف نحو الشق
   335→    const dy = topY - box.max.y;            // إنزال البطاقة حتى مستوى الشق
   336→    const dz = bracketZ + 0.004 - box.min.z; // البراكيت على مستوى الفتحات
   337→    gpuGroup.position.x += dx;
   338→    gpuGroup.position.y += dy;
   339→    gpuGroup.position.z += dz;
   340→  } else {
   341→    // bbox غير متاح (مجسم إجرائي) — استخدم MOUNTS كما هي
   342→    gpuGroup.position.set(1.47, 2.2, 1.42);
   343→  }
   344→  gpuGroup.updateMatrixWorld(true);
   345→
   346→  const box2 = new THREE.Box3().setFromObject(gpuGroup);
   347→  gpuSeat.gpuBox = { min: [box2.min.x, box2.min.y, box2.min.z], max: [box2.max.x, box2.max.y, box2.max.z] };
   348→  gpuSeat.seated = true;
   349→

   351→  const coolerBox = new THREE.Box3().setFromObject(visuals.dynamic.cooler);
   352→  const ramBox = new THREE.Box3().setFromObject(visuals.dynamic.ram);
   353→  const gapTo = (other: THREE.Box3): number => {
   354→    if (!isFinite(other.min.x)) return 99;
   355→    const dy = Math.max(other.min.y - box2.max.y, box2.min.y - other.max.y);
   356→    if (dy >= 0) return dy;
   357→    // تداخل رأسي: قس الفجوة الأفقية
   358→    const dz = Math.max(other.min.z - box2.max.z, box2.min.z - other.max.z);
   359→    const dx2 = Math.max(other.min.x - box2.max.x, box2.min.x - other.max.x);
   360→    return Math.max(dz, dx2);
   361→  };
   362→  gpuSeat.clearance = { toCooler: gapTo(coolerBox), toRam: gapTo(ramBox) };
   363→  console.info(
   364→    `[v6] GPU seated at slot (${slot.x.toFixed(2)}, ${slot.y.toFixed(2)}, ${slot.z.toFixed(2)}), bracketZ=${bracketZ.toFixed(2)}, ` +
   365→    `clearance cooler=${(gpuSeat.clearance.toCooler * 10).toFixed(1)}cm ram=${(gpuSeat.clearance.toRam * 10).toFixed(1)}cm`,
   366→  );
   367→}

   369→// ---------- (V6-G) موصل طاقة البطاقة + نهاية كابل الشاشة — اشتقاق من البطاقة الجالسة ----------
   370→
   371→/**
   372→ * يُستدعى مرة واحدة من buildAllParts بعد applyGlbParts (يغطي مسار GLB ومسار الفشل
   373→ * الإجرائي معًا — gpuSeat.gpuBox يُملأ في alignGpuToSlot في الحالتين):
   374→ *  1) موصل gpuPower كان مثبتًا على MOUNTS.gpuPower القديمة فيطفو ~1.4 وحدة فوق البطاقة
   375→ *     بعد محاذاتها — يُعاد وضعه على مقبس 8pin فوق البطاقة: عقدة anchor_gpu_power في
   376→ *     النموذج الحقيقي، أو مركز مقبس المجسم الإجرائي 'gpu-power-socket'، ثم أعلى البطاقة
   377→ *     قرب حافتها الخلفية كحل أخير.
   378→ *  2) كابل الطاقة يُعاد بناؤه لينتهي عند الموصل نفسه (النقاط الثابتة مخزنة في userData).
   379→ *  3) نهاية كابل إشارة الشاشة تُشتق من منطقة منافذ البراكيت للبطاقة الجالسة — كانت
   380→ *     معايرة على بطاقة v3 فكانت تغوص خلف الهيكل.
   381→ * حتمية وقابلة لإعادة النداء — لا تعتمد على حالة متغيرة بين الندائين.
   382→ */
   383→export function alignGpuPowerCable(visuals: PartVisuals): void {
   384→  const box = gpuSeat.gpuBox;
   385→  if (!box) return;
   386→  const gpuGroup = visuals.dynamic.gpu;
   387→  const powerGroup = visuals.dynamic.gpuPower;
   388→  if (!gpuGroup || !powerGroup) return;
   389→
   390→  // 1) موضع موصل 8pin فوق سطح البطاقة
   391→  const target = new THREE.Vector3();
   392→  if (anchorCenter(gpuGroup, 'anchor_gpu_power', target)) {
   393→    // لا يغرق الموصل داخل جسم البطاقة إن وسمت العقدة سطحًا أدنى من القمة
   394→    target.y = Math.max(target.y, box.max[1] + 0.04);
   395→  } else {
   396→    const socket = findNode(gpuGroup, 'gpu-power-socket'); // مقبس المجسم الإجرائي
   397→    if (socket) {
   398→      gpuGroup.updateMatrixWorld(true);
   399→      const sb = new THREE.Box3().setFromObject(socket);
   400→      if (isFinite(sb.min.x)) sb.getCenter(target);
   401→      else socket.getWorldPosition(target);
   402→    } else {
   403→      // حل أخير: أعلى البطاقة قرب حافتها الخلفية العلوية
   404→      target.set((box.min[0] + box.max[0]) / 2, box.max[1] + 0.05, box.min[2] + 0.35);
   405→    }
   406→  }
   407→  powerGroup.position.copy(target);
   408→  powerGroup.updateMatrixWorld(true);
   409→
   410→  // 2) إعادة بناء كابل الطاقة: مخرج PSU ثابت → صعود يسار الهيكل → فوق البطاقة → الموصل
   411→  const cable = powerGroup.getObjectByName('gpu-power-cable');
   412→  if (cable instanceof THREE.Mesh) {
   413→    const fixed = cable.userData['worldPts'] as Array<readonly [number, number, number]> | undefined;
   414→    if (fixed && fixed.length >= 2) {
   415→      const rel = (p: THREE.Vector3): THREE.Vector3 => p.clone().sub(powerGroup.position);
   416→      const pts = fixed.map((p) => rel(new THREE.Vector3(p[0], p[1], p[2])));
   417→      // نقطة الاقتراب: من يسار البطاقة وفوق قمتها — لا يخترق الكابل جسم البطاقة
   418→      pts.push(rel(new THREE.Vector3(box.min[0] - 0.25, box.max[1] + 0.3, (fixed[fixed.length - 1][2] + target.z) / 2)));
   419→      pts.push(rel(target)); // الموصل نفسه = أصل المجموعة
   420→      cable.geometry.dispose();
   421→      cable.geometry = new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts), 32, 0.04, 8, false);
   422→    }
   423→  }
   424→
   425→  // 3) نهاية كابل إشارة الشاشة على منطقة منافذ البراكيت الخلفي (F5)
   426→  const portZ = box.min[2] + 0.02; // أطراف المنافذ عند الحد الخلفي للبطاقة
   427→  let portX: number;
   428→  let portY: number;
   429→  if (gpuSeat.source === 'glb-anchors') {
   430→    // حافة الشق = حافة البراكيت (موثقة بالتدقيق: منافذ عند x≈1.8، أعلى القمة بقليل)
   431→    portX = box.max[0] - 0.03;
   432→    portY = box.max[1] - 0.06;
   433→  } else {
   434→    // المجسم الإجرائي: مركز البراكيت المسماي = منطقة منافذه تمامًا
   435→    const bracket = findNode(gpuGroup, 'gpu-bracket');
   436→    gpuGroup.updateMatrixWorld(true);
   437→    const bb = new THREE.Box3().setFromObject(bracket ?? gpuGroup);
   438→    if (isFinite(bb.min.x)) {
   439→      portX = (bb.min.x + bb.max.x) / 2;
   440→      portY = (bb.min.y + bb.max.y) / 2;
   441→    } else {
   442→      portX = gpuGroup.position.x;
   443→      portY = box.max[1] - 0.06;
   444→    }
   445→  }
   446→  const port = new THREE.Vector3(portX, portY, portZ);
   447→  const signal = visuals.monitor.getObjectByName('monitor-signal-cable');
   448→  if (signal instanceof THREE.Mesh) {
   449→    const fixed = signal.userData['worldPts'] as Array<readonly [number, number, number]> | undefined;
   450→    if (fixed) {
   451→      const mp = PERIPH.monitorPos;
   452→      const pts = fixed.map((p) => new THREE.Vector3(p[0] - mp[0], p[1] - mp[1], p[2] - mp[2]));
   453→      pts.push(port.clone().sub(new THREE.Vector3(mp[0], mp[1], mp[2])));
   454→      signal.geometry.dispose();
   455→      signal.geometry = new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts), 32, 0.028, 8, false);
   456→    }
   457→  }
   458→
   459→  console.info(
   460→    `[v6] gpuPower connector seated at (${target.x.toFixed(2)}, ${target.y.toFixed(2)}, ${target.z.toFixed(2)}), ` +
   461→    `monitor cable end at (${port.x.toFixed(2)}, ${port.y.toFixed(2)}, ${port.z.toFixed(2)})`,
   462→  );
   463→}

   467→function swapCase(visuals: PartVisuals): void {
   468→  const def = DEFS.case;
   469→  const model = cache.get(def.url);
   470→  if (!model) return;
   471→
   472→  // 1) الهيكل: كل العقد ما عدا الزجاج
   473→  const shellClone = model.clone(true);
   474→  const glassNode = findNode(shellClone, 'case_glass');
   475→  glassNode?.removeFromParent();
   476→  const shellRoot = new THREE.Group();
   477→  shellRoot.add(shellClone);
   478→  const shellN = normalize(shellRoot, def);
   479→  clearGroup(visuals.caseShell);
   480→  setShadows(shellN.root, true, true);
   481→  visuals.caseShell.add(shellN.root);
   482→
   483→  // لمبة الطاقة الحقيقية — خامتها تُستنسخ لعزل التحكم بالإطفاء/التشغيل
   484→  const led = findNode(visuals.caseShell, 'case_power_led');
   485→  if (led instanceof THREE.Mesh) {
   486→    const mats = Array.isArray(led.material) ? [led.material[0]] : [led.material];
   487→    led.material = (mats[0] as THREE.MeshStandardMaterial).clone();
   488→    visuals.flags.powerLed = led;
   489→  }
   490→
   491→  // 2) الغطاء الزجاجي داخل المجموعة المفصلية (محور المفصلة عند (0,2.25,0.06))
   492→  // الهدف: عند زاوية الدوران 0 يكون الزجاج في مكانه الحقيقي في الإحداثيات العالمية:
   493→  //   world = hinge · translate(offset − hinge) · R·S
   494→  const glassSource = findNode(model, 'case_glass');
   495→  if (glassSource) {
   496→    const hinge = new THREE.Vector3(0, 2.25, 0.06);
   497→    const wrap = new THREE.Group();
   498→    wrap.quaternion.setFromRotationMatrix(new THREE.Matrix4().makeBasis(vec3(def.basis[0]), vec3(def.basis[1]), vec3(def.basis[2])));
   499→    wrap.scale.set(def.scale[0], def.scale[1], def.scale[2]);
   500→    wrap.position.copy(shellN.offset).sub(hinge);
   501→    wrap.add(glassSource.clone(true));
   502→    clearGroup(visuals.sidePanel);
   503→    visuals.sidePanel.add(wrap);
   504→    setShadows(wrap, false);
   505→  }
   506→}

   562→      group.add(s);
   563→    }
   564→  }

   567→  const fanPlug = makePlugBox(0.08, 0.08, 0.18);

   570→  flags.fanHeaderPlug = fanPlug;
   571→  group.add(fanPlug);
   572→

   574→  fpPlug.position.set(-0.115, -1.2, 0.9);
   575→  fpPlug.name = 'flag-front-panel-plug';
   576→  flags.frontPanelPlug = fpPlug;
   577→  group.add(fpPlug);
   578→}

   732→    }
   733→  }
