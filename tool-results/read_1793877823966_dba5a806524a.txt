    58→/** عدد موصلات PCIe 8-pin التي تحتاجها البطاقة من قيمة powerConnector. */
    59→function gpuPcieConnectorsNeeded(spec: string | undefined): number {
    60→  const v = (spec ?? '').trim().toLowerCase();
    61→  if (v === '8pin+6pin' || v === '6pin+8pin') return 2;
    62→  if (v === '8pin' || v === '6pin') return 1;
    63→  return 0; // none أو أي قيمة أخرى: لا موصل خارجي
    64→}
