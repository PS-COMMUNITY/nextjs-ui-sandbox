   146→export function isWebGLAvailable(): boolean {
   147→  try {
   148→    const canvas = document.createElement('canvas');
   149→    return Boolean(canvas.getContext('webgl2') || canvas.getContext('webgl'));
   150→  } catch {
   151→    return false;
   152→  }
   153→}

   155→/** تخلّص من مادة وكل خرائطها النسيجية (يُستخدم في dispose) */
   156→function disposeMaterial(mat: THREE.Material): void {
   157→  for (const value of Object.values(mat) as unknown[]) {
   158→    if (value instanceof THREE.Texture) value.dispose();
   159→  }
   160→  mat.dispose();
   161→}

   383→  /** (13) تبديل الجودة: كثافة البكسل + تشغيل/إيقاف الظلال */
   384→  setQuality(q: SceneQuality): void {
   385→    if (this.blocked) return;
   386→    this.quality = q;
   387→    this.renderer.setPixelRatio(this.pixelRatioFor(q));
   388→    const wantShadows = q !== 'low';
   389→    if (this.renderer.shadowMap.enabled !== wantShadows) {
   390→      this.renderer.shadowMap.enabled = wantShadows;
   391→      this.renderer.shadowMap.needsUpdate = true;
   392→      if (this.dirLight) this.dirLight.castShadow = wantShadows;
   393→    }
   394→  }

   479→  setFocusMode(on: boolean): void {
   480→    if (this.blocked || !this.vos) return;
   481→    if (on === this.focusMode) return;
   482→    const screen = this.visuals.monitor.getObjectByName('monitor-screen');

   486→    this.controls.enabled = !on;

   489→      this.savedCam = { pos: this.camera.position.clone(), target: this.controls.target.clone() };
   490→      const box = new THREE.Box3().setFromObject(screen);
   491→      const center = box.getCenter(new THREE.Vector3());
   492→      const size = box.getSize(new THREE.Vector3());
   493→      // عمود السطح = أصغر بُعد — اتجاه الكاميرا من جهة المشاهد (+Z النموذج مُدار بمجموعة الشاشة)
   494→      const normal = new THREE.Vector3(0, 0, 1).applyQuaternion(screen.getWorldQuaternion(new THREE.Quaternion())).normalize();
   495→      const fov = THREE.MathUtils.degToRad(this.camera.fov);
   496→      const aspect = Math.max(0.4, this.camera.aspect);
   497→      const distW = size.x / 2 / (Math.tan(fov / 2) * aspect);
   498→      const distH = size.y / 2 / Math.tan(fov / 2);
   499→      const dist = Math.max(distW, distH) * 1.12;
   500→      this.startCamTween(center.clone().addScaledVector(normal, dist), center.clone(), 0.55);
   501→      this.renderer.domElement.style.cursor = 'none';
   502→      this.callbacks.onFocusMode?.(true);
   503→    } else {
   504→      if (this.savedCam) {
   505→        this.startCamTween(this.savedCam.pos, this.savedCam.target, 0.5);
   506→        this.savedCam = null;
   507→      }
   508→      this.renderer.domElement.style.cursor = '';
   509→      this.callbacks.onFocusMode?.(false);
   510→    }
   511→    this.drawMonitor(true);
   512→  }

   527→  /** تحويل نقطة شاشة المتصفح إلى إحداثيات سطح العرض (UV → بكسل) — null إن لم يصب السطح */
   528→  private screenHit(clientX: number, clientY: number): { x: number; y: number } | null {
   529→    const screen = this.visuals.monitor.getObjectByName('monitor-screen');
   530→    if (!(screen instanceof THREE.Mesh)) return null;
   531→    const rect = this.renderer.domElement.getBoundingClientRect();
   532→    const w = rect.width > 0 ? rect.width : 1;
   533→    const h = rect.height > 0 ? rect.height : 1;
   534→    this._screenNdc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
   535→    this.raycaster.setFromCamera(this._screenNdc, this.camera);
   536→    const hits = this.raycaster.intersectObject(screen, false);
   537→    if (hits.length === 0 || hits[0].uv === undefined) return null;
   538→    const uv = hits[0].uv!;
   539→    // PlaneGeometry: (0,0) أسفل-يسار — الكانفس: (0,0) أعلى-يسار ⇒ قلب Y فقط
   540→    return { x: uv.x * this.vos!.W, y: (1 - uv.y) * this.vos!.H };
   541→  }

   689→  /** (6) حلقة الرسم: مراوح، اقتراب القطع، تنعيم الغطاء/الكاميرا، النظام الافتراضي، ثم الرسم */
   690→  private readonly tick = (): void => {
   691→    if (this.disposed || this.failed) return;
   692→    this.rafId = requestAnimationFrame(this.tick);
   693→    const dt = Math.min(this.clock.getDelta(), 0.05); // حماية من قفزات التبويبات الخاملة
   694→    const k = 1 - Math.exp(-LERP_RATE * dt);
   695→
   696→    // (V6 §3) منظومة المراوح: سرعات ودوران كل ريش على محوره الصحيح + الحرارة المُحاكاة
   697→    if (this.fans && this.lastState) {
   698→      const s = this.lastState;
   699→      const powered = s.flags.powerOn && s.flags.acConnected && !s.peripherals.psuDead;
   700→      const conditions = {
   701→        powered,
   702→        cpuCoolerOk: isInstalled(s.parts.cooler),
   703→        gpuOk: isInstalled(s.parts.gpu) && isInstalled(s.parts.gpuPower),
   704→        caseFansOk: isInstalled(s.parts.fans) && !s.flags.fanHeadersDisconnected,
   705→        psuOk: powered,
   706→        cpuLoad: 0.42,
   707→        gpuLoad: 0.48,
   708→      };
   709→      this.fans.update(dt, conditions);
   710→
   711→      // الحرارة المُحاكاة (بيانات تعليمية معلنة): ترتفع بلا مبرد/بلا مراوح وتقترب من الهدف
   712→      this.thermalAccum += dt;
   713→      if (this.thermalAccum >= 1) {
   714→        this.thermalAccum = 0;
   715→        const cpuTarget = !powered ? 26 : isInstalled(s.parts.cooler) ? (s.flags.fanHeadersDisconnected ? 86 : 46) : 97;
   716→        const gpuTarget = !powered ? 26 : conditions.gpuOk ? 52 : isInstalled(s.parts.gpu) ? 84 : 26;
   717→        const cpu = THREE.MathUtils.lerp(s.thermal.cpuTempC, cpuTarget, 0.18);
   718→        const gpu = THREE.MathUtils.lerp(s.thermal.gpuTempC, gpuTarget, 0.18);
   719→        this.callbacks.onThermal?.(cpu, gpu);
   720→        // إطفاء حراري تعليمي عند تجاوز الحد (فصّل التبريد = الجهاز ينطفئ)
   721→        if (cpu >= 95.5 && powered) this.callbacks.onThermalShutdown?.(cpu);
   722→      }
   723→    }
   724→
   725→    // اقتراب القطع نحو أهدافها (الموضع lerp والدوران slerp) — عدا الهيكل الثابت
   726→    for (const id of COMPONENT_IDS) {
   727→      if (id === 'case') continue;
   728→      const part = this.parts[id];
   729→      part.group.position.lerp(part.targetPos, k);
   730→      part.group.quaternion.slerp(part.targetQuat, k);
   731→    }
   732→
   733→    // تطفو القطعة الملتقطة (awaiting) فوق الصينية بلطف — إشارة بصرية لجاهزية التركيب
   734→    const elapsed = this.clock.getElapsedTime();
   735→    for (const id of COMPONENT_IDS) {
   736→      if (id === 'case') continue;
   737→      if (this.lastState && this.lastState.parts[id] === 'awaiting' && this.dragId !== id) {
   738→        const part = this.parts[id];
   739→        part.group.position.y = part.trayPos.y + 0.16 + Math.sin(elapsed * 3) * 0.04;
   740→      }
   741→    }
   742→
   743→    // تنعيم دوران الغطاء الجانبي نحو الهدف
   744→    const panel = this.visuals.sidePanel;
   745→    panel.rotation.y += (this.panelTarget - panel.rotation.y) * k;
   746→
   747→    // استكمال تنعيم الكاميرا (resetCamera / focusComponent)
   748→    const tween = this.camTween;
   749→    if (tween) {
   750→      tween.t = Math.min(tween.t + dt / tween.dur, 1);
   751→      const eased = PCScene.easeInOut(tween.t);
   752→      this.camera.position.lerpVectors(tween.fromPos, tween.toPos, eased);
   753→      this.controls.target.lerpVectors(tween.fromTarget, tween.toTarget, eased);
   754→      if (tween.t >= 1) this.camTween = null;
   755→    }
   756→
   757→    this.controls.update();
   758→
   759→    // إعادة رسم نسيج الشاشة الافتراضية: فوري في وضع التحكم بالشاشة، مُقيّد في المختبر
   760→    if (this.painter) {
   761→      const interval = this.focusMode ? 90 : MONITOR_REDRAW_MS;
   762→      if (nowMs() - this.lastMonitorDrawAt >= interval) {
   763→        this.drawMonitor(false);
   764→      }
   765→    }
   766→
   767→    this.renderer.render(this.scene, this.camera);
   768→  };
   769→
   770→  /** تنعيم easeInOut: t<0.5 ? 2t² : 1 − (−2t+2)²/2 */
   771→  private static easeInOut(t: number): number {
   772→    return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
   773→  }

   775→  /** إعادة حساب أهداف القطع من آخر حالة متزامنة + علم التفجير الحالي */
   776→  private recomputeTargets(): void {
   777→    const parts = this.lastState?.parts;
   778→    for (const id of COMPONENT_IDS) {
   779→      if (id === 'case') continue;
   780→      const part = this.parts[id];
   781→      const status: string | undefined = parts ? parts[id] : undefined;
   782→      if (status === 'removed' || status === 'damaged') {
   783→        part.targetPos.copy(part.trayPos);
   784→        part.targetRot.copy(part.trayRot);
   785→        if (status === 'damaged') {
   786→          // القطعة التالفة تستلقي مائلة على الصينية
   787→          part.targetRot.z += 0.35;
   788→        }
   789→      } else if (status === 'awaiting') {
   790→        // ملتقطة من الصينية — ترتفع قليلاً (وتطفو في حلقة الرسم)
   791→        part.targetPos.copy(part.trayPos);
   792→        part.targetPos.y += 0.16;
   793→        part.targetRot.copy(part.trayRot);
   794→      } else if (status === 'separated') {
   795→        // فصل جزئي: ما زالت في موضعها مع إزاحة صغيرة نحو اتجاه الفك
   796→        part.targetPos.copy(part.basePos);
   797→        part.targetPos.addScaledVector(part.explodeVec, SEPARATE_FACTOR);
   798→        part.targetRot.set(0, 0, 0);
   799→      } else {
   800→        part.targetPos.copy(part.basePos);
   801→        if (this.exploded) part.targetPos.add(part.explodeVec);
   802→        part.targetRot.set(0, 0, 0);
   803→      }
   804→      part.targetQuat.setFromEuler(part.targetRot);
   805→    }
   806→  }

   808→  /** نسخ الأهداف فورًا إلى المجسمات (للتنعيم المعطَّل animate=false) */
   809→  private snapToTargets(): void {
   810→    for (const id of COMPONENT_IDS) {
   811→      if (id === 'case') continue;
   812→      const part = this.parts[id];
   813→      part.group.position.copy(part.targetPos);
   814→      part.group.quaternion.copy(part.targetQuat);
   815→    }
   816→    this.visuals.sidePanel.rotation.y = this.panelTarget;
   817→  }

   819→  /**
   820→   * (5) تطبيق الأعلام فورًا على المجسمات الفرعية.
   821→   * افتراضات المحاور (لأن بناء المصنع موازٍ ولم يُثبَّت بعد):
   822→   *  - مزلاج PCIe ورافعة CPU: دوران حول Y المحلي (اللوح عمودي يواجه -X).
   823→   *  - مزالج RAM: دوران حول Z بإشارة متناوبة بين الطرفين.
   824→   *  - البراغي: ترتفع على +Y (مسامير اللوحة تُسحب على -X) مع دوران طفيف حول Z.
   825→   *  - القوابس: إزاحة -X 0.12 عند الفصل.
   826→   * كل الإزاحات نسبية إلى القيم الأصلية المحفوظة عند البناء.
   827→   */
   828→  private applyFlags(flags: Record<MachineFlag, boolean>, animate: boolean): void {
   829→    // الغطاء الجانبي — عبر هدف تنعيمي تتبعه الحلقة
   830→    this.panelTarget = flags.panelOpen ? PANEL_OPEN_ANGLE : 0;
   831→    if (!animate) this.visuals.sidePanel.rotation.y = this.panelTarget;
   832→
   833→    // مزلاج PCIe
   834→    if (this.latchRec) {
   835→      this.latchRec.obj.rotation.y = this.latchRec.baseRot + (flags.pcieLatchOpen ? LATCH_OPEN_ANGLE : 0);
   836→    }
   837→
   838→    // مزالج RAM
   839→    for (const clip of this.clipRecs) {
   840→      clip.obj.rotation.z = clip.baseRotZ + (flags.ramClipsOpen ? clip.sign * CLIP_OPEN_ANGLE : 0);
   841→    }
   842→
   843→    // رافعة مقبس CPU
   844→    if (this.leverRec) {
   845→      this.leverRec.obj.rotation.y = this.leverRec.baseRot + (flags.cpuLeverOpen ? LEVER_OPEN_ANGLE : 0);
   846→    }
   847→
   848→    // البراغي (gpu/psu/fan/m2/mobo)
   849→    for (const [flag, recs] of this.screwMap) {
   850→      const out = flags[flag];
   851→      for (const rec of recs) {
   852→        rec.obj.position.copy(rec.basePos);
   853→        if (out) {
   854→          rec.obj.position[rec.axis] += rec.delta;
   855→          rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN;
   856→        } else {
   857→          rec.obj.rotation.z = rec.baseRotZ;
   858→        }
   859→      }
   860→    }
   861→
   862→    // قوابس الموصلات (fan headers / front panel)
   863→    for (const plug of this.plugRecs) {
   864→      plug.obj.position.x = plug.baseX - (flags[plug.flag] ? PLUG_PULL : 0);
   865→    }
   866→
   867→    // لمبة الطاقة — مادة خاصة بها (cached) — تنطفئ مع مزود ميت
   868→    const led = this.visuals.flags.powerLed;
   869→    if (led) {
   870→      const mat = led.material;
   871→      if (!Array.isArray(mat) && 'emissive' in mat) {
   872→        const std = mat as THREE.MeshStandardMaterial;
   873→        const alive = flags.powerOn && !(this.lastState?.peripherals.psuDead ?? false);
   874→        std.emissive.set(alive ? COLORS.green : 0x000000);
   875→        std.emissiveIntensity = LED_INTENSITY;
   876→      }
   877→    }
   878→  }

   900→  /** تطبيق شفافية على مواد مجموعة (الزجاج شفاف أصلًا فيبقى كما هو) */
   901→  private applyShellOpacity(root: THREE.Object3D, opacity: number): void {
   902→    root.traverse((obj) => {
   903→      if (!(obj instanceof THREE.Mesh)) return;
   904→      const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   905→      for (const mat of mats) {
   906→        if (mat.transparent) continue; // الزجاج/الشفاف الأصلي لا يُمسّ

   908→          this.shellOriginals.set(mat, {
   909→            opacity: mat.opacity,
   910→            transparent: mat.transparent,
   911→            depthWrite: mat.depthWrite,
   912→          });
   913→        }
   914→        mat.transparent = true;
   915→        mat.opacity = opacity;
   916→        mat.depthWrite = false;
   917→      }
   918→    });
   919→  }

   941→   */
   942→  private loadRealAssets(visuals: PartVisuals): void {
   943→    const loader = new GLTFLoader();
   944→    const root = visuals.root;
   945→
   946→    // 1) أساس PCB الحقيقي — يستبدل وجه اللوحة الأمامي فقط، ويبقى الباقي إجرائيًا تفاعليًا
   947→    loader.load(
   948→      '/models/circuit-board/circuit_board_1k.gltf',
   949→      (gltf) => {
   950→        if (this.disposed || this.failed) return;
   951→        const pcb = visuals.dynamic.motherboard.getObjectByName('mobo-pcb');
   952→        if (!pcb) return;
   953→        const scan = gltf.scene;
   954→        scan.traverse((obj) => {
   955→          if (!(obj instanceof THREE.Mesh)) return;
   956→          obj.castShadow = false;
   957→          obj.receiveShadow = true;

   959→          const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   960→          for (const m of mats) {
   961→            if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = 0;
   962→          }
   963→        });
   964→        // bbox الأصل (م): 0.404×0.078×0.398 — رأسيًا مُدار بحيث مكوّناته نحو -X (وجه المشاهد)

   966→        const size = bbox.getSize(new THREE.Vector3());
   967→        const targetH = BOARD.yMax - BOARD.yMin;      // 3.0
   968→        const targetD = BOARD.zMax - BOARD.zMin;      // 2.44
   969→        scan.rotation.z = Math.PI / 2;                // +Y المحلي → -X عالمي
   970→        scan.scale.set(
   971→          BOARD.thickness / Math.max(size.y, 1e-6),   // سماكة اللوحة
   972→          targetH / Math.max(size.x, 1e-6),           // ارتفاع اللوحة
   973→          targetD / Math.max(size.z, 1e-6),           // عمق اللوحة
   974→        );
   975→        // إعادة قياس bbox بعد الدوران ثم تمركز اللوحة على مستوى المجموعة
   976→        scan.updateMatrixWorld(true);
   977→        const bbox2 = new THREE.Box3().setFromObject(scan);
   978→        const center2 = bbox2.getCenter(new THREE.Vector3());
   979→        scan.position.sub(center2);
   980→        scan.position.x += BOARD.thickness * 0.18;    // بارز قليلًا نحو الوجه
   981→        scan.name = 'mobo-real-pcb';
   982→        pcb.add(scan);
   983→        // إخفاء الوجه الإجرائي الأخضر (يُبقى الجسم مرجعًا للسماكة والظهر)
   984→        if (pcb instanceof THREE.Mesh && !Array.isArray(pcb.material)) {
   985→          pcb.material = pcb.material.clone();
   986→          (pcb.material as THREE.MeshStandardMaterial).color.set(0x0f2a1f);
   987→        }
   988→      },
   989→      undefined,
   990→      () => {
   991→        // فشل تحميل الأصل → يبقى PCB الإجرائي كما هو (مشهد كامل بلا انقطاع)
   992→        console.warn('[PCScene] تعذر تحميل مسح اللوحة الحقيقي — استُخدم PCB الإجرائي');
   993→      },
   994→    );
   995→
   996→    // 2) مفك تدريبي على سجادة القطع (جزء بصري ثانوي — بلا تفاعل)
   997→    loader.load(
   998→      '/models/screwdriver/screwdriver_1k.gltf',
   999→      (gltf) => {
  1000→        if (this.disposed || this.failed) return;
  1001→        const tool = gltf.scene;
  1002→        const bbox = new THREE.Box3().setFromObject(tool);
  1003→        const size = bbox.getSize(new THREE.Vector3());
  1004→        const s = 1.6 / Math.max(size.y, 1e-6); // طول مستهدف ≈ 16سم
  1005→        tool.scale.setScalar(s);
  1006→        tool.updateMatrixWorld(true);
  1007→        const b2 = new THREE.Box3().setFromObject(tool);
  1008→        tool.position.set(-3.1, TRAY.tableTopY - b2.min.y, 3.55);
  1009→        tool.rotation.y = 0.5;
  1010→        tool.name = 'desk-screwdriver';
  1011→        root.add(tool);
  1012→      },
  1013→      undefined,
  1014→      () => console.warn('[PCScene] تعذر تحميل المفك التدريبي'),
  1015→    );
  1016→
  1017→    // 3..5) أدوات وزينة الورشة CC0 — كلها فشلها صامت (تفصيل ثانوي)
  1018→    const placeTool = (
  1019→      url: string,
  1020→      name: string,
  1021→      targetLen: number,          // البعد الأطول المستهدف بوحدات المشهد
  1022→      pos: [number, number, number],
  1023→      rotY: number,
  1024→      opts?: { light?: boolean },
  1025→    ): void => {
  1026→      loader.load(
  1027→        url,
  1028→        (gltf) => {
  1029→          if (this.disposed || this.failed) return;
  1030→          const obj = gltf.scene;
  1031→          // توحيد الخامات: منع أي emissive قد يأتي من الأصل
  1032→          obj.traverse((o) => {
  1033→            if (!(o instanceof THREE.Mesh)) return;
  1034→            o.castShadow = this.quality !== 'low';
  1035→            o.receiveShadow = true;
  1036→            const mats = Array.isArray(o.material) ? o.material : [o.material];
  1037→            for (const m of mats) {
  1038→              if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = 0;
  1039→            }
  1040→          });
  1041→          const bbox = new THREE.Box3().setFromObject(obj);
  1042→          const size = bbox.getSize(new THREE.Vector3());
  1043→          const s = targetLen / Math.max(size.x, size.y, size.z, 1e-6);
  1044→          obj.scale.setScalar(s);
  1045→          obj.updateMatrixWorld(true);
  1046→          const b2 = new THREE.Box3().setFromObject(obj);
  1047→          obj.position.set(pos[0], pos[1] - b2.min.y, pos[2]);
  1048→          obj.rotation.y = rotY;
  1049→          obj.name = name;
  1050→          root.add(obj);
  1051→          if (opts?.light) {
  1052→            const head = new THREE.Vector3((b2.min.x + b2.max.x) / 2, b2.max.y - 0.15, (b2.min.z + b2.max.z) / 2);
  1053→            obj.updateMatrixWorld(true);
  1054→            const worldHead = head.applyMatrix4(obj.matrixWorld);
  1055→            const warm = new THREE.PointLight(0xffe7c4, 0.7, 9, 1.4);
  1056→            warm.position.copy(worldHead);
  1057→            warm.name = `${name}-light`;
  1058→            root.add(warm);
  1059→          }
  1060→        },
  1061→        undefined,
  1062→        () => console.warn(`[PCScene] تعذر تحميل ${name}`),
  1063→      );
  1064→    };
  1065→
  1066→    // طقم مفكات على حافة السجادة اليسرى (طوله الطبيعي ≈ 2.2 وحدة)

  1068→    // كماشات قرب صينية البراغي
  1069→    placeTool('/models/pliers/pliers_1k.gltf', 'tool-pliers', 1.5, [-4.0, TRAY.tableTopY, 2.75], -0.5);
  1070→    // مصباح مكتب على الطرف الأيمن للمكتب — ذراعه يمتد فوق المكتب المفتوح

  1072→    placeTool('/models/desk_lamp_arm_01/desk_lamp_arm_01_1k.gltf', 'desk-lamp', 4.6, [7.4, 0, 3.1], -2.2, { light: true });
  1073→    // صندوق عدة معدني تحت المكتب (على أرضية المختبر)

  1077→  /**
  1078→   * رسم نسيج الشاشة الافتراضية:
  1079→   *  - حالات التمهيد (off/noSignal/booting/post/warning/success) عبر الرسّام القديم.
  1080→   *  - 'os' و 'bios' يديرهما النظام الافتراضي التفاعلي (VOS): سطح مكتب/معالج تثبيت/
  1081→   *    بيئة استرداد/إعدادات UEFI — كلها تفاعلية بالماوس والكيبورد الحقيقيين (V6-B).
  1082→   */
  1083→  private drawMonitor(force: boolean): void {
  1084→    if (!this.painter || this.blocked) return;
  1085→    if (!force && nowMs() - this.lastMonitorDrawAt < 60) return;
  1086→    this.lastMonitorDrawAt = nowMs();
  1087→    const sim = this.lastState;
  1088→    const elapsed = this.bootStartedAt !== null ? Math.max(0, Date.now() - this.bootStartedAt) : 0;
  1089→    if (!sim) {
  1090→      this.painter.draw(
  1091→        { screen: 'noSignal', bootProgress: 0, postLinesAr: [], bootDevicesAr: [], canType: false },
  1092→        this.notepad,
  1093→        Date.now(),
  1094→      );
  1095→      return;
  1096→    }
  1097→    const view = computeMonitorView(sim, elapsed);
  1098→    const now = Date.now();
  1099→
  1100→    if (this.vos && (view.screen === 'os' || view.screen === 'bios')) {
  1101→      this.drawVosScreen(sim, view.screen, now);
  1102→      return;
  1103→    }
  1104→
  1105→    this.painter.draw(view, this.notepad, now);
  1106→  }

  1108→  /** رسم بيئة النظام الافتراضي/البرنامج الثابت على كانفس الشاشة + رفع النسيج */
  1109→  private drawVosScreen(sim: SimulationState, screen: 'os' | 'bios', now: number): void {
  1110→    if (!this.vos || !this.painter) return;
  1111→    const vos = this.vos;
  1112→
  1113→    if (screen === 'bios') {
  1114→      vos.mode = 'firmware';

  1116→      vos.fwData = {

  1118→        ramAr: `DDR5 32GB${sim.firmware.xmp ? ' @5200 (XMP)' : ' @2133'} — ${sim.flags.ramClipsOpen ? 'مزالج مفتوحة!' : 'مثبتة'}`,
  1119→        boardAr: 'PC LAB Z690 (reference)',
  1120→        tempsAr: `CPU ${sim.thermal.cpuTempC.toFixed(0)}°C / GPU ${sim.thermal.gpuTempC.toFixed(0)}°C (simulated)`,
  1121→        fansAr: this.fans?.report().map((f) => `${f.labelAr}: ${f.rpm} RPM`).join(' | ') ?? '—',
  1122→        bootOrder: [...sim.firmware.bootOrder],
  1123→        secureBoot: sim.firmware.secureBoot,
  1124→        tpm: sim.firmware.tpm,
  1125→        xmp: sim.firmware.xmp,
  1126→        devicesAr: devs,
  1127→        onBootOrder: (order) => this.callbacks.onFirmwareChange?.({ bootOrder: order }),
  1128→        onToggle: (key) => this.callbacks.onFirmwareChange?.({ [key]: !sim.firmware[key] }),
  1129→        onExit: () => this.callbacks.onFirmwareChange?.({ setupRequested: false }),
  1130→        onBootDevice: (id) => this.callbacks.onBootDevice?.(id),
  1131→      };
  1132→      vos.render(now);
  1133→      this.painter.texture.needsUpdate = true;

  1135→    }
  1136→
  1137→    // screen === 'os': حسم وضع VOS من مصدر الحقيقة
  1138→    if (sim.usbMedia && sim.firmware.bootOrder[0] === 'usb') {
  1139→      // الإقلاع من وسيط التثبيت ⇒ معالج التثبيت (شاشة كاملة)
  1140→      if (vos.mode !== 'setup') {
  1141→        if (getSetupState().step === 0 || getSetupState().done) resetSetup();
  1142→        vos.mode = 'setup';
  1143→      }
  1144→      vos.drawSetupWith((ctx, os2, t) => drawInstaller(os2, ctx, t), now);
  1145→      this.painter.texture.needsUpdate = true;
  1146→      return;
  1147→    }
  1148→
  1149→    if (sim.os.bcdCorrupt && sim.os.installed) {
  1150→      // فشل إقلاع النظام (F16) ⇒ بيئة الاسترداد
  1151→      vos.mode = 'recovery';
  1152→      vos.recoveryData = {
  1153→        attempts: sim.os.repairAttempts,
  1154→        osNameAr: sim.os.installed,
  1155→        onStartupRepair: () => this.callbacks.onStartupRepair?.(),
  1156→        onCommandPrompt: () => {
  1157→          const win = vos.openApp('cmd');
  1158→          if (win) win.state['recovery'] = true;
  1159→          vos.mode = 'desktop';
  1160→        },
  1161→        onRestart: () => this.callbacks.onRecoveryRestart?.(),
  1162→      };
  1163→      vos.render(now);
  1164→      this.painter.texture.needsUpdate = true;
  1165→      return;
  1166→    }
  1167→
  1168→    // سطح المكتب الافتراضي التفاعلي
  1169→    vos.mode = 'desktop';
  1170→    vos.render(now);
  1171→    this.painter.texture.needsUpdate = true;
  1172→  }

  1174→  /** ربط أحداث النظام الافتراضي بمتجر الحالة (مصدر الحقيقة الواحد) */

  1176→    if (!this.vos) return;
  1177→    this.vos.storeReader = () => {
  1178→      const s = this.lastState;
  1179→      return {
  1180→        os: {
  1181→          installed: s?.os.installed ?? null,
  1182→          edition: s?.os.edition ?? null,
  1183→          driversMissing: s?.os.driversMissing ?? [],

  1250→  private pickAt(clientX: number, clientY: number): PickedInfo {
  1251→    const rect = this.renderer.domElement.getBoundingClientRect();
  1252→    const w = rect.width > 0 ? rect.width : 1;
  1253→    const h = rect.height > 0 ? rect.height : 1;
  1254→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1255→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1256→    const hits = this.raycaster.intersectObjects(this.visuals.root.children, true);
  1257→    for (const hit of hits) {
  1258→      if (this.isHidden(hit.object)) continue;        // تجاهل ما خفيه العزل
  1259→      if (this.isSeeThrough(hit.object)) continue;    // الزجاج/الشفاف لا يحجب الاختيار
  1260→      let cur: THREE.Object3D | null = hit.object;
  1261→      while (cur) {
  1262→        // الطرفيات (زر الشاشة / زر طاقة الشاشة) — لا كيبورد ولا ماوس (توجيه v3 أ.1)
  1263→        const per: unknown = cur.userData['peripheral'];
  1264→        if (typeof per === 'string') {
  1265→          return { type: 'periph', kind: per as PeripheralKind };
  1266→        }
  1267→        const cid: unknown = cur.userData['componentId'];
  1268→        if (typeof cid === 'string' && (COMPONENT_IDS as readonly string[]).includes(cid)) {
  1269→          return { type: 'part', id: cid as ComponentId };
  1270→        }
  1271→        cur = cur.parent;
  1272→      }
  1273→      // إصابة أولى بلا معرف (عدة/زينة) — تمرير النقر عبرها إلى ما خلفها (توجيه v4)
  1274→      continue;
  1275→    }
  1276→    return null;
  1277→  }

  1414→  /** (7) نقرة فقط إذا كانت المسافة بين down/up أقل من 6px — والسحب الفعلي يُعالج قبلها */
  1415→  private readonly onPointerUp = (event: PointerEvent): void => {
  1416→    if (this.blocked) return;
  1417→
  1418→    // (V6-B) توجيه رفع الزر إلى النظام الافتراضي في وضع الشاشة
  1419→    if (this.focusMode && this.vos) {
  1420→      const hit = this.screenHit(event.clientX, event.clientY);
  1421→      if (hit) this.vos.pointer('up', hit.x, hit.y, { button: event.button });
  1422→      this.drawMonitor(true);
  1423→      this.pointerDownActive = false;
  1424→      return;
  1425→    }
  1426→
  1427→    if (this.dragId) {
  1428→      this.finishDrag(event.clientX, event.clientY);
  1429→      this.pointerDownActive = false;
  1430→      return;
  1431→    }
  1432→    this.pendingDrag = null;
  1433→    if (!this.pointerDownActive) return;
  1434→    this.pointerDownActive = false;
  1435→    const dx = event.clientX - this.downX;
  1436→    const dy = event.clientY - this.downY;
  1437→    if (dx * dx + dy * dy >= CLICK_MAX_DIST_SQ) return; // سحب لا نقرة
  1438→
  1439→    // (V6-B §4.3) النقر على سطح الشاشة المجسمة نفسه يدخل وضع التحكم ويسلّم النقرة
  1440→    const onScreen = this.screenHit(event.clientX, event.clientY);
  1441→    if (onScreen && this.vos) {
  1442→      this.setFocusMode(true);
  1443→      this.vos.pointer('down', onScreen.x, onScreen.y, { button: event.button });
  1444→      this.vos.pointer('up', onScreen.x, onScreen.y, { button: event.button });
  1445→      this.drawMonitor(true);
  1446→      return;
  1447→    }
  1448→
  1449→    const picked = this.pickAt(event.clientX, event.clientY);
  1450→    if (picked && picked.type === 'periph') {
  1451→      this.callbacks.onPeripheral?.(picked.kind, picked.detail);
  1452→      return;
  1453→    }
  1454→    this.callbacks.onSelect(picked ? picked.id : null);
  1455→  };

  1457→  /** (7) تحويم/سحب — بفحص مُقيَّد زمنيًا للتحويم (100ms) */
