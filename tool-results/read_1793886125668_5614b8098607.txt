   177→   177→let persistTimer: ReturnType<typeof setTimeout> | null = null;
   178→   178→
   179→   179→function schedulePersist(get: () => LabStore) {
   180→   180→  if (persistTimer) clearTimeout(persistTimer);
   181→   181→  persistTimer = setTimeout(() => {
   182→   182→    const s = get();
   183→   183→    if (!s.hydrated) return;
   184→   184→    const shape: PersistShape = {
   185→   185→      version: APP_VERSION,
   186→   186→      simulation: s.sim,
   187→   187→      tech: s.tech,
   188→   188→      logs: s.logs,
   189→   189→      quizScores: s.quizScores,
   190→   190→      missionsDone: s.missionsDone,
   191→   191→      lessonsRead: s.lessonsRead,
   192→   192→      settings: { quality: s.quality },
   193→   193→      savedAt: Date.now(),
   194→   194→    };
   195→   195→    const res = saveAll(shape);
   196→   196→    if (res.ok) {
   197→   197→      useLabStore.setState({ saveStatus: { available: true, lastSavedAt: Date.now(), lastErrorAr: null } });
   198→   198→    } else {
   199→   199→      useLabStore.setState({ saveStatus: { available: true, lastSavedAt: null, lastErrorAr: res.messageAr } });
   200→   200→    }
   201→   201→  }, 400);
   202→   202→}

   357→   357→      toast.error(check.messageAr);
   358→   358→      return;
   359→   359→    }
   360→   360→    const sim = applyInstall(s.sim, id);
   361→   361→    set({ sim });
   362→   362→    s.logEvent('install', `تركيب ${nameOf(id)}`, true, 'أُعيدت القطعة إلى موضع تثبيتها — أغلق المزالج والبراغي إن وجدت');

   364→   364→    schedulePersist(get);

   372→   372→      return;
   373→   373→    }
   374→   374→    const check = canRemove(s.sim, id);
   375→   375→    if (!check.ok) {
   376→   376→      s.logEvent('remove', `${name}: فك مرفوض`, false, check.messageAr);
   377→   377→      toast.error(check.messageAr);
   378→   378→      return;
   379→   379→    }
   380→   380→    // الكابلات: فصل مباشر إلى الصينية — القطع الكبيرة: فصل جزئي ثم استخراج
   381→   381→    const sim = isCable(id) ? applyRemove(s.sim, id) : applyDetach(s.sim, id);
   382→   382→    set({ sim });
   383→   383→    if (isCable(id)) {
   384→   384→      s.logEvent('remove', `فصل ${name}`, true, 'نُقل إلى صينية القطع');
   385→   385→      toast.success(`${name}: فُصل إلى صينية القطع`);
   386→   386→    } else {
   387→   387→      s.logEvent('remove', `فصل جزئي: ${name}`, true, 'أكمل الاستخراج إلى الصينية');
   388→   388→      toast.success(`${name}: فُصل جزئيًا — أكمل الاستخراج`);
   389→   389→    }
   390→   390→    schedulePersist(get);
