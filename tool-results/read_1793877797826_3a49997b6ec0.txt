    83→export function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
    84→  const rr = Math.min(r, w / 2, h / 2);
    85→  ctx.beginPath();
    86→  ctx.moveTo(x + rr, y);
    87→  ctx.arcTo(x + w, y, x + w, y + h, rr);
    88→  ctx.arcTo(x + w, y + h, x, y + h, rr);
    89→  ctx.arcTo(x, y + h, x, y, rr);
    90→  ctx.arcTo(x, y, x + w, y, rr);
    91→  ctx.closePath();
    92→}

    96→interface HotRect {
    97→  id: string;
    98→  x: number; y: number; w: number; h: number;
    99→  kind: 'button' | 'title' | 'resize' | 'icon' | 'row' | 'custom' | 'scroll';
   100→  cursor: 'default' | 'pointer' | 'text';
   101→  onClick?: (opts: VosPointerOpts) => void;

   103→  onRight?: () => void;
   104→  onScroll?: (dy: number) => void;
   105→  z: number;
   106→}

   108→export class UiFrame {
   109→  rects: HotRect[] = [];
   110→  keyHandler: ((ev: VosKeyEvent) => boolean) | null = null; // true = استُهلك
   111→  hoverId: string | null = null;
   112→  textCursor: boolean = false;
   113→
   114→  add(r: HotRect): void {
   115→    this.rects.push(r);
   116→  }
   117→
   118→  hit(x: number, y: number): HotRect | null {
   119→    // الأعلى z أولًا ثم الأحدث (ترتيب الإضافة يعكس التراكب داخل النافذة)
   120→    let best: HotRect | null = null;
   121→    for (const r of this.rects) {
   122→      if (x >= r.x && x <= r.x + r.w && y >= r.y && y <= r.y + r.h) {
   123→        if (!best || r.z > best.z) best = r;
   124→      }
   125→    }
   126→    return best;
   127→  }
   128→}

   151→  readonly W: number;
   152→  readonly H: number;
   153→  private ctx: CanvasRenderingContext2D;
   154→
   155→  /** وضع البيئة: سطح مكتب / برنامج ثابت / معالج تثبيت / استرداد */

   160→  private nextWinId = 1;
   161→
   162→  apps = new Map<string, VosAppDef>();
   163→  desktopIcons: string[] = [];
   164→
   165→  startOpen = false;
   166→  cursor = { x: 640, y: 300, visible: false };

   180→  private dragResize: { id: string; sx: number; sy: number; w0: number; h0: number } | null = null;

   186→  constructor(ctx: CanvasRenderingContext2D, W: number, H: number) {

   192→  registerApp(app: VosAppDef): void {

   196→  // ============ النوافذ ============
   197→
   198→  openApp(appId: string, args?: Record<string, unknown>): VosWindow | null {
   199→    const app = this.apps.get(appId);
   200→    if (!app) return null;
   201→    if (app.fullscreenOnly) {
   202→      // تطبيقات ملء الشاشة (المثبّت) تُدار كأوضاع لا نوافذ
   203→      this.mode = 'setup';
   204→      this.dirty = true;
   205→      return null;
   206→    }
   207→    const existing = this.windows.find((w) => w.appId === appId);
   208→    if (existing && !args) {
   209→      existing.minimized = false;
   210→      existing.z = ++this.zCounter;
   211→      this.dirty = true;
   212→      return existing;
   213→    }
   214→    const offset = (this.windows.length % 5) * 26;
   215→    const win: VosWindow = {
   216→      id: `win-${this.nextWinId++}`,
   217→      appId,
   218→      titleAr: app.titleAr,
   219→      x: Math.round(this.W * 0.18) + offset,
   220→      y: Math.round(this.H * 0.12) + offset,
   221→      w: Math.min(app.defaultW, this.W - 40),
   222→      h: Math.min(app.defaultH, this.H - TASKBAR_H - 40),
   223→      minimized: false,
   224→      maximized: false,
   225→      z: ++this.zCounter,
   226→      state: { ...(args ?? {}) },
   227→    };
   228→    this.windows.push(win);
   229→    this.dirty = true;
   230→    return win;
   231→  }
   232→
   233→  closeWindow(id: string): void {
   234→    const idx = this.windows.findIndex((w) => w.id === id);
   235→    if (idx >= 0) {
   236→      this.windows.splice(idx, 1);
   237→      this.dirty = true;
   238→    }
   239→  }
   240→
   241→  focusWindow(id: string): void {
   242→    const win = this.windows.find((w) => w.id === id);
   243→    if (win) {
   244→      win.z = ++this.zCounter;
   245→      win.minimized = false;
   246→      this.dirty = true;
   247→    }
   248→  }
   249→
   250→  notify(titleAr: string, bodyAr: string): void {
   251→    this.toasts.push({ titleAr, bodyAr, at: Date.now() });
   252→    if (this.toasts.length > 3) this.toasts.shift();
   253→    this.dirty = true;
   254→  }

   256→  /** واجهة التطبيقات: أدوات الرسم الفوري */
   257→  get ui(): UiFrame {

   259→  }

   261→  /** أصل محتوى النافذة الحالية (تُضبط قبل رسم كل تطبيق) — يُستخدم لتسجيل مستطيلات التفاعل */
   262→  contentOrigin = { x: 0, y: 0 };
   263→
   264→  /** تسجيل مستطيل تفاعل بإحداثيات محلية للتطبيق (يُحول لأحداثيات الشاشة داخليًا) */
   265→  addRect(
   266→    kind: 'button' | 'title' | 'resize' | 'icon' | 'row' | 'custom' | 'scroll',
   267→    x: number, y: number, w: number, h: number,
   268→    handlers: { onClick?: (o?: VosPointerOpts) => void; onDbl?: () => void; onRight?: () => void; onScroll?: (dy: number) => void } ,
   269→    cursor: 'default' | 'pointer' | 'text' = 'pointer',
   270→    z = 20,
   271→  ): string {
   272→    const id = `r${this.rectCounter++}`;
   273→    this.frame.add({
   274→      id,
   275→      x: x + this.contentOrigin.x,
   276→      y: y + this.contentOrigin.y,
   277→      w, h, kind, cursor, z,
   278→      onClick: handlers.onClick ? () => handlers.onClick?.() : undefined,
   279→      onDbl: handlers.onDbl,
   280→      onRight: handlers.onRight,
   281→      onScroll: handlers.onScroll,
   282→    });
   283→    return id;
   284→  }

   288→  /** هل المؤشر فوق مستطيل محلي؟ (للتحويم داخل التطبيقات) */
   289→  hoverAt(x: number, y: number, w: number, h: number): boolean {
   290→    const cx = this.cursor.x - this.contentOrigin.x;
   291→    const cy = this.cursor.y - this.contentOrigin.y;
   292→    return cx >= x && cx <= x + w && cy >= y && cy <= y + h;
   293→  }

   307→  // ============ الإدخال (API الرسمي الوحيد) ============
   308→
   309→  pointer(type: 'move' | 'down' | 'up' | 'dbl' | 'right', x: number, y: number, opts: VosPointerOpts = {}): void {
   310→    this.cursor.x = Math.max(0, Math.min(this.W - 1, x));
   311→    this.cursor.y = Math.max(0, Math.min(this.H - 1, y));
   312→    const f = this.lastFrame;
   313→    if (!f) return;
   314→
   315→    if (type === 'move') {
   316→      const hit = f.hit(x, y);
   317→      this.cursorKind = hit ? hit.cursor : 'default';
   318→      f.hoverId = hit?.id ?? null;
   319→      this.dirty = true;
   320→      return;
   321→    }
   322→
   323→    if (type === 'down') {
   324→      const hit = f.hit(x, y);
   325→      this.activeId = hit?.id ?? null;
   326→      this.activeStart = { x, y };
   327→      this.downAt = Date.now();
   328→      if (!hit && this.startOpen) this.startOpen = false;
   329→      if (this.ctxMenu && (!hit || !hit.id.startsWith('ctx'))) this.ctxMenu = null;
   330→      if (hit?.kind === 'title') {
   331→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   332→        if (win) {
   333→          this.focusWindow(win.id);
   334→          if (!win.maximized) this.dragWin = { id: win.id, dx: x - win.x, dy: y - win.y };
   335→        }
   336→      }
   337→      if (hit?.kind === 'resize') {
   338→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   339→        if (win) this.dragResize = { id: win.id, sx: x, sy: y, w0: win.w, h0: win.h };
   340→      }
   341→      this.dirty = true;
   342→      return;
   343→    }
   344→
   345→    if (type === 'up') {
   346→      const hit = f.hit(x, y);
   347→      const now = Date.now();
   348→      if (this.dragWin) {
   349→        // إن لم يتحرك المؤشر فعليًا فهي نقرة عنوان (تركيز فقط)
   350→        const moved = Math.abs(x - this.activeStart.x) + Math.abs(y - this.activeStart.y) > 4;
   351→        if (!moved) this.dragWin = null;
   352→      }
   353→      this.dragWin = null;
   354→      this.dragResize = null;
   355→      if (this.activeId && hit && hit.id === this.activeId) {
   356→        if (this.lastClickId === hit.id && now - this.lastClickAt < 380) {
   357→          hit.onDbl?.();
   358→          this.lastClickId = null;
   359→        } else {
   360→          hit.onClick?.(opts);
   361→          this.lastClickId = hit.id;
   362→          this.lastClickAt = now;
   363→        }
   364→      }
   365→      this.activeId = null;
   366→      this.dirty = true;
   367→      return;
   368→    }
   369→
   370→    if (type === 'dbl') {
   371→      const hit = f.hit(x, y);
   372→      hit?.onDbl?.();
   373→      return;
   374→    }
   375→
   376→    if (type === 'right') {
   377→      const hit = f.hit(x, y);
   378→      if (hit?.onRight) hit.onRight();
   379→      else if (!hit) this.desktopContextMenu(x, y);
   380→      this.dirty = true;
   381→      return;
   382→    }
   383→  }
   384→
   385→  wheel(x: number, y: number, dy: number): void {
   386→    const f = this.lastFrame;
   387→    if (!f) return;
   388→    const hit = f.hit(x, y);
   389→    hit?.onScroll?.(dy);
   390→    this.dirty = true;
   391→  }

   393→  /** لوحة المفاتيح الحقيقية — توجيه V6 §4.3: Enter/Backspace/Tab/Shift/Ctrl/الأسهم/Esc */
   394→  key(ev: VosKeyEvent): void {
   395→    // مربع نص مركّز يلتقط الكتابة أولًا
   396→    if (this.focusedTextbox) {
   397→      const tb = this.focusedTextbox;
   398→      if (ev.key === 'Enter') {
   399→        tb.onEnter?.();
   400→        this.dirty = true;
   401→        return;
   402→      }
   403→      if (ev.key === 'Backspace') {
   404→        tb.set(tb.get().slice(0, -1));
   405→        this.dirty = true;
   406→        return;
   407→      }
   408→      if (ev.key.length === 1 && !ev.ctrl && !ev.alt && !ev.meta) {
   409→        tb.set(tb.get() + (ev.shift ? ev.key.toUpperCase() : ev.key));
   410→        this.dirty = true;
   411→        return;
   412→      }
   413→      if (ev.key === 'Escape') {
   414→        this.focusedTextbox = null;
   415→        this.dirty = true;
   416→        return;
   417→      }
   418→    }
   419→
   420→    // عام: Esc يغلق قائمة ابدأ
   421→    if (ev.key === 'Escape' && this.startOpen) {
   422→      this.startOpen = false;
   423→      this.dirty = true;
   424→      return;
   425→    }
   426→
   427→    // تمرير إلى معالج مفتاح الإطار (التطبيق النشط)
   428→    if (this.frame.keyHandler && this.frame.keyHandler(ev)) {
   429→      this.dirty = true;
   430→      return;
   431→    }
   432→    this.dirty = true;
   433→  }

   437→  /** يرسم إطارًا كاملًا ويعيد true إذا اكتمل رسم فعلي */
   438→  render(now: number): boolean {
   439→    this.frame = new UiFrame();
   440→    const ctx = this.ctx;
   441→    this.blinkAt = now;
   442→
   443→    ctx.save();
   444→    ctx.clearRect(0, 0, this.W, this.H);
   445→
   446→    if (this.mode === 'desktop') {
   447→      this.drawDesktop(ctx, now);
   448→    } else if (this.mode === 'firmware') {
   449→      this.drawFirmware(ctx, now);
   450→    } else if (this.mode === 'recovery') {
   451→      this.drawRecovery(ctx, now);
   452→    }
   453→    // mode 'setup' يرسمه تطبيق المثبّت عبر drawSetupWith
   454→
   455→    this.drawOverlays(ctx, now);
   456→    this.drawCursor(ctx);
   457→
   458→    ctx.restore();
   459→
   460→    this.lastFrame = this.frame;
   461→    this.dirty = false;
   462→    return true;
   463→  }

   465→  /** يفرض رسم وضع setup عبر تطبيق المثبّت — يستدعى من PCScene قبل render */
   466→  drawSetupWith(fn: (ctx: CanvasRenderingContext2D, os: VirtualOS, now: number) => void, now: number): boolean {
   467→    this.frame = new UiFrame();
   468→    const ctx = this.ctx;
   469→    ctx.save();
   470→    ctx.clearRect(0, 0, this.W, this.H);
   471→    fn(ctx, this, now);
   472→    this.drawOverlays(ctx, now);
   473→    this.drawCursor(ctx);
   474→    ctx.restore();
   475→    this.lastFrame = this.frame;
   476→    this.dirty = false;
   477→    return true;
   478→  }

   482→  private drawDesktop(ctx: CanvasRenderingContext2D, now: number): void {
   483→    // خلفية
   484→    const g = ctx.createLinearGradient(0, 0, this.W, this.H);
   485→    g.addColorStop(0, '#0f2e33');
   486→    g.addColorStop(0.5, '#0b2228');
   487→    g.addColorStop(1, '#07161b');
   488→    ctx.fillStyle = g;
   489→    ctx.fillRect(0, 0, this.W, this.H);
   490→    // شعار هندسي خفيف
   491→    ctx.strokeStyle = 'rgba(57,199,184,0.10)';
   492→    ctx.lineWidth = 2;
   493→    for (let i = 0; i < 5; i++) {
   494→      ctx.beginPath();
   495→      ctx.arc(this.W - 120, 90, 30 + i * 26, 0, Math.PI * 2);
   496→      ctx.stroke();
   497→    }
   498→    ctx.fillStyle = 'rgba(215,223,233,0.55)';
   499→    ctx.font = 'bold 20px monospace';
   500→    ctx.textAlign = 'left';
   501→    ctx.fillText('PC Lab OS — بيئة تدريب افتراضية', 18, this.H - TASKBAR_H - 14);
   502→    ctx.fillStyle = 'rgba(139,151,165,0.75)';
   503→    ctx.font = '13px sans-serif';
   504→    ctx.fillText('محاكاة تعليمية — ليست نظام Windows حقيقيًا', 18, this.H - TASKBAR_H + 6);
   505→
   506→    // أيقونات سطح المكتب
   507→    let col = 0;

   582→  private drawWindow(ctx: CanvasRenderingContext2D, win: VosWindow, now: number): void {

   632→        win.prev = { x: win.x, y: win.y, w: win.w, h: win.h };
   633→        win.maximized = true;
   634→      } else if (win.prev) {
   635→        win.maximized = false;

   637→      }

   661→    ctx.save();

  1107→  wrapText(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, maxW: number, lineH: number): number {
  1108→    const words = text.split(/\s+/);
  1109→    let line = '';
  1110→    let yy = y;
  1111→    for (const w of words) {
  1112→      const test = line ? `${line} ${w}` : w;
  1113→      if (ctx.measureText(test).width > maxW && line) {
  1114→        ctx.fillText(line, x, yy);
  1115→        line = w;
  1116→        yy += lineH;
  1117→      } else {
  1118→        line = test;
  1119→      }
  1120→    }
  1121→    if (line) ctx.fillText(line, x, yy);
  1122→    return yy;
  1123→  }
