   146→export function isWebGLAvailable(): boolean {
   147→  try {
   148→    const canvas = document.createElement('canvas');
   149→    return Boolean(canvas.getContext('webgl2') || canvas.getContext('webgl'));
   150→  } catch {
   151→    return false;
   152→  }

   155→/** تخلّص من مادة وكل خرائطها النسيجية (يُستخدم في dispose) */
   156→function disposeMaterial(mat: THREE.Material): void {
   157→  for (const value of Object.values(mat) as unknown[]) {
   158→    if (value instanceof THREE.Texture) value.dispose();
   159→  }
   160→  mat.dispose();
   161→}

   384→  /** (13) تبديل الجودة: كثافة البكسل + تشغيل/إيقاف الظلال */
   385→  setQuality(q: SceneQuality): void {
   386→    if (this.blocked) return;
   387→    this.quality = q;
   388→    this.renderer.setPixelRatio(this.pixelRatioFor(q));
   389→    const wantShadows = q !== 'low';
   390→    if (this.renderer.shadowMap.enabled !== wantShadows) {
   391→      this.renderer.shadowMap.enabled = wantShadows;
   392→      this.renderer.shadowMap.needsUpdate = true;
   393→      if (this.dirLight) this.dirLight.castShadow = wantShadows;
   394→    }
   395→  }

   479→  setFocusMode(on: boolean): void {
   480→    if (this.blocked || !this.vos) return;
   481→    if (on === this.focusMode) return;
   482→    const screen = this.visuals.monitor.getObjectByName('monitor-screen');
   483→    if (!(screen instanceof THREE.Mesh)) return;
   484→    this.focusMode = on;
   485→    this.vos.cursor.visible = on;
   486→    this.controls.enabled = !on;
   487→    if (on) {
   488→      // تأطير الشاشة: مسافة تحسب من أبعاد السطح الفعلية وزاوية الرؤية
   489→      this.savedCam = { pos: this.camera.position.clone(), target: this.controls.target.clone() };
   490→      const box = new THREE.Box3().setFromObject(screen);
   491→      const center = box.getCenter(new THREE.Vector3());
   492→      const size = box.getSize(new THREE.Vector3());
   493→      // عمود السطح = أصغر بُعد — اتجاه الكاميرا من جهة المشاهد (+Z النموذج مُدار بمجموعة الشاشة)
   494→      const normal = new THREE.Vector3(0, 0, 1).applyQuaternion(screen.getWorldQuaternion(new THREE.Quaternion())).normalize();
   495→      const fov = THREE.MathUtils.degToRad(this.camera.fov);
   496→      const aspect = Math.max(0.4, this.camera.aspect);
   497→      const distW = size.x / 2 / (Math.tan(fov / 2) * aspect);
   498→      const distH = size.y / 2 / Math.tan(fov / 2);
   499→      const dist = Math.max(distW, distH) * 1.12;
   500→      this.startCamTween(center.clone().addScaledVector(normal, dist), center.clone(), 0.55);
   501→      this.renderer.domElement.style.cursor = 'none';
   502→      this.callbacks.onFocusMode?.(true);
   503→    } else {
   504→      if (this.savedCam) {
   505→        this.startCamTween(this.savedCam.pos, this.savedCam.target, 0.5);
   506→        this.savedCam = null;
   507→      }
   508→      this.renderer.domElement.style.cursor = '';
   509→      this.callbacks.onFocusMode?.(false);
   510→    }
   511→    this.drawMonitor(true);
   512→  }

   527→  /** تحويل نقطة شاشة المتصفح إلى إحداثيات سطح العرض (UV → بكسل) — null إن لم يصب السطح */
   528→  private screenHit(clientX: number, clientY: number): { x: number; y: number } | null {
   529→    const screen = this.visuals.monitor.getObjectByName('monitor-screen');
   530→    if (!(screen instanceof THREE.Mesh)) return null;
   531→    const rect = this.renderer.domElement.getBoundingClientRect();
   532→    const w = rect.width > 0 ? rect.width : 1;
   533→    const h = rect.height > 0 ? rect.height : 1;
   534→    this._screenNdc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
   535→    this.raycaster.setFromCamera(this._screenNdc, this.camera);
   536→    const hits = this.raycaster.intersectObject(screen, false);
   537→    if (hits.length === 0 || hits[0].uv === undefined) return null;
   538→    if (!this.vos) return null; // (إصلاح تدقيق V6) حارس: كان غير آمن null هنا
   539→    const uv = hits[0].uv!;
   540→    // PlaneGeometry: (0,0) أسفل-يسار — الكانفس: (0,0) أعلى-يسار ⇒ قلب Y فقط
   541→    return { x: uv.x * this.vos.W, y: (1 - uv.y) * this.vos.H };
   542→  }

   666→      if (this.scene.background instanceof THREE.Texture) this.scene.background.dispose();

   673→    this.shellOriginals.clear();
   674→    this.damagedOriginals.clear();
   675→    this.painter?.dispose();
   676→    this.painter = null;
   677→  }

   692→  /** (6) حلقة الرسم: مراوح، اقتراب القطع، تنعيم الغطاء/الكاميرا، النظام الافتراضي، ثم الرسم */
   693→  private readonly tick = (): void => {
   694→    if (this.disposed || this.failed) return;
   695→    this.rafId = requestAnimationFrame(this.tick);
   696→    const dt = Math.min(this.clock.getDelta(), 0.05); // حماية من قفزات التبويبات الخاملة
   697→    const k = 1 - Math.exp(-LERP_RATE * dt);
   698→
   699→    // (V6 §3) منظومة المراوح: سرعات ودوران كل ريش على محوره الصحيح + الحرارة المُحاكاة
   700→    if (this.fans && this.lastState) {
   701→      const s = this.lastState;
   702→      const powered = s.flags.powerOn && s.flags.acConnected && !s.peripherals.psuDead;
   703→      // (إصلاح تدقيق V6) الأحمال مشتقة من الحالة والحرارة المُحاكاة — كانت ثوابت مكتوبة يدويًا
   704→      // حلقة تغذية راجعة تعليمية: غياب التبريد يرفع الحرارة فيرفع الأحمال فيتسارع ما تبقى من مراوح
   705→      const heatCpu = Math.min(1, Math.max(0, (s.thermal.cpuTempC - 50) / 45));
   706→      const heatGpu = Math.min(1, Math.max(0, (s.thermal.gpuTempC - 50) / 40));
   707→      if (!this.fanConditions) this.fanConditions = { powered: false, cpuCoolerOk: false, gpuOk: false, caseFansOk: false, psuOk: false, cpuLoad: 0, gpuLoad: 0 };
   708→      const conditions = this.fanConditions;
   709→      conditions.powered = powered;
   710→      conditions.cpuCoolerOk = isInstalled(s.parts.cooler) && !s.flags.fanHeadersDisconnected;
   711→      conditions.gpuOk = isInstalled(s.parts.gpu) && isInstalled(s.parts.gpuPower);
   712→      conditions.caseFansOk = isInstalled(s.parts.fans) && !s.flags.fanHeadersDisconnected;
   713→      conditions.psuOk = powered;
   714→      conditions.cpuLoad = powered ? Math.min(1, 0.35 + 0.65 * heatCpu) : 0;
   715→      conditions.gpuLoad = powered && conditions.gpuOk ? Math.min(1, 0.3 + 0.7 * heatGpu) : 0;
   716→      this.fans.update(dt, conditions);
   717→
   718→      // الحرارة المُحاكاة (بيانات تعليمية معلنة): ترتفع بلا مبرد/بلا مراوح وتقترب من الهدف
   719→      this.thermalAccum += dt;
   720→      if (this.thermalAccum >= 1) {
   721→        this.thermalAccum = 0;
   722→        const cpuTarget = !powered ? 26 : isInstalled(s.parts.cooler) ? (s.flags.fanHeadersDisconnected ? 86 : 46) : 97;
   723→        const gpuTarget = !powered ? 26 : conditions.gpuOk ? 52 : isInstalled(s.parts.gpu) ? 84 : 26;
   724→        const cpu = THREE.MathUtils.lerp(s.thermal.cpuTempC, cpuTarget, 0.18);
   725→        const gpu = THREE.MathUtils.lerp(s.thermal.gpuTempC, gpuTarget, 0.18);
   726→        this.callbacks.onThermal?.(cpu, gpu);
   727→        // إطفاء حراري تعليمي عند تجاوز الحد (فصّل التبريد = الجهاز ينطفئ)
   728→        if (cpu >= 95.5 && powered) this.callbacks.onThermalShutdown?.(cpu);
   729→      }
   730→    }
   731→
   732→    // اقتراب القطع نحو أهدافها (الموضع lerp والدوران slerp) — عدا الهيكل الثابت
   733→    for (const id of COMPONENT_IDS) {
   734→      if (id === 'case') continue;
   735→      const part = this.parts[id];
   736→      part.group.position.lerp(part.targetPos, k);
   737→      part.group.quaternion.slerp(part.targetQuat, k);
   738→    }
   739→
   740→    // تطفو القطعة الملتقطة (awaiting) فوق الصينية بلطف — إشارة بصرية لجاهزية التركيب
   741→    const elapsed = this.clock.getElapsedTime();
   742→    for (const id of COMPONENT_IDS) {
   743→      if (id === 'case') continue;
   744→      if (this.lastState && this.lastState.parts[id] === 'awaiting' && this.dragId !== id) {
   745→        const part = this.parts[id];
   746→        part.group.position.y = part.trayPos.y + 0.16 + Math.sin(elapsed * 3) * 0.04;
   747→      }
   748→    }
   749→
   750→    // تنعيم دوران الغطاء الجانبي نحو الهدف
   751→    const panel = this.visuals.sidePanel;
   752→    panel.rotation.y += (this.panelTarget - panel.rotation.y) * k;
   753→
   754→    // استكمال تنعيم الكاميرا (resetCamera / focusComponent)
   755→    const tween = this.camTween;
   756→    if (tween) {
   757→      tween.t = Math.min(tween.t + dt / tween.dur, 1);
   758→      const eased = PCScene.easeInOut(tween.t);
   759→      this.camera.position.lerpVectors(tween.fromPos, tween.toPos, eased);
   760→      this.controls.target.lerpVectors(tween.fromTarget, tween.toTarget, eased);
   761→      if (tween.t >= 1) this.camTween = null;
   762→    }
   763→
   764→    this.controls.update();
   765→
   766→    // إعادة رسم نسيج الشاشة الافتراضية: فوري في وضع التحكم بالشاشة، مُقيّد في المختبر
   767→    if (this.painter) {
   768→      const interval = this.focusMode ? 90 : MONITOR_REDRAW_MS;
   769→      if (nowMs() - this.lastMonitorDrawAt >= interval) {
   770→        this.drawMonitor(false);
   771→      }
   772→    }
   773→
   774→    this.renderer.render(this.scene, this.camera);
   775→  };
   776→
   777→  /** تنعيم easeInOut: t<0.5 ? 2t² : 1 − (−2t+2)²/2 */
   778→  private static easeInOut(t: number): number {
   779→    return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
   780→  }

   782→  /** إعادة حساب أهداف القطع من آخر حالة متزامنة + علم التفجير الحالي */
   783→  private recomputeTargets(): void {
   784→    const parts = this.lastState?.parts;
   785→    for (const id of COMPONENT_IDS) {
   786→      if (id === 'case') continue;
   787→      const part = this.parts[id];
   788→      const status: string | undefined = parts ? parts[id] : undefined;
   789→      if (status === 'removed' || status === 'damaged') {
   790→        part.targetPos.copy(part.trayPos);
   791→        part.targetRot.copy(part.trayRot);
   792→        if (status === 'damaged') {
   793→          // القطعة التالفة تستلقي مائلة على الصينية
   794→          part.targetRot.z += 0.35;
   795→        }
   796→      } else if (status === 'awaiting') {
   797→        // ملتقطة من الصينية — ترتفع قليلاً (وتطفو في حلقة الرسم)
   798→        part.targetPos.copy(part.trayPos);
   799→        part.targetPos.y += 0.16;
   800→        part.targetRot.copy(part.trayRot);
   801→      } else if (status === 'separated') {
   802→        // فصل جزئي: ما زالت في موضعها مع إزاحة صغيرة نحو اتجاه الفك
   803→        part.targetPos.copy(part.basePos);
   804→        part.targetPos.addScaledVector(part.explodeVec, SEPARATE_FACTOR);
   805→        part.targetRot.set(0, 0, 0);
   806→      } else {
   807→        part.targetPos.copy(part.basePos);
   808→        if (this.exploded) part.targetPos.add(part.explodeVec);
   809→        part.targetRot.set(0, 0, 0);
   810→      }
   811→      part.targetQuat.setFromEuler(part.targetRot);
   812→    }
   813→  }
   814→
   815→  /** نسخ الأهداف فورًا إلى المجسمات (للتنعيم المعطَّل animate=false) */
   816→  private snapToTargets(): void {
   817→    for (const id of COMPONENT_IDS) {
   818→      if (id === 'case') continue;
   819→      const part = this.parts[id];
   820→      part.group.position.copy(part.targetPos);
   821→      part.group.quaternion.copy(part.targetQuat);
   822→    }
   823→    this.visuals.sidePanel.rotation.y = this.panelTarget;
   824→  }
   825→
   826→  /**
   827→   * (5) تطبيق الأعلام فورًا على المجسمات الفرعية.
   828→   * افتراضات المحاور (لأن بناء المصنع موازٍ ولم يُثبَّت بعد):
   829→   *  - مزلاج PCIe ورافعة CPU: دوران حول Y المحلي (اللوح عمودي يواجه -X).
   830→   *  - مزالج RAM: دوران حول Z بإشارة متناوبة بين الطرفين.
   831→   *  - البراغي: ترتفع على +Y (مسامير اللوحة تُسحب على -X) مع دوران طفيف حول Z.
   832→   *  - القوابس: إزاحة -X 0.12 عند الفصل.
   833→   * كل الإزاحات نسبية إلى القيم الأصلية المحفوظة عند البناء.
   834→   */
   835→  private applyFlags(flags: Record<MachineFlag, boolean>, animate: boolean): void {
   836→    // الغطاء الجانبي — عبر هدف تنعيمي تتبعه الحلقة
   837→    this.panelTarget = flags.panelOpen ? PANEL_OPEN_ANGLE : 0;
   838→    if (!animate) this.visuals.sidePanel.rotation.y = this.panelTarget;
   839→
   840→    // مزلاج PCIe
   841→    if (this.latchRec) {
   842→      this.latchRec.obj.rotation.y = this.latchRec.baseRot + (flags.pcieLatchOpen ? LATCH_OPEN_ANGLE : 0);
   843→    }
   844→
   845→    // مزالج RAM
   846→    for (const clip of this.clipRecs) {
   847→      clip.obj.rotation.z = clip.baseRotZ + (flags.ramClipsOpen ? clip.sign * CLIP_OPEN_ANGLE : 0);
   848→    }
   849→
   850→    // رافعة مقبس CPU
   851→    if (this.leverRec) {
   852→      this.leverRec.obj.rotation.y = this.leverRec.baseRot + (flags.cpuLeverOpen ? LEVER_OPEN_ANGLE : 0);
   853→    }
   854→
   855→    // البراغي (gpu/psu/fan/m2/mobo)
   856→    for (const [flag, recs] of this.screwMap) {
   857→      const out = flags[flag];
   858→      for (const rec of recs) {
   859→        rec.obj.position.copy(rec.basePos);
   860→        if (out) {
   861→          rec.obj.position[rec.axis] += rec.delta;
   862→          rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN;
   863→        } else {
   864→          rec.obj.rotation.z = rec.baseRotZ;
   865→        }
   866→      }
   867→    }
   868→
   869→    // قوابس الموصلات (fan headers / front panel)
   870→    for (const plug of this.plugRecs) {
   871→      plug.obj.position.x = plug.baseX - (flags[plug.flag] ? PLUG_PULL : 0);
   872→    }
   873→
   874→    // لمبة الطاقة — مادة خاصة بها (cached) — تنطفئ مع مزود ميت
   875→    const led = this.visuals.flags.powerLed;
   876→    if (led) {
   877→      const mat = led.material;
   878→      if (!Array.isArray(mat) && 'emissive' in mat) {
   879→        const std = mat as THREE.MeshStandardMaterial;
   880→        const alive = flags.powerOn && !(this.lastState?.peripherals.psuDead ?? false);
   881→        std.emissive.set(alive ? COLORS.green : 0x000000);
   882→        std.emissiveIntensity = LED_INTENSITY;
   883→      }
   884→    }
   885→  }

   891→  /** إعادة حساب مواد الهيكل من الحالة المجمّعة (xray يتقدم على العزل) */
   892→  private refreshShellMaterials(): void {

   895→      mat.transparent = saved.transparent;

   900→      this.applyShellOpacity(this.visuals.caseShell, 0.12);

   902→    } else if (this.isolatedId !== null && this.isolatedId !== 'case') {
   903→      this.applyShellOpacity(this.visuals.caseShell, 0.08);
   904→    }
   905→  }

   907→  /** تطبيق شفافية على مواد مجموعة (الزجاج شفاف أصلًا فيبقى كما هو) */
   908→  private applyShellOpacity(root: THREE.Object3D, opacity: number): void {
   909→    root.traverse((obj) => {
   910→      if (!(obj instanceof THREE.Mesh)) return;
   911→      const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   912→      for (const mat of mats) {
   913→        if (mat.transparent) continue; // الزجاج/الشفاف الأصلي لا يُمسّ
   914→        if (!this.shellOriginals.has(mat)) {
   915→          this.shellOriginals.set(mat, {
   916→            opacity: mat.opacity,
   917→            transparent: mat.transparent,
   918→            depthWrite: mat.depthWrite,
   919→          });
   920→        }
   921→        mat.transparent = true;
   922→        mat.opacity = opacity;
   923→        mat.depthWrite = false;
   924→      }
   925→    });
   926→  }

   941→  /**

   949→  private loadRealAssets(visuals: PartVisuals): void {
   950→    const loader = new GLTFLoader();
   951→    const root = visuals.root;
   952→
   953→    // 1) أساس PCB الحقيقي — يستبدل وجه اللوحة الأمامي فقط، ويبقى الباقي إجرائيًا تفاعليًا
   954→    loader.load(
   955→      '/models/circuit-board/circuit_board_1k.gltf',
   956→      (gltf) => {
   957→        if (this.disposed || this.failed) return;
   958→        const pcb = visuals.dynamic.motherboard.getObjectByName('mobo-pcb');
   959→        if (!pcb) return;
   960→        const scan = gltf.scene;
   961→        scan.traverse((obj) => {
   962→          if (!(obj instanceof THREE.Mesh)) return;

   966→          const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   967→          for (const m of mats) {
   968→            if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = 0;

   971→        // bbox الأصل (م): 0.404×0.078×0.398 — رأسيًا مُدار بحيث مكوّناته نحو -X (وجه المشاهد)

   974→        const targetH = BOARD.yMax - BOARD.yMin;      // 3.0
   975→        const targetD = BOARD.zMax - BOARD.zMin;      // 2.44

   977→        scan.scale.set(
   978→          BOARD.thickness / Math.max(size.y, 1e-6),   // سماكة اللوحة
   979→          targetH / Math.max(size.x, 1e-6),           // ارتفاع اللوحة
   980→          targetD / Math.max(size.z, 1e-6),           // عمق اللوحة

   983→        scan.updateMatrixWorld(true);
   984→        const bbox2 = new THREE.Box3().setFromObject(scan);
   985→        const center2 = bbox2.getCenter(new THREE.Vector3());
   986→        scan.position.sub(center2);
   987→        scan.position.x += BOARD.thickness * 0.18;    // بارز قليلًا نحو الوجه
   988→        scan.name = 'mobo-real-pcb';
   989→        pcb.add(scan);
   990→        // إخفاء الوجه الإجرائي الأخضر (يُبقى الجسم مرجعًا للسماكة والظهر)
   991→        if (pcb instanceof THREE.Mesh && !Array.isArray(pcb.material)) {
   992→          pcb.material = pcb.material.clone();
   993→          (pcb.material as THREE.MeshStandardMaterial).color.set(0x0f2a1f);
   994→        }
   995→      },
   996→      undefined,
   997→      () => {
   998→        // فشل تحميل الأصل → يبقى PCB الإجرائي كما هو (مشهد كامل بلا انقطاع)
   999→        console.warn('[PCScene] تعذر تحميل مسح اللوحة الحقيقي — استُخدم PCB الإجرائي');
  1000→      },
  1001→    );
  1002→
  1003→    // 2) مفك تدريبي على سجادة القطع (جزء بصري ثانوي — بلا تفاعل)
  1004→    loader.load(
  1005→      '/models/screwdriver/screwdriver_1k.gltf',
  1006→      (gltf) => {
  1007→        if (this.disposed || this.failed) return;
  1008→        const tool = gltf.scene;
  1009→        const bbox = new THREE.Box3().setFromObject(tool);
  1010→        const size = bbox.getSize(new THREE.Vector3());
  1011→        const s = 1.6 / Math.max(size.y, 1e-6); // طول مستهدف ≈ 16سم
  1012→        tool.scale.setScalar(s);
  1013→        tool.updateMatrixWorld(true);
  1014→        const b2 = new THREE.Box3().setFromObject(tool);
  1015→        tool.position.set(-3.1, TRAY.tableTopY - b2.min.y, 3.55);
  1016→        tool.rotation.y = 0.5;
  1017→        tool.name = 'desk-screwdriver';
  1018→        root.add(tool);
  1019→      },
  1020→      undefined,
  1021→      () => console.warn('[PCScene] تعذر تحميل المفك التدريبي'),
  1022→    );
  1023→
  1024→    // 3..5) أدوات وزينة الورشة CC0 — كلها فشلها صامت (تفصيل ثانوي)
  1025→    const placeTool = (
  1026→      url: string,
  1027→      name: string,
  1028→      targetLen: number,          // البعد الأطول المستهدف بوحدات المشهد
  1029→      pos: [number, number, number],
  1030→      rotY: number,
  1031→      opts?: { light?: boolean },
  1032→    ): void => {
  1033→      loader.load(
  1034→        url,
  1035→        (gltf) => {
  1036→          if (this.disposed || this.failed) return;
  1037→          const obj = gltf.scene;
  1038→          // توحيد الخامات: منع أي emissive قد يأتي من الأصل
  1039→          obj.traverse((o) => {
  1040→            if (!(o instanceof THREE.Mesh)) return;
  1041→            o.castShadow = this.quality !== 'low';
  1042→            o.receiveShadow = true;
  1043→            const mats = Array.isArray(o.material) ? o.material : [o.material];
  1044→            for (const m of mats) {
  1045→              if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = 0;
  1046→            }
  1047→          });
  1048→          const bbox = new THREE.Box3().setFromObject(obj);
  1049→          const size = bbox.getSize(new THREE.Vector3());
  1050→          const s = targetLen / Math.max(size.x, size.y, size.z, 1e-6);
  1051→          obj.scale.setScalar(s);
  1052→          obj.updateMatrixWorld(true);
  1053→          const b2 = new THREE.Box3().setFromObject(obj);
  1054→          obj.position.set(pos[0], pos[1] - b2.min.y, pos[2]);
  1055→          obj.rotation.y = rotY;
  1056→          obj.name = name;
  1057→          root.add(obj);
  1058→          if (opts?.light) {
  1059→            const head = new THREE.Vector3((b2.min.x + b2.max.x) / 2, b2.max.y - 0.15, (b2.min.z + b2.max.z) / 2);
  1060→            obj.updateMatrixWorld(true);
  1061→            const worldHead = head.applyMatrix4(obj.matrixWorld);
  1062→            const warm = new THREE.PointLight(0xffe7c4, 0.7, 9, 1.4);
  1063→            warm.position.copy(worldHead);
  1064→            warm.name = `${name}-light`;
  1065→            root.add(warm);
  1066→          }
  1067→        },
  1068→        undefined,
  1069→        () => console.warn(`[PCScene] تعذر تحميل ${name}`),
  1070→      );
  1071→    };
  1072→
  1073→    // طقم مفكات على حافة السجادة اليسرى (طوله الطبيعي ≈ 2.2 وحدة)

  1080→    // صندوق عدة معدني تحت المكتب (على أرضية المختبر)
  1081→    placeTool('/models/metal_tool_chest/metal_tool_chest_1k.gltf', 'under-desk-tool-chest', 5.4, [5.4, DESK.floorY, 5.2], 0.15);
  1082→  }

  1084→  /**
  1085→   * رسم نسيج الشاشة الافتراضية:
  1086→   *  - حالات التمهيد (off/noSignal/booting/post/warning/success) عبر الرسّام القديم.
  1087→   *  - 'os' و 'bios' يديرهما النظام الافتراضي التفاعلي (VOS): سطح مكتب/معالج تثبيت/
  1088→   *    بيئة استرداد/إعدادات UEFI — كلها تفاعلية بالماوس والكيبورد الحقيقيين (V6-B).
  1089→   */
  1090→  private drawMonitor(force: boolean): void {
  1091→    if (!this.painter || this.blocked) return;
  1092→    if (!force && nowMs() - this.lastMonitorDrawAt < 60) return;
  1093→    this.lastMonitorDrawAt = nowMs();
  1094→    const sim = this.lastState;
  1095→    const elapsed = this.bootStartedAt !== null ? Math.max(0, Date.now() - this.bootStartedAt) : 0;
  1096→    if (!sim) {
  1097→      this.painter.draw(
  1098→        { screen: 'noSignal', bootProgress: 0, postLinesAr: [], bootDevicesAr: [], canType: false },
  1099→        this.notepad,
  1100→        Date.now(),
  1101→      );
  1102→      return;
  1103→    }
  1104→    const view = computeMonitorView(sim, elapsed);
  1105→    const now = Date.now();
  1106→
  1107→    if (this.vos && (view.screen === 'os' || view.screen === 'bios')) {
  1108→      this.drawVosScreen(sim, view.screen, now);
  1109→      return;
  1110→    }
  1111→
  1112→    this.painter.draw(view, this.notepad, now);
  1113→  }

  1115→  /** رسم بيئة النظام الافتراضي/البرنامج الثابت على كانفس الشاشة + رفع النسيج */
  1116→  private drawVosScreen(sim: SimulationState, screen: 'os' | 'bios', now: number): void {
  1117→    if (!this.vos || !this.painter) return;
  1118→    const vos = this.vos;
  1119→
  1120→    if (screen === 'bios') {
  1121→      vos.mode = 'firmware';
  1122→      // (إصلاح تدقيق V6) قائمة الأجهزة الفعلية فقط — كانت تعرض أقراصًا غير مركّبة
  1123→      const devs = bootableDevices(sim).map((d) => `${d.id} — ${d.labelAr}`);
  1124→      vos.fwData = {
  1125→        cpuAr: 'LGA1700 — 6C/12T (reference)',
  1126→        ramAr: `DDR5 32GB${sim.firmware.xmp ? ' @5200 (XMP)' : ' @2133'} — ${sim.flags.ramClipsOpen ? 'مزالج مفتوحة!' : 'مثبتة'}`,
  1127→        boardAr: 'PC LAB Z690 (reference)',
  1128→        tempsAr: `CPU ${sim.thermal.cpuTempC.toFixed(0)}°C / GPU ${sim.thermal.gpuTempC.toFixed(0)}°C (simulated)`,
  1129→        fansAr: this.fans?.report().map((f) => `${f.labelAr}: ${f.rpm} RPM`).join(' | ') ?? '—',
  1130→        bootMode: sim.firmware.bootMode,
  1131→        bootOrder: [...sim.firmware.bootOrder],
  1132→        secureBoot: sim.firmware.secureBoot,
  1133→        tpm: sim.firmware.tpm,
  1134→        xmp: sim.firmware.xmp,
  1135→        devicesAr: devs,
  1136→        onBootOrder: (order) => this.callbacks.onFirmwareChange?.({ bootOrder: order }),
  1137→        onToggle: (key) => this.callbacks.onFirmwareChange?.({ [key]: !sim.firmware[key] }),
  1138→        onBootMode: (mode) => this.callbacks.onFirmwareChange?.({ bootMode: mode }),
  1139→        onExit: () => this.callbacks.onFirmwareChange?.({ setupRequested: false }),
  1140→        onBootDevice: (id) => this.callbacks.onBootDevice?.(id),
  1141→      };
  1142→      vos.render(now);
  1143→      this.painter.texture.needsUpdate = true;
  1144→      return;
  1145→    }
  1146→
  1147→    // screen === 'os': حسم وضع VOS من مصدر الحقيقة
  1148→    if (sim.usbMedia && sim.firmware.bootOrder[0] === 'usb') {
  1149→      // الإقلاع من وسيط التثبيت ⇒ معالج التثبيت (شاشة كاملة)
  1150→      if (vos.mode !== 'setup') {
  1151→        if (getSetupState().step === 0 || getSetupState().done) resetSetup();
  1152→        vos.mode = 'setup';
  1153→      }
  1154→      vos.drawSetupWith((ctx, os2, t) => drawInstaller(os2, ctx, t), now);

  1274→  private pickAt(clientX: number, clientY: number): PickedInfo {
  1275→    const rect = this.renderer.domElement.getBoundingClientRect();
  1276→    const w = rect.width > 0 ? rect.width : 1;
  1277→    const h = rect.height > 0 ? rect.height : 1;
  1278→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1279→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1280→    const hits = this.raycaster.intersectObjects(this.visuals.root.children, true);
  1281→    for (const hit of hits) {
  1282→      if (this.isHidden(hit.object)) continue;        // تجاهل ما خفيه العزل
  1283→      if (this.isSeeThrough(hit.object)) continue;    // الزجاج/الشفاف لا يحجب الاختيار
  1284→      let cur: THREE.Object3D | null = hit.object;
  1285→      while (cur) {
  1286→        // الطرفيات (زر الشاشة / زر طاقة الشاشة) — لا كيبورد ولا ماوس (توجيه v3 أ.1)
  1287→        const per: unknown = cur.userData['peripheral'];
  1288→        if (typeof per === 'string') {
  1289→          return { type: 'periph', kind: per as PeripheralKind };
  1290→        }
  1291→        const cid: unknown = cur.userData['componentId'];
  1292→        if (typeof cid === 'string' && (COMPONENT_IDS as readonly string[]).includes(cid)) {
  1293→          return { type: 'part', id: cid as ComponentId };
  1294→        }
  1295→        cur = cur.parent;
  1296→      }
  1297→      // إصابة أولى بلا معرف (عدة/زينة) — تمرير النقر عبرها إلى ما خلفها (توجيه v4)
  1298→      continue;
  1299→    }
  1300→    return null;
  1301→  }

  1558→    const hit = this.rayPlaneHit(clientX, clientY);
  1559→    if (hit) this.dragOffset.copy(part.group.position).sub(hit);
  1560→    this.renderer.domElement.style.cursor = 'grabbing';
  1561→  }
  1562→
  1563→  /** تحديث موضع القطعة أثناء السحب (بصريًا فقط — الحالة المنطقية لا تتغير هنا) */
  1564→  private updateDrag(clientX: number, clientY: number): void {
  1565→    const id = this.dragId;
  1566→    if (!id) return;
  1567→    const hit = this.rayPlaneHit(clientX, clientY);
  1568→    if (!hit) return;
  1569→    const part = this.parts[id];
  1570→    part.group.position.set(hit.x + this.dragOffset.x, part.group.position.y, hit.z + this.dragOffset.z);
  1571→    // الهدف يتبع اليد — يمنع حلقة الرسم من محاربة السحب
  1572→    part.targetPos.copy(part.group.position);
  1573→  }
  1574→
  1575→  /** إنهاء السحب: تحديد منطقة الإفلات ثم إبلاغ المنطق (التحقق أولًا ثم الحركة — توجيه v3 §6) */
  1576→  private finishDrag(clientX: number, clientY: number): void {
  1577→    const id = this.dragId;
  1578→    this.dragId = null;
  1579→    this.controls.enabled = true;
  1580→    this.renderer.domElement.style.cursor = '';
  1581→    if (!id) return;
  1582→    const hit = this.rayPlaneHit(clientX, clientY);
  1583→    let zone: DropZone = null;
  1584→    if (hit) {
  1585→      const inTray = hit.x >= TRAY.minX && hit.x <= TRAY.maxX && hit.z >= TRAY.minZ && hit.z <= TRAY.maxZ;
  1586→      const inCase = hit.x >= CASE.minX && hit.x <= CASE.maxX && hit.z >= CASE.minZ && hit.z <= CASE.maxZ;
  1587→      zone = inTray ? 'tray' : inCase ? 'case' : null;
  1588→    }
  1589→    // المرور عبر منطق المتجر: إن رُفض الإجراء تعاد القطعة لموضع راحتها عبر syncState
  1590→    this.callbacks.onPartDrop?.(id, zone);
  1591→  }
  1592→
  1593→  /** إلغاء أي سحب جارٍ (Esc / إيقاف الوضع) */
  1594→  private cancelDrag(): void {
  1595→    if (this.dragId) {
  1596→      const part = this.parts[this.dragId];
  1597→      this.recomputeTargets(); // يعيد الهدف الصحيح من آخر حالة
  1598→      if (part) part.group.position.copy(part.targetPos);
  1599→    }
  1600→    this.dragId = null;
  1601→    this.pendingDrag = null;
  1602→    if (this.controls) this.controls.enabled = true;
  1603→    if (this.renderer) this.renderer.domElement.style.cursor = '';
  1604→  }

  1606→  /** تقاطع شعاع المؤشر مع سطح السحب الحالي */
  1607→  private rayPlaneHit(clientX: number, clientY: number): THREE.Vector3 | null {
  1608→    const rect = this.renderer.domElement.getBoundingClientRect();
  1609→    const w = rect.width > 0 ? rect.width : 1;
  1610→    const h = rect.height > 0 ? rect.height : 1;
  1611→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1612→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1613→    const hit = this._v2;
  1614→    return this.raycaster.ray.intersectPlane(this.dragPlane, hit) ? hit : null;
  1615→  }
