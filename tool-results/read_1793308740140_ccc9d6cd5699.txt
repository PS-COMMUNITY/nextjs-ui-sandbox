    48→interface MatOpts {
    49→  map?: THREE.Texture | null;
    50→  emissive?: number;
    51→  emissiveIntensity?: number;
    52→  transparent?: boolean;
    53→  opacity?: number;
    54→  side?: THREE.Side;
    55→}

    57→/** cache للمواد القياسية — مفاتيح بنطاق القطعة */
    58→const matCache = new Map<string, THREE.MeshStandardMaterial>();
    59→
    60→function mat(color: number, metalness: number, roughness: number, opts?: MatOpts): THREE.MeshStandardMaterial {
    61→  const mapId = opts?.map ? opts.map.uuid : 'n';
    62→  const key = `${partScope}|${color}|${metalness}|${roughness}|${mapId}|${opts?.emissive ?? ''}|${opts?.emissiveIntensity ?? ''}|${opts?.transparent ?? ''}|${opts?.opacity ?? ''}|${opts?.side ?? ''}`;
    63→  const cached = matCache.get(key);
    64→  if (cached) return cached;
    65→  const m = new THREE.MeshStandardMaterial({
    66→    color,
    67→    metalness,
    68→    roughness,
    69→    map: opts?.map ?? null,
    70→    emissive: opts?.emissive ?? 0x000000,
    71→    emissiveIntensity: opts?.emissiveIntensity ?? 1,
    72→    transparent: opts?.transparent ?? false,
    73→    opacity: opts?.opacity ?? 1,
    74→    side: opts?.side ?? THREE.FrontSide,
    75→    flatShading: false,
    76→  });
    77→  matCache.set(key, m);
    78→  return m;
    79→}

   102→/** صندوق سريع مع ضبط الظلال حسب الجودة */
   103→function box(
   104→  w: number, h: number, d: number, material: THREE.Material,
   105→  x = 0, y = 0, z = 0, cast = true, receive = false,
   106→): THREE.Mesh {
   107→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), material);
   108→  m.position.set(x, y, z);
   109→  if (castEnabled && cast) m.castShadow = true;
   110→  if (receive) m.receiveShadow = true;
   111→  return m;
   112→}

   114→type Axis = 'x' | 'y' | 'z';

   116→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
   117→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
   118→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
   119→  if (axis === 'x') m.rotation.z = Math.PI / 2;
   120→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
   121→  m.position.set(x, y, z);
   122→  if (castEnabled && cast) m.castShadow = true;
   123→  return m;
   124→}

   126→/** برغي معدني صغير — محور حسب الاستخدام (رأس سداسي + فتحة تصالب) */
   127→function makeScrew(r: number, length: number, axis: Axis = 'y'): THREE.Mesh {
   128→  return cyl(r, length, mat(COLORS.aluminum, 0.85, 0.35), axis);

   131→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
   132→function makeCable(points: THREE.Vector3[], radius: number): THREE.Mesh {
   133→  const curve = new THREE.CatmullRomCurve3(points);
   134→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(0x181d24, 0.3, 0.6));
   135→  if (castEnabled) m.castShadow = true;
   136→  return m;
   137→}

   139→/** تحويل نقاط كابل عالمية إلى محلية بالنسبة لموضع تثبيت القطعة */
   140→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
   141→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));
   142→}

   144→/** مستطيل مستوٍ مواجه لمحور محدد (للملصقات والشبكات) */
   145→function plane(w: number, h: number, material: THREE.Material, axis: 'x' | 'y' | 'z', x: number, y: number, z: number): THREE.Mesh {
   146→  const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), material);
   147→  if (axis === 'x') m.rotation.y = -Math.PI / 2; // مواجه -X
   148→  else if (axis === 'y') m.rotation.x = -Math.PI / 2;
   149→  m.position.set(x, y, z);
   150→  return m;
   151→}

   153→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   154→
   155→/**
   156→ * مروحة واقعية: إطار مربع كامل بحواف وأركان دائرية + فتحات براغي + حلقة داخلية
   157→ * + 9 شفرات مائلة + محور بغطاء. محور الدوران الافتراضي = Y المحلي.
   158→ */
   159→function makeFan(radius: number): FanParts {
   160→  const depth = Math.max(0.14, radius * 0.5);
   161→  const half = radius * 1.32;
   162→
   163→  // الإطار: 4 حواف كاملة + 4 أركان أسطوانية
   164→  const frame = new THREE.Group();
   165→  const frameMat = mat(COLORS.plastic, 0.3, 0.62);
   166→  const barGeoX = new THREE.BoxGeometry(half * 2, depth, 0.1);
   167→  const barGeoZ = new THREE.BoxGeometry(0.1, depth, half * 2);
   168→  for (const s of [-1, 1]) {
   169→    const barX = new THREE.Mesh(barGeoX, frameMat);
   170→    barX.position.set(0, 0, s * half);
   171→    const barZ = new THREE.Mesh(barGeoZ, frameMat);
   172→    barZ.position.set(s * half, 0, 0);
   173→    if (castEnabled) { barX.castShadow = true; barZ.castShadow = true; }
   174→    frame.add(barX, barZ);
   175→  }
   176→  const cornerGeo = new THREE.CylinderGeometry(0.055, 0.055, depth, 10);
   177→  const holeGeo = new THREE.CylinderGeometry(0.026, 0.026, 0.02, 8);
   178→  const holeMat = mat(COLORS.black, 0.2, 0.6);
   179→  for (const sx of [-1, 1]) {
   180→    for (const sz of [-1, 1]) {
   181→      const corner = new THREE.Mesh(cornerGeo, frameMat);
   182→      corner.position.set(sx * (half - 0.05), 0, sz * (half - 0.05));
   183→      if (castEnabled) corner.castShadow = true;
   184→      frame.add(corner);
   185→      // فتحة برغي مظلمة على الوجه العلوي
   186→      const hole = new THREE.Mesh(holeGeo, holeMat);
   187→      hole.position.set(sx * half, depth / 2, sz * half);
   188→      frame.add(hole);
   189→    }
   190→  }
   191→  // حلقة داخلية محيطية
   192→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.99, 0.016, 6, 28), mat(COLORS.metalDark, 0.5, 0.5));
   193→  ring.rotation.x = Math.PI / 2;
   194→  if (castEnabled) ring.castShadow = true;
   195→  frame.add(ring);
   196→  // دعامات خلفية (ستراهنة بين الحواف خلف الشفرات)
   197→  const strutGeo = new THREE.BoxGeometry(0.05, 0.02, radius * 1.7);
   198→  for (const sz of [-1, 1]) {
   199→    const strut = new THREE.Mesh(strutGeo, frameMat);
   200→    strut.position.set(sz * radius * 0.72, -depth * 0.28, 0);
   201→    strut.rotation.y = Math.PI / 2;
   202→    frame.add(strut);
   203→  }
   204→
   205→  // الشفرات: 9 شفرات مائلة 35° داخل مجموعة blades (تُدفع إلى spinners)
   206→  const blades = new THREE.Group();
   207→  const bladeGeo = new THREE.BoxGeometry(0.016, radius * 0.8, radius * 0.32);
   208→  const bladeMat = mat(0x242c36, 0.25, 0.45);
   209→  for (let i = 0; i < 9; i++) {
   210→    const arm = new THREE.Object3D();
   211→    arm.rotation.y = (i / 9) * Math.PI * 2;
   212→    const pitch = new THREE.Object3D();
   213→    pitch.position.set(radius * 0.52, 0, 0);
   214→    pitch.rotation.x = 0.62; // ميلان الشفرة ≈ 35°
   215→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   216→    blade.rotation.z = Math.PI / 2;
   217→    if (castEnabled) blade.castShadow = true;
   218→    pitch.add(blade);
   219→    arm.add(pitch);
   220→    blades.add(arm);
   221→  }
   222→  const hub = new THREE.Mesh(
   223→    new THREE.CylinderGeometry(radius * 0.17, radius * 0.17, depth * 0.72, 14),
   224→    mat(COLORS.metalDark, 0.5, 0.4),
   225→  );
   226→  if (castEnabled) hub.castShadow = true;
   227→  blades.add(hub);
   228→  // غطاء محور بحلقة داكنة
   229→  const cap = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.1, radius * 0.1, depth * 0.76, 12), mat(0x11161c, 0.3, 0.5));
   230→  blades.add(cap);
   231→  const capRing = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.11, 0.008, 6, 16), mat(COLORS.metalMid, 0.6, 0.4));
   232→  capRing.rotation.x = Math.PI / 2;
   233→  capRing.position.y = depth * 0.38;
   234→  blades.add(capRing);
   235→
   236→  return { frame, blades };
   237→}

   257→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   258→  scene.background = new THREE.Color(COLORS.bg);
   259→  scene.fog = new THREE.Fog(COLORS.bg, 18, 42);
   260→
   261→  // أرضية دائرية داكنة تستقبل الظلال
   262→  const ground = new THREE.Mesh(
   263→    new THREE.CircleGeometry(22, 24),
   264→    new THREE.MeshStandardMaterial({ color: 0x141a22, metalness: 0, roughness: 0.92, flatShading: false }),
   265→  );
   266→  ground.rotation.x = -Math.PI / 2;
   267→  ground.position.y = 0;
   268→  ground.receiveShadow = true;
   269→  scene.add(ground);
   270→
   271→  // شبكة أرضية خفيفة
   272→  const grid = new THREE.GridHelper(30, 30, 0x2a3340, 0x1a212b);
   273→  grid.position.y = 0.01;
   274→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   275→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   276→  for (const gm of gridMats) {
   277→    gm.transparent = true;
   278→    gm.opacity = 0.4;
   279→  }
   280→  scene.add(grid);
   281→
   282→  scene.add(new THREE.AmbientLight(0xffffff, 0.45));
   283→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x20242c, 0.5));
   284→
   285→  // شمس دافئة مع ظلال (حسب الجودة)
   286→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.0);
   287→  dir.position.set(-6, 8, 6);
   288→  if (quality !== 'low') {
   289→    dir.castShadow = true;
   290→    const size = quality === 'high' ? 2048 : 1024;
   291→    dir.shadow.mapSize.set(size, size);
   292→    dir.shadow.camera.left = -6;
   293→    dir.shadow.camera.right = 6;
   294→    dir.shadow.camera.top = 6;
   295→    dir.shadow.camera.bottom = -6;
   296→    dir.shadow.camera.near = 2;
   297→    dir.shadow.camera.far = 30;
   298→    dir.shadow.camera.updateProjectionMatrix();
   299→    dir.shadow.bias = -0.0004;
   300→    dir.shadow.normalBias = 0.02;
   301→  }
   302→  scene.add(dir);
   303→
   304→  const fill = new THREE.PointLight(0xffffff, 0.5, 0, 0);
   305→  fill.position.set(3, 3, -2);
   306→  scene.add(fill);
   307→  const front = new THREE.PointLight(0xfff4e0, 1.0, 0, 0);
   308→  front.position.set(-5, 3.5, 3);
   309→  scene.add(front);
   310→}

   316→function buildCaseShell(flags: FlagVisuals): THREE.Group {
   317→  scope('case');
   318→  const g = new THREE.Group();
   319→  g.name = 'case-shell';
   320→
   321→  const cx = (CASE.minX + CASE.maxX) / 2; // 1.05
   322→  const cy = (CASE.minY + CASE.maxY) / 2; // 2.25
   323→  const cz = (CASE.minZ + CASE.maxZ) / 2; // 2.3
   324→  const wX = CASE.maxX - CASE.minX;       // 2.1
   325→  const hY = CASE.maxY - CASE.minY;       // 4.5
   326→  const dZ = CASE.maxZ - CASE.minZ;       // 4.6
   327→
   328→  // جدران بخامة معدن مصقول (فرشات)
   329→  const wallMat = mat(COLORS.metalDark, 0.5, 0.5, { map: brushedTexture('#2a323c') });
   330→  const trayWallMat = mat(0x2c343e, 0.5, 0.48, { map: brushedTexture('#333c46') });
   331→
   332→  g.add(box(0.05, hY, dZ, trayWallMat, CASE.maxX - 0.025, cy, cz, true, true)); // يمين (صينية اللوحة)
   333→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.maxY - 0.025, cz, true, true));     // سقف
   334→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.minY + 0.025, cz, true, true));     // أرضية
   335→  g.add(box(wX, hY, 0.05, wallMat, cx, cy, CASE.minZ + 0.025, true, true));     // خلف
   336→  g.add(box(wX, hY, 0.05, mat(0x262d36, 0.35, 0.6), cx, cy, CASE.maxZ - 0.025, true, true)); // أمام
   337→
   338→  // حافتا شطف عموديتان على حافة الصينية (أمام/خلف)

   340→  g.add(box(0.02, hY * 0.92, 0.02, mat(0x414c58, 0.6, 0.4), CASE.maxX - 0.18, cy, CASE.minZ + 0.62, false));
   341→
   342→  // ===== الواجهة الأمامية =====

   352→  // ===== منطقة التحكم العلوية الأمامية (على السقف) =====

   356→  g.add(cyl(0.02, 0.012, mat(0xd8dade, 0.2, 0.5), 'y', 0.28, CASE.maxY + 0.028, CASE.maxZ - 0.45, false));

   358→    color: 0x10151b, emissive: 0x000000, emissiveIntensity: 1.2, roughness: 0.35, metalness: 0.1, flatShading: false,
   359→  });
   360→  const powerLed = new THREE.Mesh(new THREE.SphereGeometry(0.03, 12, 8), ledMat); // مادة فريدة يتحكم بها علم powerOn
   361→  powerLed.position.set(0.78, CASE.maxY + 0.01, CASE.maxZ - 0.45);

   366→  const usbMat = mat(COLORS.black, 0.3, 0.55);

   369→    g.add(box(0.15, 0.016, 0.09, usbMat, ux, CASE.maxY - 0.002, CASE.maxZ - 0.45, false));
   370→    g.add(box(0.11, 0.006, 0.06, tongueMat, ux, CASE.maxY + 0.004, CASE.maxZ - 0.45, false));
   371→  }
   372→  g.add(cyl(0.028, 0.014, mat(0x2f6b4f, 0.4, 0.45), 'y', 1.08, CASE.maxY + 0.002, CASE.maxZ - 0.45, false));
   373→  g.add(cyl(0.028, 0.014, mat(0x7a4040, 0.4, 0.45), 'y', 1.2, CASE.maxY + 0.002, CASE.maxZ - 0.45, false));
   374→
   375→  // ===== السقف: فلتر غبار شبكي =====

   593→  g.add(box(0.08, 0.5, 0.25, black, onFace(0.08), 1.0, 1.05, true));   // ATX24 (عالمي 3.75, 2.62)

   617→    const h = new THREE.Mesh(headerGeo, headerMat);

   621→  const fanPlug = box(0.08, 0.08, 0.18, mat(0x101418, 0.2, 0.6), onFace(0.08) - 0.03, 1.44, 0.775, false);

   625→
   626→  // 12) كتلة front-panel + قابسها
   627→  g.add(box(0.06, 0.14, 0.22, headerMat, onFace(0.06), -1.2, 0.9, false));

   629→  fpPlug.name = 'flag-front-panel-plug';
   630→  flags.frontPanelPlug = fpPlug;
   631→  g.add(fpPlug);
   632→
   633→  setComponentId(g, 'motherboard');
   634→  return g;
   635→}

   638→// المعالج (dynamic.cpu) — متمركز عند MOUNTS.cpu، الوجه نحو -X
   639→// ============================================================
   640→
   641→function buildCpu(): THREE.Group {
   642→  scope('cpu');
   643→  const g = new THREE.Group();
   644→  g.name = 'part-cpu';
   645→
   646→  // قاعدة خضراء (substrate)
   647→  g.add(box(0.05, 0.4, 0.4, mat(0x1e6b3a, 0.3, 0.55), 0, 0, 0, true, true));
   648→  // أطراف ذهبية محيطية على الوجه الأمامي (إشارة إلى تلامسات LGA)
   649→  const goldEdge = mat(0xc9a23f, 0.9, 0.3);
   650→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, -0.193, false));
   651→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, 0.193, false));
   652→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, -0.193, 0, false));
   653→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, 0.193, 0, false));
   654→
   655→  // غطاء IHS بأركان مشطوفة (مستطيل مشطوف عبر Cylinder ثماني مسطح؟) — نستخدم Extrude
   656→  const ihsShape = new THREE.Shape();
   657→  const s = 0.17;
   658→  const r = 0.045;
   659→  ihsShape.moveTo(-s + r, -s);
   660→  ihsShape.lineTo(s - r, -s);
   661→  ihsShape.quadraticCurveTo(s, -s, s, -s + r);
   662→  ihsShape.lineTo(s, s - r);
   663→  ihsShape.quadraticCurveTo(s, s, s - r, s);
   664→  ihsShape.lineTo(-s + r, s);
   665→  ihsShape.quadraticCurveTo(-s, s, -s, s - r);
   666→  ihsShape.lineTo(-s, -s + r);
   667→  ihsShape.quadraticCurveTo(-s, -s, -s + r, -s);
   668→  const ihsGeo = new THREE.ExtrudeGeometry(ihsShape, { depth: 0.035, bevelEnabled: false });
   669→  const ihs = new THREE.Mesh(ihsGeo, mat(0xc2c9d1, 0.9, 0.3));
   670→  ihs.rotation.y = -Math.PI / 2; // البثق على المحور X نحو -X
   671→  ihs.position.set(-0.045, 0, 0);
   672→  if (castEnabled) ihs.castShadow = true;
