    57→/** cache للمواد القياسية — مفاتيح بنطاق القطعة */
    58→const matCache = new Map<string, THREE.MeshStandardMaterial>();

    60→function mat(color: number, metalness: number, roughness: number, opts?: MatOpts): THREE.MeshStandardMaterial {
    61→  const mapId = opts?.map ? opts.map.uuid : 'n';
    62→  const key = `${partScope}|${color}|${metalness}|${roughness}|${mapId}|${opts?.emissive ?? ''}|${opts?.emissiveIntensity ?? ''}|${opts?.transparent ?? ''}|${opts?.opacity ?? ''}|${opts?.side ?? ''}`;
    63→  const cached = matCache.get(key);
    64→  if (cached) return cached;
    65→  const m = new THREE.MeshStandardMaterial({
    66→    color,
    67→    metalness,
    68→    roughness,
    69→    map: opts?.map ?? null,
    70→    emissive: opts?.emissive ?? 0x000000,
    71→    emissiveIntensity: opts?.emissiveIntensity ?? 1,
    72→    transparent: opts?.transparent ?? false,
    73→    opacity: opts?.opacity ?? 1,
    74→    side: opts?.side ?? THREE.FrontSide,
    75→    flatShading: false,
    76→  });
    77→  matCache.set(key, m);
    78→  return m;
    79→}

   123→  return m;

   153→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   154→
   155→/**
   156→ * مروحة واقعية: إطار مربع كامل بحواف وأركان دائرية + فتحات براغي + حلقة داخلية
   157→ * + 9 شفرات مائلة + محور بغطاء. محور الدوران الافتراضي = Y المحلي.
   158→ */
   159→function makeFan(radius: number): FanParts {
   160→  const depth = Math.max(0.14, radius * 0.5);
   161→  const half = radius * 1.32;
   162→
   163→  // الإطار: 4 حواف كاملة + 4 أركان أسطوانية
   164→  const frame = new THREE.Group();
   165→  const frameMat = mat(COLORS.plastic, 0.3, 0.62);
   166→  const barGeoX = new THREE.BoxGeometry(half * 2, depth, 0.1);
   167→  const barGeoZ = new THREE.BoxGeometry(0.1, depth, half * 2);
   168→  for (const s of [-1, 1]) {
   169→    const barX = new THREE.Mesh(barGeoX, frameMat);
   170→    barX.position.set(0, 0, s * half);
   171→    const barZ = new THREE.Mesh(barGeoZ, frameMat);
   172→    barZ.position.set(s * half, 0, 0);
   173→    if (castEnabled) { barX.castShadow = true; barZ.castShadow = true; }
   174→    frame.add(barX, barZ);
   175→  }
   176→  const cornerGeo = new THREE.CylinderGeometry(0.055, 0.055, depth, 10);
   177→  const holeGeo = new THREE.CylinderGeometry(0.026, 0.026, 0.02, 8);
   178→  const holeMat = mat(COLORS.black, 0.2, 0.6);
   179→  for (const sx of [-1, 1]) {
   180→    for (const sz of [-1, 1]) {
   181→      const corner = new THREE.Mesh(cornerGeo, frameMat);
   182→      corner.position.set(sx * (half - 0.05), 0, sz * (half - 0.05));
   183→      if (castEnabled) corner.castShadow = true;
   184→      frame.add(corner);
   185→      // فتحة برغي مظلمة على الوجه العلوي
   186→      const hole = new THREE.Mesh(holeGeo, holeMat);
   187→      hole.position.set(sx * half, depth / 2, sz * half);
   188→      frame.add(hole);
   189→    }
   190→  }
   191→  // حلقة داخلية محيطية
   192→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.99, 0.016, 6, 28), mat(COLORS.metalDark, 0.5, 0.5));
   193→  ring.rotation.x = Math.PI / 2;
   194→  if (castEnabled) ring.castShadow = true;
   195→  frame.add(ring);
   196→  // دعامات خلفية (ستراهنة بين الحواف خلف الشفرات)
   197→  const strutGeo = new THREE.BoxGeometry(0.05, 0.02, radius * 1.7);
   198→  for (const sz of [-1, 1]) {
   199→    const strut = new THREE.Mesh(strutGeo, frameMat);
   200→    strut.position.set(sz * radius * 0.72, -depth * 0.28, 0);
   201→    strut.rotation.y = Math.PI / 2;
   202→    frame.add(strut);
   203→  }
   204→
   205→  // الشفرات: 9 شفرات مائلة 35° داخل مجموعة blades (تُدفع إلى spinners)
   206→  const blades = new THREE.Group();
   207→  const bladeGeo = new THREE.BoxGeometry(0.016, radius * 0.8, radius * 0.32);
   208→  const bladeMat = mat(0x242c36, 0.25, 0.45);
   209→  for (let i = 0; i < 9; i++) {
   210→    const arm = new THREE.Object3D();
   211→    arm.rotation.y = (i / 9) * Math.PI * 2;
   212→    const pitch = new THREE.Object3D();
   213→    pitch.position.set(radius * 0.52, 0, 0);
   214→    pitch.rotation.x = 0.62; // ميلان الشفرة ≈ 35°
   215→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   216→    blade.rotation.z = Math.PI / 2;
   217→    if (castEnabled) blade.castShadow = true;
   218→    pitch.add(blade);
   219→    arm.add(pitch);
   220→    blades.add(arm);
   221→  }
   222→  const hub = new THREE.Mesh(
   223→    new THREE.CylinderGeometry(radius * 0.17, radius * 0.17, depth * 0.72, 14),
   224→    mat(COLORS.metalDark, 0.5, 0.4),
   225→  );
   226→  if (castEnabled) hub.castShadow = true;
   227→  blades.add(hub);
   228→  // غطاء محور بحلقة داكنة
   229→  const cap = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.1, radius * 0.1, depth * 0.76, 12), mat(0x11161c, 0.3, 0.5));
   230→  blades.add(cap);
   231→  const capRing = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.11, 0.008, 6, 16), mat(COLORS.metalMid, 0.6, 0.4));
   232→  capRing.rotation.x = Math.PI / 2;
   233→  capRing.position.y = depth * 0.38;
   234→  blades.add(capRing);
   235→
   236→  return { frame, blades };
   237→}

   308→  front.position.set(-5, 3.5, 3);
   309→  scene.add(front);

   641→function buildCpu(): THREE.Group {
   642→  scope('cpu');

   644→  g.name = 'part-cpu';
   645→
   646→  // قاعدة خضراء (substrate)
   647→  g.add(box(0.05, 0.4, 0.4, mat(0x1e6b3a, 0.3, 0.55), 0, 0, 0, true, true));
   648→  // أطراف ذهبية محيطية على الوجه الأمامي (إشارة إلى تلامسات LGA)
   649→  const goldEdge = mat(0xc9a23f, 0.9, 0.3);
   650→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, -0.193, false));
   651→  g.add(box(0.008, 0.4, 0.014, goldEdge, -0.026, 0, 0.193, false));
   652→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, -0.193, 0, false));
   653→  g.add(box(0.008, 0.014, 0.4, goldEdge, -0.026, 0.193, 0, false));
   654→
   655→  // غطاء IHS بأركان مشطوفة (مستطيل مشطوف عبر Cylinder ثماني مسطح؟) — نستخدم Extrude
   656→  const ihsShape = new THREE.Shape();
   657→  const s = 0.17;
   658→  const r = 0.045;
   659→  ihsShape.moveTo(-s + r, -s);
   660→  ihsShape.lineTo(s - r, -s);
   661→  ihsShape.quadraticCurveTo(s, -s, s, -s + r);
   662→  ihsShape.lineTo(s, s - r);
   663→  ihsShape.quadraticCurveTo(s, s, s - r, s);
   664→  ihsShape.lineTo(-s + r, s);
   665→  ihsShape.quadraticCurveTo(-s, s, -s, s - r);
   666→  ihsShape.lineTo(-s, -s + r);
   667→  ihsShape.quadraticCurveTo(-s, -s, -s + r, -s);
   668→  const ihsGeo = new THREE.ExtrudeGeometry(ihsShape, { depth: 0.035, bevelEnabled: false });
   669→  const ihs = new THREE.Mesh(ihsGeo, mat(0xc2c9d1, 0.9, 0.3));
   670→  ihs.rotation.y = -Math.PI / 2; // البثق على المحور X نحو -X
   671→  ihs.position.set(-0.045, 0, 0);
   672→  if (castEnabled) ihs.castShadow = true;
   673→  g.add(ihs);
   674→
   675→  // شقّا اتجاه على حافتين (توجيه المعالج) — على وجه IHS الأمامي
   676→  const notchMat = mat(0x6b7280, 0.6, 0.4);
   677→  g.add(box(0.02, 0.03, 0.07, notchMat, -0.082, 0.155, 0.05, false));
   678→  g.add(box(0.02, 0.03, 0.07, notchMat, -0.082, -0.155, -0.05, false));
   679→
   680→  // مثلث اتجاه ذهبي مسطح (prism ثلاثي) على وجه IHS
   681→  const tri = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.05, 0.008, 3), mat(0xd4af37, 0.85, 0.3));
   682→  tri.rotation.z = Math.PI / 2;
   683→  tri.position.set(-0.086, 0.1, -0.1);
   684→  g.add(tri);
   685→
   686→  setComponentId(g, 'cpu');
   687→  return g;
   688→}

   694→function buildCooler(spinners: THREE.Object3D[]): THREE.Group {
   695→  scope('cooler');
   696→  const g = new THREE.Group();
   697→  g.name = 'part-cooler';
   698→  const copper = mat(COLORS.copper, 0.9, 0.3);
   699→  const alu = mat(COLORS.aluminum, 0.85, 0.4, { map: brushedTexture('#a7b0ba') });
   700→
   701→  // قاعدة نحاسية على وجه المعالج
   702→  g.add(box(0.05, 0.3, 0.3, copper, 0.33, 0, 0, true));
   703→
   704→  // قائمتا تثبيت جانبيتان (من القاعدة إلى كتلة الزعانف)
   705→  g.add(box(0.24, 0.6, 0.015, alu, 0.2, 0, 0.3, false));
   706→  g.add(box(0.24, 0.6, 0.015, alu, 0.2, 0, -0.3, false));
   707→
   708→  // 4 أنابيب حرارية نحاسية بقوس علوي (TubeGeometry على منحنى)
