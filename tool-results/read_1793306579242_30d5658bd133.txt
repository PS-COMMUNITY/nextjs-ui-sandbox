    25→/** cache للمواد القياسية — لا تنشئ مواد جديدة لكل إطار */
    26→const matCache = new Map<string, THREE.MeshStandardMaterial>();
    27→
    28→function mat(color: number, metalness: number, roughness: number): THREE.MeshStandardMaterial {
    29→  const key = `${color}-${metalness}-${roughness}`;
    30→  const cached = matCache.get(key);
    31→  if (cached) return cached;
    32→  const m = new THREE.MeshStandardMaterial({ color, metalness, roughness, flatShading: false });
    33→  matCache.set(key, m);
    34→  return m;
    35→}

    37→/** زجاج الغطاء الجانبي — مادة فيزيائية واحدة مشتركة */
    38→let glassMat: THREE.MeshPhysicalMaterial | null = null;
    39→function glass(): THREE.MeshPhysicalMaterial {
    40→  if (!glassMat) {
    41→    glassMat = new THREE.MeshPhysicalMaterial({
    42→      color: COLORS.glass,
    43→      metalness: 0.1,
    44→      roughness: 0.08,
    45→      transparent: true,
    46→      opacity: 0.16,
    47→      side: THREE.DoubleSide,
    48→      flatShading: false,
    49→    });
    50→  }
    51→  return glassMat;
    52→}

    58→/** صندوق سريع مع ضبط الظلال حسب الجودة */
    59→function box(
    60→  w: number, h: number, d: number, material: THREE.Material,
    61→  x = 0, y = 0, z = 0, cast = true, receive = false,
    62→): THREE.Mesh {
    63→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), material);
    64→  m.position.set(x, y, z);
    65→  if (castEnabled && cast) m.castShadow = true;
    66→  if (receive) m.receiveShadow = true;
    67→  return m;
    68→}

    70→type Axis = 'x' | 'y' | 'z';
    71→
    72→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
    73→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
    74→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
    75→  if (axis === 'x') m.rotation.z = Math.PI / 2;
    76→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
    77→  m.position.set(x, y, z);
    78→  if (castEnabled && cast) m.castShadow = true;
    79→  return m;
    80→}

    83→function makeScrew(r: number, length: number, axis: Axis = 'y'): THREE.Mesh {
    84→  return cyl(r, length, mat(COLORS.aluminum, 0.85, 0.35), axis);

    87→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
    88→function makeCable(points: THREE.Vector3[], radius: number): THREE.Mesh {
    89→  const curve = new THREE.CatmullRomCurve3(points);
    90→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(0x181d24, 0.3, 0.6));
    91→  if (castEnabled) m.castShadow = true;
    92→  return m;
    93→}

    95→/** تحويل نقاط كابل عالمية إلى محلية بالنسبة لموضع تثبيت القطعة */
    96→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
    97→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));
    98→}

   100→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   101→
   102→/**
   103→ * مروحة عامة: إطار (4 أركان + حلقة Torus) + شفرات (7 ألواح مائلة 30°) + محور مركزي.
   104→ * محور الدوران الافتراضي = Y المحلي (تُلف المروحة لاحقًا حسب موضعها).
   105→ */
   106→function makeFan(radius: number): FanParts {
   107→  const depth = Math.max(0.12, radius * 0.55);
   108→  const half = radius + 0.06;
   109→
   110→  // الإطار
   111→  const frame = new THREE.Group();
   112→  const frameMat = mat(COLORS.plastic, 0.3, 0.6);
   113→  const postGeo = new THREE.BoxGeometry(0.09, depth, 0.09);
   114→  for (const sx of [-1, 1]) {
   115→    for (const sz of [-1, 1]) {
   116→      const post = new THREE.Mesh(postGeo, frameMat);
   117→      post.position.set(sx * half, 0, sz * half);
   118→      if (castEnabled) post.castShadow = true;
   119→      frame.add(post);
   120→    }
   121→  }
   122→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.97, 0.018, 6, 24), mat(COLORS.metalDark, 0.5, 0.5));
   123→  ring.rotation.x = Math.PI / 2; // الحلقة في مستوى XZ (عمودية على محور Y)
   124→  if (castEnabled) ring.castShadow = true;
   125→  frame.add(ring);
   126→
   127→  // الشفرات: 7 ألواح موزعة دائريًا داخل مجموعة blades (تُدفع إلى spinners)
   128→  const blades = new THREE.Group();
   129→  const bladeGeo = new THREE.BoxGeometry(0.02, radius * 0.8, radius * 0.35);
   130→  const bladeMat = mat(COLORS.metalMid, 0.4, 0.5);
   131→  for (let i = 0; i < 7; i++) {
   132→    const arm = new THREE.Object3D();
   133→    arm.rotation.y = (i / 7) * Math.PI * 2;
   134→    const pitch = new THREE.Object3D();
   135→    pitch.position.set(radius * 0.5, 0, 0);
   136→    pitch.rotation.x = Math.PI / 6; // ميلان الشفرة 30° حول المحور الشعاعي
   137→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   138→    blade.rotation.z = Math.PI / 2; // الطول شعاعيًا
   139→    if (castEnabled) blade.castShadow = true;
   140→    pitch.add(blade);
   141→    arm.add(pitch);
   142→    blades.add(arm);
   143→  }
   144→  const hub = new THREE.Mesh(
   145→    new THREE.CylinderGeometry(radius * 0.16, radius * 0.16, depth * 0.7, 12),
   146→    mat(COLORS.metalDark, 0.5, 0.4),
   147→  );
   148→  if (castEnabled) hub.castShadow = true;
   149→  blades.add(hub);
   150→
   151→  return { frame, blades };
   152→}

   172→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   173→  scene.background = new THREE.Color(COLORS.bg);
   174→  scene.fog = new THREE.Fog(COLORS.bg, 18, 42);
   175→
   176→  // أرضية دائرية داكنة تستقبل الظلال
   177→  const ground = new THREE.Mesh(
   178→    new THREE.CircleGeometry(22, 24),
   179→    new THREE.MeshStandardMaterial({ color: 0x141a22, metalness: 0, roughness: 0.92, flatShading: false }),
   180→  );
   181→  ground.rotation.x = -Math.PI / 2;
   182→  ground.position.y = 0;
   183→  ground.receiveShadow = true;
   184→  scene.add(ground);
   185→
   186→  // شبكة أرضية خفيفة
   187→  const grid = new THREE.GridHelper(30, 30, 0x2a3340, 0x1a212b);
   188→  grid.position.y = 0.01;
   189→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   190→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   191→  for (const gm of gridMats) {
   192→    gm.transparent = true;
   193→    gm.opacity = 0.4;
   194→  }
   195→  scene.add(grid);
   196→
   197→  // إضاءة مباشرة مخفّضة — الخريطة البيئية (scene.environment) تتكفل بالملء العام
   198→  scene.add(new THREE.AmbientLight(0xffffff, 0.45));
   199→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x20242c, 0.5));
   200→
   201→  // شمس دافئة مع ظلال (حسب الجودة)
   202→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.0);
   203→  dir.position.set(-6, 8, 6);
   204→  if (quality !== 'low') {
   205→    dir.castShadow = true;
   206→    const size = quality === 'high' ? 2048 : 1024;
   207→    dir.shadow.mapSize.set(size, size);
   208→    dir.shadow.camera.left = -6;
   209→    dir.shadow.camera.right = 6;
   210→    dir.shadow.camera.top = 6;
   211→    dir.shadow.camera.bottom = -6;
   212→    dir.shadow.camera.near = 2;
   213→    dir.shadow.camera.far = 30;
   214→    dir.shadow.camera.updateProjectionMatrix();
   215→    dir.shadow.bias = -0.0004;
   216→    dir.shadow.normalBias = 0.02;
   217→  }
   218→  scene.add(dir);
   219→
   220→  // ضوء تعبئة خفيف خلفي (بلا ظلال؛ decay=0 ليبقى «خفيفًا» بمقدار 0.5 فعليًا)
   221→  const fill = new THREE.PointLight(0xffffff, 0.5, 0, 0);
   222→  fill.position.set(3, 3, -2);
   223→  scene.add(fill);
   224→  // ضوء أمامي موجّه من جهة الزجاج لإبراز وجه اللوحة الأم
   225→  const front = new THREE.PointLight(0xfff4e0, 1.0, 0, 0);
   226→  front.position.set(-5, 3.5, 3);
   227→  scene.add(front);
   228→}

   234→function buildCaseShell(flags: FlagVisuals): THREE.Group {
   235→  const g = new THREE.Group();
   236→  g.name = 'case-shell';
   237→
   238→  const cx = (CASE.minX + CASE.maxX) / 2; // 1.05
   239→  const cy = (CASE.minY + CASE.maxY) / 2; // 2.25
   240→  const cz = (CASE.minZ + CASE.maxZ) / 2; // 2.3
   241→  const wX = CASE.maxX - CASE.minX;       // 2.1
   242→  const hY = CASE.maxY - CASE.minY;       // 4.5
   243→  const dZ = CASE.maxZ - CASE.minZ;       // 4.6
   244→  const wallMat = mat(COLORS.metalDark, 0.45, 0.55);
   245→  const trayWallMat = mat(0x2c343e, 0.5, 0.5); // جدار صينية اللوحة الأم — أفتح قليلًا لتمييزه
   246→
   247→  // الجدران (سماكة 0.05) — الجدار الأيمن عند x≈2.1 يستقبل اللوحة الأم
   248→  g.add(box(0.05, hY, dZ, trayWallMat, CASE.maxX - 0.025, cy, cz, true, true)); // يمين
   249→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.maxY - 0.025, cz, true, true)); // سقف
   250→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.minY + 0.025, cz, true, true)); // أرضية
   251→  g.add(box(wX, hY, 0.05, wallMat, cx, cy, CASE.minZ + 0.025, true, true)); // خلف
   252→  g.add(box(wX, hY, 0.05, mat(0x262d36, 0.35, 0.6), cx, cy, CASE.maxZ - 0.025, true, true)); // أمام
   253→
   254→  // حافة مشطوفة على مخطط الصينية (شريط معدني فاتح يفصل الصينية عن قصورة الكابلات)
   255→  g.add(box(0.02, hY * 0.92, 0.02, mat(0x414c58, 0.6, 0.4), CASE.maxX - 0.18, cy, CASE.maxZ - 0.62, false));
   256→
   257→  // قصورة I/O الخلفية (مستطيل داكن بارز قليلًا خارج اللوحة الخلفية)
   258→  g.add(box(0.55, 1.1, 0.03, mat(COLORS.black, 0.3, 0.6), 1.35, 3.35, CASE.minZ - 0.005, false));
   259→
   260→  // 7 مقابس توسعة أفقية (محور X) مكدسة رأسيًا أسفل I/O
   261→  const slotStripMat = mat(COLORS.metalMid, 0.6, 0.45);
   262→  for (let i = 0; i < 7; i++) {
   263→    g.add(box(0.72, 0.05, 0.03, slotStripMat, 1.35, 2.55 - i * 0.13, CASE.minZ - 0.005, false));
   264→  }
   265→
   266→  // شريط شبكة تهوية على الواجهة الأمامية (شرائط أفقية رفيعة)
   267→  const ventMat = mat(0x11151b, 0.3, 0.6);
   268→  for (let i = 0; i < 9; i++) {
   269→    g.add(box(1.5, 0.035, 0.03, ventMat, 1.05, 1.1 + i * 0.275, CASE.maxZ + 0.035, false));
   270→  }
   271→
   272→  // زر التشغيل (أسطوانة على السطح العلوي قرب الواجهة) + لمبة LED
   273→  g.add(cyl(0.07, 0.06, mat(COLORS.metalMid, 0.5, 0.4), 'y', 0.5, CASE.maxY + 0.005, CASE.maxZ - 0.45, false));
   274→  const ledMat = new THREE.MeshStandardMaterial({
   275→    color: 0x10151b, emissive: 0x000000, emissiveIntensity: 1.2, roughness: 0.35, metalness: 0.1, flatShading: false,
   276→  });
   277→  const powerLed = new THREE.Mesh(new THREE.SphereGeometry(0.03, 12, 8), ledMat); // مادة فريدة يتحكم بها علم powerOn
   278→  powerLed.position.set(0.78, CASE.maxY + 0.01, CASE.maxZ - 0.45);
   279→  flags.powerLed = powerLed;
   280→  g.add(powerLed);
   281→
   282→  // قاعدتان سفليتان
   283→  const footMat = mat(COLORS.black, 0.3, 0.6);
   284→  g.add(box(wX, 0.06, 0.45, footMat, cx, 0.03, CASE.minZ + 0.45, true));
   285→  g.add(box(wX, 0.06, 0.45, footMat, cx, 0.03, CASE.maxZ - 0.45, true));
   286→
   287→  // 6 مسامير تثبيت اللوحة الأم — أبناء الهيكل (تبقى في الهيكل عند فك اللوحة)
   288→  const ys = [BOARD.yMin + 0.1, (BOARD.yMin + BOARD.yMax) / 2, BOARD.yMax - 0.1];
   289→  const zs = [BOARD.zMin + 0.1, BOARD.zMax - 0.1];
   290→  for (const wy of ys) {
   291→    for (const wz of zs) {
   292→      const s = makeScrew(0.035, 0.3, 'x');
   293→      s.position.set(BOARD.x + 0.05, wy, wz);
   294→      flags.moboScrews.push(s);
   295→      g.add(s);
   296→    }
   297→  }
   298→
   299→  setComponentId(g, 'case');
   300→  return g;
   301→}

   329→// اللوحة الأم (dynamic.motherboard) — متمركزة عند MOUNTS.motherboard
   330→// محليًا: الوجه الأمامي نحو -X عند x = -thickness/2

   332→
   333→function buildMotherboard(flags: FlagVisuals): THREE.Group {
   334→  const g = new THREE.Group();
   335→  g.name = 'part-motherboard';
   336→  const face = -BOARD.thickness / 2;                  // -0.03
   337→  const onFace = (t: number): number => face - t / 2; // مركز قطعة سماكتها t على الوجه
   338→
   339→  // 1) جسم PCB (السماكة على X، الأبعاد من BOARD)
   340→  g.add(box(BOARD.thickness, BOARD.yMax - BOARD.yMin, BOARD.zMax - BOARD.zMin, mat(COLORS.pcb, 0.35, 0.6), 0, 0, 0, true, true));
   341→
   342→  // 1ب) تفاصيل SMD: رقائق صغيرة + مكثفات VRM تملأ الوجه الأمامي (مواقع خارج كل القطع الأخرى)
   343→  const chipMat = mat(0x11171e, 0.25, 0.5);
   344→  const capMat = mat(0x3a4a58, 0.55, 0.4);
   345→  const chipSpots: Array<readonly [number, number]> = [
   346→    [1.0, -0.8], [1.0, -0.65], [1.0, -0.5], [1.04, -0.35],          // صف VRM فوق المقبس
   347→    [0.45, 0.5], [0.55, 0.68], [0.4, 0.82], [0.7, 0.55], [0.85, 0.75], // يمين فتحات RAM
   348→    [-0.3, 0.82], [-0.42, 0.88],                                     // تحت فتحات RAM
   349→    [-1.38, -0.62], [-1.38, -0.34],                                  // أسفل فتحات PCIe
   350→  ];
   351→  for (const [sy, sz] of chipSpots) {
   352→    g.add(box(0.045, 0.09, 0.15, chipMat, onFace(0.045), sy, sz, false));
   353→  }
   354→  const capGeo = new THREE.CylinderGeometry(0.026, 0.026, 0.09, 10);
   355→  for (const cz of [-0.98, -0.84, -0.7, -0.56]) {
   356→    const c = new THREE.Mesh(capGeo, capMat);
   357→    c.rotation.z = Math.PI / 2;
   358→    c.position.set(onFace(0.09), 0.86, cz);
   359→    g.add(c);
   360→  }
   361→
   362→  // 2) مقبس المعالج: إطار + لوح معدني داخلي (عالمي y=3.45, z=0.95)
   363→  g.add(box(0.06, 0.42, 0.42, mat(COLORS.aluminum, 0.8, 0.45), onFace(0.06), 0.7, -0.62, true));
   364→  g.add(box(0.04, 0.36, 0.36, mat(0xc2c9d1, 0.9, 0.3), onFace(0.04) - 0.005, 0.7, -0.62, false));
   365→
   366→  // رافعة المقبس: pivot عند حافة المقبس (يدور حول X المحلي عند الفتح)
   367→  const lever = new THREE.Object3D();

   735→  g.add(box(0.12, 0.26, 0.2, mat(COLORS.black, 0.3, 0.5), 0, 0, 0, true, true));
   736→
   737→  // 8 أسلاك: صفان على Z × 4 على Y
   738→  const wireGeo = new THREE.BoxGeometry(0.04, 0.035, 0.035);
   739→  let idx = 1;
   740→  for (let i = 0; i < 4; i++) {
   741→    for (const zr of [-0.05, 0.05]) {
   742→      const wire = new THREE.Mesh(wireGeo, mat(WIRE_COLORS[idx % WIRE_COLORS.length], 0.2, 0.6));
   743→      wire.position.set(-0.08, -0.09 + i * 0.06, zr);
   744→      g.add(wire);
   745→      idx += 2;
   746→    }
   747→  }
   748→
   749→  // كابل منحدر من PSU (1.3,0.8,1.0) عبر (1.0,3.2,0.5) إلى الموضع
   750→  g.add(makeCable(cablePoints(MOUNTS.eps, [
   751→    [1.3, 0.8, 1.0],
   752→    [1.0, 3.2, 0.5],
   753→    [1.72, 4.02, 0.55],
   754→  ]), 0.04));
   755→
   756→  setComponentId(g, 'eps');
   757→  return g;
   758→}

   788→// ============================================================
   789→// صينية القطع (داخل root مباشرة — بلا componentId)
   790→// ============================================================
   791→
   792→function buildTrayTable(): THREE.Group {
   793→  const g = new THREE.Group();
   794→  g.name = 'tray-table'; // غير قابلة للاختيار عمدًا
   795→
   796→  const cx = (TRAY.minX + TRAY.maxX) / 2; // -3.35
   797→  const cz = (TRAY.minZ + TRAY.maxZ) / 2; // 2.4
   798→
   799→  // السطح (أعلى = TRAY.tableTopY = 0.9)
   800→  g.add(box(TRAY.maxX - TRAY.minX, 0.08, TRAY.maxZ - TRAY.minZ, mat(0x2b333d, 0.4, 0.6), cx, 0.86, cz, true, true));
   801→
   802→  // 4 أرجل أسطوانية
   803→  const legGeo = new THREE.CylinderGeometry(0.05, 0.05, 0.82, 12);
   804→  const legMat = mat(COLORS.metalMid, 0.6, 0.45);
   805→  for (const lx of [TRAY.minX + 0.15, TRAY.maxX - 0.15]) {
   806→    for (const lz of [TRAY.minZ + 0.15, TRAY.maxZ - 0.15]) {
   807→      const leg = new THREE.Mesh(legGeo, legMat);
   808→      leg.position.set(lx, 0.41, lz);
   809→      if (castEnabled) leg.castShadow = true;
   810→      g.add(leg);
   811→    }
   812→  }
   813→
   814→  // خطوط خانات باهتة لكل مكوّن (عدا case) — تُثبَّت داخل حدود الطاولة
   815→  const markers: Array<{ id: ComponentId; size: readonly [number, number] }> = [
   816→    { id: 'ram', size: [0.55, 1.55] },
   817→    { id: 'gpu', size: [1.15, 2.45] },
   818→    { id: 'cpu', size: [0.55, 0.55] },
   819→    { id: 'gpuPower', size: [0.4, 0.45] },
   820→    { id: 'cooler', size: [0.7, 0.95] },
   821→    { id: 'ssd', size: [0.3, 0.9] },
   822→    { id: 'fans', size: [1.0, 1.0] },
   823→    { id: 'psu', size: [1.5, 1.5] },
   824→    { id: 'motherboard', size: [2.05, 2.4] },
   825→    { id: 'atx24', size: [0.4, 0.6] },
   826→    { id: 'eps', size: [0.35, 0.5] },
   827→  ];
   828→  const markerMat = mat(0x24303a, 0.2, 0.8);
   829→  for (const mk of markers) {
   830→    const p = TRAY.slots[mk.id];
   831→    const px = Math.min(Math.max(p[0], TRAY.minX + mk.size[0] / 2 + 0.02), TRAY.maxX - mk.size[0] / 2 - 0.02);
   832→    const pz = Math.min(Math.max(p[2], TRAY.minZ + mk.size[1] / 2 + 0.02), TRAY.maxZ - mk.size[1] / 2 - 0.02);
   833→    g.add(box(mk.size[0], 0.012, mk.size[1], markerMat, px, 0.906, pz, false));
   834→  }
   835→
   836→  return g;
   837→}

   843→export function buildAllParts(quality: SceneQuality): PartVisuals {
   844→  castEnabled = quality !== 'low';
   845→
   846→  const flags: FlagVisuals = {
   847→    ramClips: [],
   848→    gpuScrews: [],
   849→    psuScrews: [],
   850→    fanScrews: [],
   851→    moboScrews: [],
   852→  };
   853→  const spinners: THREE.Object3D[] = [];
   854→
   855→  // بناء الأجزاء
   856→  const caseShell = buildCaseShell(flags);
   857→  const sidePanel = buildSidePanel();

   859→  const cpu = buildCpu();
   860→  const cooler = buildCooler(spinners);
   861→  const ram = buildRam();
   862→  const gpu = buildGpu(flags, spinners);
   863→  const psu = buildPsu(flags);
   864→  const ssd = buildSsd();
   865→  const fans = buildFans(flags, spinners);
   866→  const atx24 = buildAtx24();
   867→  const eps = buildEps();

   869→  const trayTable = buildTrayTable();
   870→
   871→  // مواضع التثبيت الابتدائية (المدير يعيد تموضعها عند الفك/التفجير)
   872→  const place = (grp: THREE.Group, id: ComponentId): void => {
   873→    grp.position.set(...MOUNTS[id]);
   874→  };
   875→  place(motherboard, 'motherboard');
   876→  place(cpu, 'cpu');
   877→  place(cooler, 'cooler');
   878→  place(ram, 'ram');
   879→  place(gpu, 'gpu');
   880→  place(psu, 'psu');
   881→  place(ssd, 'ssd');
   882→  place(fans, 'fans');
   883→  place(atx24, 'atx24');
   884→  place(eps, 'eps');
   885→  place(gpuPower, 'gpuPower');
   886→
   887→  // مجموعة placeholder للمكوّن 'case' (الهيكل نفسه لا يُحرك أبدًا — فارغة للأمان)
   888→  const caseAnchor = new THREE.Group();
   889→  caseAnchor.name = 'case-anchor';
   890→  place(caseAnchor, 'case');
   891→
   892→  const root = new THREE.Group();
   893→  root.name = 'pc-parts-root';
   894→  root.add(caseShell, sidePanel, trayTable, caseAnchor);
   895→  root.add(motherboard, cpu, cooler, ram, gpu, psu, ssd, fans, atx24, eps, gpuPower);

   900→    cpu,
   901→    cooler,
   902→    ram,
   903→    gpu,
   904→    psu,
   905→    ssd,
   906→    fans,
   907→    atx24,
   908→    eps,
   909→    gpuPower,
   910→  };
   911→
   912→  return { root, dynamic, caseShell, sidePanel, flags, spinners };
   913→}
