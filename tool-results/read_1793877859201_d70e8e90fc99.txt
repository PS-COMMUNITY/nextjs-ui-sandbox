    16→function btn(
    17→  os: VirtualOS, ctx: CanvasRenderingContext2D,
    18→  x: number, y: number, w: number, h: number, label: string,
    19→  onClick: () => void,
    20→  opts: { color?: string; fg?: string; disabled?: boolean; small?: boolean } = {},
    21→): void {
    22→  const hovered = !opts.disabled && os.hoverAt(x, y, w, h);
    23→  ctx.fillStyle = opts.disabled ? '#222b35' : hovered ? (opts.color ?? OS_COLORS.accent) : 'rgba(255,255,255,0.09)';
    24→  roundRect(ctx, x, y, w, h, 6);
    25→  ctx.fill();
    26→  ctx.strokeStyle = opts.disabled ? '#2a3440' : hovered ? 'rgba(255,255,255,0.35)' : OS_COLORS.winBorder;
    27→  roundRect(ctx, x, y, w, h, 6);
    28→  ctx.stroke();
    29→  ctx.fillStyle = opts.disabled ? '#5a6774' : (opts.fg ?? (hovered ? '#06231f' : OS_COLORS.text));
    30→  ctx.font = opts.small ? '12px sans-serif' : 'bold 13px sans-serif';
    31→  ctx.textAlign = 'center';
    32→  ctx.fillText(label, x + w / 2, y + h / 2 + 4.5);
    33→  ctx.textAlign = 'left';
    34→  if (!opts.disabled) {
    35→    os.addRect('button', x, y, w, h, { onClick }, 'pointer');
    36→  }
    37→}

    41→function textbox(
    42→  os: VirtualOS, ctx: CanvasRenderingContext2D,
    43→  x: number, y: number, w: number, h: number,
    44→  state: TextboxState,
    45→  opts: { mono?: boolean; onEnter?: () => void; placeholder?: string; autoFocus?: boolean } = {},
    46→): void {
    47→  const focused = os.focusedTextbox?.get === (() => state.text);
    48→  const hovered = os.hoverAt(x, y, w, h);
    49→  ctx.fillStyle = OS_COLORS.inputBg;
    50→  roundRect(ctx, x, y, w, h, 6);
    51→  ctx.fill();
    52→  ctx.strokeStyle = focused ? OS_COLORS.accent : hovered ? '#3d4c5c' : OS_COLORS.winBorder;
    53→  roundRect(ctx, x, y, w, h, 6);
    54→  ctx.stroke();
    55→  ctx.save();
    56→  ctx.beginPath();
    57→  roundRect(ctx, x + 2, y + 2, w - 4, h - 4, 5);
    58→  ctx.clip();
    59→  ctx.fillStyle = OS_COLORS.text;
    60→  ctx.font = opts.mono ? '13px monospace' : '13.5px sans-serif';
    61→  const show = state.text || (opts.placeholder && !focused ? opts.placeholder : '');
    62→  if (show) ctx.fillText(show, x + 10, y + h / 2 + 4.5);
    63→  if (focused && os.blink) {
    64→    const tw = ctx.measureText(state.text).width;
    65→    ctx.fillStyle = OS_COLORS.accent;
    66→    ctx.fillRect(x + 10 + tw + 1, y + 6, 2, h - 12);
    67→  }
    68→  ctx.restore();
    69→  const focus = (): void => {
    70→    os.focusedTextbox = {
    71→      get: () => state.text,
    72→      set: (t) => { state.text = t; },
    73→      onEnter: opts.onEnter,
    74→    };
    75→    os.dirty = true;
    76→  };
    77→  os.addRect('custom', x, y, w, h, { onClick: focus }, 'text');
    78→  if (opts.autoFocus && !os.focusedTextbox) focus();
    79→}

    81→function panelHeader(ctx: CanvasRenderingContext2D, w: number, titleAr: string, subAr?: string): number {
    82→  ctx.fillStyle = OS_COLORS.text;
    83→  ctx.font = 'bold 15px sans-serif';

    86→    ctx.fillStyle = OS_COLORS.dim;
    87→    ctx.font = '12px sans-serif';
    88→    ctx.fillText(subAr, 14, 42);

    90→  ctx.strokeStyle = OS_COLORS.winBorder;
    91→  ctx.beginPath();
    92→  ctx.moveTo(0, 52.5);
    93→  ctx.lineTo(w, 52.5);
    94→  ctx.stroke();
    95→  return 64;

   135→function fsGet(root: FsNode, path: string[]): FsNode | null {
   136→  let cur: FsNode = root;
   137→  for (const seg of path) {
   138→    const next = cur.children?.find((c) => c.name === seg);
   139→    if (!next) return null;
   140→    cur = next;
   141→  }
   142→  return cur;
   143→}

   389→/** منفذ الأوامر التعليمي — كل الأوامر داخل المحاكاة فقط (توجيه §12: لا أوامر نظام حقيقية) */
   390→function runCommand(os: VirtualOS, win: VosWindow, st: { lines: string[]; input: string; cwd: string }, raw: string): void {
   391→  const store = useLabStore.getState();
   392→  const [cmd, ...rest] = raw.split(/\s+/);
   393→  const arg = rest.join(' ');
   394→  const out = (...lines: string[]): void => { st.lines.push(...lines); };
   395→  switch ((cmd ?? '').toLowerCase()) {
   396→    case 'help':
   397→      out('الأوامر المتاحة (محاكاة تعليمية):',
   398→        '  help, cls, echo, ver, systeminfo, ipconfig, ping',
   399→        '  dir, cd <folder>, type <file.txt>',
   400→        '  driverquery, diskpart, tasklist',
   401→        '  sfc /scannow, chkdsk C:',
   402→        '  bootrec /fixmbr | /fixboot | /rebuildbcd   (فقط من بيئة الاسترداد)',
   403→        '  shutdown /r — إعادة تشغيل الجهاز الافتراضي, exit');
   404→      break;
   405→    case 'cls':
   406→      st.lines = [];
   407→      break;
   408→    case 'echo':
   409→      out(arg || '');
   410→      break;
   411→    case 'ver':
   412→      out(store.sim.os.installed ? `PC Lab OS (محاكاة تعليمية بنمط ${store.sim.os.installed}) — Build 6.0.0-sim` : 'لا نظام مثبت — وسيط التثبيت مطلوب');
   413→      break;
   414→    case 'systeminfo': {
   415→      const t = store.sim.thermal;
   416→      out('Host Name:                 PC-LAB-01',
   417→        `OS Name:                   ${store.sim.os.installed ? 'Virtual Windows (' + store.sim.os.installed + ')' : '—'}`,
