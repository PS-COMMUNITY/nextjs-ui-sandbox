    86→    86→function StatusBadge({ id }: { id: ComponentId }) {
    87→    87→  const sim = useLabStore((s) => s.sim);
    88→    88→  if (id === 'case') {
    89→    89→    return <Badge variant="outline" className="gap-1">هيكل الجهاز</Badge>;
    90→    90→  }
    91→    91→  const secureOpen = (SECURE_FLAGS[id] ?? []).some((f) => sim.flags[f]);
    92→    92→  const state = assemblyStateOf(sim.parts[id], secureOpen);
    93→    93→  const info = ASSEMBLY_STATE_INFO[state];
    94→    94→  const colorClass: Record<string, string> = {
    95→    95→    emerald: 'bg-emerald-600/15 text-emerald-400 border-emerald-600/40',
    96→    96→    amber: 'bg-amber-600/15 text-amber-400 border-amber-600/40',
    97→    97→    slate: 'bg-slate-600/15 text-slate-300 border-slate-600/40',
    98→    98→    cyan: 'bg-cyan-600/15 text-cyan-400 border-cyan-600/40',
    99→    99→    rose: 'bg-rose-600/15 text-rose-400 border-rose-600/40',
   100→   100→  };
   101→   101→  const icon =
   102→   102→    state === 'INSTALLED_CORRECTLY' ? <CheckCircle2 className="h-3 w-3" aria-hidden /> :
   103→   103→    state === 'AWAITING_INSTALL' ? <HandMetal className="h-3 w-3" aria-hidden /> :
   104→   104→    state === 'TRAINING_DAMAGE' ? <TriangleAlert className="h-3 w-3" aria-hidden /> :
   105→   105→    <Wrench className="h-3 w-3" aria-hidden />;
   106→   106→  return (
   107→   107→    <Badge variant="outline" className={cn('gap-1', colorClass[info.colorAr] ?? '')}>
   108→   108→      {icon}
   109→   109→      {state === 'INSTALLED_CORRECTLY' ? 'مثبتة بإحكام' : state === 'INSTALLED_SECURED' ? 'مثبتة (مزالج مفتوحة)' : info.labelAr}
   110→   110→    </Badge>
   111→   111→  );
   112→   112→}

   115→   115→function ChecklistRow({
   116→   116→  item,
   117→   117→  mode,
   118→   118→  onFlag,
   119→   119→  onPart,
   120→   120→}: {
   121→   121→  item: ChecklistItem;
   122→   122→  mode: 'remove' | 'install';
   123→   123→  onFlag: (f: MachineFlag) => void;
   124→   124→  onPart: (id: ComponentId) => void;
   125→   125→}) {
   126→   126→  const isFlag = item.source !== 'safetyOff' && item.source !== 'safetyUnplug' && item.source !== 'base';
   127→   127→  const canRunFlag = !item.done && isFlag && item.available;
   128→   128→  const canRunPart = !item.done && !isFlag && item.actionPart && item.available;
   129→   129→  return (
   130→   130→    <li
   131→   131→      className={cn(
   132→   132→        'flex items-center gap-2 rounded-md border px-2.5 py-2 text-xs',
   133→   133→        item.done ? 'border-emerald-600/30 bg-emerald-600/10' : 'border-border bg-card',
   134→   134→        !item.done && !item.available && 'opacity-60',
   135→   135→      )}
   136→   136→    >
   137→   137→      {item.done ? (
   138→   138→        <CheckCircle2 className="h-4 w-4 shrink-0 text-emerald-500" aria-hidden />
   139→   139→      ) : (
   140→   140→        <CircleDashed className="h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   141→   141→      )}
   142→   142→      <div className="min-w-0 flex-1">
   143→   143→        <p className={cn('font-medium leading-tight', item.done && 'line-through opacity-70')}>{item.labelAr}</p>
   144→   144→        {!item.done && item.hintAr ? (
   145→   145→          <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">{item.hintAr}</p>
   146→   146→        ) : null}
   147→   147→      </div>
   148→   148→      {canRunFlag ? (
