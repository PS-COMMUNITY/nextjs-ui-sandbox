   148→export function isWebGLAvailable(): boolean {
   149→  try {
   150→    const canvas = document.createElement('canvas');
   151→    return Boolean(canvas.getContext('webgl2') || canvas.getContext('webgl'));
   152→  } catch {
   153→    return false;
   154→  }
   155→}

   157→/** تخلّص من مادة وكل خرائطها النسيجية (يُستخدم في dispose) */
   158→function disposeMaterial(mat: THREE.Material): void {
   159→  for (const value of Object.values(mat) as unknown[]) {
   160→    if (value instanceof THREE.Texture) value.dispose();
   161→  }
   162→  mat.dispose();
   163→}

   244→  constructor(init: { container: HTMLDivElement; quality: SceneQuality; callbacks: PCSceneCallbacks }) {
   245→    this.container = init.container;
   246→    this.quality = init.quality;
   247→    this.callbacks = init.callbacks;
   248→
   249→    // (1) فحص WebGL قبل أي شيء — عند الفشل نعلّم failed ونبلّغ دون بناء
   250→    if (!isWebGLAvailable()) {
   251→      this.fail(WEBGL_ERROR_AR);
   252→      return;
   253→    }
   254→
   255→    try {
   256→      const w = Math.max(1, this.container.clientWidth);
   257→      const h = Math.max(1, this.container.clientHeight);
   258→
   259→      // (2) المُصيِّر — نغمة سينمائية ACES لدرجات لون واقعية
   260→      const renderer = new THREE.WebGLRenderer({ antialias: true });
   261→      renderer.setClearColor(COLORS.bg);
   262→      renderer.shadowMap.enabled = this.quality !== 'low';
   263→      renderer.shadowMap.type = THREE.PCFSoftShadowMap;
   264→      renderer.toneMapping = THREE.ACESFilmicToneMapping;
   265→      renderer.toneMappingExposure = TONE_EXPOSURE;
   266→      renderer.setSize(w, h, false);
   267→      renderer.domElement.style.width = '100%';
   268→      renderer.domElement.style.height = '100%';
   269→      renderer.domElement.style.display = 'block';
   270→      this.renderer = renderer;
   271→      // الحاوية position:relative يديرها الاستدعاء — اللوحة تملأها بالكامل
   272→      this.container.appendChild(renderer.domElement);
   273→
   274→      // (3) المشهد والكاميرا والتحكم المداري
   275→      this.scene = new THREE.Scene();
   276→      // خريطة بيئية (IBL) عبر PMREM — تجعل المعادن تعكس مثل الواقع بدل أن تبدو سوداء
   277→      const pmrem = new THREE.PMREMGenerator(renderer);
   278→      this.scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
   279→      this.scene.environmentIntensity = ENV_INTENSITY;
   280→      pmrem.dispose();
   281→      this.camera = new THREE.PerspectiveCamera(40, w / h, 0.1, 100);
   282→      this.camera.position.set(...DEFAULT_CAMERA.pos);
   283→      this.controls = new OrbitControls(this.camera, renderer.domElement);
   284→      this.controls.target.set(...DEFAULT_CAMERA.target);
   285→      this.controls.enableDamping = true;
   286→      this.controls.dampingFactor = 0.08;
   287→      this.controls.minDistance = 2.2;
   288→      this.controls.maxDistance = 22;
   289→      this.controls.maxPolarAngle = 1.45;
   290→      this.controls.update();
   291→      // تفاعل المستخدم يلغي أي تنعيم كاميرا جارٍ
   292→      this.controls.addEventListener('start', () => {
   293→        this.camTween = null;
   294→      });
   295→
   296→      // (4) البيئة ثم القطع — بنداءات المصنع المتفق عليها
   297→      buildEnvironment(this.scene, this.quality);
   298→      const visuals = buildAllParts(this.quality);
   299→      this.scene.add(visuals.root);
   300→      this.visuals = visuals;
   301→
   302→      // ضوء اتجاهي (أضافه buildEnvironment) للتحكم به عند تبديل الظلال
   303→      this.scene.traverse((obj) => {
   304→        if (!this.dirLight && obj instanceof THREE.DirectionalLight) this.dirLight = obj;
   305→      });
   306→
   307→      // تسجيل المعرفات دفاعيًا (المصنع يفعل ذلك عادةً — إعادة النداء غير ضارة)
   308→      for (const id of COMPONENT_IDS) setComponentId(visuals.dynamic[id], id);
   309→      setComponentId(visuals.caseShell, 'case');
   310→      setComponentId(visuals.sidePanel, 'case');
   311→
   312→      this.initDynamicState(visuals);
   313→      this.initFlagRecords(visuals);
   314→
   315→      // (الجلسة 4) الشاشة الافتراضية: نسيج ديناميكي + مرجع كابل الإشارة
   316→      this.painter = createScreenPainter();
   317→      if (this.painter) {
   318→        const screenMesh = visuals.monitor.getObjectByName('monitor-screen');
   319→        if (screenMesh instanceof THREE.Mesh) {
   320→          const smat = screenMesh.material;
   321→          if (!Array.isArray(smat) && 'map' in smat) {
   322→            (smat as THREE.MeshBasicMaterial).map = this.painter.texture;
   323→            (smat as THREE.MeshBasicMaterial).color.set(0xffffff);
   324→            (smat as THREE.MeshBasicMaterial).needsUpdate = true;
   325→          }
   326→        }
   327→        this.drawMonitor(true);
   328→      }
   329→      this.monitorCableGroup = visuals.monitor.getObjectByName('monitor-cable') ?? null;
   330→
   331→      // أصل حقيقي CC0 (Poly Haven): مسح لوحة دوائر كأساس PCB واقعي + مفك تدريبي
   332→      this.loadRealAssets(visuals);
   333→
   334→      // (7)(15) مستمعو المؤشر + مراقب الحجم
   335→      renderer.domElement.addEventListener('pointerdown', this.onPointerDown);
   336→      renderer.domElement.addEventListener('pointerup', this.onPointerUp);
   337→      renderer.domElement.addEventListener('pointermove', this.onPointerMove);
   338→      this.resizeObserver = new ResizeObserver(() => this.onResize());
   339→      this.resizeObserver.observe(this.container);
   340→
   341→      // (13) ضبط كثافة البكسل حسب الجودة (الظلال مضبوطة أعلاه بنفس القيمة)
   342→      this.setQuality(this.quality);
   343→
   344→      // خطاف تشخيصي للتطوير فقط (لا يتغير سلوك التطبيق)
   345→      if (typeof window !== 'undefined') {
   346→        (window as unknown as Record<string, unknown>)['__pcScene'] = this;
   347→      }
   348→
   349→      // (6) بدء حلقة الرسم
   350→      this.rafId = requestAnimationFrame(this.tick);
   351→      this.callbacks.onReady?.();
   352→    } catch {
   353→      // المعالجة نفسها: رسالة + failed=true
   354→      this.fail(WEBGL_ERROR_AR);
   355→    }
   356→  }

   362→  /** (13) تبديل الجودة: كثافة البكسل + تشغيل/إيقاف الظلال */
   363→  setQuality(q: SceneQuality): void {
   364→    if (this.blocked) return;
   365→    this.quality = q;
   366→    this.renderer.setPixelRatio(this.pixelRatioFor(q));
   367→    const wantShadows = q !== 'low';
   368→    if (this.renderer.shadowMap.enabled !== wantShadows) {
   369→      this.renderer.shadowMap.enabled = wantShadows;
   370→      this.renderer.shadowMap.needsUpdate = true;
   371→      if (this.dirLight) this.dirLight.castShadow = wantShadows;
   372→    }
   373→  }
   374→
   375→  /** (10) شفافية الهيكل والغطاء (X-Ray) مع استعادة الأصول عند الإيقاف */
   376→  setXray(on: boolean): void {
   377→    if (this.blocked) return;
   378→    if (this.xrayOn === on) return;
   379→    this.xrayOn = on;
   380→    this.refreshShellMaterials();
   381→  }
   382→
   383→  /** (5) فتح/إغلاق الغطاء الجانبي حول مفصلته الخلفية */
   384→  setPanelOpen(on: boolean, animate?: boolean): void {
   385→    if (this.blocked) return;
   386→    this.panelTarget = on ? PANEL_OPEN_ANGLE : 0;
   387→    if (this.lastState) this.lastState.flags.panelOpen = on;
   388→    if (animate === false) this.visuals.sidePanel.rotation.y = this.panelTarget;
   389→  }
   390→
   391→  /** (5) تشغيل/إيقاف العرض التفجيري — يعيد حساب الأهداف من آخر حالة متزامنة */
   392→  setExploded(on: boolean, animate?: boolean): void {
   393→    if (this.blocked) return;
   394→    this.exploded = on;
   395→    this.recomputeTargets();
   396→    if (animate === false) this.snapToTargets();
   397→  }
   398→
   399→  /** (5) مزامنة حالة المحاكاة: أهداف الحركة (تنعيم) + تطبيق الأعلام فورًا */
   400→  syncState(s: SimulationState): void {
   401→    if (this.blocked) return;
   402→    // نسخة دفاعية حتى لا تؤثر طفرات المتصل على أهدافنا
   403→    this.lastState = {
   404→      flags: { ...s.flags },
   405→      parts: { ...s.parts },
   406→      peripherals: { ...s.peripherals },
   407→    };
   408→    this.recomputeTargets();
   409→    this.applyFlags(s.flags, true);
   410→    this.updateRuntimeVisuals(s);
   411→    this.drawMonitor(true);
   412→  }

   568→  /** (6) حلقة الرسم: دوّارات، اقتراب القطع، تنعيم الغطاء، تنعيم الكاميرا، ثم الرسم */
   569→  private readonly tick = (): void => {
   570→    if (this.disposed || this.failed) return;
   571→    this.rafId = requestAnimationFrame(this.tick);
   572→    const dt = Math.min(this.clock.getDelta(), 0.05); // حماية من قفزات التبويبات الخاملة
   573→    const k = 1 - Math.exp(-LERP_RATE * dt);
   574→
   575→    // دوران شفرات المراوح حول محورها المحفوظ — فقط والجهاز يعمل فعلاً
   576→    if (this.spinning) {
   577→      for (const spinner of this.spinners) {
   578→        spinner.obj.rotateOnAxis(spinner.axis, SPIN_SPEED * dt);
   579→      }
   580→    }
   581→
   582→    // اقتراب القطع نحو أهدافها (الموضع lerp والدوران slerp) — عدا الهيكل الثابت
   583→    for (const id of COMPONENT_IDS) {
   584→      if (id === 'case') continue;
   585→      const part = this.parts[id];
   586→      part.group.position.lerp(part.targetPos, k);
   587→      part.group.quaternion.slerp(part.targetQuat, k);
   588→    }
   589→
   590→    // تطفو القطعة الملتقطة (awaiting) فوق الصينية بلطف — إشارة بصرية لجاهزية التركيب
   591→    const elapsed = this.clock.getElapsedTime();
   592→    for (const id of COMPONENT_IDS) {
   593→      if (id === 'case') continue;
   594→      if (this.lastState && this.lastState.parts[id] === 'awaiting' && this.dragId !== id) {
   595→        const part = this.parts[id];
   596→        part.group.position.y = part.trayPos.y + 0.16 + Math.sin(elapsed * 3) * 0.04;
   597→      }
   598→    }
   599→
   600→    // تنعيم دوران الغطاء الجانبي نحو الهدف
   601→    const panel = this.visuals.sidePanel;
   602→    panel.rotation.y += (this.panelTarget - panel.rotation.y) * k;
   603→
   604→    // استكمال تنعيم الكاميرا (resetCamera / focusComponent)
   605→    const tween = this.camTween;
   606→    if (tween) {
   607→      tween.t = Math.min(tween.t + dt / tween.dur, 1);
   608→      const eased = PCScene.easeInOut(tween.t);
   609→      this.camera.position.lerpVectors(tween.fromPos, tween.toPos, eased);
   610→      this.controls.target.lerpVectors(tween.fromTarget, tween.toTarget, eased);
   611→      if (tween.t >= 1) this.camTween = null;
   612→    }
   613→
   614→    this.controls.update();
   615→
   616→    // إعادة رسم نسيج الشاشة الافتراضية بمعدل مقيد
   617→    if (this.painter && nowMs() - this.lastMonitorDrawAt >= MONITOR_REDRAW_MS) {
   618→      this.drawMonitor(false);
   619→    }
   620→
   621→    this.renderer.render(this.scene, this.camera);
   622→  };
   623→
   624→  /** تنعيم easeInOut: t<0.5 ? 2t² : 1 − (−2t+2)²/2 */
   625→  private static easeInOut(t: number): number {
   626→    return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
   627→  }

   629→  /** إعادة حساب أهداف القطع من آخر حالة متزامنة + علم التفجير الحالي */
   630→  private recomputeTargets(): void {
   631→    const parts = this.lastState?.parts;
   632→    for (const id of COMPONENT_IDS) {
   633→      if (id === 'case') continue;
   634→      const part = this.parts[id];
   635→      const status: string | undefined = parts ? parts[id] : undefined;
   636→      if (status === 'removed' || status === 'damaged') {
   637→        part.targetPos.copy(part.trayPos);
   638→        part.targetRot.copy(part.trayRot);
   639→        if (status === 'damaged') {
   640→          // القطعة التالفة تستلقي مائلة على الصينية
   641→          part.targetRot.z += 0.35;
   642→        }
   643→      } else if (status === 'awaiting') {
   644→        // ملتقطة من الصينية — ترتفع قليلاً (وتطفو في حلقة الرسم)
   645→        part.targetPos.copy(part.trayPos);
   646→        part.targetPos.y += 0.16;
   647→        part.targetRot.copy(part.trayRot);
   648→      } else if (status === 'separated') {
   649→        // فصل جزئي: ما زالت في موضعها مع إزاحة صغيرة نحو اتجاه الفك
   650→        part.targetPos.copy(part.basePos);
   651→        part.targetPos.addScaledVector(part.explodeVec, SEPARATE_FACTOR);
   652→        part.targetRot.set(0, 0, 0);
   653→      } else {
   654→        part.targetPos.copy(part.basePos);
   655→        if (this.exploded) part.targetPos.add(part.explodeVec);
   656→        part.targetRot.set(0, 0, 0);
   657→      }
   658→      part.targetQuat.setFromEuler(part.targetRot);
   659→    }
   660→  }

   673→  /**
   674→   * (5) تطبيق الأعلام فورًا على المجسمات الفرعية.
   675→   * افتراضات المحاور (لأن بناء المصنع موازٍ ولم يُثبَّت بعد):
   676→   *  - مزلاج PCIe ورافعة CPU: دوران حول Y المحلي (اللوح عمودي يواجه -X).
   677→   *  - مزالج RAM: دوران حول Z بإشارة متناوبة بين الطرفين.
   678→   *  - البراغي: ترتفع على +Y (مسامير اللوحة تُسحب على -X) مع دوران طفيف حول Z.
   679→   *  - القوابس: إزاحة -X 0.12 عند الفصل.
   680→   * كل الإزاحات نسبية إلى القيم الأصلية المحفوظة عند البناء.
   681→   */
   682→  private applyFlags(flags: Record<MachineFlag, boolean>, animate: boolean): void {
   683→    // الغطاء الجانبي — عبر هدف تنعيمي تتبعه الحلقة
   684→    this.panelTarget = flags.panelOpen ? PANEL_OPEN_ANGLE : 0;
   685→    if (!animate) this.visuals.sidePanel.rotation.y = this.panelTarget;
   686→
   687→    // مزلاج PCIe
   688→    if (this.latchRec) {
   689→      this.latchRec.obj.rotation.y = this.latchRec.baseRot + (flags.pcieLatchOpen ? LATCH_OPEN_ANGLE : 0);
   690→    }
   691→
   692→    // مزالج RAM
   693→    for (const clip of this.clipRecs) {
   694→      clip.obj.rotation.z = clip.baseRotZ + (flags.ramClipsOpen ? clip.sign * CLIP_OPEN_ANGLE : 0);
   695→    }
   696→
   697→    // رافعة مقبس CPU
   698→    if (this.leverRec) {
   699→      this.leverRec.obj.rotation.y = this.leverRec.baseRot + (flags.cpuLeverOpen ? LEVER_OPEN_ANGLE : 0);
   700→    }
   701→
   702→    // البراغي (gpu/psu/fan/m2/mobo)
   703→    for (const [flag, recs] of this.screwMap) {
   704→      const out = flags[flag];
   705→      for (const rec of recs) {
   706→        rec.obj.position.copy(rec.basePos);
   707→        if (out) {
   708→          rec.obj.position[rec.axis] += rec.delta;
   709→          rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN;
   710→        } else {
   711→          rec.obj.rotation.z = rec.baseRotZ;
   712→        }
   713→      }
   714→    }
   715→
   716→    // قوابس الموصلات (fan headers / front panel)
   717→    for (const plug of this.plugRecs) {
   718→      plug.obj.position.x = plug.baseX - (flags[plug.flag] ? PLUG_PULL : 0);
   719→    }
   720→
   721→    // لمبة الطاقة — مادة خاصة بها (cached) — تنطفئ مع مزود ميت
   722→    const led = this.visuals.flags.powerLed;
   723→    if (led) {
   724→      const mat = led.material;
   725→      if (!Array.isArray(mat) && 'emissive' in mat) {
   726→        const std = mat as THREE.MeshStandardMaterial;
   727→        const alive = flags.powerOn && !(this.lastState?.peripherals.psuDead ?? false);
   728→        std.emissive.set(alive ? COLORS.green : 0x000000);
   729→        std.emissiveIntensity = LED_INTENSITY;
   730→      }
   731→    }
   732→  }

   754→  /** تطبيق شفافية على مواد مجموعة (الزجاج شفاف أصلًا فيبقى كما هو) */
   755→  private applyShellOpacity(root: THREE.Object3D, opacity: number): void {
   756→    root.traverse((obj) => {
   757→      if (!(obj instanceof THREE.Mesh)) return;
   758→      const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   759→      for (const mat of mats) {
   760→        if (mat.transparent) continue; // الزجاج/الشفاف الأصلي لا يُمسّ
   761→        if (!this.shellOriginals.has(mat)) {
   762→          this.shellOriginals.set(mat, {
   763→            opacity: mat.opacity,
   764→            transparent: mat.transparent,
   765→            depthWrite: mat.depthWrite,
   766→          });
   767→        }
   768→        mat.transparent = true;
   769→        mat.opacity = opacity;
   770→        mat.depthWrite = false;
   771→      }
   772→    });
   773→  }

   788→  /**
   789→   * تحميل الأصول الحقيقية المرخصة CC0 (توجيه v3 §3 + v4 أ.4) — فشلها لا يعطل المشهد:
   790→   *  1) مسح لوحة دوائر (Poly Haven — circuit_board): يستبدل وجه PCB الإجرائي للوحة الأم.
   791→   *  2) مفك مسطّح (Poly Haven — screwdriver): أداة تدريب على سجادة القطع.
   792→   *  3) طقم مفكات (screwdrivers_02) + كماشات (pliers): أدوات حقيقية على السجادة.
   793→   *  4) مصباح مكتب (desk_lamp_arm_01): إضاءة مكتب واقعية فوق السجادة + ضوء نقطي دافئ.
   794→   *  5) صندوق عدة معدني (metal_tool_chest): تحت المكتب — تفصيل ورشة.
   795→   */
   796→  private loadRealAssets(visuals: PartVisuals): void {
   797→    const loader = new GLTFLoader();
   798→    const root = visuals.root;
   799→
   800→    // 1) أساس PCB الحقيقي — يستبدل وجه اللوحة الأمامي فقط، ويبقى الباقي إجرائيًا تفاعليًا
   801→    loader.load(
   802→      '/models/circuit-board/circuit_board_1k.gltf',
   803→      (gltf) => {
   804→        if (this.disposed || this.failed) return;
   805→        const pcb = visuals.dynamic.motherboard.getObjectByName('mobo-pcb');
   806→        if (!pcb) return;
   807→        const scan = gltf.scene;
   808→        scan.traverse((obj) => {
   809→          if (!(obj instanceof THREE.Mesh)) return;
   810→          obj.castShadow = false;
   811→          obj.receiveShadow = true;
   812→          // توحيد الخامات: منع أي emissive قد يأتي من الأصل (يبقى الإبراز صادقًا)
   813→          const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   814→          for (const m of mats) {
   815→            if ('emissiveIntensity' in m) (m as THREE.MeshStandardMaterial).emissiveIntensity = 0;
   816→          }

  1010→  /** إيجاد الهدف عند نقطة شاشة: أول إصابة مرئية غير شفافة — قطعة PC أو طرفية */
  1011→  private pickAt(clientX: number, clientY: number): PickedInfo {
  1012→    const rect = this.renderer.domElement.getBoundingClientRect();
  1013→    const w = rect.width > 0 ? rect.width : 1;
  1014→    const h = rect.height > 0 ? rect.height : 1;
  1015→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1016→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1017→    const hits = this.raycaster.intersectObjects(this.visuals.root.children, true);
  1018→    for (const hit of hits) {
  1019→      if (this.isHidden(hit.object)) continue;        // تجاهل ما خفيه العزل
  1020→      if (this.isSeeThrough(hit.object)) continue;    // الزجاج/الشفاف لا يحجب الاختيار
  1021→      let cur: THREE.Object3D | null = hit.object;
  1022→      while (cur) {
  1023→        // الطرفيات (زر الشاشة / زر طاقة الشاشة) — لا كيبورد ولا ماوس (توجيه v3 أ.1)
  1024→        const per: unknown = cur.userData['peripheral'];
  1025→        if (typeof per === 'string') {
  1026→          return { type: 'periph', kind: per as PeripheralKind };
  1027→        }
  1028→        const cid: unknown = cur.userData['componentId'];
  1029→        if (typeof cid === 'string' && (COMPONENT_IDS as readonly string[]).includes(cid)) {
  1030→          return { type: 'part', id: cid as ComponentId };
  1031→        }
  1032→        cur = cur.parent;
  1033→      }
  1034→      // إصابة أولى بلا معرف (عدة/زينة) — تمرير النقر عبرها إلى ما خلفها (توجيه v4)
  1035→      continue;
  1036→    }
  1037→    return null;
  1038→  }

  1130→  /** استنتاج محور الدوران: أصغر بُعد في صندوق الإحاطة للشبك الأول هو عمود القرص */
  1131→  private computeSpinnerAxis(spinner: THREE.Object3D): THREE.Vector3 {
  1132→    const fallback = new THREE.Vector3(0, 1, 0);
  1133→    const mesh = this.firstMesh(spinner);
  1134→    if (!mesh || !mesh.geometry) return fallback;
  1135→    const geometry = mesh.geometry;
  1136→    if (!geometry.boundingBox) geometry.computeBoundingBox();
  1137→    const bb = geometry.boundingBox;
  1138→    if (!bb) return fallback;
  1139→    const ex = bb.max.x - bb.min.x;
  1140→    const ey = bb.max.y - bb.min.y;
  1141→    const ez = bb.max.z - bb.min.z;
  1142→    const axis = new THREE.Vector3(
  1143→      ex <= ey && ex <= ez ? 1 : 0,
  1144→      ey <= ex && ey <= ez ? 1 : 0,
  1145→      ez <= ex && ey <= ez ? 1 : 0,
  1146→    );
  1147→    if (axis.lengthSq() === 0) return fallback;
  1148→    // تحويل المحور من فضاء الشبك إلى الفضاء المحلي للدوّار إن كان ابنًا مُدارًا
  1149→    if (mesh !== spinner) {
  1150→      spinner.updateWorldMatrix(true, false);
  1151→      mesh.updateWorldMatrix(true, false);
  1152→      const qSpinner = new THREE.Quaternion();
  1153→      const qMesh = new THREE.Quaternion();
  1154→      spinner.getWorldQuaternion(qSpinner);
  1155→      mesh.getWorldQuaternion(qMesh);
  1156→      axis.applyQuaternion(qSpinner.invert().multiply(qMesh));
  1157→    }
  1158→    return axis.normalize();
  1159→  }

  1209→  private readonly onPointerUp = (event: PointerEvent): void => {
  1210→    if (this.blocked) return;
  1211→    if (this.dragId) {
  1212→      this.finishDrag(event.clientX, event.clientY);
  1213→      this.pointerDownActive = false;
  1214→      return;
  1215→    }
  1216→    this.pendingDrag = null;
  1217→    if (!this.pointerDownActive) return;
  1218→    this.pointerDownActive = false;
  1219→    const dx = event.clientX - this.downX;
  1220→    const dy = event.clientY - this.downY;
  1221→    if (dx * dx + dy * dy >= CLICK_MAX_DIST_SQ) return; // سحب لا نقرة
  1222→    const picked = this.pickAt(event.clientX, event.clientY);
  1223→    if (picked && picked.type === 'periph') {
  1224→      this.callbacks.onPeripheral?.(picked.kind, picked.detail);
  1225→      return;
  1226→    }
  1227→    this.callbacks.onSelect(picked ? picked.id : null);
  1228→  };
  1229→
  1230→  /** (7) تحويم/سحب — بفحص مُقيَّد زمنيًا للتحويم (100ms) */
  1231→  private readonly onPointerMove = (event: PointerEvent): void => {
  1232→    if (this.blocked) return;
  1233→    if (this.dragId) {
  1234→      this.updateDrag(event.clientX, event.clientY);
  1235→      return;
  1236→    }
  1237→    if (this.pendingDrag) {
  1238→      const dx = event.clientX - this.pendingDrag.startX;
  1239→      const dy = event.clientY - this.pendingDrag.startY;
  1240→      if (dx * dx + dy * dy >= DRAG_START_DIST_SQ) {
  1241→        this.beginDrag(this.pendingDrag.id, event.clientX, event.clientY);
  1242→        this.updateDrag(event.clientX, event.clientY);
  1243→      }
  1244→      return;
  1245→    }
  1246→    const now = performance.now();
  1247→    if (now - this.lastHoverAt < HOVER_THROTTLE_MS) return;
  1248→    this.lastHoverAt = now;
  1249→    this.renderer.domElement.style.cursor = this.pickAt(event.clientX, event.clientY) ? 'pointer' : '';
  1250→  };
  1251→
  1252→  /** دبل-كليك: إعادة الكاميرا إلى الوضع الافتراضي (موثقة في الإعدادات) */
  1253→  private readonly onDoubleClick = (): void => {
  1254→    if (this.blocked) return;
  1255→    this.resetCamera();
  1256→  };

  1262→  /** بدء سحب فعلي: تعطيل المداري + سطح أفقي عند ارتفاع القطعة */
  1263→  private beginDrag(id: ComponentId, clientX: number, clientY: number): void {
  1264→    const part = this.parts[id];
  1265→    if (!part) return;
  1266→    this.pendingDrag = null;
  1267→    this.dragId = id;
  1268→    this.controls.enabled = false;
  1269→    this.dragPlane.set(new THREE.Vector3(0, 1, 0), -part.group.position.y);
  1270→    // إزاحة الإمساك: فرق نقطة النقر عن مركز القطعة على السطح نفسه
  1271→    const hit = this.rayPlaneHit(clientX, clientY);
  1272→    if (hit) this.dragOffset.copy(part.group.position).sub(hit);
  1273→    this.renderer.domElement.style.cursor = 'grabbing';
  1274→  }
  1275→
  1276→  /** تحديث موضع القطعة أثناء السحب (بصريًا فقط — الحالة المنطقية لا تتغير هنا) */
  1277→  private updateDrag(clientX: number, clientY: number): void {
  1278→    const id = this.dragId;
  1279→    if (!id) return;
  1280→    const hit = this.rayPlaneHit(clientX, clientY);
  1281→    if (!hit) return;
  1282→    const part = this.parts[id];
  1283→    part.group.position.set(hit.x + this.dragOffset.x, part.group.position.y, hit.z + this.dragOffset.z);
  1284→    // الهدف يتبع اليد — يمنع حلقة الرسم من محاربة السحب
  1285→    part.targetPos.copy(part.group.position);
  1286→  }
  1287→
  1288→  /** إنهاء السحب: تحديد منطقة الإفلات ثم إبلاغ المنطق (التحقق أولًا ثم الحركة — توجيه v3 §6) */
  1289→  private finishDrag(clientX: number, clientY: number): void {
  1290→    const id = this.dragId;
  1291→    this.dragId = null;
  1292→    this.controls.enabled = true;
  1293→    this.renderer.domElement.style.cursor = '';
  1294→    if (!id) return;
  1295→    const hit = this.rayPlaneHit(clientX, clientY);
  1296→    let zone: DropZone = null;
  1297→    if (hit) {
  1298→      const inTray = hit.x >= TRAY.minX && hit.x <= TRAY.maxX && hit.z >= TRAY.minZ && hit.z <= TRAY.maxZ;
  1299→      const inCase = hit.x >= CASE.minX && hit.x <= CASE.maxX && hit.z >= CASE.minZ && hit.z <= CASE.maxZ;
  1300→      zone = inTray ? 'tray' : inCase ? 'case' : null;
  1301→    }
  1302→    // المرور عبر منطق المتجر: إن رُفض الإجراء تعاد القطعة لموضع راحتها عبر syncState
  1303→    this.callbacks.onPartDrop?.(id, zone);
  1304→  }
  1305→
  1306→  /** إلغاء أي سحب جارٍ (Esc / إيقاف الوضع) */
  1307→  private cancelDrag(): void {
  1308→    if (this.dragId) {
  1309→      const part = this.parts[this.dragId];
  1310→      this.recomputeTargets(); // يعيد الهدف الصحيح من آخر حالة
  1311→      if (part) part.group.position.copy(part.targetPos);
  1312→    }
  1313→    this.dragId = null;
  1314→    this.pendingDrag = null;
  1315→    if (this.controls) this.controls.enabled = true;
  1316→    if (this.renderer) this.renderer.domElement.style.cursor = '';
  1317→  }
  1318→
  1319→  /** تقاطع شعاع المؤشر مع سطح السحب الحالي */
  1320→  private rayPlaneHit(clientX: number, clientY: number): THREE.Vector3 | null {
  1321→    const rect = this.renderer.domElement.getBoundingClientRect();
  1322→    const w = rect.width > 0 ? rect.width : 1;
  1323→    const h = rect.height > 0 ? rect.height : 1;
  1324→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1325→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1326→    const hit = this._v2;
  1327→    return this.raycaster.ray.intersectPlane(this.dragPlane, hit) ? hit : null;
  1328→  }
  1329→}

  1332→function nowMs(): number {
  1333→  return typeof performance !== 'undefined' ? performance.now() : Date.now();
