    48→function StageStepper({ current, onJump }: { current: TechStage; onJump: (s: TechStage) => void }) {
    49→  return (
    50→    <div className="flex flex-wrap gap-1.5">
    51→      {TECH_STAGES.map((st) => {
    52→        const active = st.id === current;
    53→        const passed = st.id < current;
    54→        return (
    55→          <button
    56→            key={st.id}
    57→            type="button"
    58→            onClick={() => passed && onJump(st.id)}
    59→            disabled={!passed}
    60→            className={cn(
    61→              'flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] font-medium transition-colors min-h-[30px]',
    62→              active && 'border-primary bg-primary text-primary-foreground shadow-sm',
    63→              passed && !active && 'border-emerald-600/40 bg-emerald-600/10 text-emerald-400 hover:bg-emerald-600/20',
    64→              !active && !passed && 'border-border text-muted-foreground',
    65→            )}
    66→          >
    67→            <span className="font-mono">{st.id}</span>
    68→            {st.titleAr}
    69→          </button>
    70→        );
    71→      })}
    72→    </div>
    73→  );
    74→}
