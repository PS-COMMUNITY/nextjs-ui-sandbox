    98→function StatusBadge({ id }: { id: ComponentId }) {
    99→  const sim = useLabStore((s) => s.sim);
   100→  if (id === 'case') {
   101→    return <Badge variant="outline" className="gap-1">هيكل الجهاز</Badge>;
   102→  }
   103→  const secureOpen = (SECURE_FLAGS[id] ?? []).some((f) => sim.flags[f]);
   104→  const state = assemblyStateOf(sim.parts[id], secureOpen);
   105→  const info = ASSEMBLY_STATE_INFO[state];
   106→  const colorClass: Record<string, string> = {
   107→    emerald: 'bg-emerald-600/15 text-emerald-400 border-emerald-600/40',
   108→    amber: 'bg-amber-600/15 text-amber-400 border-amber-600/40',
   109→    slate: 'bg-slate-600/15 text-slate-300 border-slate-600/40',
   110→    cyan: 'bg-cyan-600/15 text-cyan-400 border-cyan-600/40',
   111→    rose: 'bg-rose-600/15 text-rose-400 border-rose-600/40',
   112→  };
   113→  const icon =
   114→    state === 'INSTALLED_CORRECTLY' ? <CheckCircle2 className="h-3 w-3" aria-hidden /> :
   115→    state === 'AWAITING_INSTALL' ? <HandMetal className="h-3 w-3" aria-hidden /> :
   116→    state === 'TRAINING_DAMAGE' ? <TriangleAlert className="h-3 w-3" aria-hidden /> :
   117→    <Wrench className="h-3 w-3" aria-hidden />;
   118→  return (
   119→    <Badge variant="outline" className={cn('gap-1', colorClass[info.colorAr] ?? '')}>
   120→      {icon}
   121→      {state === 'INSTALLED_CORRECTLY' ? 'مثبتة بإحكام' : state === 'INSTALLED_SECURED' ? 'مثبتة (مزالج مفتوحة)' : info.labelAr}
   122→    </Badge>
   123→  );
   124→}

   126→/** خطوة قائمة تحقق قابلة للتنفيذ (علم ميكانيكي أو فك/تركيب قطعة) */
   127→function ChecklistRow({
   128→  item,
   129→  mode,
   130→  onFlag,
   131→  onPart,
   132→}: {
   133→  item: ChecklistItem;
   134→  mode: 'remove' | 'install';
   135→  onFlag: (f: MachineFlag) => void;
   136→  onPart: (id: ComponentId) => void;
   137→}) {
   138→  const isFlag = item.source !== 'safetyOff' && item.source !== 'safetyUnplug' && item.source !== 'base';
   139→  const canRunFlag = !item.done && isFlag && item.available;
   140→  const canRunPart = !item.done && !isFlag && item.actionPart && item.available;
   141→  return (
   142→    <li
   143→      className={cn(
   144→        'flex items-center gap-2 rounded-md border px-2.5 py-2 text-xs',
   145→        item.done ? 'border-emerald-600/30 bg-emerald-600/10' : 'border-border bg-card',
   146→        !item.done && !item.available && 'opacity-60',
   147→      )}
   148→    >
   149→      {item.done ? (
   150→        <CheckCircle2 className="h-4 w-4 shrink-0 text-emerald-500" aria-hidden />
   151→      ) : (
   152→        <CircleDashed className="h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   153→      )}
   154→      <div className="min-w-0 flex-1">
   155→        <p className={cn('font-medium leading-tight', item.done && 'line-through opacity-70')}>{item.labelAr}</p>
   156→        {!item.done && item.hintAr ? (
   157→          <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">{item.hintAr}</p>
   158→        ) : null}
   159→      </div>
   160→      {canRunFlag ? (
   161→        <Button size="sm" variant="secondary" className="h-8 gap-1 px-2.5 text-[11px] shrink-0 max-sm:h-11 max-sm:px-3" onClick={() => onFlag(item.source as MachineFlag)}>
   162→          <Hammer className="h-3 w-3" aria-hidden /> نفّذ
   163→        </Button>
   164→      ) : null}
   165→      {canRunPart ? (
   166→        <Button size="sm" variant="secondary" className="h-8 gap-1 px-2.5 text-[11px] shrink-0 max-sm:h-11 max-sm:px-3" onClick={() => onPart(item.actionPart!)}>
   167→          <Hammer className="h-3 w-3" aria-hidden />
