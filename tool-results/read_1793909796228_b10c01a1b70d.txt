   151→export function isWebGLAvailable(): boolean {
   152→  try {
   153→    const canvas = document.createElement('canvas');
   154→    return Boolean(canvas.getContext('webgl2') || canvas.getContext('webgl'));
   155→  } catch {
   156→    return false;
   157→  }
   158→}

   160→/** تخلّص من مادة وكل خرائطها النسيجية (يُستخدم في dispose) */
   161→function disposeMaterial(mat: THREE.Material): void {
   162→  for (const value of Object.values(mat) as unknown[]) {
   163→    if (value instanceof THREE.Texture) value.dispose();
   164→  }
   165→  mat.dispose();
   166→}

   260→  constructor(init: { container: HTMLDivElement; quality: SceneQuality; callbacks: PCSceneCallbacks }) {

   271→    try {
   272→      const w = Math.max(1, this.container.clientWidth);

   294→      this.scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
   295→      this.scene.environmentIntensity = ENV_INTENSITY;
   296→      pmrem.dispose();
   297→      this.camera = new THREE.PerspectiveCamera(40, w / h, 0.1, 100);
   298→      this.camera.position.set(...DEFAULT_CAMERA.pos);
   299→      this.controls = new OrbitControls(this.camera, renderer.domElement);
   300→      this.controls.target.set(...DEFAULT_CAMERA.target);
   301→      this.controls.enableDamping = true;
   302→      this.controls.dampingFactor = 0.08;
   303→      this.controls.minDistance = 2.2;
   304→      this.controls.maxDistance = 22;
   305→      this.controls.maxPolarAngle = 1.45;
   306→      this.controls.update();
   307→      // (V8.1) حجز الأزيموت ديناميكيًا: الكاميرا تبقى داخل الورشة ولا تعبر جدار المكتب

   310→        const dist = this.camera.position.distanceTo(this.controls.target);
   311→        const cosMin = THREE.MathUtils.clamp(
   312→          (DESK.minZ + 0.9 - this.controls.target.z) / Math.max(dist, 0.001),
   313→          -1, 0.35,
   314→        );
   315→        const azMax = Math.acos(cosMin);
   316→        this.controls.minAzimuthAngle = -azMax;
   317→        this.controls.maxAzimuthAngle = azMax;

   357→            (smat as THREE.MeshBasicMaterial).color.set(0xffffff);

   395→      // خطاف تشخيصي للتطوير فقط (لا يتغير سلوك التطبيق)

   398→      }
   399→

   402→      this.callbacks.onReady?.();
   403→    } catch {

   405→      this.fail(WEBGL_ERROR_AR);
   406→    }
   407→  }

   506→  /** تفعيل/إنهاء وضع التحكم بالشاشة: أولوية الإدخال تنتقل إلى النظام الافتراضي */
   507→  setFocusMode(on: boolean): void {
   508→    if (this.blocked || !this.vos) return;
   509→    if (on === this.focusMode) return;
   510→    const screen = this.visuals.monitor.getObjectByName('monitor-screen');
   511→    if (!(screen instanceof THREE.Mesh)) return;
   512→    this.focusMode = on;
   513→    this.vos.cursor.visible = on;
   514→    this.controls.enabled = !on;
   515→    if (on) {
   516→      // تأطير الشاشة: مسافة تحسب من أبعاد السطح الفعلية وزاوية الرؤية
   517→      this.savedCam = { pos: this.camera.position.clone(), target: this.controls.target.clone() };
   518→      const box = new THREE.Box3().setFromObject(screen);
   519→      const center = box.getCenter(new THREE.Vector3());
   520→      const size = box.getSize(new THREE.Vector3());
   521→      // عمود السطح = أصغر بُعد — اتجاه الكاميرا من جهة المشاهد (+Z النموذج مُدار بمجموعة الشاشة)
   522→      const normal = new THREE.Vector3(0, 0, 1).applyQuaternion(screen.getWorldQuaternion(new THREE.Quaternion())).normalize();
   523→      const fov = THREE.MathUtils.degToRad(this.camera.fov);
   524→      const aspect = Math.max(0.4, this.camera.aspect);
   525→      const distW = size.x / 2 / (Math.tan(fov / 2) * aspect);
   526→      const distH = size.y / 2 / Math.tan(fov / 2);
   527→      const dist = Math.max(distW, distH) * 1.12;
   528→      this.startCamTween(center.clone().addScaledVector(normal, dist), center.clone(), 0.55);
   529→      this.renderer.domElement.style.cursor = 'none';
   530→      this.callbacks.onFocusMode?.(true);
   531→    } else {
   532→      if (this.savedCam) {
   533→        this.startCamTween(this.savedCam.pos, this.savedCam.target, 0.5);
   534→        this.savedCam = null;
   535→      }
   536→      this.renderer.domElement.style.cursor = '';
   537→      this.callbacks.onFocusMode?.(false);
   538→    }
   539→    this.drawMonitor(true);
   540→  }

   542→  /** توجيه مفتاح لوحة المفاتيح الحقيقي إلى النظام الافتراضي — يعيد true إن استُهلك */
   543→  handleMonitorKey(ev: { key: string; code: string; ctrlKey: boolean; shiftKey: boolean; altKey: boolean }): boolean {
   544→    if (!this.focusMode || !this.vos) return false;
   545→    if (ev.key === 'Escape' || ev.code === 'Escape') {

   548→      // ولا يخرج من وضع الشاشة — الاختيارية تجعل الكود صالحًا قبل وصول الدالة
   549→      const v = this.vos as unknown as { hasKeyboardFocus?: () => boolean };
   550→      if (v?.hasKeyboardFocus?.()) {

   560→    this.drawMonitor(true);
   561→    return true;
   562→  }

   564→  /** تحويل نقطة شاشة المتصفح إلى إحداثيات سطح العرض (UV → بكسل) — null إن لم يصب السطح */
   565→  private screenHit(clientX: number, clientY: number): { x: number; y: number } | null {
   566→    const screen = this.visuals.monitor.getObjectByName('monitor-screen');
   567→    if (!(screen instanceof THREE.Mesh)) return null;
   568→    const rect = this.renderer.domElement.getBoundingClientRect();
   569→    const w = rect.width > 0 ? rect.width : 1;
   570→    const h = rect.height > 0 ? rect.height : 1;
   571→    this._screenNdc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
   572→    this.raycaster.setFromCamera(this._screenNdc, this.camera);
   573→    const hits = this.raycaster.intersectObject(screen, false);
   574→    if (hits.length === 0 || hits[0].uv === undefined) return null;
   575→    if (!this.vos) return null; // (إصلاح تدقيق V6) حارس: كان غير آمن null هنا
   576→    const uv = hits[0].uv!;
   577→    // PlaneGeometry: (0,0) أسفل-يسار — الكانفس: (0,0) أعلى-يسار ⇒ قلب Y فقط
   578→    return { x: uv.x * this.vos.W, y: (1 - uv.y) * this.vos.H };
   579→  }

   740→  /** (6) حلقة الرسم: مراوح، اقتراب القطع، تنعيم الغطاء/الكاميرا، النظام الافتراضي، ثم الرسم */
   741→  private readonly tick = (): void => {
   742→    if (this.disposed || this.failed) return;
   743→    this.rafId = requestAnimationFrame(this.tick);
   744→    const dt = Math.min(this.clock.getDelta(), 0.05); // حماية من قفزات التبويبات الخاملة
   745→    const k = 1 - Math.exp(-LERP_RATE * dt);
   746→
   747→    // (V6 §3) منظومة المراوح: سرعات ودوران كل ريش على محوره الصحيح + الحرارة المُحاكاة
   748→    if (this.fans && this.lastState) {
   749→      const s = this.lastState;
   750→      const powered = s.flags.powerOn && s.flags.acConnected && !s.peripherals.psuDead;
   751→      // (إصلاح تدقيق V6) الأحمال مشتقة من الحالة والحرارة المُحاكاة — كانت ثوابت مكتوبة يدويًا
   752→      // حلقة تغذية راجعة تعليمية: غياب التبريد يرفع الحرارة فيرفع الأحمال فيتسارع ما تبقى من مراوح
   753→      const heatCpu = Math.min(1, Math.max(0, (s.thermal.cpuTempC - 50) / 45));
   754→      const heatGpu = Math.min(1, Math.max(0, (s.thermal.gpuTempC - 50) / 40));
   755→      if (!this.fanConditions) this.fanConditions = { powered: false, cpuCoolerOk: false, gpuOk: false, caseFansOk: false, psuOk: false, cpuLoad: 0, gpuLoad: 0 };
   756→      const conditions = this.fanConditions;
   757→      conditions.powered = powered;
   758→      conditions.cpuCoolerOk = isInstalled(s.parts.cooler) && !s.flags.fanHeadersDisconnected;
   759→      conditions.gpuOk = isInstalled(s.parts.gpu) && isInstalled(s.parts.gpuPower);
   760→      conditions.caseFansOk = isInstalled(s.parts.fans) && !s.flags.fanHeadersDisconnected;
   761→      conditions.psuOk = powered;
   762→      conditions.cpuLoad = powered ? Math.min(1, 0.35 + 0.65 * heatCpu) : 0;
   763→      conditions.gpuLoad = powered && conditions.gpuOk ? Math.min(1, 0.3 + 0.7 * heatGpu) : 0;
   764→      this.fans.update(dt, conditions);
   765→
   766→      // الحرارة المُحاكاة (بيانات تعليمية معلنة): ترتفع بلا مبرد/بلا مراوح وتقترب من الهدف
   767→      this.thermalAccum += dt;
   768→      if (this.thermalAccum >= 1) {
   769→        this.thermalAccum = 0;
   770→        const cpuTarget = !powered ? 26 : isInstalled(s.parts.cooler) ? (s.flags.fanHeadersDisconnected ? 86 : 46) : 97;
   771→        const gpuTarget = !powered ? 26 : conditions.gpuOk ? 52 : isInstalled(s.parts.gpu) ? 84 : 26;
   772→        const cpu = THREE.MathUtils.lerp(s.thermal.cpuTempC, cpuTarget, 0.18);
   773→        const gpu = THREE.MathUtils.lerp(s.thermal.gpuTempC, gpuTarget, 0.18);
   774→        this.callbacks.onThermal?.(cpu, gpu);
   775→        // إطفاء حراري تعليمي عند تجاوز الحد (فصّل التبريد = الجهاز ينطفئ)
   776→        if (cpu >= 95.5 && powered) this.callbacks.onThermalShutdown?.(cpu);
   777→      }
   778→    }
   779→
   780→    // اقتراب القطع نحو أهدافها (الموضع lerp والدوران slerp) — عدا الهيكل الثابت
   781→    for (const id of COMPONENT_IDS) {
   782→      if (id === 'case') continue;
   783→      const part = this.parts[id];
   784→      part.group.position.lerp(part.targetPos, k);
   785→      part.group.quaternion.slerp(part.targetQuat, k);
   786→    }
   787→
   788→    // تطفو القطعة الملتقطة (awaiting) فوق الصينية بلطف — إشارة بصرية لجاهزية التركيب
   789→    const elapsed = this.clock.getElapsedTime();
   790→    for (const id of COMPONENT_IDS) {
   791→      if (id === 'case') continue;
   792→      if (this.lastState && this.lastState.parts[id] === 'awaiting' && this.dragId !== id) {
   793→        const part = this.parts[id];
   794→        part.group.position.y = part.trayPos.y + 0.16 + Math.sin(elapsed * 3) * 0.04;
   795→      }
   796→    }
   797→
   798→    // تنعيم دوران الغطاء الجانبي نحو الهدف
   799→    const panel = this.visuals.sidePanel;
   800→    panel.rotation.y += (this.panelTarget - panel.rotation.y) * k;
   801→
   802→    // (V8) استكمال حركة فك/تثبيت البراغي — لفّة ورفعة ناعمة بدل القفز المفاجئ
   803→    if (this.screwAnims.size > 0) {
   804→      for (const [rec, anim] of this.screwAnims) {
   805→        anim.t = Math.min(1, anim.t + dt / 0.38);
   806→        const kk = PCScene.easeInOut(anim.t);
   807→        rec.obj.position.copy(rec.basePos);
   808→        rec.obj.position[rec.axis] += anim.out ? rec.delta * kk : rec.delta * (1 - kk);
   809→        rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN * (anim.out ? kk : 1 - kk);
   810→        if (anim.t >= 1) this.screwAnims.delete(rec);
   811→      }
   812→    }
   813→
   814→    // استكمال تنعيم الكاميرا (resetCamera / focusComponent)
   815→    const tween = this.camTween;
   816→    if (tween) {
   817→      tween.t = Math.min(tween.t + dt / tween.dur, 1);
   818→      const eased = PCScene.easeInOut(tween.t);
   819→      this.camera.position.lerpVectors(tween.fromPos, tween.toPos, eased);
   820→      this.controls.target.lerpVectors(tween.fromTarget, tween.toTarget, eased);
   821→      if (tween.t >= 1) this.camTween = null;
   822→    }
   823→
   824→    this.controls.update();
   825→
   826→    // إعادة رسم نسيج الشاشة الافتراضية: فوري في وضع التحكم بالشاشة، مُقيّد في المختبر
   827→    if (this.painter) {
   828→      const interval = this.focusMode ? 90 : MONITOR_REDRAW_MS;
   829→      if (nowMs() - this.lastMonitorDrawAt >= interval) {
   830→        this.drawMonitor(false);
   831→      }
   832→    }
   833→
   834→    this.renderer.render(this.scene, this.camera);
   835→  };
   836→
   837→  /** تنعيم easeInOut: t<0.5 ? 2t² : 1 − (−2t+2)²/2 */
   838→  private static easeInOut(t: number): number {
   839→    return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
   840→  }

   842→  /** إعادة حساب أهداف القطع من آخر حالة متزامنة + علم التفجير الحالي */
   843→  private recomputeTargets(): void {
   844→    const parts = this.lastState?.parts;
   845→    for (const id of COMPONENT_IDS) {
   846→      if (id === 'case') continue;
   847→      const part = this.parts[id];
   848→      const status: string | undefined = parts ? parts[id] : undefined;
   849→      if (status === 'removed' || status === 'damaged') {
   850→        part.targetPos.copy(part.trayPos);
   851→        part.targetRot.copy(part.trayRot);
   852→        if (status === 'damaged') {
   853→          // القطعة التالفة تستلقي مائلة على الصينية
   854→          part.targetRot.z += 0.35;
   855→        }
   856→      } else if (status === 'awaiting') {
   857→        // ملتقطة من الصينية — ترتفع قليلاً (وتطفو في حلقة الرسم)
   858→        part.targetPos.copy(part.trayPos);
   859→        part.targetPos.y += 0.16;
   860→        part.targetRot.copy(part.trayRot);
   861→      } else if (status === 'separated') {
   862→        // فصل جزئي: ما زالت في موضعها مع إزاحة صغيرة نحو اتجاه الفك
   863→        part.targetPos.copy(part.basePos);
   864→        part.targetPos.addScaledVector(part.explodeVec, SEPARATE_FACTOR);
   865→        part.targetRot.set(0, 0, 0);
   866→      } else {
   867→        part.targetPos.copy(part.basePos);
   868→        if (this.exploded) part.targetPos.add(part.explodeVec);
   869→        part.targetRot.set(0, 0, 0);
   870→      }
   871→      part.targetQuat.setFromEuler(part.targetRot);
   872→    }
   873→  }

   888→   * افتراضات المحاور (لأن بناء المصنع موازٍ ولم يُثبَّت بعد):
   889→   *  - مزلاج PCIe ورافعة CPU: دوران حول Y المحلي (اللوح عمودي يواجه -X).
   890→   *  - مزالج RAM: دوران حول Z بإشارة متناوبة بين الطرفين.
   891→   *  - البراغي: ترتفع على +Y (مسامير اللوحة تُسحب على -X) مع دوران طفيف حول Z.
   892→   *  - القوابس: إزاحة -X 0.12 عند الفصل.
   893→   * كل الإزاحات نسبية إلى القيم الأصلية المحفوظة عند البناء.
   894→   */
   895→  private applyFlags(flags: Record<MachineFlag, boolean>, animate: boolean): void {
   896→    // (V8) لقطة سابقة للأعلام لكشف تغيّر البراغي بين المزامنات (لتوليد حركة الفك)
   897→    const prev = this.flagPrev;
   898→    this.flagPrev = { ...flags };
   899→
   900→    // الغطاء الجانبي — عبر هدف تنعيمي تتبعه الحلقة
   901→    this.panelTarget = flags.panelOpen ? PANEL_OPEN_ANGLE : 0;
   902→    if (!animate) this.visuals.sidePanel.rotation.y = this.panelTarget;
   903→
   904→    // مزلاج PCIe
   905→    if (this.latchRec) {
   906→      this.latchRec.obj.rotation.y = this.latchRec.baseRot + (flags.pcieLatchOpen ? LATCH_OPEN_ANGLE : 0);
   907→    }
   908→
   909→    // مزالج RAM
   910→    for (const clip of this.clipRecs) {
   911→      clip.obj.rotation.z = clip.baseRotZ + (flags.ramClipsOpen ? clip.sign * CLIP_OPEN_ANGLE : 0);
   912→    }
   913→
   914→    // رافعة مقبس CPU
   915→    if (this.leverRec) {
   916→      this.leverRec.obj.rotation.y = this.leverRec.baseRot + (flags.cpuLeverOpen ? LEVER_OPEN_ANGLE : 0);
   917→    }
   918→
   919→    // البراغي (gpu/psu/fan/m2/mobo) — (V8) فك/تثبيت متحرك: البرغي الذي تغيّر علمه يستلم أنيميشن،
   920→    // وما تحت أنيميشن جارٍ لا تُطبّق عليه القيمة النهائية مباشرة (الأنيميشن يملكه)
   921→    for (const [flag, recs] of this.screwMap) {
   922→      const out = flags[flag];
   923→      const changed = prev ? prev[flag] !== out : false;
   924→      for (const rec of recs) {
   925→        if (changed) this.screwAnims.set(rec, { out, t: 0 });
   926→        if (this.screwAnims.has(rec)) continue;
   927→        rec.obj.position.copy(rec.basePos);
   928→        if (out) {
   929→          rec.obj.position[rec.axis] += rec.delta;
   930→          rec.obj.rotation.z = rec.baseRotZ + SCREW_SPIN;
   931→        } else {
   932→          rec.obj.rotation.z = rec.baseRotZ;
   933→        }
   934→      }
   935→    }
   936→
   937→    // قوابس الموصلات (fan headers / front panel)
   938→    for (const plug of this.plugRecs) {
   939→      plug.obj.position.x = plug.baseX - (flags[plug.flag] ? PLUG_PULL : 0);
   940→    }
   941→
   942→    // لمبة الطاقة — مادة خاصة بها (cached) — تنطفئ مع مزود ميت
   943→    const led = this.visuals.flags.powerLed;
   944→    if (led) {
   945→      const mat = led.material;
   946→      if (!Array.isArray(mat) && 'emissive' in mat) {
   947→        const std = mat as THREE.MeshStandardMaterial;
   948→        const alive = flags.powerOn && !(this.lastState?.peripherals.psuDead ?? false);
   949→        std.emissive.set(alive ? COLORS.green : 0x000000);
   950→        std.emissiveIntensity = LED_INTENSITY;
   951→      }
   952→    }
   953→  }

   975→  /** تطبيق شفافية على مواد مجموعة (الزجاج شفاف أصلًا فيبقى كما هو) */
   976→  private applyShellOpacity(root: THREE.Object3D, opacity: number): void {
   977→    root.traverse((obj) => {
   978→      if (!(obj instanceof THREE.Mesh)) return;
   979→      const mats = Array.isArray(obj.material) ? obj.material : [obj.material];
   980→      for (const mat of mats) {
   981→        if (mat.transparent) continue; // الزجاج/الشفاف الأصلي لا يُمسّ

   983→          this.shellOriginals.set(mat, {
   984→            opacity: mat.opacity,
   985→            transparent: mat.transparent,
   986→            depthWrite: mat.depthWrite,
   987→          });
   988→        }
   989→        mat.transparent = true;
   990→        mat.opacity = opacity;
   991→        mat.depthWrite = false;
   992→      }
   993→    });
   994→  }

  1354→  /** إيجاد الهدف عند نقطة شاشة: أول إصابة مرئية غير شفافة — قطعة PC أو طرفية */
  1355→  private pickAt(clientX: number, clientY: number): PickedInfo {
  1356→    const rect = this.renderer.domElement.getBoundingClientRect();
  1357→    const w = rect.width > 0 ? rect.width : 1;
  1358→    const h = rect.height > 0 ? rect.height : 1;
  1359→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1360→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1361→    const hits = this.raycaster.intersectObjects(this.visuals.root.children, true);
  1362→    for (const hit of hits) {
  1363→      if (this.isHidden(hit.object)) continue;        // تجاهل ما خفيه العزل
  1364→      // (V8) النقاط الفعالة أسبق من القطعة الحاملة: برغي/مزلاج/رافعة/قابس/أداة/زر فيزيائي
  1365→      let cur: THREE.Object3D | null = hit.object;
  1366→      let hotspot: PickedInfo = null;
  1367→      let panelGlassClosed = false;
  1368→      while (cur) {
  1369→        const per: unknown = cur.userData['peripheral'];
  1370→        if (typeof per === 'string') {
  1371→          hotspot = { type: 'periph', kind: per as PeripheralKind };
  1372→          break;
  1373→        }
  1374→        const ft: unknown = cur.userData['flagTarget'];
  1375→        if (typeof ft === 'string' && (MACHINE_FLAGS as readonly string[]).includes(ft)) {
  1376→          if (ft === 'panelOpen' && this.isSeeThrough(hit.object) && !(this.lastState?.flags.panelOpen ?? false)) {
  1377→            // زجاج الغطاء المغلق يمرّر النقر إلى القطع الداخلية — الإطار/الحواف فقط تفتح وتغلق
  1378→            panelGlassClosed = true;
  1379→            break;
  1380→          }
  1381→          hotspot = { type: 'flag', flag: ft as MachineFlag };
  1382→          break;
  1383→        }
  1384→        if (cur.userData['benchTool'] === 'screwdriver') {
  1385→          hotspot = { type: 'tool', tool: 'screwdriver' };
  1386→          break;
  1387→        }
  1388→        if (cur.userData['powerButton'] === true) {
  1389→          hotspot = { type: 'powerButton' };
  1390→          break;
  1391→        }
  1392→        if (cur.userData['acSwitch'] === true) {
  1393→          hotspot = { type: 'flag', flag: 'acConnected' };
  1394→          break;
  1395→        }
  1396→        cur = cur.parent;
  1397→      }
  1398→      if (hotspot) return hotspot;
  1399→      if (panelGlassClosed) continue;
  1400→      if (this.isSeeThrough(hit.object)) continue;    // الزجاج/الشفاف لا يحجب الاختيار
  1401→      let cur2: THREE.Object3D | null = hit.object;
  1402→      while (cur2) {
  1403→        const cid: unknown = cur2.userData['componentId'];
  1404→        if (typeof cid === 'string' && (COMPONENT_IDS as readonly string[]).includes(cid)) {
  1405→          return { type: 'part', id: cid as ComponentId };
  1406→        }
  1407→        cur2 = cur2.parent;
  1408→      }
  1409→      // إصابة أولى بلا معرف (عدة/زينة) — تمرير النقر عبرها إلى ما خلفها (توجيه v4)
  1410→      continue;
  1411→    }
  1412→    return null;
  1413→  }

  1552→  private readonly onPointerDown = (event: PointerEvent): void => {
  1553→    if (this.blocked) return;
  1554→    this.pointerDownActive = true;
  1555→    this.downX = event.clientX;
  1556→    this.downY = event.clientY;
  1557→
  1558→    // (V6-B) وضع التحكم بالشاشة: كل إدخال المؤشر يوجّه إلى النظام الافتراضي
  1559→    if (this.focusMode && this.vos) {
  1560→      const hit = this.screenHit(event.clientX, event.clientY);
  1561→      if (hit) this.vos.pointer('down', hit.x, hit.y, { button: event.button });
  1562→      this.drawMonitor(true);
  1563→      return;
  1564→    }
  1565→
  1566→    // بدء سحب محتمل: فقط بالزر الأيسر وعند تفعيل وضع الفك/التركيب
  1567→    if (this.dragEnabled && event.button === 0) {
  1568→      const picked = this.pickAt(event.clientX, event.clientY);
  1569→      if (picked && picked.type === 'part' && DRAGGABLE.has(this.lastState?.parts[picked.id] ?? '')) {
  1570→        this.pendingDrag = { id: picked.id, startX: event.clientX, startY: event.clientY };
  1571→      }
  1572→    }
  1573→  };

  1575→  /** (7) نقرة فقط إذا كانت المسافة بين down/up أقل من 6px — والسحب الفعلي يُعالج قبلها */
  1576→  private readonly onPointerUp = (event: PointerEvent): void => {
  1577→    if (this.blocked) return;
  1578→
  1579→    // (V6-B) توجيه رفع الزر إلى النظام الافتراضي في وضع الشاشة
  1580→    if (this.focusMode && this.vos) {
  1581→      const hit = this.screenHit(event.clientX, event.clientY);
  1582→      if (hit) this.vos.pointer('up', hit.x, hit.y, { button: event.button });
  1583→      this.drawMonitor(true);
  1584→      this.pointerDownActive = false;
  1585→      return;
  1586→    }
  1587→
  1588→    if (this.dragId) {
  1589→      this.finishDrag(event.clientX, event.clientY);
  1590→      this.pointerDownActive = false;
  1591→      return;
  1592→    }
  1593→    this.pendingDrag = null;
  1594→    if (!this.pointerDownActive) return;
  1595→    this.pointerDownActive = false;
  1596→    const dx = event.clientX - this.downX;
  1597→    const dy = event.clientY - this.downY;
  1598→    if (dx * dx + dy * dy >= CLICK_MAX_DIST_SQ) return; // سحب لا نقرة
  1599→
  1600→    // (V6-B §4.3) النقر على سطح الشاشة المجسمة نفسه يدخل وضع التحكم ويسلّم النقرة
  1601→    const onScreen = this.screenHit(event.clientX, event.clientY);
  1602→    if (onScreen && this.vos) {
  1603→      this.setFocusMode(true);
  1604→      this.vos.pointer('down', onScreen.x, onScreen.y, { button: event.button });
  1605→      this.vos.pointer('up', onScreen.x, onScreen.y, { button: event.button });
  1606→      this.drawMonitor(true);
  1607→      return;
  1608→    }
  1609→
  1610→    const picked = this.pickAt(event.clientX, event.clientY);
  1611→    if (picked && picked.type === 'periph') {
  1612→      this.callbacks.onPeripheral?.(picked.kind, picked.detail);
  1613→      return;
  1614→    }
  1615→    // (V8) التحكم المباشر كفني حقيقي: برغي/مزلاج/رافعة/قابس/زر طاقة/مفتاح PSU
  1616→    if (picked && picked.type === 'flag') {
  1617→      if (SCREW_FLAGS.has(picked.flag) && this.activeTool !== 'screwdriver') {
  1618→        // البراغي لا تُفك بالأصابع — التقط المفك أولًا (كما في الواقع)
  1619→        this.callbacks.onToolRequired?.('screwdriver');
  1620→        return;
  1621→      }
  1622→      this.callbacks.onFlagClick?.(picked.flag);
  1623→      return;
  1624→    }
  1625→    if (picked && picked.type === 'tool') {
  1626→      this.setActiveTool(picked.tool);
  1627→      this.callbacks.onToolPick?.(picked.tool);
  1628→      return;
  1629→    }
  1630→    if (picked && picked.type === 'powerButton') {
  1631→      this.callbacks.onPowerButton?.();
  1632→      return;
  1633→    }
  1634→    this.callbacks.onSelect(picked && picked.type === 'part' ? picked.id : null);
  1635→  };

  1637→  /** (7) تحويم/سحب — بفحص مُقيَّد زمنيًا للتحويم (100ms) */
  1638→  private readonly onPointerMove = (event: PointerEvent): void => {
  1639→    if (this.blocked) return;
  1640→
  1641→    // (V6-B) حركة المؤشر الحقيقي → المؤشر الافتراضي داخل الشاشة المجسمة
  1642→    if (this.focusMode && this.vos) {
  1643→      const hit = this.screenHit(event.clientX, event.clientY);
  1644→      if (hit) this.vos.pointer('move', hit.x, hit.y);
  1645→      this.drawMonitor(true);
  1646→      return;
  1647→    }
  1648→
  1649→    if (this.dragId) {
  1650→      this.updateDrag(event.clientX, event.clientY);
  1651→      return;
  1652→    }
  1653→    if (this.pendingDrag) {
  1654→      const dx = event.clientX - this.pendingDrag.startX;
  1655→      const dy = event.clientY - this.pendingDrag.startY;
  1656→      if (dx * dx + dy * dy >= DRAG_START_DIST_SQ) {
  1657→        this.beginDrag(this.pendingDrag.id, event.clientX, event.clientY);
  1658→        this.updateDrag(event.clientX, event.clientY);
  1659→      }
  1660→      return;
  1661→    }
  1662→    const now = performance.now();
  1663→    if (now - this.lastHoverAt < HOVER_THROTTLE_MS) return;
  1664→    this.lastHoverAt = now;
  1665→    const hovering = this.pickAt(event.clientX, event.clientY);
  1666→    // (V8) مؤشر المفك:crosshair عندما تكون الأداة باليد — إحساس أدوات حقيقي
  1667→    this.renderer.domElement.style.cursor = hovering
  1668→      ? 'pointer'
  1669→      : this.activeTool === 'screwdriver' ? 'crosshair' : '';
  1670→  };

  1709→  /** بدء سحب فعلي: تعطيل المداري + سطح أفقي عند ارتفاع القطعة */
  1710→  private beginDrag(id: ComponentId, clientX: number, clientY: number): void {
  1711→    const part = this.parts[id];
  1712→    if (!part) return;
  1713→    this.pendingDrag = null;
  1714→    this.dragId = id;
  1715→    this.controls.enabled = false;
  1716→    this.dragPlane.set(new THREE.Vector3(0, 1, 0), -part.group.position.y);
  1717→    // إزاحة الإمساك: فرق نقطة النقر عن مركز القطعة على السطح نفسه
  1718→    const hit = this.rayPlaneHit(clientX, clientY);
  1719→    if (hit) this.dragOffset.copy(part.group.position).sub(hit);
  1720→    this.renderer.domElement.style.cursor = 'grabbing';
  1721→  }
  1722→
  1723→  /** تحديث موضع القطعة أثناء السحب (بصريًا فقط — الحالة المنطقية لا تتغير هنا) */
  1724→  private updateDrag(clientX: number, clientY: number): void {
  1725→    const id = this.dragId;
  1726→    if (!id) return;
  1727→    const hit = this.rayPlaneHit(clientX, clientY);
  1728→    if (!hit) return;
  1729→    const part = this.parts[id];
  1730→    part.group.position.set(hit.x + this.dragOffset.x, part.group.position.y, hit.z + this.dragOffset.z);
  1731→    // الهدف يتبع اليد — يمنع حلقة الرسم من محاربة السحب
  1732→    part.targetPos.copy(part.group.position);
  1733→  }

  1735→  /** إنهاء السحب: تحديد منطقة الإفلات ثم إبلاغ المنطق (التحقق أولًا ثم الحركة — توجيه v3 §6) */
  1736→  private finishDrag(clientX: number, clientY: number): void {
  1737→    const id = this.dragId;
  1738→    this.dragId = null;
  1739→    this.controls.enabled = true;
  1740→    this.renderer.domElement.style.cursor = '';
  1741→    if (!id) return;
  1742→    const hit = this.rayPlaneHit(clientX, clientY);
  1743→    let zone: DropZone = null;
  1744→    if (hit) {
  1745→      const inTray = hit.x >= TRAY.minX && hit.x <= TRAY.maxX && hit.z >= TRAY.minZ && hit.z <= TRAY.maxZ;
  1746→      const inCase = hit.x >= CASE.minX && hit.x <= CASE.maxX && hit.z >= CASE.minZ && hit.z <= CASE.maxZ;
  1747→      zone = inTray ? 'tray' : inCase ? 'case' : null;
  1748→    }
  1749→    // المرور عبر منطق المتجر: إن رُفض الإجراء تعاد القطعة لموضع راحتها عبر syncState
  1750→    this.callbacks.onPartDrop?.(id, zone);
  1751→  }

  1753→  /** إلغاء أي سحب جارٍ (Esc / إيقاف الوضع) */
  1754→  private cancelDrag(): void {

  1756→      const part = this.parts[this.dragId];
  1757→      this.recomputeTargets(); // يعيد الهدف الصحيح من آخر حالة
  1758→      if (part) part.group.position.copy(part.targetPos);
  1759→    }
  1760→    this.dragId = null;
  1761→    this.pendingDrag = null;
  1762→    if (this.controls) this.controls.enabled = true;
  1763→    if (this.renderer) this.renderer.domElement.style.cursor = '';
  1764→  }

  1766→  /** تقاطع شعاع المؤشر مع سطح السحب الحالي */
  1767→  private rayPlaneHit(clientX: number, clientY: number): THREE.Vector3 | null {
  1768→    const rect = this.renderer.domElement.getBoundingClientRect();
  1769→    const w = rect.width > 0 ? rect.width : 1;
  1770→    const h = rect.height > 0 ? rect.height : 1;
  1771→    this._ndc.set(((clientX - rect.left) / w) * 2 - 1, -((clientY - rect.top) / h) * 2 + 1);
  1772→    this.raycaster.setFromCamera(this._ndc, this.camera);
  1773→    const hit = this._v2;
  1774→    return this.raycaster.ray.intersectPlane(this.dragPlane, hit) ? hit : null;
  1775→  }
