   176→   176→let persistTimer: ReturnType<typeof setTimeout> | null = null;
   177→   177→
   178→   178→function schedulePersist(get: () => LabStore) {
   179→   179→  if (persistTimer) clearTimeout(persistTimer);
   180→   180→  persistTimer = setTimeout(() => {
   181→   181→    const s = get();
   182→   182→    if (!s.hydrated) return;
   183→   183→    const shape: PersistShape = {
   184→   184→      version: APP_VERSION,
   185→   185→      simulation: s.sim,
   186→   186→      tech: s.tech,
   187→   187→      logs: s.logs,
   188→   188→      quizScores: s.quizScores,
   189→   189→      missionsDone: s.missionsDone,
   190→   190→      lessonsRead: s.lessonsRead,
   191→   191→      settings: { quality: s.quality },
   192→   192→      savedAt: Date.now(),
   193→   193→    };
   194→   194→    const res = saveAll(shape);
   195→   195→    if (res.ok) {
   196→   196→      useLabStore.setState({ saveStatus: { available: true, lastSavedAt: Date.now(), lastErrorAr: null } });
   197→   197→    } else {
   198→   198→      useLabStore.setState({ saveStatus: { available: true, lastSavedAt: null, lastErrorAr: res.messageAr } });
   199→   199→    }
   200→   200→  }, 400);
   201→   201→}

   374→   374→    if (!check.ok) {
   375→   375→      s.logEvent('remove', `${name}: فك مرفوض`, false, check.messageAr);

   384→   384→      toast.success(`${name}: فُصل إلى صينية القطع`);

   388→   388→    }
