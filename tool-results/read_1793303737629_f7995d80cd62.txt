    25→/** cache للمواد القياسية — لا تنشئ مواد جديدة لكل إطار */
    26→const matCache = new Map<string, THREE.MeshStandardMaterial>();
    27→
    28→function mat(color: number, metalness: number, roughness: number): THREE.MeshStandardMaterial {
    29→  const key = `${color}-${metalness}-${roughness}`;
    30→  const cached = matCache.get(key);
    31→  if (cached) return cached;
    32→  const m = new THREE.MeshStandardMaterial({ color, metalness, roughness, flatShading: false });
    33→  matCache.set(key, m);
    34→  return m;
    35→}

    37→/** زجاج الغطاء الجانبي — مادة فيزيائية واحدة مشتركة */
    38→let glassMat: THREE.MeshPhysicalMaterial | null = null;
    39→function glass(): THREE.MeshPhysicalMaterial {
    40→  if (!glassMat) {
    41→    glassMat = new THREE.MeshPhysicalMaterial({
    42→      color: COLORS.glass,
    43→      metalness: 0.1,
    44→      roughness: 0.08,
    45→      transparent: true,
    46→      opacity: 0.16,
    47→      side: THREE.DoubleSide,
    48→      flatShading: false,
    49→    });
    50→  }
    51→  return glassMat;
    52→}

    58→/** صندوق سريع مع ضبط الظلال حسب الجودة */
    59→function box(
    60→  w: number, h: number, d: number, material: THREE.Material,
    61→  x = 0, y = 0, z = 0, cast = true, receive = false,
    62→): THREE.Mesh {
    63→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), material);
    64→  m.position.set(x, y, z);
    65→  if (castEnabled && cast) m.castShadow = true;
    66→  if (receive) m.receiveShadow = true;
    67→  return m;
    68→}

    70→type Axis = 'x' | 'y' | 'z';
    71→
    72→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
    73→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
    74→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
    75→  if (axis === 'x') m.rotation.z = Math.PI / 2;
    76→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
    77→  m.position.set(x, y, z);
    78→  if (castEnabled && cast) m.castShadow = true;
    79→  return m;
    80→}

    82→/** برغي معدني صغير — محور حسب الاستخدام */
    83→function makeScrew(r: number, length: number, axis: Axis = 'y'): THREE.Mesh {
    84→  return cyl(r, length, mat(COLORS.aluminum, 0.85, 0.35), axis);

    87→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
    88→function makeCable(points: THREE.Vector3[], radius: number): THREE.Mesh {
    89→  const curve = new THREE.CatmullRomCurve3(points);
    90→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(0x101418, 0.3, 0.6));
    91→  if (castEnabled) m.castShadow = true;
    92→  return m;
    93→}

    95→/** تحويل نقاط كابل عالمية إلى محلية بالنسبة لموضع تثبيت القطعة */
    96→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
    97→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));
    98→}

   100→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   101→
   102→/**
   103→ * مروحة عامة: إطار (4 أركان + حلقة Torus) + شفرات (7 ألواح مائلة 30°) + محور مركزي.
   104→ * محور الدوران الافتراضي = Y المحلي (تُلف المروحة لاحقًا حسب موضعها).
   105→ */
   106→function makeFan(radius: number): FanParts {
   107→  const depth = Math.max(0.12, radius * 0.55);
   108→  const half = radius + 0.06;
   109→
   110→  // الإطار
   111→  const frame = new THREE.Group();
   112→  const frameMat = mat(COLORS.plastic, 0.3, 0.6);
   113→  const postGeo = new THREE.BoxGeometry(0.09, depth, 0.09);
   114→  for (const sx of [-1, 1]) {
   115→    for (const sz of [-1, 1]) {
   116→      const post = new THREE.Mesh(postGeo, frameMat);
   117→      post.position.set(sx * half, 0, sz * half);
   118→      if (castEnabled) post.castShadow = true;
   119→      frame.add(post);
   120→    }
   121→  }
   122→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.97, 0.018, 6, 24), mat(COLORS.metalDark, 0.5, 0.5));
   123→  ring.rotation.x = Math.PI / 2; // الحلقة في مستوى XZ (عمودية على محور Y)
   124→  if (castEnabled) ring.castShadow = true;
   125→  frame.add(ring);
   126→
   127→  // الشفرات: 7 ألواح موزعة دائريًا داخل مجموعة blades (تُدفع إلى spinners)
   128→  const blades = new THREE.Group();
   129→  const bladeGeo = new THREE.BoxGeometry(0.02, radius * 0.8, radius * 0.35);
   130→  const bladeMat = mat(COLORS.metalMid, 0.4, 0.5);
   131→  for (let i = 0; i < 7; i++) {
   132→    const arm = new THREE.Object3D();
   133→    arm.rotation.y = (i / 7) * Math.PI * 2;
   134→    const pitch = new THREE.Object3D();
   135→    pitch.position.set(radius * 0.5, 0, 0);
   136→    pitch.rotation.x = Math.PI / 6; // ميلان الشفرة 30° حول المحور الشعاعي
   137→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   138→    blade.rotation.z = Math.PI / 2; // الطول شعاعيًا
   139→    if (castEnabled) blade.castShadow = true;
   140→    pitch.add(blade);
   141→    arm.add(pitch);
   142→    blades.add(arm);
   143→  }
   144→  const hub = new THREE.Mesh(
   145→    new THREE.CylinderGeometry(radius * 0.16, radius * 0.16, depth * 0.7, 12),
   146→    mat(COLORS.metalDark, 0.5, 0.4),
   147→  );
   148→  if (castEnabled) hub.castShadow = true;
   149→  blades.add(hub);
   150→
   151→  return { frame, blades };
   152→}

   172→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   173→  scene.background = new THREE.Color(COLORS.bg);
   174→  scene.fog = new THREE.Fog(COLORS.bg, 14, 30);
   175→
   176→  // أرضية دائرية داكنة تستقبل الظلال
   177→  const ground = new THREE.Mesh(
   178→    new THREE.CircleGeometry(22, 24),
   179→    new THREE.MeshStandardMaterial({ color: 0x0e1319, metalness: 0, roughness: 0.95, flatShading: false }),
   180→  );
   181→  ground.rotation.x = -Math.PI / 2;
   182→  ground.position.y = 0;
   183→  ground.receiveShadow = true;
   184→  scene.add(ground);
   185→
   186→  // شبكة أرضية خفيفة
   187→  const grid = new THREE.GridHelper(30, 30, 0x242b34, 0x161b22);
   188→  grid.position.y = 0.01;
   189→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   190→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   191→  for (const gm of gridMats) {
   192→    gm.transparent = true;
   193→    gm.opacity = 0.35;
   194→  }
   195→  scene.add(grid);
   196→
   197→  // إضاءة محيطية أساسية + ضوء نصف كروي لملء الفراغات
   198→  scene.add(new THREE.AmbientLight(0xffffff, 1.35));
   199→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x1a1d24, 0.7));
   200→
   201→  // شمس دافئة مع ظلال (حسب الجودة)
   202→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.4);
   203→  dir.position.set(-6, 8, 6);
   204→  if (quality !== 'low') {
   205→    dir.castShadow = true;
   206→    const size = quality === 'high' ? 2048 : 1024;
   207→    dir.shadow.mapSize.set(size, size);
   208→    dir.shadow.camera.left = -6;
   209→    dir.shadow.camera.right = 6;
   210→    dir.shadow.camera.top = 6;
   211→    dir.shadow.camera.bottom = -6;
   212→    dir.shadow.camera.near = 2;
   213→    dir.shadow.camera.far = 30;
   214→    dir.shadow.camera.updateProjectionMatrix();
   215→    dir.shadow.bias = -0.0004;
   216→    dir.shadow.normalBias = 0.02;
   217→  }
   218→  scene.add(dir);
   219→
   220→  // ضوء تعبئة خفيف خلفي (بلا ظلال؛ decay=0 ليبقى «خفيفًا» بمقدار 0.8 فعليًا)
   221→  const fill = new THREE.PointLight(0xffffff, 0.8, 0, 0);
   222→  fill.position.set(3, 3, -2);
   223→  scene.add(fill);
   224→  // ضوء أمامي موجّه من جهة الزجاج لإبراز وجه اللوحة الأم
   225→  const front = new THREE.PointLight(0xfff4e0, 1.4, 0, 0);
   226→  front.position.set(-5, 3.5, 3);
   227→  scene.add(front);
   228→}

   234→function buildCaseShell(flags: FlagVisuals): THREE.Group {
   235→  const g = new THREE.Group();
   236→  g.name = 'case-shell';
   237→
   238→  const cx = (CASE.minX + CASE.maxX) / 2; // 1.05
   239→  const cy = (CASE.minY + CASE.maxY) / 2; // 2.25
   240→  const cz = (CASE.minZ + CASE.maxZ) / 2; // 2.3
   241→  const wX = CASE.maxX - CASE.minX;       // 2.1
   242→  const hY = CASE.maxY - CASE.minY;       // 4.5
   243→  const dZ = CASE.maxZ - CASE.minZ;       // 4.6
   244→  const wallMat = mat(COLORS.metalDark, 0.45, 0.55);
   245→
   246→  // الجدران (سماكة 0.05) — الجدار الأيمن عند x≈2.1 يستقبل اللوحة الأم
   247→  g.add(box(0.05, hY, dZ, wallMat, CASE.maxX - 0.025, cy, cz, true, true)); // يمين
   248→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.maxY - 0.025, cz, true, true)); // سقف
   249→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.minY + 0.025, cz, true, true)); // أرضية
   250→  g.add(box(wX, hY, 0.05, wallMat, cx, cy, CASE.minZ + 0.025, true, true)); // خلف
   251→  g.add(box(wX, hY, 0.05, mat(0x1a1e24, 0.35, 0.6), cx, cy, CASE.maxZ - 0.025, true, true)); // أمام
   252→
   253→  // قصورة I/O الخلفية (مستطيل داكن بارز قليلًا خارج اللوحة الخلفية)
   254→  g.add(box(0.55, 1.1, 0.03, mat(COLORS.black, 0.3, 0.6), 1.35, 3.35, CASE.minZ - 0.005, false));
   255→
   256→  // 7 مقابس توسعة أفقية (محور X) مكدسة رأسيًا أسفل I/O
   257→  const slotStripMat = mat(COLORS.metalMid, 0.6, 0.45);
   258→  for (let i = 0; i < 7; i++) {
   259→    g.add(box(0.72, 0.05, 0.03, slotStripMat, 1.35, 2.55 - i * 0.13, CASE.minZ - 0.005, false));
   260→  }
   261→
   262→  // شريط شبكة تهوية على الواجهة الأمامية (شرائط أفقية رفيعة)
   263→  const ventMat = mat(0x11151b, 0.3, 0.6);
   264→  for (let i = 0; i < 9; i++) {
   265→    g.add(box(1.5, 0.035, 0.03, ventMat, 1.05, 1.1 + i * 0.275, CASE.maxZ + 0.035, false));
   266→  }
   267→
   268→  // زر التشغيل (أسطوانة على السطح العلوي قرب الواجهة) + لمبة LED
   269→  g.add(cyl(0.07, 0.06, mat(COLORS.metalMid, 0.5, 0.4), 'y', 0.5, CASE.maxY + 0.005, CASE.maxZ - 0.45, false));
   270→  const ledMat = new THREE.MeshStandardMaterial({
   271→    color: 0x10151b, emissive: 0x000000, emissiveIntensity: 1.2, roughness: 0.35, metalness: 0.1, flatShading: false,
   272→  });
   273→  const powerLed = new THREE.Mesh(new THREE.SphereGeometry(0.03, 12, 8), ledMat); // مادة فريدة يتحكم بها علم powerOn
   274→  powerLed.position.set(0.78, CASE.maxY + 0.01, CASE.maxZ - 0.45);
   275→  flags.powerLed = powerLed;
   276→  g.add(powerLed);
   277→
   278→  // قاعدتان سفليتان
   279→  const footMat = mat(COLORS.black, 0.3, 0.6);
   280→  g.add(box(wX, 0.06, 0.45, footMat, cx, 0.03, CASE.minZ + 0.45, true));
   281→  g.add(box(wX, 0.06, 0.45, footMat, cx, 0.03, CASE.maxZ - 0.45, true));
   282→
   283→  // 6 مسامير تثبيت اللوحة الأم — أبناء الهيكل (تبقى في الهيكل عند فك اللوحة)
   284→  const ys = [BOARD.yMin + 0.1, (BOARD.yMin + BOARD.yMax) / 2, BOARD.yMax - 0.1];
   285→  const zs = [BOARD.zMin + 0.1, BOARD.zMax - 0.1];
   286→  for (const wy of ys) {
   287→    for (const wz of zs) {
   288→      const s = makeScrew(0.035, 0.3, 'x');
   289→      s.position.set(BOARD.x + 0.05, wy, wz);
   290→      flags.moboScrews.push(s);
   291→      g.add(s);
   292→    }
   293→  }
   294→
   295→  setComponentId(g, 'case');
   296→  return g;
   297→}

   324→// ============================================================
   325→// اللوحة الأم (dynamic.motherboard) — متمركزة عند MOUNTS.motherboard
   326→// محليًا: الوجه الأمامي نحو -X عند x = -thickness/2
   327→// ============================================================
   328→
   329→function buildMotherboard(flags: FlagVisuals): THREE.Group {
   330→  const g = new THREE.Group();
   331→  g.name = 'part-motherboard';
   332→  const face = -BOARD.thickness / 2;                  // -0.03
   333→  const onFace = (t: number): number => face - t / 2; // مركز قطعة سماكتها t على الوجه
   334→
   335→  // 1) جسم PCB (السماكة على X، الأبعاد من BOARD)
   336→  g.add(box(BOARD.thickness, BOARD.yMax - BOARD.yMin, BOARD.zMax - BOARD.zMin, mat(COLORS.pcb, 0.35, 0.6), 0, 0, 0, true, true));
   337→
   338→  // 2) مقبس المعالج: إطار + لوح معدني داخلي (عالمي y=3.45, z=0.95)
   339→  g.add(box(0.06, 0.42, 0.42, mat(COLORS.aluminum, 0.8, 0.45), onFace(0.06), 0.7, -0.62, true));
   340→  g.add(box(0.04, 0.36, 0.36, mat(0xc2c9d1, 0.9, 0.3), onFace(0.04) - 0.005, 0.7, -0.62, false));
   341→
   342→  // رافعة المقبس: pivot عند حافة المقبس (يدور حول X المحلي عند الفتح)
   343→  const lever = new THREE.Object3D();
   344→  lever.name = 'flag-cpu-lever';
   345→  lever.position.set(onFace(0.06) - 0.04, 0.45, -0.87);
   346→  lever.add(box(0.025, 0.42, 0.045, mat(COLORS.plastic, 0.2, 0.5), 0, 0.21, 0.02, false));
   347→  flags.cpuLever = lever;
   348→  g.add(lever);
   349→
   350→  // 3) أربع فتحات RAM عمودية + مزلاجان لكل فتحة (8 pivots تدور حول X)
   351→  const ramSlotGeo = new THREE.BoxGeometry(0.05, 1.35, 0.08);
   352→  const ramSlotMat = mat(0x1b1f26, 0.2, 0.55);
   353→  const clipGeo = new THREE.BoxGeometry(0.07, 0.07, 0.1);
   354→  const clipMat = mat(0x39414c, 0.3, 0.5);
   355→  for (const z of [-0.27, -0.07, 0.13, 0.33]) { // عالمي z = 1.30 .. 1.90
   356→    const slot = new THREE.Mesh(ramSlotGeo, ramSlotMat);
   357→    slot.position.set(onFace(0.05), 0.55, z);
   358→    g.add(slot);
   359→    for (const end of [1, -1]) {
   360→      const clip = new THREE.Object3D();
   361→      clip.name = 'flag-ram-clip';
   362→      clip.position.set(onFace(0.05) - 0.005, 0.55 + end * 0.675, z);
   363→      const cm = new THREE.Mesh(clipGeo, clipMat);
   364→      cm.position.set(0, end * 0.05, 0);
   365→      clip.add(cm);
   366→      flags.ramClips.push(clip);
   367→      g.add(clip);
   368→    }
   369→  }
   370→
   371→  // 4) فتحتا PCIe أفقيتان على Z (x16 عند عالمي y=2.0، x4 عند 1.65)
   372→  const pcieMat = mat(0x242a33, 0.2, 0.55);
   373→  g.add(box(0.05, 0.07, 1.85, pcieMat, onFace(0.05), -0.75, -0.15, true));
   374→  g.add(box(0.05, 0.07, 1.85, pcieMat, onFace(0.05), -1.1, -0.15, true));
   375→

   763→  const cz = (TRAY.minZ + TRAY.maxZ) / 2; // 2.4
   764→
   765→  // السطح (أعلى = TRAY.tableTopY = 0.9)
   766→  g.add(box(TRAY.maxX - TRAY.minX, 0.08, TRAY.maxZ - TRAY.minZ, mat(COLORS.metalDark, 0.4, 0.65), cx, 0.86, cz, true, true));
   767→
   768→  // 4 أرجل أسطوانية
   769→  const legGeo = new THREE.CylinderGeometry(0.05, 0.05, 0.82, 12);
   770→  const legMat = mat(COLORS.metalMid, 0.6, 0.45);
   771→  for (const lx of [TRAY.minX + 0.15, TRAY.maxX - 0.15]) {
   772→    for (const lz of [TRAY.minZ + 0.15, TRAY.maxZ - 0.15]) {
   773→      const leg = new THREE.Mesh(legGeo, legMat);
   774→      leg.position.set(lx, 0.41, lz);
   775→      if (castEnabled) leg.castShadow = true;
   776→      g.add(leg);
   777→    }
   778→  }
   779→
   780→  // خطوط خانات باهتة لكل مكوّن (عدا case) — تُثبَّت داخل حدود الطاولة
   781→  const markers: Array<{ id: ComponentId; size: readonly [number, number] }> = [
   782→    { id: 'ram', size: [0.55, 1.55] },
   783→    { id: 'gpu', size: [1.15, 2.45] },
   784→    { id: 'cpu', size: [0.55, 0.55] },
   785→    { id: 'gpuPower', size: [0.4, 0.45] },
   786→    { id: 'cooler', size: [0.7, 0.95] },
   787→    { id: 'ssd', size: [0.3, 0.9] },
   788→    { id: 'fans', size: [1.0, 1.0] },
   789→    { id: 'psu', size: [1.5, 1.5] },
   790→    { id: 'motherboard', size: [2.05, 2.4] },
   791→    { id: 'atx24', size: [0.4, 0.6] },
   792→    { id: 'eps', size: [0.35, 0.5] },
   793→  ];
   794→  const markerMat = mat(0x1a2029, 0.2, 0.8);
   795→  for (const mk of markers) {
   796→    const p = TRAY.slots[mk.id];
   797→    const px = Math.min(Math.max(p[0], TRAY.minX + mk.size[0] / 2 + 0.02), TRAY.maxX - mk.size[0] / 2 - 0.02);
   798→    const pz = Math.min(Math.max(p[2], TRAY.minZ + mk.size[1] / 2 + 0.02), TRAY.maxZ - mk.size[1] / 2 - 0.02);
   799→    g.add(box(mk.size[0], 0.012, mk.size[1], markerMat, px, 0.906, pz, false));
   800→  }
   801→
   802→  return g;
   803→}

   809→export function buildAllParts(quality: SceneQuality): PartVisuals {
   810→  castEnabled = quality !== 'low';
   811→
   812→  const flags: FlagVisuals = {
   813→    ramClips: [],
   814→    gpuScrews: [],
   815→    psuScrews: [],
   816→    fanScrews: [],
   817→    moboScrews: [],
   818→  };
   819→  const spinners: THREE.Object3D[] = [];
   820→
   821→  // بناء الأجزاء
   822→  const caseShell = buildCaseShell(flags);
   823→  const sidePanel = buildSidePanel();
   824→  const motherboard = buildMotherboard(flags);
   825→  const cpu = buildCpu();
   826→  const cooler = buildCooler(spinners);
   827→  const ram = buildRam();
   828→  const gpu = buildGpu(flags, spinners);
   829→  const psu = buildPsu(flags);
   830→  const ssd = buildSsd();
   831→  const fans = buildFans(flags, spinners);
   832→  const atx24 = buildAtx24();
   833→  const eps = buildEps();
   834→  const gpuPower = buildGpuPower();
   835→  const trayTable = buildTrayTable();
   836→
   837→  // مواضع التثبيت الابتدائية (المدير يعيد تموضعها عند الفك/التفجير)
   838→  const place = (grp: THREE.Group, id: ComponentId): void => {
   839→    grp.position.set(...MOUNTS[id]);
   840→  };
   841→  place(motherboard, 'motherboard');
   842→  place(cpu, 'cpu');
   843→  place(cooler, 'cooler');
   844→  place(ram, 'ram');
   845→  place(gpu, 'gpu');
   846→  place(psu, 'psu');
   847→  place(ssd, 'ssd');
   848→  place(fans, 'fans');
   849→  place(atx24, 'atx24');
   850→  place(eps, 'eps');
   851→  place(gpuPower, 'gpuPower');
   852→
   853→  // مجموعة placeholder للمكوّن 'case' (الهيكل نفسه لا يُحرك أبدًا — فارغة للأمان)
   854→  const caseAnchor = new THREE.Group();
   855→  caseAnchor.name = 'case-anchor';
   856→  place(caseAnchor, 'case');
   857→
   858→  const root = new THREE.Group();
   859→  root.name = 'pc-parts-root';
   860→  root.add(caseShell, sidePanel, trayTable, caseAnchor);
   861→  root.add(motherboard, cpu, cooler, ram, gpu, psu, ssd, fans, atx24, eps, gpuPower);
   862→
   863→  const dynamic: Record<ComponentId, THREE.Group> = {
   864→    case: caseAnchor,
   865→    motherboard,
   866→    cpu,
   867→    cooler,
   868→    ram,
   869→    gpu,
   870→    psu,
   871→    ssd,
   872→    fans,
   873→    atx24,
   874→    eps,
   875→    gpuPower,
   876→  };
   877→
   878→  return { root, dynamic, caseShell, sidePanel, flags, spinners };
   879→}
