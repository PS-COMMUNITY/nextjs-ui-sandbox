   100→function StatusBadge({ id }: { id: ComponentId }) {
   101→  const sim = useLabStore((s) => s.sim);
   102→  if (id === 'case') {
   103→    return <Badge variant="outline" className="gap-1">هيكل الجهاز</Badge>;
   104→  }
   105→  const secureOpen = (SECURE_FLAGS[id] ?? []).some((f) => sim.flags[f]);
   106→  const state = assemblyStateOf(sim.parts[id], secureOpen);
   107→  const info = ASSEMBLY_STATE_INFO[state];
   108→  const colorClass: Record<string, string> = {
   109→    emerald: 'bg-emerald-600/15 text-emerald-400 border-emerald-600/40',
   110→    amber: 'bg-amber-600/15 text-amber-400 border-amber-600/40',
   111→    slate: 'bg-slate-600/15 text-slate-300 border-slate-600/40',
   112→    cyan: 'bg-cyan-600/15 text-cyan-400 border-cyan-600/40',
   113→    rose: 'bg-rose-600/15 text-rose-400 border-rose-600/40',
   114→  };
   115→  const icon =
   116→    state === 'INSTALLED_CORRECTLY' ? <CheckCircle2 className="h-3 w-3" aria-hidden /> :
   117→    state === 'AWAITING_INSTALL' ? <HandMetal className="h-3 w-3" aria-hidden /> :
   118→    state === 'TRAINING_DAMAGE' ? <TriangleAlert className="h-3 w-3" aria-hidden /> :
   119→    <Wrench className="h-3 w-3" aria-hidden />;
   120→  return (
   121→    <Badge variant="outline" className={cn('gap-1', colorClass[info.colorAr] ?? '')}>
   122→      {icon}
   123→      {state === 'INSTALLED_CORRECTLY' ? 'مثبتة بإحكام' : state === 'INSTALLED_SECURED' ? 'مثبتة (مزالج مفتوحة)' : info.labelAr}
   124→    </Badge>
   125→  );
   126→}

   128→/** خطوة قائمة تحقق قابلة للتنفيذ (علم ميكانيكي أو فك/تركيب قطعة) */
   129→function ChecklistRow({
   130→  item,
   131→  mode,
   132→  onFlag,
   133→  onPart,
   134→}: {
   135→  item: ChecklistItem;
   136→  mode: 'remove' | 'install';
   137→  onFlag: (f: MachineFlag) => void;
   138→  onPart: (id: ComponentId) => void;
   139→}) {
   140→  const isFlag = item.source !== 'safetyOff' && item.source !== 'safetyUnplug' && item.source !== 'base';
   141→  const canRunFlag = !item.done && isFlag && item.available;
   142→  const canRunPart = !item.done && !isFlag && item.actionPart && item.available;
   143→  return (
   144→    <li
   145→      className={cn(
   146→        'flex items-center gap-2 rounded-md border px-2.5 py-2 text-xs',
   147→        item.done ? 'border-emerald-600/30 bg-emerald-600/10' : 'border-border bg-card',
   148→        !item.done && !item.available && 'opacity-60',
   149→      )}
   150→    >
   151→      {item.done ? (
   152→        <CheckCircle2 className="h-4 w-4 shrink-0 text-emerald-500" aria-hidden />
   153→      ) : (
   154→        <CircleDashed className="h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   155→      )}
   156→      <div className="min-w-0 flex-1">
   157→        <p className={cn('font-medium leading-tight', item.done && 'line-through opacity-70')}>{item.labelAr}</p>
   158→        {!item.done && item.hintAr ? (
   159→          <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">{item.hintAr}</p>
   160→        ) : null}
   161→      </div>
   162→      {canRunFlag ? (
   163→        <Button size="sm" variant="secondary" className="h-8 gap-1 px-2.5 text-[11px] shrink-0 max-sm:h-11 max-sm:px-3" onClick={() => onFlag(item.source as MachineFlag)}>
   164→          <Hammer className="h-3 w-3" aria-hidden /> نفّذ
   165→        </Button>
   166→      ) : null}
   167→      {canRunPart ? (
   168→        <Button size="sm" variant="secondary" className="h-8 gap-1 px-2.5 text-[11px] shrink-0 max-sm:h-11 max-sm:px-3" onClick={() => onPart(item.actionPart!)}>
   169→          <Hammer className="h-3 w-3" aria-hidden />
   170→          {mode === 'remove' ? 'افصل' : 'وصّل'}
   171→        </Button>
   172→      ) : null}
   173→    </li>
   174→  );
   175→}
