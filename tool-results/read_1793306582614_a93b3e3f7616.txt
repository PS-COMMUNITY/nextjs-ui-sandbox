    25→    25→/** cache للمواد القياسية — لا تنشئ مواد جديدة لكل إطار */
    26→    26→const matCache = new Map<string, THREE.MeshStandardMaterial>();
    27→    27→
    28→    28→function mat(color: number, metalness: number, roughness: number): THREE.MeshStandardMaterial {
    29→    29→  const key = `${color}-${metalness}-${roughness}`;
    30→    30→  const cached = matCache.get(key);
    31→    31→  if (cached) return cached;
    32→    32→  const m = new THREE.MeshStandardMaterial({ color, metalness, roughness, flatShading: false });
    33→    33→  matCache.set(key, m);
    34→    34→  return m;
    35→    35→}

    38→    38→let glassMat: THREE.MeshPhysicalMaterial | null = null;
    39→    39→function glass(): THREE.MeshPhysicalMaterial {
    40→    40→  if (!glassMat) {
    41→    41→    glassMat = new THREE.MeshPhysicalMaterial({

    44→    44→      roughness: 0.08,
    45→    45→      transparent: true,
    46→    46→      opacity: 0.16,
    47→    47→      side: THREE.DoubleSide,
    48→    48→      flatShading: false,
    49→    49→    });
    50→    50→  }
    51→    51→  return glassMat;
    52→    52→}

    58→    58→/** صندوق سريع مع ضبط الظلال حسب الجودة */
    59→    59→function box(
    60→    60→  w: number, h: number, d: number, material: THREE.Material,
    61→    61→  x = 0, y = 0, z = 0, cast = true, receive = false,
    62→    62→): THREE.Mesh {
    63→    63→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), material);
    64→    64→  m.position.set(x, y, z);
    65→    65→  if (castEnabled && cast) m.castShadow = true;
    66→    66→  if (receive) m.receiveShadow = true;
    67→    67→  return m;
    68→    68→}

    70→    70→type Axis = 'x' | 'y' | 'z';

    72→    72→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
    73→    73→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
    74→    74→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
    75→    75→  if (axis === 'x') m.rotation.z = Math.PI / 2;
    76→    76→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
    77→    77→  m.position.set(x, y, z);
    78→    78→  if (castEnabled && cast) m.castShadow = true;
    79→    79→  return m;
    80→    80→}

    83→    83→function makeScrew(r: number, length: number, axis: Axis = 'y'): THREE.Mesh {
    84→    84→  return cyl(r, length, mat(COLORS.aluminum, 0.85, 0.35), axis);
    85→    85→}

    87→    87→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
    88→    88→function makeCable(points: THREE.Vector3[], radius: number): THREE.Mesh {
    89→    89→  const curve = new THREE.CatmullRomCurve3(points);
    90→    90→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(0x181d24, 0.3, 0.6));
    91→    91→  if (castEnabled) m.castShadow = true;
    92→    92→  return m;
    93→    93→}

    95→    95→/** تحويل نقاط كابل عالمية إلى محلية بالنسبة لموضع تثبيت القطعة */
    96→    96→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
    97→    97→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));
    98→    98→}

   100→   100→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   101→   101→
   102→   102→/**
   103→   103→ * مروحة عامة: إطار (4 أركان + حلقة Torus) + شفرات (7 ألواح مائلة 30°) + محور مركزي.
   104→   104→ * محور الدوران الافتراضي = Y المحلي (تُلف المروحة لاحقًا حسب موضعها).
   105→   105→ */
   106→   106→function makeFan(radius: number): FanParts {
   107→   107→  const depth = Math.max(0.12, radius * 0.55);
   108→   108→  const half = radius + 0.06;
   109→   109→
   110→   110→  // الإطار
   111→   111→  const frame = new THREE.Group();
   112→   112→  const frameMat = mat(COLORS.plastic, 0.3, 0.6);
   113→   113→  const postGeo = new THREE.BoxGeometry(0.09, depth, 0.09);
   114→   114→  for (const sx of [-1, 1]) {
   115→   115→    for (const sz of [-1, 1]) {
   116→   116→      const post = new THREE.Mesh(postGeo, frameMat);
   117→   117→      post.position.set(sx * half, 0, sz * half);
   118→   118→      if (castEnabled) post.castShadow = true;
   119→   119→      frame.add(post);
   120→   120→    }
   121→   121→  }
   122→   122→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.97, 0.018, 6, 24), mat(COLORS.metalDark, 0.5, 0.5));
   123→   123→  ring.rotation.x = Math.PI / 2; // الحلقة في مستوى XZ (عمودية على محور Y)
   124→   124→  if (castEnabled) ring.castShadow = true;
   125→   125→  frame.add(ring);
   126→   126→
   127→   127→  // الشفرات: 7 ألواح موزعة دائريًا داخل مجموعة blades (تُدفع إلى spinners)
   128→   128→  const blades = new THREE.Group();
   129→   129→  const bladeGeo = new THREE.BoxGeometry(0.02, radius * 0.8, radius * 0.35);
   130→   130→  const bladeMat = mat(COLORS.metalMid, 0.4, 0.5);
   131→   131→  for (let i = 0; i < 7; i++) {
   132→   132→    const arm = new THREE.Object3D();
   133→   133→    arm.rotation.y = (i / 7) * Math.PI * 2;
   134→   134→    const pitch = new THREE.Object3D();
   135→   135→    pitch.position.set(radius * 0.5, 0, 0);
   136→   136→    pitch.rotation.x = Math.PI / 6; // ميلان الشفرة 30° حول المحور الشعاعي
   137→   137→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   138→   138→    blade.rotation.z = Math.PI / 2; // الطول شعاعيًا
   139→   139→    if (castEnabled) blade.castShadow = true;
   140→   140→    pitch.add(blade);
   141→   141→    arm.add(pitch);
   142→   142→    blades.add(arm);
   143→   143→  }
   144→   144→  const hub = new THREE.Mesh(
   145→   145→    new THREE.CylinderGeometry(radius * 0.16, radius * 0.16, depth * 0.7, 12),
   146→   146→    mat(COLORS.metalDark, 0.5, 0.4),
   147→   147→  );
   148→   148→  if (castEnabled) hub.castShadow = true;
   149→   149→  blades.add(hub);
   150→   150→
   151→   151→  return { frame, blades };
   152→   152→}

   172→   172→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   173→   173→  scene.background = new THREE.Color(COLORS.bg);
   174→   174→  scene.fog = new THREE.Fog(COLORS.bg, 18, 42);
   175→   175→
   176→   176→  // أرضية دائرية داكنة تستقبل الظلال
   177→   177→  const ground = new THREE.Mesh(
   178→   178→    new THREE.CircleGeometry(22, 24),
   179→   179→    new THREE.MeshStandardMaterial({ color: 0x141a22, metalness: 0, roughness: 0.92, flatShading: false }),
   180→   180→  );
   181→   181→  ground.rotation.x = -Math.PI / 2;
   182→   182→  ground.position.y = 0;
   183→   183→  ground.receiveShadow = true;
   184→   184→  scene.add(ground);
   185→   185→
   186→   186→  // شبكة أرضية خفيفة
   187→   187→  const grid = new THREE.GridHelper(30, 30, 0x2a3340, 0x1a212b);
   188→   188→  grid.position.y = 0.01;
   189→   189→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   190→   190→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   191→   191→  for (const gm of gridMats) {
   192→   192→    gm.transparent = true;
   193→   193→    gm.opacity = 0.4;
   194→   194→  }
   195→   195→  scene.add(grid);
   196→   196→
   197→   197→  // إضاءة مباشرة مخفّضة — الخريطة البيئية (scene.environment) تتكفل بالملء العام
   198→   198→  scene.add(new THREE.AmbientLight(0xffffff, 0.45));
   199→   199→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x20242c, 0.5));
   200→   200→
   201→   201→  // شمس دافئة مع ظلال (حسب الجودة)
   202→   202→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.0);
   203→   203→  dir.position.set(-6, 8, 6);
   204→   204→  if (quality !== 'low') {
   205→   205→    dir.castShadow = true;
   206→   206→    const size = quality === 'high' ? 2048 : 1024;
   207→   207→    dir.shadow.mapSize.set(size, size);
   208→   208→    dir.shadow.camera.left = -6;
   209→   209→    dir.shadow.camera.right = 6;
   210→   210→    dir.shadow.camera.top = 6;
   211→   211→    dir.shadow.camera.bottom = -6;
   212→   212→    dir.shadow.camera.near = 2;
   213→   213→    dir.shadow.camera.far = 30;
   214→   214→    dir.shadow.camera.updateProjectionMatrix();
   215→   215→    dir.shadow.bias = -0.0004;
   216→   216→    dir.shadow.normalBias = 0.02;
   217→   217→  }
   218→   218→  scene.add(dir);
   219→   219→
   220→   220→  // ضوء تعبئة خفيف خلفي (بلا ظلال؛ decay=0 ليبقى «خفيفًا» بمقدار 0.5 فعليًا)
   221→   221→  const fill = new THREE.PointLight(0xffffff, 0.5, 0, 0);
   222→   222→  fill.position.set(3, 3, -2);
   223→   223→  scene.add(fill);
   224→   224→  // ضوء أمامي موجّه من جهة الزجاج لإبراز وجه اللوحة الأم
   225→   225→  const front = new THREE.PointLight(0xfff4e0, 1.0, 0, 0);
   226→   226→  front.position.set(-5, 3.5, 3);
   227→   227→  scene.add(front);
   228→   228→}

   234→   234→function buildCaseShell(flags: FlagVisuals): THREE.Group {
   235→   235→  const g = new THREE.Group();
   236→   236→  g.name = 'case-shell';
   237→   237→
   238→   238→  const cx = (CASE.minX + CASE.maxX) / 2; // 1.05
   239→   239→  const cy = (CASE.minY + CASE.maxY) / 2; // 2.25
   240→   240→  const cz = (CASE.minZ + CASE.maxZ) / 2; // 2.3
   241→   241→  const wX = CASE.maxX - CASE.minX;       // 2.1
   242→   242→  const hY = CASE.maxY - CASE.minY;       // 4.5
   243→   243→  const dZ = CASE.maxZ - CASE.minZ;       // 4.6
   244→   244→  const wallMat = mat(COLORS.metalDark, 0.45, 0.55);
   245→   245→  const trayWallMat = mat(0x2c343e, 0.5, 0.5); // جدار صينية اللوحة الأم — أفتح قليلًا لتمييزه
   246→   246→
   247→   247→  // الجدران (سماكة 0.05) — الجدار الأيمن عند x≈2.1 يستقبل اللوحة الأم
   248→   248→  g.add(box(0.05, hY, dZ, trayWallMat, CASE.maxX - 0.025, cy, cz, true, true)); // يمين
   249→   249→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.maxY - 0.025, cz, true, true)); // سقف
   250→   250→  g.add(box(wX, 0.05, dZ, wallMat, cx, CASE.minY + 0.025, cz, true, true)); // أرضية
   251→   251→  g.add(box(wX, hY, 0.05, wallMat, cx, cy, CASE.minZ + 0.025, true, true)); // خلف
   252→   252→  g.add(box(wX, hY, 0.05, mat(0x262d36, 0.35, 0.6), cx, cy, CASE.maxZ - 0.025, true, true)); // أمام
   253→   253→
   254→   254→  // حافة مشطوفة على مخطط الصينية (شريط معدني فاتح يفصل الصينية عن قصورة الكابلات)
   255→   255→  g.add(box(0.02, hY * 0.92, 0.02, mat(0x414c58, 0.6, 0.4), CASE.maxX - 0.18, cy, CASE.maxZ - 0.62, false));
   256→   256→
   257→   257→  // قصورة I/O الخلفية (مستطيل داكن بارز قليلًا خارج اللوحة الخلفية)
   258→   258→  g.add(box(0.55, 1.1, 0.03, mat(COLORS.black, 0.3, 0.6), 1.35, 3.35, CASE.minZ - 0.005, false));
   259→   259→
   260→   260→  // 7 مقابس توسعة أفقية (محور X) مكدسة رأسيًا أسفل I/O
   261→   261→  const slotStripMat = mat(COLORS.metalMid, 0.6, 0.45);
   262→   262→  for (let i = 0; i < 7; i++) {
   263→   263→    g.add(box(0.72, 0.05, 0.03, slotStripMat, 1.35, 2.55 - i * 0.13, CASE.minZ - 0.005, false));
   264→   264→  }
   265→   265→
   266→   266→  // شريط شبكة تهوية على الواجهة الأمامية (شرائط أفقية رفيعة)
   267→   267→  const ventMat = mat(0x11151b, 0.3, 0.6);
   268→   268→  for (let i = 0; i < 9; i++) {
   269→   269→    g.add(box(1.5, 0.035, 0.03, ventMat, 1.05, 1.1 + i * 0.275, CASE.maxZ + 0.035, false));
   270→   270→  }
   271→   271→
   272→   272→  // زر التشغيل (أسطوانة على السطح العلوي قرب الواجهة) + لمبة LED

   275→   275→    color: 0x10151b, emissive: 0x000000, emissiveIntensity: 1.2, roughness: 0.35, metalness: 0.1, flatShading: false,
   276→   276→  });
   277→   277→  const powerLed = new THREE.Mesh(new THREE.SphereGeometry(0.03, 12, 8), ledMat); // مادة فريدة يتحكم بها علم powerOn
   278→   278→  powerLed.position.set(0.78, CASE.maxY + 0.01, CASE.maxZ - 0.45);
   279→   279→  flags.powerLed = powerLed;
   280→   280→  g.add(powerLed);
   281→   281→
   282→   282→  // قاعدتان سفليتان
   283→   283→  const footMat = mat(COLORS.black, 0.3, 0.6);
   284→   284→  g.add(box(wX, 0.06, 0.45, footMat, cx, 0.03, CASE.minZ + 0.45, true));
   285→   285→  g.add(box(wX, 0.06, 0.45, footMat, cx, 0.03, CASE.maxZ - 0.45, true));
   286→   286→
   287→   287→  // 6 مسامير تثبيت اللوحة الأم — أبناء الهيكل (تبقى في الهيكل عند فك اللوحة)
   288→   288→  const ys = [BOARD.yMin + 0.1, (BOARD.yMin + BOARD.yMax) / 2, BOARD.yMax - 0.1];
   289→   289→  const zs = [BOARD.zMin + 0.1, BOARD.zMax - 0.1];
   290→   290→  for (const wy of ys) {
   291→   291→    for (const wz of zs) {
   292→   292→      const s = makeScrew(0.035, 0.3, 'x');
   293→   293→      s.position.set(BOARD.x + 0.05, wy, wz);
   294→   294→      flags.moboScrews.push(s);
   295→   295→      g.add(s);
   296→   296→    }
   297→   297→  }
   298→   298→
   299→   299→  setComponentId(g, 'case');
   300→   300→  return g;
   301→   301→}

   329→   329→// اللوحة الأم (dynamic.motherboard) — متمركزة عند MOUNTS.motherboard

   449→   449→
   450→   450→  setComponentId(g, 'motherboard');
   451→   451→  return g;
   452→   452→}

   661→   661→function buildFans(flags: FlagVisuals, spinners: THREE.Object3D[]): THREE.Group {
   662→   662→  const g = new THREE.Group();
   663→   663→  g.name = 'part-fans';
   664→   664→  const mount = MOUNTS.fans;
   665→   665→  const defs: Array<{ pos: readonly [number, number, number]; rotX: number }> = [
   666→   666→    { pos: [1.05, 2.9, 4.42], rotX: Math.PI / 2 },  // أمامية علوية (محور Z)
   667→   667→    { pos: [1.05, 1.5, 4.42], rotX: Math.PI / 2 },  // أمامية سفلية
   668→   668→    { pos: [1.05, 3.9, 0.12], rotX: -Math.PI / 2 }, // خلفية علوية (تدفع للخلف)
   669→   669→  ];
   670→   670→  for (const d of defs) {
   671→   671→    const wrapper = new THREE.Group();
   672→   672→    wrapper.position.set(d.pos[0] - mount[0], d.pos[1] - mount[1], d.pos[2] - mount[2]);
   673→   673→    wrapper.rotation.x = d.rotX; // محور الدوران Y المحلي → Z عالمي
   674→   674→    const fan = makeFan(0.42);
   675→   675→    wrapper.add(fan.frame, fan.blades);
   676→   676→    spinners.push(fan.blades);
   677→   677→    // 4 براغي زوايا لكل مروحة (على المحور المحلي Y = محور المروحة)
   678→   678→    const half = 0.42 + 0.06;
   679→   679→    const sy = 0.42 * 0.55 / 2 + 0.015;
   680→   680→    for (const sx of [-1, 1]) {
   681→   681→      for (const sz of [-1, 1]) {
   682→   682→        const s = makeScrew(0.03, 0.06, 'y');
   683→   683→        s.position.set(sx * half, sy, sz * half);
   684→   684→        s.name = 'flag-fan-screw';
   685→   685→        flags.fanScrews.push(s);
   686→   686→        wrapper.add(s);
   687→   687→      }
   688→   688→    }
   689→   689→    g.add(wrapper);
   690→   690→  }
   691→   691→  setComponentId(g, 'fans');
   692→   692→  return g;
   693→   693→}

   789→   789→// صينية القطع (داخل root مباشرة — بلا componentId)
   790→   790→// ============================================================
   791→   791→
   792→   792→function buildTrayTable(): THREE.Group {
   793→   793→  const g = new THREE.Group();
   794→   794→  g.name = 'tray-table'; // غير قابلة للاختيار عمدًا
   795→   795→
   796→   796→  const cx = (TRAY.minX + TRAY.maxX) / 2; // -3.35
   797→   797→  const cz = (TRAY.minZ + TRAY.maxZ) / 2; // 2.4
   798→   798→
   799→   799→  // السطح (أعلى = TRAY.tableTopY = 0.9)
   800→   800→  g.add(box(TRAY.maxX - TRAY.minX, 0.08, TRAY.maxZ - TRAY.minZ, mat(0x2b333d, 0.4, 0.6), cx, 0.86, cz, true, true));
   801→   801→
   802→   802→  // 4 أرجل أسطوانية
   803→   803→  const legGeo = new THREE.CylinderGeometry(0.05, 0.05, 0.82, 12);
   804→   804→  const legMat = mat(COLORS.metalMid, 0.6, 0.45);
   805→   805→  for (const lx of [TRAY.minX + 0.15, TRAY.maxX - 0.15]) {
   806→   806→    for (const lz of [TRAY.minZ + 0.15, TRAY.maxZ - 0.15]) {
   807→   807→      const leg = new THREE.Mesh(legGeo, legMat);
   808→   808→      leg.position.set(lx, 0.41, lz);
   809→   809→      if (castEnabled) leg.castShadow = true;
   810→   810→      g.add(leg);
   811→   811→    }
   812→   812→  }
   813→   813→
   814→   814→  // خطوط خانات باهتة لكل مكوّن (عدا case) — تُثبَّت داخل حدود الطاولة
   815→   815→  const markers: Array<{ id: ComponentId; size: readonly [number, number] }> = [
   816→   816→    { id: 'ram', size: [0.55, 1.55] },
   817→   817→    { id: 'gpu', size: [1.15, 2.45] },
   818→   818→    { id: 'cpu', size: [0.55, 0.55] },
   819→   819→    { id: 'gpuPower', size: [0.4, 0.45] },
   820→   820→    { id: 'cooler', size: [0.7, 0.95] },
   821→   821→    { id: 'ssd', size: [0.3, 0.9] },
   822→   822→    { id: 'fans', size: [1.0, 1.0] },
   823→   823→    { id: 'psu', size: [1.5, 1.5] },
   824→   824→    { id: 'motherboard', size: [2.05, 2.4] },
   825→   825→    { id: 'atx24', size: [0.4, 0.6] },
   826→   826→    { id: 'eps', size: [0.35, 0.5] },
   827→   827→  ];
   828→   828→  const markerMat = mat(0x24303a, 0.2, 0.8);
   829→   829→  for (const mk of markers) {
   830→   830→    const p = TRAY.slots[mk.id];
   831→   831→    const px = Math.min(Math.max(p[0], TRAY.minX + mk.size[0] / 2 + 0.02), TRAY.maxX - mk.size[0] / 2 - 0.02);
   832→   832→    const pz = Math.min(Math.max(p[2], TRAY.minZ + mk.size[1] / 2 + 0.02), TRAY.maxZ - mk.size[1] / 2 - 0.02);
   833→   833→    g.add(box(mk.size[0], 0.012, mk.size[1], markerMat, px, 0.906, pz, false));
   834→   834→  }
   835→   835→
   836→   836→  return g;
   837→   837→}

   859→   859→  const cpu = buildCpu();
   860→   860→  const cooler = buildCooler(spinners);
   861→   861→  const ram = buildRam();

   863→   863→  const psu = buildPsu(flags);

   872→   872→  const place = (grp: THREE.Group, id: ComponentId): void => {
   873→   873→    grp.position.set(...MOUNTS[id]);

   875→   875→  place(motherboard, 'motherboard');

   883→   883→  place(atx24, 'atx24');
   884→   884→  place(eps, 'eps');
   885→   885→  place(gpuPower, 'gpuPower');
   886→   886→

   888→   888→  const caseAnchor = new THREE.Group();
   889→   889→  caseAnchor.name = 'case-anchor';

   898→   898→    case: caseAnchor,
   899→   899→    motherboard,

   904→   904→    psu,
   905→   905→    ssd,
   906→   906→    fans,
   907→   907→    atx24,
   908→   908→    eps,
   909→   909→    gpuPower,
   910→   910→  };
   911→   911→
   912→   912→  return { root, dynamic, caseShell, sidePanel, flags, spinners };
