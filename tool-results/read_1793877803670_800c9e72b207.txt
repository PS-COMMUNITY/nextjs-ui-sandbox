    83→    83→export function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
    84→    84→  const rr = Math.min(r, w / 2, h / 2);
    85→    85→  ctx.beginPath();
    86→    86→  ctx.moveTo(x + rr, y);
    87→    87→  ctx.arcTo(x + w, y, x + w, y + h, rr);
    88→    88→  ctx.arcTo(x + w, y + h, x, y + h, rr);
    89→    89→  ctx.arcTo(x, y + h, x, y, rr);
    90→    90→  ctx.arcTo(x, y, x + w, y, rr);
    91→    91→  ctx.closePath();
    92→    92→}

   108→   108→export class UiFrame {
   109→   109→  rects: HotRect[] = [];
   110→   110→  keyHandler: ((ev: VosKeyEvent) => boolean) | null = null; // true = استُهلك
   111→   111→  hoverId: string | null = null;
   112→   112→  textCursor: boolean = false;
   113→   113→
   114→   114→  add(r: HotRect): void {
   115→   115→    this.rects.push(r);
   116→   116→  }
   117→   117→
   118→   118→  hit(x: number, y: number): HotRect | null {
   119→   119→    // الأعلى z أولًا ثم الأحدث (ترتيب الإضافة يعكس التراكب داخل النافذة)
   120→   120→    let best: HotRect | null = null;
   121→   121→    for (const r of this.rects) {
   122→   122→      if (x >= r.x && x <= r.x + r.w && y >= r.y && y <= r.y + r.h) {
   123→   123→        if (!best || r.z > best.z) best = r;
   124→   124→      }
   125→   125→    }
   126→   126→    return best;
   127→   127→  }
   128→   128→}

   198→   198→  openApp(appId: string, args?: Record<string, unknown>): VosWindow | null {
   199→   199→    const app = this.apps.get(appId);
   200→   200→    if (!app) return null;
   201→   201→    if (app.fullscreenOnly) {
   202→   202→      // تطبيقات ملء الشاشة (المثبّت) تُدار كأوضاع لا نوافذ
   203→   203→      this.mode = 'setup';
   204→   204→      this.dirty = true;
   205→   205→      return null;
   206→   206→    }
   207→   207→    const existing = this.windows.find((w) => w.appId === appId);
   208→   208→    if (existing && !args) {
   209→   209→      existing.minimized = false;
   210→   210→      existing.z = ++this.zCounter;
   211→   211→      this.dirty = true;
   212→   212→      return existing;
   213→   213→    }
   214→   214→    const offset = (this.windows.length % 5) * 26;
   215→   215→    const win: VosWindow = {
   216→   216→      id: `win-${this.nextWinId++}`,
   217→   217→      appId,
   218→   218→      titleAr: app.titleAr,
   219→   219→      x: Math.round(this.W * 0.18) + offset,
   220→   220→      y: Math.round(this.H * 0.12) + offset,
   221→   221→      w: Math.min(app.defaultW, this.W - 40),
   222→   222→      h: Math.min(app.defaultH, this.H - TASKBAR_H - 40),
   223→   223→      minimized: false,
   224→   224→      maximized: false,
   225→   225→      z: ++this.zCounter,
   226→   226→      state: { ...(args ?? {}) },
   227→   227→    };
   228→   228→    this.windows.push(win);
   229→   229→    this.dirty = true;
   230→   230→    return win;

   309→   309→  pointer(type: 'move' | 'down' | 'up' | 'dbl' | 'right', x: number, y: number, opts: VosPointerOpts = {}): void {
   310→   310→    this.cursor.x = Math.max(0, Math.min(this.W - 1, x));
   311→   311→    this.cursor.y = Math.max(0, Math.min(this.H - 1, y));
   312→   312→    const f = this.lastFrame;
   313→   313→    if (!f) return;
   314→   314→
   315→   315→    if (type === 'move') {
   316→   316→      const hit = f.hit(x, y);
   317→   317→      this.cursorKind = hit ? hit.cursor : 'default';
   318→   318→      f.hoverId = hit?.id ?? null;
   319→   319→      this.dirty = true;
   320→   320→      return;
   321→   321→    }
   322→   322→
   323→   323→    if (type === 'down') {
   324→   324→      const hit = f.hit(x, y);
   325→   325→      this.activeId = hit?.id ?? null;
   326→   326→      this.activeStart = { x, y };
   327→   327→      this.downAt = Date.now();
   328→   328→      if (!hit && this.startOpen) this.startOpen = false;
   329→   329→      if (this.ctxMenu && (!hit || !hit.id.startsWith('ctx'))) this.ctxMenu = null;
   330→   330→      if (hit?.kind === 'title') {
   331→   331→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   332→   332→        if (win) {
   333→   333→          this.focusWindow(win.id);
   334→   334→          if (!win.maximized) this.dragWin = { id: win.id, dx: x - win.x, dy: y - win.y };
   335→   335→        }
   336→   336→      }
   337→   337→      if (hit?.kind === 'resize') {
   338→   338→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   339→   339→        if (win) this.dragResize = { id: win.id, sx: x, sy: y, w0: win.w, h0: win.h };
   340→   340→      }
   341→   341→      this.dirty = true;
   342→   342→      return;
   343→   343→    }
   344→   344→
   345→   345→    if (type === 'up') {
   346→   346→      const hit = f.hit(x, y);
   347→   347→      const now = Date.now();
   348→   348→      if (this.dragWin) {
   349→   349→        // إن لم يتحرك المؤشر فعليًا فهي نقرة عنوان (تركيز فقط)
   350→   350→        const moved = Math.abs(x - this.activeStart.x) + Math.abs(y - this.activeStart.y) > 4;
   351→   351→        if (!moved) this.dragWin = null;
   352→   352→      }
   353→   353→      this.dragWin = null;
   354→   354→      this.dragResize = null;
   355→   355→      if (this.activeId && hit && hit.id === this.activeId) {
   356→   356→        if (this.lastClickId === hit.id && now - this.lastClickAt < 380) {
   357→   357→          hit.onDbl?.();
   358→   358→          this.lastClickId = null;
   359→   359→        } else {
   360→   360→          hit.onClick?.(opts);
   361→   361→          this.lastClickId = hit.id;
   362→   362→          this.lastClickAt = now;
   363→   363→        }
   364→   364→      }
   365→   365→      this.activeId = null;
   366→   366→      this.dirty = true;
   367→   367→      return;
   368→   368→    }
   369→   369→
   370→   370→    if (type === 'dbl') {
   371→   371→      const hit = f.hit(x, y);
   372→   372→      hit?.onDbl?.();
   373→   373→      return;
   374→   374→    }
   375→   375→
   376→   376→    if (type === 'right') {
   377→   377→      const hit = f.hit(x, y);
   378→   378→      if (hit?.onRight) hit.onRight();
   379→   379→      else if (!hit) this.desktopContextMenu(x, y);
   380→   380→      this.dirty = true;
   381→   381→      return;
   382→   382→    }
   383→   383→  }
   384→   384→
   385→   385→  wheel(x: number, y: number, dy: number): void {
   386→   386→    const f = this.lastFrame;
   387→   387→    if (!f) return;
   388→   388→    const hit = f.hit(x, y);
   389→   389→    hit?.onScroll?.(dy);
   390→   390→    this.dirty = true;
   391→   391→  }

   393→   393→  /** لوحة المفاتيح الحقيقية — توجيه V6 §4.3: Enter/Backspace/Tab/Shift/Ctrl/الأسهم/Esc */
   394→   394→  key(ev: VosKeyEvent): void {
   395→   395→    // مربع نص مركّز يلتقط الكتابة أولًا
   396→   396→    if (this.focusedTextbox) {
   397→   397→      const tb = this.focusedTextbox;
   398→   398→      if (ev.key === 'Enter') {
   399→   399→        tb.onEnter?.();
   400→   400→        this.dirty = true;
   401→   401→        return;
   402→   402→      }
   403→   403→      if (ev.key === 'Backspace') {
   404→   404→        tb.set(tb.get().slice(0, -1));
   405→   405→        this.dirty = true;
   406→   406→        return;
   407→   407→      }
   408→   408→      if (ev.key.length === 1 && !ev.ctrl && !ev.alt && !ev.meta) {
   409→   409→        tb.set(tb.get() + (ev.shift ? ev.key.toUpperCase() : ev.key));
   410→   410→        this.dirty = true;
   411→   411→        return;
   412→   412→      }
   413→   413→      if (ev.key === 'Escape') {
   414→   414→        this.focusedTextbox = null;
   415→   415→        this.dirty = true;
   416→   416→        return;
   417→   417→      }
   418→   418→    }
   419→   419→
   420→   420→    // عام: Esc يغلق قائمة ابدأ
   421→   421→    if (ev.key === 'Escape' && this.startOpen) {
   422→   422→      this.startOpen = false;
   423→   423→      this.dirty = true;
   424→   424→      return;
   425→   425→    }
   426→   426→
   427→   427→    // تمرير إلى معالج مفتاح الإطار (التطبيق النشط)
