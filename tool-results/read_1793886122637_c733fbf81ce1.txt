    83→export function roundRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
    84→  const rr = Math.min(r, w / 2, h / 2);
    85→  ctx.beginPath();
    86→  ctx.moveTo(x + rr, y);
    87→  ctx.arcTo(x + w, y, x + w, y + h, rr);
    88→  ctx.arcTo(x + w, y + h, x, y + h, rr);
    89→  ctx.arcTo(x, y + h, x, y, rr);
    90→  ctx.arcTo(x, y, x + w, y, rr);
    91→  ctx.closePath();
    92→}

   108→export class UiFrame {
   109→  rects: HotRect[] = [];
   110→  keyHandler: ((ev: VosKeyEvent) => boolean) | null = null; // true = استُهلك
   111→  hoverId: string | null = null;
   112→  textCursor: boolean = false;
   113→
   114→  add(r: HotRect): void {
   115→    this.rects.push(r);
   116→  }
   117→
   118→  hit(x: number, y: number): HotRect | null {
   119→    // الأعلى z أولًا وعند التساوي الأحدث (ترتيب الإضافة يطابق ترتيب الرسم فوق بعضه)
   120→    let best: HotRect | null = null;
   121→    for (const r of this.rects) {
   122→      if (x >= r.x && x <= r.x + r.w && y >= r.y && y <= r.y + r.h) {
   123→        if (!best || r.z >= best.z) best = r;
   124→      }
   125→    }
   126→    return best;
   127→  }
   128→}

   202→  openApp(appId: string, args?: Record<string, unknown>): VosWindow | null {
   203→    const app = this.apps.get(appId);
   204→    if (!app) return null;
   205→    if (app.fullscreenOnly) {
   206→      // تطبيقات ملء الشاشة (المثبّت) تُدار كأوضاع لا نوافذ
   207→      this.mode = 'setup';
   208→      this.dirty = true;
   209→      return null;
   210→    }
   211→    const existing = this.windows.find((w) => w.appId === appId);
   212→    if (existing && !args) {
   213→      existing.minimized = false;
   214→      existing.z = ++this.zCounter;
   215→      this.dirty = true;
   216→      return existing;
   217→    }
   218→    const offset = (this.windows.length % 5) * 26;
   219→    const win: VosWindow = {
   220→      id: `win-${this.nextWinId++}`,
   221→      appId,
   222→      titleAr: app.titleAr,
   223→      x: Math.round(this.W * 0.18) + offset,
   224→      y: Math.round(this.H * 0.12) + offset,
   225→      w: Math.min(app.defaultW, this.W - 40),
   226→      h: Math.min(app.defaultH, this.H - TASKBAR_H - 40),
   227→      minimized: false,
   228→      maximized: false,
   229→      z: ++this.zCounter,
   230→      state: { ...(args ?? {}) },
   231→    };
   232→    this.windows.push(win);
   233→    this.dirty = true;
   234→    return win;
   235→  }
   236→
   237→  closeWindow(id: string): void {
   238→    const idx = this.windows.findIndex((w) => w.id === id);
   239→    if (idx >= 0) {
   240→      this.windows.splice(idx, 1);
   241→      this.dirty = true;
   242→    }
   243→  }
   244→
   245→  focusWindow(id: string): void {
   246→    const win = this.windows.find((w) => w.id === id);
   247→    if (win) {
   248→      win.z = ++this.zCounter;
   249→      win.minimized = false;
   250→      this.dirty = true;
   251→    }
   252→  }

   257→    this.dirty = true;

   284→      onRight: handlers.onRight,

   287→    return id;
   288→  }

   313→  pointer(type: 'move' | 'down' | 'up' | 'dbl' | 'right', x: number, y: number, opts: VosPointerOpts = {}): void {
   314→    this.cursor.x = Math.max(0, Math.min(this.W - 1, x));
   315→    this.cursor.y = Math.max(0, Math.min(this.H - 1, y));
   316→    const f = this.lastFrame;
   317→    if (!f) return;
   318→
   319→    if (type === 'move') {
   320→      const hit = f.hit(x, y);
   321→      this.cursorKind = hit ? hit.cursor : 'default';
   322→      f.hoverId = hit?.id ?? null;
   323→      this.dirty = true;
   324→      return;
   325→    }
   326→
   327→    if (type === 'down') {
   328→      const hit = f.hit(x, y);
   329→      this.activeId = hit?.id ?? null;
   330→      this.activeStart = { x, y };
   331→      this.downAt = Date.now();
   332→      if (!hit && this.startOpen) this.startOpen = false;
   333→      if (this.ctxMenu && (!hit || !hit.id.startsWith('ctx'))) this.ctxMenu = null;
   334→      if (hit?.kind === 'title') {
   335→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   336→        if (win) {
   337→          this.focusWindow(win.id);
   338→          if (!win.maximized) this.dragWin = { id: win.id, dx: x - win.x, dy: y - win.y };
   339→        }
   340→      }
   341→      if (hit?.kind === 'resize') {
   342→        const win = this.windows.find((w) => w.id === hit.id.split(':')[1]);
   343→        if (win) this.dragResize = { id: win.id, sx: x, sy: y, w0: win.w, h0: win.h };
   344→      }
   345→      this.dirty = true;
   346→      return;
   347→    }
   348→
   349→    if (type === 'up') {
   350→      const hit = f.hit(x, y);
   351→      const now = Date.now();
   352→      if (this.dragWin) {
   353→        // إن لم يتحرك المؤشر فعليًا فهي نقرة عنوان (تركيز فقط)
   354→        const moved = Math.abs(x - this.activeStart.x) + Math.abs(y - this.activeStart.y) > 4;
   355→        if (!moved) this.dragWin = null;
   356→      }
   357→      this.dragWin = null;
   358→      this.dragResize = null;
   359→      if (this.activeId && hit && hit.id === this.activeId) {
   360→        if (this.lastClickId === hit.id && now - this.lastClickAt < 380) {
   361→          hit.onDbl?.();
   362→          this.lastClickId = null;
   363→        } else {
   364→          hit.onClick?.(opts);
   365→          this.lastClickId = hit.id;
   366→          this.lastClickAt = now;
   367→        }
   368→      }
   369→      this.activeId = null;
   370→      this.dirty = true;
   371→      return;
   372→    }
   373→
   374→    if (type === 'dbl') {
   375→      const hit = f.hit(x, y);
   376→      hit?.onDbl?.();
   377→      return;
   378→    }
   379→
   380→    if (type === 'right') {
   381→      const hit = f.hit(x, y);
   382→      if (hit?.onRight) hit.onRight();
   383→      else if (!hit) this.desktopContextMenu(x, y);
   384→      this.dirty = true;
   385→      return;
   386→    }
   387→  }
   388→
   389→  wheel(x: number, y: number, dy: number): void {
   390→    const f = this.lastFrame;
   391→    if (!f) return;
   392→    const hit = f.hit(x, y);
   393→    hit?.onScroll?.(dy);
   394→    this.dirty = true;
   395→  }

   397→  /** لوحة المفاتيح الحقيقية — توجيه V6 §4.3: Enter/Backspace/Tab/Shift/Ctrl/الأسهم/Esc */
   398→  key(ev: VosKeyEvent): void {
   399→    // مربع نص مركّز يلتقط الكتابة أولًا
   400→    if (this.focusedTextbox) {
   401→      const tb = this.focusedTextbox;
   402→      if (ev.key === 'Enter') {
   403→        tb.onEnter?.();
   404→        this.dirty = true;
   405→        return;
   406→      }
   407→      if (ev.key === 'Backspace') {
   408→        tb.set(tb.get().slice(0, -1));
   409→        this.dirty = true;
   410→        return;
   411→      }
   412→      if (ev.key.length === 1 && !ev.ctrl && !ev.alt && !ev.meta) {
   413→        tb.set(tb.get() + (ev.shift ? ev.key.toUpperCase() : ev.key));
   414→        this.dirty = true;
   415→        return;
   416→      }
   417→      if (ev.key === 'Escape') {
   418→        this.focusedTextbox = null;
   419→        this.dirty = true;
   420→        return;
   421→      }
   422→    }
   423→
   424→    // عام: Esc يغلق قائمة ابدأ
   425→    if (ev.key === 'Escape' && this.startOpen) {
   426→      this.startOpen = false;
   427→      this.dirty = true;
   428→      return;
   429→    }
   430→
   431→    // تمرير إلى معالج مفتاح الإطار (التطبيق النشط)
   432→    if (this.frame.keyHandler && this.frame.keyHandler(ev)) {
   433→      this.dirty = true;
   434→      return;
   435→    }
   436→    this.dirty = true;
   437→  }

   441→  /** يرسم إطارًا كاملًا ويعيد true إذا اكتمل رسم فعلي */
   442→  render(now: number): boolean {
   443→    this.frame = new UiFrame();
   444→    // (إصلاح تدقيق V6) معرّفات مستقرة بين الإطارات: كانت r1، r2… تتزايد عالميًا عبر الرسوم،
   445→    // فتتغير هوية المستطيل نفسه بين pointerdown وpointerup (الرسم يحدث بينهما)
   446→    // ويسقط النقر كله على الشاشة المجسمة. التصفير مع كل إطار يجعل الإطارات المتطابقة
   447→// تنتج معرّفات متطابقة — والمطابقة في activeId تنجح.
   448→    this.rectCounter = 1;
   449→    const ctx = this.ctx;
   450→    this.blinkAt = now;
   451→
   452→    ctx.save();
   453→    ctx.clearRect(0, 0, this.W, this.H);
   454→
   455→    if (this.mode === 'desktop') {
   456→      this.drawDesktop(ctx, now);
   457→    } else if (this.mode === 'firmware') {
   458→      this.drawFirmware(ctx, now);
   459→    } else if (this.mode === 'recovery') {
   460→      this.drawRecovery(ctx, now);
   461→    }
   462→    // mode 'setup' يرسمه تطبيق المثبّت عبر drawSetupWith
   463→
   464→    this.drawOverlays(ctx, now);
   465→    this.drawCursor(ctx);
   466→
   467→    ctx.restore();
   468→
   469→    this.lastFrame = this.frame;
   470→    this.dirty = false;
   471→    return true;
   472→  }

   474→  /** يفرض رسم وضع setup عبر تطبيق المثبّت — يستدعى من PCScene قبل render */
   475→  drawSetupWith(fn: (ctx: CanvasRenderingContext2D, os: VirtualOS, now: number) => void, now: number): boolean {
   476→    this.frame = new UiFrame();
   477→    this.rectCounter = 1; // (إصلاح تدقيق V6) نفس سبب render() — معرّفات مستقرة بين الإطارات
   478→    const ctx = this.ctx;
   479→    ctx.save();
   480→    ctx.clearRect(0, 0, this.W, this.H);
   481→    fn(ctx, this, now);
   482→    this.drawOverlays(ctx, now);
   483→    this.drawCursor(ctx);
   484→    ctx.restore();
   485→    this.lastFrame = this.frame;
   486→    this.dirty = false;
   487→    return true;
   488→  }

   492→  private drawDesktop(ctx: CanvasRenderingContext2D, now: number): void {
   493→    // خلفية

   592→  private drawWindow(ctx: CanvasRenderingContext2D, win: VosWindow, now: number): void {
   593→    const app = this.apps.get(win.appId);

   641→      if (!win.maximized) {
   642→        win.prev = { x: win.x, y: win.y, w: win.w, h: win.h };
   643→        win.maximized = true;
   644→      } else if (win.prev) {

   647→      }

   650→    const btns: Array<{ id: string; label: string; color: string; act: () => void }> = [

   666→      bx -= btnW;

   669→

   671→    ctx.save();
   672→    roundRect(ctx, x + 1, y + titleH, w - 2, h - titleH - 1, 6);
   673→    ctx.clip();

  1125→  wrapText(ctx: CanvasRenderingContext2D, text: string, x: number, y: number, maxW: number, lineH: number): number {

  1127→    let line = '';
  1128→    let yy = y;
  1129→    for (const w of words) {
  1130→      const test = line ? `${line} ${w}` : w;
  1131→      if (ctx.measureText(test).width > maxW && line) {
  1132→        ctx.fillText(line, x, yy);
  1133→        line = w;
  1134→        yy += lineH;
  1135→      } else {
  1136→        line = test;
  1137→      }
  1138→    }
  1139→    if (line) ctx.fillText(line, x, yy);
  1140→    return yy;
  1141→  }
