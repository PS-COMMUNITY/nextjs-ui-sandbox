    61→    61→function mat(color: number, metalness: number, roughness: number, opts?: MatOpts): THREE.MeshStandardMaterial {
    62→    62→  const mapId = opts?.map ? opts.map.uuid : 'n';
    63→    63→  const key = `${partScope}|${color}|${metalness}|${roughness}|${mapId}|${opts?.emissive ?? ''}|${opts?.emissiveIntensity ?? ''}|${opts?.transparent ?? ''}|${opts?.opacity ?? ''}|${opts?.side ?? ''}`;
    64→    64→  const cached = matCache.get(key);
    65→    65→  if (cached) return cached;
    66→    66→  const m = new THREE.MeshStandardMaterial({
    67→    67→    color,
    68→    68→    metalness,
    69→    69→    roughness,
    70→    70→    map: opts?.map ?? null,
    71→    71→    emissive: opts?.emissive ?? 0x000000,
    72→    72→    emissiveIntensity: opts?.emissiveIntensity ?? 1,
    73→    73→    transparent: opts?.transparent ?? false,
    74→    74→    opacity: opts?.opacity ?? 1,
    75→    75→    side: opts?.side ?? THREE.FrontSide,
    76→    76→    flatShading: false,
    77→    77→  });
    78→    78→  matCache.set(key, m);
    79→    79→  return m;
    80→    80→}

   104→   104→function box(
   105→   105→  w: number, h: number, d: number, material: THREE.Material,
   106→   106→  x = 0, y = 0, z = 0, cast = true, receive = false,
   107→   107→): THREE.Mesh {

   109→   109→  m.position.set(x, y, z);
   110→   110→  if (castEnabled && cast) m.castShadow = true;
   111→   111→  if (receive) m.receiveShadow = true;
   112→   112→  return m;

   117→   117→/** أسطوانة سريعة بمحور محدد (12 قطعة قطاعية — اقتصادية) */
   118→   118→function cyl(radius: number, length: number, material: THREE.Material, axis: Axis, x = 0, y = 0, z = 0, cast = true): THREE.Mesh {
   119→   119→  const m = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 12), material);
   120→   120→  if (axis === 'x') m.rotation.z = Math.PI / 2;
   121→   121→  else if (axis === 'z') m.rotation.x = Math.PI / 2;
   122→   122→  m.position.set(x, y, z);
   123→   123→  if (castEnabled && cast) m.castShadow = true;
   124→   124→  return m;
   125→   125→}

   132→   132→/** كابل أنبوبي على منحنى CatmullRom (نقاط محلية داخل مجموعة القطعة) */
   133→   133→function makeCable(points: THREE.Vector3[], radius: number, color = 0x181d24): THREE.Mesh {
   134→   134→  const curve = new THREE.CatmullRomCurve3(points);
   135→   135→  const m = new THREE.Mesh(new THREE.TubeGeometry(curve, 32, radius, 8, false), mat(color, 0.3, 0.6));
   136→   136→  if (castEnabled) m.castShadow = true;
   137→   137→  return m;
   138→   138→}

   141→   141→function cablePoints(mount: readonly [number, number, number], pts: Array<readonly [number, number, number]>): THREE.Vector3[] {
   142→   142→  return pts.map((p) => new THREE.Vector3(p[0] - mount[0], p[1] - mount[1], p[2] - mount[2]));

   146→   146→function plane(w: number, h: number, material: THREE.Material, axis: 'x' | 'y' | 'z', x: number, y: number, z: number): THREE.Mesh {
   147→   147→  const m = new THREE.Mesh(new THREE.PlaneGeometry(w, h), material);
   148→   148→  if (axis === 'x') m.rotation.y = -Math.PI / 2; // مواجه -X
   149→   149→  else if (axis === 'y') m.rotation.x = -Math.PI / 2;
   150→   150→  m.position.set(x, y, z);
   151→   151→  return m;
   152→   152→}

   154→   154→interface FanParts { frame: THREE.Group; blades: THREE.Group; }
   155→   155→
   156→   156→/**
   157→   157→ * مروحة واقعية: إطار مربع كامل بحواف وأركان دائرية + فتحات براغي + حلقة داخلية
   158→   158→ * + 9 شفرات مائلة + محور بغطاء. محور الدوران الافتراضي = Y المحلي.
   159→   159→ */
   160→   160→function makeFan(radius: number): FanParts {
   161→   161→  const depth = Math.max(0.14, radius * 0.5);
   162→   162→  const half = radius * 1.32;
   163→   163→
   164→   164→  // الإطار: 4 حواف كاملة + 4 أركان أسطوانية
   165→   165→  const frame = new THREE.Group();
   166→   166→  const frameMat = mat(COLORS.plastic, 0.3, 0.62);
   167→   167→  const barGeoX = new THREE.BoxGeometry(half * 2, depth, 0.1);
   168→   168→  const barGeoZ = new THREE.BoxGeometry(0.1, depth, half * 2);
   169→   169→  for (const s of [-1, 1]) {
   170→   170→    const barX = new THREE.Mesh(barGeoX, frameMat);
   171→   171→    barX.position.set(0, 0, s * half);
   172→   172→    const barZ = new THREE.Mesh(barGeoZ, frameMat);
   173→   173→    barZ.position.set(s * half, 0, 0);
   174→   174→    if (castEnabled) { barX.castShadow = true; barZ.castShadow = true; }
   175→   175→    frame.add(barX, barZ);
   176→   176→  }
   177→   177→  const cornerGeo = new THREE.CylinderGeometry(0.055, 0.055, depth, 10);
   178→   178→  const holeGeo = new THREE.CylinderGeometry(0.026, 0.026, 0.02, 8);
   179→   179→  const holeMat = mat(COLORS.black, 0.2, 0.6);
   180→   180→  for (const sx of [-1, 1]) {
   181→   181→    for (const sz of [-1, 1]) {
   182→   182→      const corner = new THREE.Mesh(cornerGeo, frameMat);
   183→   183→      corner.position.set(sx * (half - 0.05), 0, sz * (half - 0.05));
   184→   184→      if (castEnabled) corner.castShadow = true;
   185→   185→      frame.add(corner);
   186→   186→      // فتحة برغي مظلمة على الوجه العلوي
   187→   187→      const hole = new THREE.Mesh(holeGeo, holeMat);
   188→   188→      hole.position.set(sx * half, depth / 2, sz * half);
   189→   189→      frame.add(hole);
   190→   190→    }
   191→   191→  }
   192→   192→  // حلقة داخلية محيطية
   193→   193→  const ring = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.99, 0.016, 6, 28), mat(COLORS.metalDark, 0.5, 0.5));
   194→   194→  ring.rotation.x = Math.PI / 2;
   195→   195→  if (castEnabled) ring.castShadow = true;
   196→   196→  frame.add(ring);
   197→   197→  // دعامات خلفية (ستراهنة بين الحواف خلف الشفرات)
   198→   198→  const strutGeo = new THREE.BoxGeometry(0.05, 0.02, radius * 1.7);
   199→   199→  for (const sz of [-1, 1]) {
   200→   200→    const strut = new THREE.Mesh(strutGeo, frameMat);
   201→   201→    strut.position.set(sz * radius * 0.72, -depth * 0.28, 0);
   202→   202→    strut.rotation.y = Math.PI / 2;
   203→   203→    frame.add(strut);
   204→   204→  }
   205→   205→
   206→   206→  // الشفرات: 9 شفرات مائلة 35° داخل مجموعة blades (تُدفع إلى spinners)
   207→   207→  const blades = new THREE.Group();
   208→   208→  const bladeGeo = new THREE.BoxGeometry(0.016, radius * 0.8, radius * 0.32);
   209→   209→  const bladeMat = mat(0x242c36, 0.25, 0.45);
   210→   210→  for (let i = 0; i < 9; i++) {
   211→   211→    const arm = new THREE.Object3D();
   212→   212→    arm.rotation.y = (i / 9) * Math.PI * 2;
   213→   213→    const pitch = new THREE.Object3D();
   214→   214→    pitch.position.set(radius * 0.52, 0, 0);
   215→   215→    pitch.rotation.x = 0.62; // ميلان الشفرة ≈ 35°
   216→   216→    const blade = new THREE.Mesh(bladeGeo, bladeMat);
   217→   217→    blade.rotation.z = Math.PI / 2;
   218→   218→    if (castEnabled) blade.castShadow = true;
   219→   219→    pitch.add(blade);
   220→   220→    arm.add(pitch);
   221→   221→    blades.add(arm);
   222→   222→  }
   223→   223→  const hub = new THREE.Mesh(
   224→   224→    new THREE.CylinderGeometry(radius * 0.17, radius * 0.17, depth * 0.72, 14),
   225→   225→    mat(COLORS.metalDark, 0.5, 0.4),
   226→   226→  );
   227→   227→  if (castEnabled) hub.castShadow = true;
   228→   228→  blades.add(hub);
   229→   229→  // غطاء محور بحلقة داكنة
   230→   230→  const cap = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.1, radius * 0.1, depth * 0.76, 12), mat(0x11161c, 0.3, 0.5));
   231→   231→  blades.add(cap);
   232→   232→  const capRing = new THREE.Mesh(new THREE.TorusGeometry(radius * 0.11, 0.008, 6, 16), mat(COLORS.metalMid, 0.6, 0.4));
   233→   233→  capRing.rotation.x = Math.PI / 2;
   234→   234→  capRing.position.y = depth * 0.38;
   235→   235→  blades.add(capRing);
   236→   236→
   237→   237→  return { frame, blades };
   238→   238→}

   258→   258→export function buildEnvironment(scene: THREE.Scene, quality: SceneQuality): void {
   259→   259→  scene.background = new THREE.Color(COLORS.bg);
   260→   260→  scene.fog = new THREE.Fog(COLORS.bg, 18, 42);
   261→   261→
   262→   262→  // أرضية دائرية داكنة تستقبل الظلال
   263→   263→  const ground = new THREE.Mesh(
   264→   264→    new THREE.CircleGeometry(22, 24),
   265→   265→    new THREE.MeshStandardMaterial({ color: 0x141a22, metalness: 0, roughness: 0.92, flatShading: false }),
   266→   266→  );
   267→   267→  ground.rotation.x = -Math.PI / 2;
   268→   268→  ground.position.y = 0;
   269→   269→  ground.receiveShadow = true;
   270→   270→  scene.add(ground);
   271→   271→
   272→   272→  // شبكة أرضية خفيفة
   273→   273→  const grid = new THREE.GridHelper(30, 30, 0x2a3340, 0x1a212b);
   274→   274→  grid.position.y = 0.01;
   275→   275→  const gridMat: THREE.Material | THREE.Material[] = grid.material;
   276→   276→  const gridMats = Array.isArray(gridMat) ? gridMat : [gridMat];
   277→   277→  for (const gm of gridMats) {
   278→   278→    gm.transparent = true;
   279→   279→    gm.opacity = 0.4;
   280→   280→  }
   281→   281→  scene.add(grid);
   282→   282→
   283→   283→  scene.add(new THREE.AmbientLight(0xffffff, 0.45));
   284→   284→  scene.add(new THREE.HemisphereLight(0xcfe8ff, 0x20242c, 0.5));
   285→   285→
   286→   286→  // شمس دافئة مع ظلال (حسب الجودة)
   287→   287→  const dir = new THREE.DirectionalLight(0xfff2e0, 2.0);
   288→   288→  dir.position.set(-6, 8, 6);
   289→   289→  if (quality !== 'low') {
   290→   290→    dir.castShadow = true;
   291→   291→    const size = quality === 'high' ? 2048 : 1024;
   292→   292→    dir.shadow.mapSize.set(size, size);
   293→   293→    dir.shadow.camera.left = -6;
   294→   294→    dir.shadow.camera.right = 6;
   295→   295→    dir.shadow.camera.top = 6;
   296→   296→    dir.shadow.camera.bottom = -6;
   297→   297→    dir.shadow.camera.near = 2;
   298→   298→    dir.shadow.camera.far = 30;
   299→   299→    dir.shadow.camera.updateProjectionMatrix();
   300→   300→    dir.shadow.bias = -0.0004;
   301→   301→    dir.shadow.normalBias = 0.02;
   302→   302→  }
   303→   303→  scene.add(dir);
   304→   304→
   305→   305→  const fill = new THREE.PointLight(0xffffff, 0.5, 0, 0);
   306→   306→  fill.position.set(3, 3, -2);
   307→   307→  scene.add(fill);
   308→   308→  const front = new THREE.PointLight(0xfff4e0, 1.0, 0, 0);
   309→   309→  front.position.set(-5, 3.5, 3);
   310→   310→  scene.add(front);
   311→   311→}
